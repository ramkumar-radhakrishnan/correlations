\section*{Group II two-$\rho$ terms with like logarithms grouped}
\noindent Only two different logarithms occur, $\log\frac{(x-z)^{2}}{(x-w)^{2}}$ and $\log\frac{(x^{\prime}-z)^{2}}{(x^{\prime}-w^{\prime})^{2}}$. The $N_{2a}$ logarithm splits into these two:
\begin{equation*}
\log\frac{(x-z)^{2}}{(x^{\prime}-z)^{2}}-\log\frac{(x-w)^{2}}{(x^{\prime}-w^{\prime})^{2}}=\log\frac{(x-z)^{2}}{(x-w)^{2}}-\log\frac{(x^{\prime}-z)^{2}}{(x^{\prime}-w^{\prime})^{2}}.
\end{equation*}
Two other rewritings are used:
\begin{itemize}
\item the $N_{2b}$ denominators factorise as $(x^{\prime}-w^{\prime})^{2}(x-z)^{2}(x-w)^{2}-(x-w)^{4}(x^{\prime}-z)^{2}=(x-w)^{2}\big[(x^{\prime}-w^{\prime})^{2}(x-z)^{2}-(x-w)^{2}(x^{\prime}-z)^{2}\big]$, and $(x^{\prime}-w^{\prime})^{4}(x-z)^{2}-(x-w)^{2}(x^{\prime}-z)^{2}(x^{\prime}-w^{\prime})^{2}=(x^{\prime}-w^{\prime})^{2}\big[(x^{\prime}-w^{\prime})^{2}(x-z)^{2}-(x-w)^{2}(x^{\prime}-z)^{2}\big]$;
\item every logarithmic term carries the common factor $1/\big[(x^{\prime}-w^{\prime})^{2}(x-z)^{2}-(x-w)^{2}(x^{\prime}-z)^{2}\big]$, which is taken out. The $N_{2d}$ and $N_{2e}$ terms share the denominator $(x-z)^{2}-(x-w)^{2}$ (and $(x^{\prime}-w^{\prime})^{2}-(x^{\prime}-z)^{2}$), so they are put over it.
\end{itemize}
The one term without a logarithm, $-\frac{d_{\perp}}{4}\frac{1}{(z-w)^{2}(z-w^{\prime})^{2}}$, stays separate. Row IV has no logarithm and is unchanged.

\noindent\textbf{Checks.}
\begin{itemize}
\item The grouped kernel was compared with the original sum of $N_{2a},\dots,N_{2f}$ at random points, with $d_{\perp}$ kept general; they agree to 15 digits.
\item The full coefficient of $\log\frac{(x^{\prime}-z)^{2}}{(x^{\prime}-w^{\prime})^{2}}$, including the prefactor $1/[\cdots]$, equals the full coefficient of $\log\frac{(x-z)^{2}}{(x-w)^{2}}$ with $(x,w)\leftrightarrow(x^{\prime},w^{\prime})$. Under this exchange the prefactor and the square bracket each change sign, so the second bracket is minus the first one with primed and unprimed points exchanged (verified to $10^{-16}$). This is the hermiticity of Group II seen at the level of the logarithms.
\end{itemize}
