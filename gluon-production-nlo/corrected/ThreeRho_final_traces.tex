\section*{Three-$\rho$ terms in trace form (limiting case, $\rho$'s non-commuting)}
\noindent\textbf{Conventions.}
\begin{itemize}
\item $U(x)$ is the adjoint Wilson line, a real $(N_{c}^{2}-1)\times(N_{c}^{2}-1)$ matrix, so $U^{\dagger}=U^{T}$. $(T^{e})_{ab}=-if^{eab}$ are the adjoint generators, $(D^{e})_{ab}=d^{eab}$, the index $e$ is summed, and ${\rm Tr}$ is the trace over adjoint indices. In particular $T^{e}T^{e}=N_{c}$.
\item The projectile average is colour-singlet. The $\rho$'s are operators; their order inside $\langle\cdots\rangle$ is the operator order.
\begin{align*}
\langle\rho^{a}(u)\rho^{b}(v)\rangle&=\frac{\delta^{ab}}{N_{c}^{2}-1}\langle\rho^{e}(u)\rho^{e}(v)\rangle,\\
\langle\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\rangle&=\frac{f^{abc}}{N_{c}(N_{c}^{2}-1)}\langle f^{a'b'c'}\rho^{a'}(u)\rho^{b'}(v)\rho^{c'}(t)\rangle+\frac{N_{c}\,d^{abc}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\langle d^{a'b'c'}\rho^{a'}(u)\rho^{b'}(v)\rho^{c'}(t)\rangle.
\end{align*}
\item The commutators are evaluated with $[\rho^{a}(u),\rho^{b}(v)]=i\kappa f^{abc}\rho^{c}(u)\,\delta^{(2)}(u-v)$; the evolution check used $\kappa=1$. They give the two-$\rho$ terms (B). Their $\delta$-functions have been integrated: $y=x'$ in the first bracket of (B), and $y=x$ everywhere else in (B).
\item For operators, c.c.\ means the hermitian conjugate.
\end{itemize}
\begin{align}
\frac{d^{3}N^{3\rho}_{\rm NLO}}{d^{3}k}&=\frac{d^{3}N^{(3)}}{d^{3}k}+\frac{d^{3}N^{(2)}}{d^{3}k}+{\rm c.c.}
\end{align}
\textbf{(A) Genuine three-$\rho$ terms.}
\begin{align}
\frac{d^{3}N^{(3)}}{d^{3}k}&=-\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,x',w'}e^{-ik\cdot w'}\int_{r}\int_{0}^{1}d\xi\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(y-z)\,(x-r)\cdot(z-r)}{(x'-w')^{2}(y-z)^{2}(x-r)^{2}(z-r)^{2}}\notag\\
&\qquad\times\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')U(x)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(w')T^{e}U^{\dagger}(w')U(x)\}\Big]\notag\\&\qquad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(z)-U(y)\big)D^{e}U^{\dagger}(x)\}-i\,{\rm Tr}\{U(w')T^{e}U^{\dagger}(w')\big(U(z)-U(y)\big)D^{e}U^{\dagger}(x)\}\Big]\bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\log\frac{(z-w)^{2}}{(x-w)^{2}}\frac{(x'-w')\cdot(y-w)\,(x-z)\cdot(w-z)}{(x'-w')^{2}(y-w)^{2}(x-z)^{2}(w-z)^{2}}\notag\\
&\qquad\times\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')U(x)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(w')T^{e}U^{\dagger}(w')U(x)\}\Big]\notag\\&\qquad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(w)-U(y)\big)D^{e}U^{\dagger}(x)\}-i\,{\rm Tr}\{U(w')T^{e}U^{\dagger}(w')\big(U(w)-U(y)\big)D^{e}U^{\dagger}(x)\}\Big]\bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\qquad\times\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')U(x)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(w')T^{e}U^{\dagger}(w')U(x)\}\Big]\notag\\&\qquad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(z)-U(y)\big)D^{e}U^{\dagger}(x)\}-i\,{\rm Tr}\{U(w')T^{e}U^{\dagger}(w')\big(U(z)-U(y)\big)D^{e}U^{\dagger}(x)\}\Big]\bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{32\pi^{5}}\,\frac{1}{k^{+}}\int_{0}^{1}d\xi\int_{x,z,y,x',y',w'}e^{-ik\cdot(w'-z)}\,e^{-ik\cdot(z-x)\,\xi}\notag\\
&\qquad\times\bigg(\frac{(x'-w')\cdot(z-x)\,(y-z)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}-2\Big[\frac{1}{\xi}\Big]_{+}\frac{(x'-w')\cdot(y-z)\,(z-x)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}\bigg)\notag\\
&\qquad\times\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\Big[\big\langle f^{abc}\rho^{a}(x')\rho^{b}(y')\rho^{c}(y)\big\rangle-\big\langle f^{abc}\rho^{a}(x')\rho^{b}(y)\rho^{c}(y')\big\rangle\Big]\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')\big(U(x)-U(y')\big)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(z)-U^{\dagger}(y)\big)U(w')T^{e}U^{\dagger}(w')\big(U(x)-U(y')\big)\}\Big]\notag\\
&\qquad\quad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\Big[\big\langle d^{abc}\rho^{a}(x')\rho^{b}(y')\rho^{c}(y)\big\rangle+\big\langle d^{abc}\rho^{a}(x')\rho^{b}(y)\rho^{c}(y')\big\rangle\Big]\Big[i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(z)-U(y)\big)D^{e}\big(U^{\dagger}(x)-U^{\dagger}(y')\big)\}-i\,{\rm Tr}\{U(w')T^{e}U^{\dagger}(w')\big(U(z)-U(y)\big)D^{e}\big(U^{\dagger}(x)-U^{\dagger}(y')\big)\}\Big]\bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{0}^{\frac{\vee}{k^{+}}-1}d\xi\int_{w,x',y',z,y,x}e^{-ik\cdot(y'-w)}\,e^{-ik\cdot(y'-z)\,\xi}\frac{(y'-z)^{m}\,(y-w)^{i}\,(x-z)^{j}}{(y'-z)^{2}\,(y-w)^{2}\,(x-z)^{2}}\notag\\
&\qquad\times\Bigg\{\bigg[\frac{\delta_{k'm}\delta_{ij}}{1+\xi}-\Big(\Big[\frac{1}{\xi(1+\xi)}\Big]_{+}+\frac{1}{1+\xi}\Big)\delta_{jm}\delta_{ik'}-\delta_{im}\delta_{jk'}\bigg]\frac{(x'-y')^{k'}}{(x'-y')^{2}}\notag\\
&\qquad\qquad\times\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(y')T^{e}U^{\dagger}(y')\big(U(z)-U(x)\big)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')\big(U(z)-U(x)\big)\}\Big]\notag\\&\qquad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[i\,{\rm Tr}\{U(y')T^{e}U^{\dagger}(y')\big(U(w)-U(y)\big)D^{e}\big(U^{\dagger}(z)-U^{\dagger}(x)\big)\}-i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(w)-U(y)\big)D^{e}\big(U^{\dagger}(z)-U^{\dagger}(x)\big)\}\Big]\bigg\}\notag\\
&\qquad\quad+\delta_{im}\delta_{jk'}\frac{(x'-z)^{k'}}{(x'-z)^{2}}\bigg\{\frac{1}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(z)T^{e}U^{\dagger}(z)\big(U(z)-U(x)\big)\}-{\rm Tr}\{T^{e}\big(U^{\dagger}(w)-U^{\dagger}(y)\big)U(x')T^{e}U^{\dagger}(x')\big(U(z)-U(x)\big)\}\Big]\notag\\&\qquad+\frac{N_{c}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x')\rho^{b}(x)\rho^{c}(y)\big\rangle\Big[i\,{\rm Tr}\{U(z)T^{e}U^{\dagger}(z)\big(U(w)-U(y)\big)D^{e}\big(U^{\dagger}(z)-U^{\dagger}(x)\big)\}-i\,{\rm Tr}\{U(x')T^{e}U^{\dagger}(x')\big(U(w)-U(y)\big)D^{e}\big(U^{\dagger}(z)-U^{\dagger}(x)\big)\}\Big]\bigg\}\Bigg\}
\end{align}
\textbf{(B) Two-$\rho$ terms from the commutators.}
\begin{align}
\frac{d^{3}N^{(2)}}{d^{3}k}&=-\frac{1}{(2\pi)^{3}}\,\frac{\kappa\,g^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\,\frac{1}{N_{c}^{2}-1}\int_{x,z,w,x',w'}e^{-ik\cdot(w'-w)}\log\frac{(x-z)^{2}}{(x-w)^{2}}\big\langle\rho^{a}(x')\rho^{a}(x)\big\rangle\Bigg\{\frac{(x'-w')^{i}(x'-z)^{j}}{(x'-w')^{2}(x'-z)^{2}}\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\qquad\times\Big[\tfrac12{\rm Tr}\{T^{e}U^{\dagger}(w)U(x')T^{e}U^{\dagger}(x')U(w)U^{\dagger}(z)U(x')\}-\tfrac12{\rm Tr}\{T^{e}U^{\dagger}(x)U(w')T^{e}U^{\dagger}(w')U(z)\}+{\rm Tr}\{T^{e}U^{\dagger}(x)U(x')T^{e}U^{\dagger}(x')U(z)\}\notag\\
&\qquad\quad+\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(w)U(w')\}-N_{c}{\rm Tr}\{U^{\dagger}(w)U(x')\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(x')\}\Big]\notag\\
&\quad+\frac{(x'-w')^{i}(x-z)^{j}}{(x'-w')^{2}(x-z)^{2}}\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\qquad\times\Big[-{\rm Tr}\{T^{e}U^{\dagger}(x)U(x')T^{e}U^{\dagger}(x')U(z)\}+{\rm Tr}\{T^{e}U^{\dagger}(x)U(w')T^{e}U^{\dagger}(w')U(z)\}\notag\\
&\qquad\quad+\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(x')\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(w')\}+\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(w)U(x')\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(w)U(w')\}\Big]\Bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{\kappa\,g^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\,\frac{1}{N_{c}^{2}-1}\int_{x,z,w,x',w'}\frac{e^{-ik\cdot(w'-w)}}{(x'-w')^{2}(x-z)^{2}}\big\langle\rho^{a}(x')\rho^{a}(x)\big\rangle\notag\\
&\qquad\times\Bigg\{\frac{\mathrm{arc\,cos}\left(\frac{(x-w)^{2}-(x-w)\cdot (x-z)}{\sqrt{(x-w)^{2}\,(z-w)^{2}}}\right)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\frac{1}{(z-w)^{2}}\Big[(x'-w')\cdot(z-w)\,(x-z)\cdot(x-w)-\frac{(x-w)\cdot(x-z)}{(x-z)^{2}}(x'-w')\cdot(z-w)\,(x-z)\cdot(x-z)\notag\\
&\qquad+(x'-w')\cdot(x-z)\,(x-z)\cdot(z-w)-\frac{(x-w)\cdot(x-z)}{(x-w)^{2}}(x'-w')\cdot(x-w)\,(x-z)\cdot(z-w)\Big]\notag\\
&\qquad+\frac{1}{2(z-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}\Big[\frac{(x'-w')\cdot(x-w)\,(x-z)\cdot(z-w)}{(x-w)^{2}}-\frac{(x'-w')\cdot(z-w)\,(x-z)\cdot(x-z)}{(x-z)^{2}}\Big]\Bigg\}\notag\\
&\qquad\times\Big[{\rm Tr}\{T^{e}U^{\dagger}(z)U(x')T^{e}U^{\dagger}(x')U(x)\}-N_{c}{\rm Tr}\{U^{\dagger}(x)U(x')\}-{\rm Tr}\{T^{e}U^{\dagger}(z)U(w')T^{e}U^{\dagger}(w')U(x)\}+N_{c}{\rm Tr}\{U^{\dagger}(x)U(w')\}\Big]\notag\\[6pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{\kappa\,g^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\,\frac{1}{N_{c}^{2}-1}\int_{x,z,x',w'}e^{-ik\cdot w'}\int_{r}\int_{0}^{1}d\xi\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(x-z)\,(x-r)\cdot(z-r)}{(x'-w')^{2}(x-z)^{2}(x-r)^{2}(z-r)^{2}}\big\langle\rho^{a}(x')\rho^{a}(x)\big\rangle\notag\\
&\qquad\times\Big[{\rm Tr}\{T^{e}U^{\dagger}(z)U(x')T^{e}U^{\dagger}(x')U(x)\}-N_{c}{\rm Tr}\{U^{\dagger}(x)U(x')\}-{\rm Tr}\{T^{e}U^{\dagger}(z)U(w')T^{e}U^{\dagger}(w')U(x)\}+N_{c}{\rm Tr}\{U^{\dagger}(x)U(w')\}\Big]\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{\kappa\,g^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\,\frac{1}{N_{c}^{2}-1}\int_{0}^{\frac{\vee}{k^{+}}-1}d\xi\int_{w,x',y',z,x}e^{-ik\cdot(y'-w)}\,e^{-ik\cdot(y'-z)\,\xi}\frac{(y'-z)^{m}\,(x-w)^{i}\,(x-z)^{j}}{(y'-z)^{2}\,(x-w)^{2}\,(x-z)^{2}}\big\langle\rho^{a}(x')\rho^{a}(x)\big\rangle\notag\\
&\qquad\times\Bigg\{\bigg[\frac{\delta_{k'm}\delta_{ij}}{1+\xi}-\Big(\Big[\frac{1}{\xi(1+\xi)}\Big]_{+}+\frac{1}{1+\xi}\Big)\delta_{jm}\delta_{ik'}-\delta_{im}\delta_{jk'}\bigg]\frac{(x'-y')^{k'}}{(x'-y')^{2}}\notag\\
&\qquad\qquad\times\Big[\tfrac12{\rm Tr}\{T^{e}U^{\dagger}(w)U(y')T^{e}U^{\dagger}(y')U(z)\}-\tfrac12{\rm Tr}\{T^{e}U^{\dagger}(w)U(x')T^{e}U^{\dagger}(x')U(z)\}-{\rm Tr}\{T^{e}U^{\dagger}(x)U(y')T^{e}U^{\dagger}(y')U(z)\}\notag\\
&\qquad\qquad\quad+{\rm Tr}\{T^{e}U^{\dagger}(x)U(x')T^{e}U^{\dagger}(x')U(z)\}+\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(y')\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(x')\}\Big]\notag\\
&\qquad\quad+\delta_{im}\delta_{jk'}\frac{(x'-z)^{k'}}{(x'-z)^{2}}\Big[\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(w)U(z)\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(z)\}-\tfrac{N_{c}}{2}{\rm Tr}\{U^{\dagger}(x)U(x')\}\notag\\
&\qquad\qquad\quad-\tfrac12{\rm Tr}\{T^{e}U^{\dagger}(w)U(x')T^{e}U^{\dagger}(x')U(z)\}+{\rm Tr}\{T^{e}U^{\dagger}(x)U(x')T^{e}U^{\dagger}(x')U(z)\}\Big]\Bigg\}
\end{align}
\noindent\textbf{Checks.}
\begin{itemize}
\item Every trace bracket was checked against the index contraction of the corresponding colour-free term, with exact SU(3) adjoint Wilson lines.
\item The averaging and commutator steps were checked against direct expectation values: $\rho^{a}$ was realised as SU(3) generators acting on colour-singlet states, so $\kappa=1$ and the $\rho$'s really do not commute. Each two-$\rho$ bracket of (B), including all the coefficients $\tfrac12$, $N_{c}$ and the signs, agrees to $10^{-10}$.
\item The d-part of (A) was checked in the same way with a baryon-like singlet state.
\end{itemize}
\textbf{Remarks.} For a Gaussian (MV) projectile, $\langle f\rho\rho\rho\rangle=\langle d\rho\rho\rho\rangle=0$, so only the two-$\rho$ terms (B) survive. The $d$-correlator is odd under charge conjugation and vanishes for a C-even projectile ensemble.
