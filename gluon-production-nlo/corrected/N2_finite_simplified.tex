\section*{Notation}
\begin{equation}
a=x'-w',\qquad r=x-z,\qquad u=y-x,\qquad n=y-z,\qquad \xi=\frac{p^{+}}{k^{+}} .
\end{equation}
These satisfy
\begin{equation}
n=u+r,\qquad n^{2}-u^{2}=2\,n\cdot r-r^{2},\qquad u\cdot r=n\cdot r-r^{2},\qquad u\cdot n=n^{2}-n\cdot r .
\end{equation}
The two colour structures are
\begin{align}
\mathcal{C}_{2}&=\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[-f^{adc}f^{fbe}U^{db}(x)U^{ce}(z)+N_{c}U^{af}(y)\Big]\rho^{b'}(x')\rho^{f}(y),\\
\mathcal{C}_{z}&=\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}U^{af}(y)-N_{c}U^{af}(z)\Big]\rho^{b'}(x')\rho^{f}(y).
\end{align}
Every tensor bracket is contracted with $T^{im}=\frac{a^{i}(z-x)^{m}}{a^{2}r^{2}}=-\frac{a^{i}r^{m}}{a^{2}r^{2}}$; below we multiply by $a^{2}r^{2}$. Note that $a^{2}r^{2}\,\delta^{im}T^{im}=-a\cdot r$. In the finite term $d_{\perp}=2$.

\section*{Step 1: contract each bracket}
\begin{align}
\text{$d_{\perp}$ term:}\quad&\xi(1-\xi)\,\frac{u^{2}-n^{2}}{r^{2}}\,(-a\cdot r)=\xi(1-\xi)\,\frac{(a\cdot r)(2\,n\cdot r-r^{2})}{r^{2}},\\
\text{$(1-\xi)\delta$ term:}\quad&\Big[\frac{n\cdot r}{r^{2}}+\frac{u\cdot n}{2u^{2}}-\frac{u^{2}-n^{2}}{2r^{2}}\Big]=\frac{2\,n\cdot r}{r^{2}}-\frac12+\frac{u\cdot n}{2u^{2}}
\ \Rightarrow\ -(1-\xi)(a\cdot r)\Big[\frac{2\,n\cdot r}{r^{2}}-\frac12+\frac{u\cdot n}{2u^{2}}\Big],\\
\text{$\xi\delta$ term:}\quad&\Big[\frac{u\cdot r}{r^{2}}-\frac{u\cdot n}{2n^{2}}-\frac{u^{2}-n^{2}}{2r^{2}}\Big]=\frac{2\,n\cdot r}{r^{2}}-\frac32-\frac{u\cdot n}{2n^{2}}
\ \Rightarrow\ -\xi(a\cdot r)\Big[\frac{2\,n\cdot r}{r^{2}}-\frac32-\frac{u\cdot n}{2n^{2}}\Big],\\
\text{last bracket:}\quad&(a\cdot r)\Big[\frac{2\,n\cdot r}{r^{2}}-1\Big]-\frac{(a\cdot n)(u\cdot r)}{2n^{2}}+\frac{(a\cdot u)(n\cdot r)}{2u^{2}},\\
\text{$+B$ term:}\quad&a^{2}r^{2}\,T^{im}B^{im}=-(a\cdot n)-\frac{(a\cdot n)(u\cdot r)}{2u^{2}},\\
\text{$[k^{+}/p^{+}]_{+}B$ term:}\quad&-\Big[\frac{k^{+}}{p^{+}}\Big]_{+}a^{2}r^{2}\,T^{im}B^{im}=+\Big[\frac{k^{+}}{p^{+}}\Big]_{+}G_{p},\qquad G_{p}=(a\cdot n)\Big[1+\frac{u\cdot r}{2u^{2}}\Big],\\
\text{$q^{+}$ block:}\quad&a^{2}r^{2}\,T^{im}A^{im}=-G_{q},\qquad G_{q}=(a\cdot u)\Big[1-\frac{n\cdot r}{2n^{2}}\Big].
\end{align}

\section*{Step 2: collect by powers of $\xi$}
The $\frac{2n\cdot r}{r^{2}}$ pieces of the $(1-\xi)\delta$, $\xi\delta$ and last brackets cancel, because $-(1-\xi)-\xi+1=0$. The constants give $(a\cdot r)\big[\frac{1-\xi}{2}+\frac{3\xi}{2}-1\big]=\xi(a\cdot r)-\frac12(a\cdot r)$. The regular part of the kernel is therefore
\begin{equation}
\mathcal{R}(\xi)=\xi(1-\xi)\,\mathcal{A}+\xi\,\mathcal{B}+\mathcal{C},
\end{equation}
with
\begin{align}
\mathcal{A}&=\frac{(a\cdot r)\,(2\,n\cdot r-r^{2})}{r^{2}},\\
\mathcal{B}&=(a\cdot r)\Big[1+\frac{u\cdot n}{2u^{2}}+\frac{u\cdot n}{2n^{2}}\Big],\\
\mathcal{C}&=-\frac{a\cdot r}{2}-a\cdot n-\frac{(a\cdot r)(u\cdot n)+(a\cdot n)(u\cdot r)-(a\cdot u)(n\cdot r)}{2u^{2}}-\frac{(a\cdot n)(u\cdot r)}{2n^{2}} .
\end{align}

\section*{Step 3: simplify $\mathcal{C}$}
Use $a\cdot u=a\cdot n-a\cdot r$, $u\cdot n=n^{2}-n\cdot r$ and $u\cdot r=n\cdot r-r^{2}$:
\begin{align}
&(a\cdot r)(u\cdot n)+(a\cdot n)(u\cdot r)-(a\cdot u)(n\cdot r)\notag\\
&=(a\cdot r)n^{2}-(a\cdot r)(n\cdot r)+(a\cdot n)(n\cdot r)-(a\cdot n)r^{2}-(a\cdot n)(n\cdot r)+(a\cdot r)(n\cdot r)\notag\\
&=(a\cdot r)\,n^{2}-(a\cdot n)\,r^{2},
\end{align}
so that
\begin{equation}
\mathcal{C}=-\frac{a\cdot r}{2}-a\cdot n-\frac{(a\cdot r)\,n^{2}-(a\cdot n)\,r^{2}}{2u^{2}}-\frac{(a\cdot n)(u\cdot r)}{2n^{2}} .
\end{equation}
Steps 1--3 were checked numerically at random points: $\xi(1-\xi)\mathcal{A}+\xi\mathcal{B}+\mathcal{C}$ equals the original contracted bracket (including the $+B$ term) to $10^{-15}$.

\section*{Step 4: the log terms}
\begin{align}
-\frac{T^{im}B^{im}}{(y-z)^{2}}&=\frac{G_{p}}{a^{2}r^{2}n^{2}},\qquad -\frac{T^{im}A^{im}}{(y-x)^{2}}=\frac{G_{q}}{a^{2}r^{2}u^{2}},\qquad
K_{S}=\frac{(a\cdot n)(u\cdot n)}{a^{2}\,n^{2}\,u^{2}\,r^{2}} .
\end{align}

\section*{Result}
In the original variables:
\begin{align}
\mathcal{A}&=\frac{(x'-w')\cdot(x-z)\,\big[2\,(y-z)\cdot(x-z)-(x-z)^{2}\big]}{(x-z)^{2}},\notag\\
\mathcal{B}&=(x'-w')\cdot(x-z)\left[1+\frac{(y-x)\cdot(y-z)}{2(y-x)^{2}}+\frac{(y-x)\cdot(y-z)}{2(y-z)^{2}}\right],\notag\\
\mathcal{C}&=-\frac{(x'-w')\cdot(x-z)}{2}-(x'-w')\cdot(y-z)-\frac{(x'-w')\cdot(x-z)\,(y-z)^{2}-(x'-w')\cdot(y-z)\,(x-z)^{2}}{2(y-x)^{2}}\notag\\
&\quad-\frac{(x'-w')\cdot(y-z)\,(y-x)\cdot(x-z)}{2(y-z)^{2}},\notag\\
G_{p}&=(x'-w')\cdot(y-z)\left[1+\frac{(y-x)\cdot(x-z)}{2(y-x)^{2}}\right],\qquad
G_{q}=(x'-w')\cdot(y-x)\left[1-\frac{(y-z)\cdot(x-z)}{2(y-z)^{2}}\right],
\end{align}
and
\begin{align}
&\frac{d^{3}N_{2a\,\rm NLO}}{d^{3}k}+\frac{d^{3}N_{2b\,\rm finite\,NLO}}{d^{3}k}=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{0}^{k^{+}}\frac{dp^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\Bigg\{\notag\\
&\quad\frac{e^{-ik\cdot(z-x)\frac{p^{+}}{k^{+}}}}{\big[p^{+}(y-x)^{2}+(k^{+}-p^{+})(y-z)^{2}\big]\,(x'-w')^{2}(x-z)^{2}}
\left[\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}\,\mathcal{A}+\frac{p^{+}}{k^{+}}\,\mathcal{B}+\mathcal{C}+\Big[\frac{k^{+}}{p^{+}}\Big]_{+}G_{p}\right]\notag\\
&\qquad\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[-f^{adc}\,f^{fbe}\,U^{db}(x)\,U^{ce}(z)+N_{c}\,U^{af}(y)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag\\
&\quad-\frac{1}{k^{+}}\left[\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}+\left[\frac{k^{+}}{k^{+}-p^{+}}\right]_{+}+\left[\frac{k^{+}}{p^{+}}\right]_{+}-2\right]\frac{(x'-w')\cdot(y-z)\,(y-x)\cdot(y-z)}{(x'-w')^{2}\,(y-z)^{2}\,(y-x)^{2}\,(x-z)^{2}}\notag\\
&\qquad\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(z)\Big]\rho^{b'}(x')\,\rho^{f}(y)\Bigg\}\notag\\
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{0}^{k^{+}}\frac{dq^{+}}{2\pi}\int_{x,y,z,x',w'}
\frac{e^{-ik\cdot(w'-x)}\,e^{-ik\cdot(x-z)\frac{q^{+}}{k^{+}}}}{\big[(k^{+}-q^{+})(y-x)^{2}+q^{+}(y-z)^{2}\big]\,(x'-w')^{2}(x-z)^{2}}\;G_{q}\left\{\Big[\frac{k^{+}}{q^{+}}\Big]_{+}-1\right\}\notag\\
&\qquad\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[-f^{adc}\,f^{fbe}\,U^{db}(x)\,U^{ce}(z)+N_{c}\,U^{af}(y)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag\\
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\,{\color{blue}\log\left[\frac{k^{+}-\Lambda}{\Lambda}\right]}\int_{x,y,z,x',w'}\Bigg\{\left[\frac{e^{-ik\cdot(w'-z)}\,G_{p}}{(x'-w')^{2}(x-z)^{2}(y-z)^{2}}+\frac{e^{-ik\cdot(w'-x)}\,G_{q}}{(x'-w')^{2}(x-z)^{2}(y-x)^{2}}\right]\notag\\
&\qquad\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[-f^{adc}\,f^{fbe}\,U^{db}(x)\,U^{ce}(z)+N_{c}\,U^{af}(y)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag\\
&\quad-2\,e^{-ik\cdot(w'-z)}\,\frac{(x'-w')\cdot(y-z)\,(y-x)\cdot(y-z)}{(x'-w')^{2}\,(y-z)^{2}\,(y-x)^{2}\,(x-z)^{2}}\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(z)\Big]\rho^{b'}(x')\,\rho^{f}(y)\Bigg\}.
\end{align}

\section*{Notes}
\begin{itemize}
\item The $q^{+}$ integral has been turned around: $\int_{k^{+}}^{0}dq^{+}\,T^{im}A^{im}(\cdots)=+\int_{0}^{k^{+}}dq^{+}\,\frac{G_{q}}{a^{2}r^{2}}(\cdots)$.
\item The two log signs flip to $+$ because $-T^{im}B^{im}=+G_{p}/(a^{2}r^{2})$ and $-T^{im}A^{im}=+G_{q}/(a^{2}r^{2})$.
\item The subtraction log keeps $-2\times\frac{g^{4}}{16\pi^{5}}=-\frac{g^{4}}{8\pi^{5}}$.
\item The full $N_{2}$ is this result plus $N_{2b\,\rm UV}$, which is unchanged.
\end{itemize}
