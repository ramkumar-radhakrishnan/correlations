\section*{Three-$\rho$ terms without colour indices}
\noindent\textbf{Notation.} $\rho(x)$ is the colour vector with components $\rho^{a}(x)$. $U(x)$ is the adjoint Wilson line, a real orthogonal $(N_{c}^{2}-1)\times(N_{c}^{2}-1)$ matrix, and $U^{T}$ is its transpose. $U(x)\rho(y)$ is the vector with components $U^{ab}(x)\rho^{b}(y)$, and $\big[U(x)-U(y)\big]\rho(z)=U(x)\rho(z)-U(y)\rho(z)$. For colour vectors,
\begin{equation*}
A\cdot B\equiv A^{a}B^{a},\qquad (A\times B)^{a}\equiv f^{abc}A^{b}B^{c},\qquad A\cdot(B\times C)=f^{abc}A^{a}B^{b}C^{c}.
\end{equation*}
The $\times$ binds more strongly than the $\cdot$: $A\cdot B\times C$ means $A\cdot(B\times C)$. The colour $\cdot$ never multiplies two transverse vectors, and the transverse $\cdot$ (between differences of positions) never multiplies two colour vectors.

\noindent\textbf{Operator order.} The $\rho$'s are operators and do not commute. In every product they act in the order in which they are written, from left to right. For an expression $E$ containing $\rho(u)$ and $\rho(v)$,
\begin{equation*}
\big[E\big]_{[\rho(u),\rho(v)]}\equiv E\big|_{\rho(u)\text{ left of }\rho(v)}-E\big|_{\rho(v)\text{ left of }\rho(u)},\qquad
\big[E\big]_{\{\rho(u),\rho(v)\}}\equiv E\big|_{\rho(u)\text{ left of }\rho(v)}+E\big|_{\rho(v)\text{ left of }\rho(u)},
\end{equation*}
with all colour contractions, and the position of every other $\rho$, unchanged. For example,
\begin{equation*}
\big[A\cdot(U(z)\rho(y)\times U(x)\rho(x))\big]_{[\rho(y),\rho(x)]}=f^{abd}A^{a}U^{bc}(z)U^{de}(x)\big[\rho^{c}(y),\rho^{e}(x)\big].
\end{equation*}
In every term, $\rho(x')$ stands leftmost. For operators, c.c.\ is the hermitian conjugate.
