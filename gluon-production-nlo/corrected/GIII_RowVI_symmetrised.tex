\section{Starting point and identities}
The term is the Group III Row VI three-$\rho$ term (with c.c.), after contraction of all transverse indices. The ordering of the $\rho$'s is kept, with $[\rho^{a}(u),\rho^{b}(v)]=i\kappa f^{abc}\rho^{c}(u)\,\delta^{(2)}(u-v)$.

\noindent\textbf{Step 1: colour.} With $U^{d'b}(z)U^{cb}(z)=\delta^{d'c}$ (checked numerically, residual $6\times10^{-12}$),
\begin{equation}
U^{d'b}(z)\Big[U^{cb}(z)\,U^{ad}(y)\,U^{ce}(x)\,\rho^{d}(y)\,\rho^{e}(x)-U^{ad}(y)\,\rho^{d}(y)\,\rho^{b}(x)\Big]=U^{ad}(y)\Big[U^{d'e}(x)-U^{d'e}(z)\Big]\rho^{d}(y)\rho^{e}(x).
\end{equation}
Each of the three terms therefore contains the single ordered product $\rho^{e'}(x')\rho^{d}(y)\rho^{e}(x)$ multiplied by the colour tensor
\begin{equation}
f^{c'd'a}\Big[U^{c'e'}(q)-U^{c'e'}(x')\Big]U^{ad}(y)\Big[U^{d'e}(x)-U^{d'e}(z)\Big],\qquad q=y'\ \text{(first two terms)},\quad q=z\ \text{(third term)}.
\end{equation}

\noindent\textbf{Step 2: symmetrise.} Repeated use of $\rho\rho=\frac12\{\rho,\rho\}+\frac12[\rho,\rho]$ gives (operator identity, checked numerically)
\begin{align}
\rho^{e'}(x')\rho^{d}(y)\rho^{e}(x)&=\big\{\rho^{e'}(x')\rho^{d}(y)\rho^{e}(x)\big\}_{\rm sym}+\frac{i\kappa}{4}\Big[f^{e'dg}\,\delta^{(2)}(x'-y)\big\{\rho^{g}(x'),\rho^{e}(x)\big\}+f^{e'eg}\,\delta^{(2)}(x'-x)\big\{\rho^{g}(x'),\rho^{d}(y)\big\}\notag\\
&\quad+f^{deg}\,\delta^{(2)}(y-x)\big\{\rho^{e'}(x'),\rho^{g}(y)\big\}\Big]+O(\kappa^{2})\ \text{one-}\rho\ \text{terms},
\end{align}
where $\{\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\}_{\rm sym}$ is the average over the six orderings. The $O(\kappa^{2})$ one-$\rho$ terms are dropped.

\noindent\textbf{Step 3: integrate the $\delta$'s.} The point $x'$ is kept: $\delta^{(2)}(x'-y)$ sets $y\to x'$, $\delta^{(2)}(x'-x)$ sets $x\to x'$, and $\delta^{(2)}(y-x)$ sets $y\to x$. The substitution is made in the kernel, the colour factor and the $\rho$'s, and the point is removed from the measure (checked automatically). Each two-$\rho$ block is labelled by the $\delta$ that was integrated.
