\begin{center}{\large\bf Group I, Row III ($\Lambda\to0$, $\vee\to\infty$): simplification of $N_{2}+N_{3a}$}\end{center}

\section*{Step 0: notation and common factor}
Every term has the same prefactor, phase and colour structure:
\begin{equation}
\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}f^{abd}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{w,w'}e^{-ik\cdot(w'-w)}\int_{x,y,z,x'}\frac{1}{(x'-w')^{2}(y-z)^{2}}\;(\ldots)\;\mathcal C,
\end{equation}
\begin{equation}
\mathcal C=\left[U^{ab'}(x')\rho^{b'}(x')-U^{ab'}(w')\rho^{b'}(x')\right]\left[U^{de}(x)U^{bc}(z)\rho^{e}(x)\rho^{c}(y)-U^{de}(x)U^{bc}(y)\rho^{e}(x)\rho^{c}(y)\right].
\end{equation}
Below we write only $(\ldots)$. We abbreviate
\begin{equation}
X=x-w,\quad Z=x-z,\quad W=z-w=X-Z,\quad X'=x'-w',\quad Y=y-z,\quad D=X^{2}-Z^{2},\quad u=\frac{Z^{2}}{X^{2}},
\end{equation}
\begin{equation}
L_{2}=\log\frac{-(x-z)^{2}}{(x-w)^{2}-(x-z)^{2}}=\log\frac{-Z^{2}}{D},\qquad L_{3}=\log\frac{(x-w)^{2}-(x-z)^{2}}{(x-w)^{2}}=\log\frac{D}{X^{2}}.
\end{equation}

\section*{Step 1: the terms}
$N_{2}$ has an overall minus sign in front of its braces. Taking it inside, $N_{2}=L_{2}\,(a_{1}+\dots+a_{6})$ with
\begin{align}
a_{1}&=-\frac{(X'\cdot W)(Y\cdot X)}{W^{2}D}, &
a_{2}&=+\frac{(X'\cdot X)(Y\cdot W)\,Z^{2}}{W^{2}D\,X^{2}}, &
a_{3}&=-\frac{(X'\cdot Y)(W\cdot X)\,Z^{2}}{W^{2}D^{2}},\notag\\
a_{4}&=+\frac{(X'\cdot W)(Y\cdot Z)\,X^{2}}{W^{2}D\,Z^{2}}, &
a_{5}&=-\frac{(X'\cdot Z)(Y\cdot W)}{W^{2}D}, &
a_{6}&=+\frac{(X'\cdot Y)(W\cdot Z)\,X^{2}}{W^{2}D^{2}}.
\end{align}
$N_{3a}=L_{3}\,(b_{1}+\dots+b_{6})+\mathcal A_{0}\,(c_{1}+c_{2}+c_{3})$ with
\begin{align}
b_{1}&=-\frac{(X'\cdot Y)(W\cdot X)\,Z^{2}}{W^{2}D^{2}}, &
b_{2}&=+\frac{(X'\cdot Y)(W\cdot Z)\,X^{2}}{W^{2}D^{2}}, &
b_{3}&=-\frac{(X'\cdot W)(Y\cdot X)}{W^{2}D},\notag\\
b_{4}&=+\frac{(X'\cdot W)(Y\cdot Z)\,X^{2}}{W^{2}D\,Z^{2}}, &
b_{5}&=+\frac{(Y\cdot W)(X'\cdot X)\,Z^{2}}{W^{2}D\,X^{2}}, &
b_{6}&=-\frac{(Y\cdot W)(X'\cdot Z)}{W^{2}D},
\end{align}
\begin{equation}
c_{1}=+\frac{(X'\cdot W)(Y\cdot X)}{W^{2}},\qquad c_{2}=+\frac{(Y\cdot W)(X'\cdot Z)}{W^{2}},\qquad c_{3}=-\frac{(X\cdot Z)(X'\cdot W)(Y\cdot Z)}{W^{2}Z^{2}}.
\end{equation}

\section*{Step 2: the $N_{2}$ and $N_{3a}$ tensors are the same}
Comparing term by term:
\begin{equation}
a_{1}=b_{3},\quad a_{2}=b_{5},\quad a_{3}=b_{1},\quad a_{4}=b_{4},\quad a_{5}=b_{6},\quad a_{6}=b_{2}.
\end{equation}
Hence, with $\mathcal T\equiv b_{1}+\dots+b_{6}$,
\begin{equation}
N_{2}+N_{3a}\big|_{\log}=(L_{2}+L_{3})\,\mathcal T.
\end{equation}

\section*{Step 3: adding the two logarithms}
Both logarithms come from the same energy denominator and are on the principal branch, $\log(-A)=\log A+i\pi$ for $A>0$.
\begin{itemize}
\item $u<1$ ($D>0$): $L_{2}=\log\frac{u}{1-u}+i\pi$ and $L_{3}=\log(1-u)$.
\item $u>1$ ($D<0$): $L_{2}=\log\frac{u}{u-1}$ and $L_{3}=\log(u-1)+i\pi$.
\end{itemize}
In both cases
\begin{equation}
L_{2}+L_{3}=\log\frac{-Z^{2}}{X^{2}}=\log\frac{(x-z)^{2}}{(x-w)^{2}}+i\pi .
\end{equation}
The logarithms $\log|1-u|$, which are singular at $|x-z|=|x-w|$, have cancelled.

\section*{Step 4: reducing the tensor $\mathcal T$}
Write $\mathcal T=X'^{i}Y^{j}\,\mathcal T^{ij}$. From Step 1,
\begin{equation}
W^{2}D\;\mathcal T^{ij}=\underbrace{-W^{i}X^{j}+u\,X^{i}W^{j}+\frac{1}{u}\,W^{i}Z^{j}-Z^{i}W^{j}}_{\text{non-}\delta\ (b_{3},b_{5},b_{4},b_{6})}
+\underbrace{\delta^{ij}\,\frac{(W\cdot Z)X^{2}-(W\cdot X)Z^{2}}{D}}_{\delta\ (b_{1},b_{2})}.
\end{equation}

\noindent\textbf{Non-$\delta$ part.} Use $X=W+Z$ in the first two terms:
\begin{align}
-W^{i}X^{j}+uX^{i}W^{j}&=-W^{i}W^{j}-W^{i}Z^{j}+u\,W^{i}W^{j}+u\,Z^{i}W^{j},\notag\\
\Rightarrow\ \text{non-}\delta&=(u-1)W^{i}W^{j}+(u-1)Z^{i}W^{j}+\Big(\frac1u-1\Big)W^{i}Z^{j}
=(u-1)\Big[X^{i}W^{j}-\frac{1}{u}W^{i}Z^{j}\Big].
\end{align}
Since $D=X^{2}(1-u)$, the factor $(u-1)$ cancels the pole:
\begin{equation}
\frac{\text{non-}\delta}{W^{2}D}=-\frac{X^{i}W^{j}}{W^{2}X^{2}}+\frac{W^{i}Z^{j}}{W^{2}Z^{2}}.
\end{equation}

\noindent\textbf{$\delta$ part.} Use $W=X-Z$:
\begin{equation}
(W\cdot Z)X^{2}-(W\cdot X)Z^{2}=(X\cdot Z)(X^{2}+Z^{2})-2X^{2}Z^{2}.
\end{equation}
Then use $2\,X\cdot Z=X^{2}+Z^{2}-W^{2}$:
\begin{equation}
(X\cdot Z)(X^{2}+Z^{2})-2X^{2}Z^{2}=\tfrac12\big[(X^{2}+Z^{2})^{2}-4X^{2}Z^{2}\big]-\tfrac12W^{2}(X^{2}+Z^{2})=\tfrac12\big[D^{2}-W^{2}(X^{2}+Z^{2})\big].
\end{equation}
Dividing by $W^{2}D^{2}$:
\begin{equation}
\frac{\delta\text{ part}}{W^{2}D}=\delta^{ij}\Big[\frac{1}{2W^{2}}-\frac{X^{2}+Z^{2}}{2D^{2}}\Big].
\end{equation}

\noindent\textbf{Result.}
\begin{equation}
\mathcal T^{ij}=\frac{\delta^{ij}}{2(z-w)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{\delta^{ij}\,\big[(x-w)^{2}+(x-z)^{2}\big]}{2\big[(x-w)^{2}-(x-z)^{2}\big]^{2}}.
\end{equation}
In terms of the Row IV kernel $\mathcal K^{ij}_{\rm IV}=\frac{\delta^{ij}}{2W^{2}}+\frac{W^{i}Z^{j}}{W^{2}Z^{2}}-\frac{X^{i}W^{j}}{W^{2}X^{2}}-\frac{X^{i}Z^{j}}{X^{2}Z^{2}}$,
\begin{equation}
\mathcal T^{ij}=\mathcal K^{ij}_{\rm IV}+\frac{X^{i}Z^{j}}{X^{2}Z^{2}}-\frac{\delta^{ij}(X^{2}+Z^{2})}{2D^{2}}.
\end{equation}

\section*{Step 5: the $\mathcal A_{0}$ terms}
$c_{1}$ and $c_{3}$ share $(X'\cdot W)$:
\begin{equation}
c_{1}+c_{3}=\frac{(X'\cdot W)}{W^{2}}\Big[Y\cdot X-\frac{(X\cdot Z)(Y\cdot Z)}{Z^{2}}\Big]=\frac{(X'\cdot W)\,(Y\cdot X_{\perp})}{W^{2}},\qquad X_{\perp}\equiv X-\frac{X\cdot Z}{Z^{2}}\,Z .
\end{equation}
$X_{\perp}$ is the component of $x-w$ perpendicular to $x-z$. In geometric form, $\mathcal A_{0}=\frac{\theta_{w}}{|(x-w)\times(z-w)|}$, where $\theta_{w}=\arccos\frac{(x-w)\cdot(z-w)}{|x-w||z-w|}$ is the angle of the triangle $(x,z,w)$ at $w$. So
\begin{equation}
\mathcal A_{0}(c_{1}+c_{2}+c_{3})=\frac{\theta_{w}}{|(x-w)\times(z-w)|\,(z-w)^{2}}\Big[(x'-w')\cdot(z-w)\;(y-z)\cdot X_{\perp}+(x'-w')\cdot(x-z)\;(y-z)\cdot(z-w)\Big].
\end{equation}

\section*{Step 6: final result}
\begin{align}
\frac{d^{3}N_{2\,\rm NLO}}{d^{3}k}+\frac{d^{3}N_{3a\,\rm NLO}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}f^{abd}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{w,w'}e^{-ik\cdot(w'-w)}\int_{x,y,z,x'}\frac{1}{(x'-w')^{2}(y-z)^{2}}\Bigg\{\notag\\
&\Big[\log\frac{(x-z)^{2}}{(x-w)^{2}}+i\pi\Big](x'-w')^{i}(y-z)^{j}\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}\notag\\&\qquad+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{\delta^{ij}\,\big[(x-w)^{2}+(x-z)^{2}\big]}{2\big[(x-w)^{2}-(x-z)^{2}\big]^{2}}\bigg]\notag\\
&+\frac{\theta_{w}}{|(x-w)\times(z-w)|\,(z-w)^{2}}\Big[(x'-w')\cdot(z-w)\;(y-z)\cdot X_{\perp}+(x'-w')\cdot(x-z)\;(y-z)\cdot(z-w)\Big]\notag\\
&\Bigg\}\times\left[U^{ab'}(x')\rho^{b'}(x')-U^{ab'}(w')\rho^{b'}(x')\right]\left[U^{de}(x)U^{bc}(z)\rho^{e}(x)\rho^{c}(y)-U^{de}(x)U^{bc}(y)\rho^{e}(x)\rho^{c}(y)\right].
\end{align}
This was checked numerically against the original $6+6+3$ terms at random points, for $u<1$ and $u>1$. The agreement is $10^{-14}$.

\section*{Remarks}
\begin{itemize}
\item \textbf{Reality.} $i\big[\log u+i\pi\big]=i\log u-\pi$. The $i\log u$ part and the $\mathcal A_{0}$ part give $\sin(k\cdot(w'-w))$ with the c.c. The $-\pi$ part gives a real term, which becomes $\cos(k\cdot(w'-w))$.
\item \textbf{The remaining pole.} The last $\delta^{ij}$ term has a double pole at $(x-z)^{2}=(x-w)^{2}$. The $i\pi$ in front does not vanish there, so $N_{2}+N_{3a}$ is \emph{not} integrable by itself. The pole cancels against $N_{1}$. Its two log terms with kernel $\frac{(X^{2}+Z^{2})\delta^{ij}}{(Z^{2}-X^{2})^{2}}$ add up to $\log u+i\pi$, and its third term is $-2\frac{X^{i}Z^{j}}{X^{2}Z^{2}}\log\frac{-Z^{2}}{X^{2}}$. In the normalisation $\frac{ig^{4}f^{abd}}{8\pi^{5}}$ above, $N_{1}$ is
\begin{equation}
(\log u+i\pi)\Big[\frac{\delta^{ij}(X^{2}+Z^{2})}{2D^{2}}-\frac{X^{i}Z^{j}}{X^{2}Z^{2}}\Big],
\end{equation}
which removes exactly the two extra pieces in $\mathcal T^{ij}=\mathcal K_{\rm IV}^{ij}+\frac{X^{i}Z^{j}}{X^{2}Z^{2}}-\frac{\delta^{ij}(X^{2}+Z^{2})}{2D^{2}}$. Therefore
\begin{equation}
N_{1}+N_{2}+N_{3a}=\Big[\log\frac{(x-z)^{2}}{(x-w)^{2}}+i\pi\Big]\,(x'-w')^{i}(y-z)^{j}\,\mathcal K^{ij}_{\rm IV}\;+\;\mathcal A_{0}\text{ terms},
\end{equation}
which is free of all $\frac{1}{(x-w)^{2}-(x-z)^{2}}$ poles. This agrees with the Row III result in Limiting\_Cases\_updated.
\end{itemize}
