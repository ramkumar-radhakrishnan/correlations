{\color{red}\noindent\textbf{Summary of corrections.}
\begin{itemize}
\item[1.] In the $\mathcal P_{gg}$ rewriting (eq.~\eqref{eq:pggform}) the prefactor must be
$\frac{g^{4}}{16\pi^{4}}$, not $\frac{g^{4}N_{c}}{16\pi^{4}}$: $\mathcal P_{gg}$ carries its own
$2N_{c}$, so $-\frac{g^{4}N_{c}}{8\pi^{4}}\hat p=-\frac{g^{4}}{16\pi^{4}}\mathcal P_{gg,+}$. The
subsequent DGLAP/RC blocks (violet/green) were already written correctly, so only this one
intermediate equation was off (by $N_{c}$).
\item[2.] The JIMWLK term in $z$-integral form (eq.~\eqref{eq:jzform}) must have a $+$ sign:
by eq.~\eqref{eq:div}, $[-\frac{2}{\epsilon}-2\gamma_{E}-\log]=+\frac{1}{\pi}\int_{z}K$, so no
extra minus appears. (The analogous substitution for the DGLAP soft channel,
eq.~\eqref{eq:second}, was done correctly: there
$[+\frac{2}{\epsilon}+\dots]=-\frac{1}{\pi}\int_{z}K$ turned the $-$ prefactor into $+$.)
The sign is also fixed in the final sum and in the JIMWLK category.
\item[3.] In the final sum and in the ``Finite terms'' category, the first entry
($+\frac{g^{4}N_{c}}{8\pi^{5}}\int\frac{d\xi}{\xi^{2}}\frac{1-\xi}{\xi}\int K(y,y',z)\,
\mathcal K\!\cdot\!\mathcal K'\,W(y,y')$) must be \textbf{deleted}: it is the DGLAP soft channel
itself (eq.~\eqref{eq:second}), which is already contained in the DGLAP category through the
full $\mathcal P_{gg}$. Keeping both counts that channel twice. What is added to the
cross-section is only the exact zero $=-$(subtraction integral)$+$(its closed value). The
bookkeeping chosen here places the subtraction $S$ \emph{inside the DGLAP category}, combined
with the soft channel $2N_{c}\frac{1-\xi}{\xi}$ it protects (the curly bracket of
\eqref{eq:addsub}, free of $\frac{2}{\epsilon}$, $\mu$ and $\log\frac{\vee}{k^{+}}$); the
finite terms are then only the boundary, endpoint and azimuthal entries, none of which carries
a large longitudinal logarithm, and $({\rm boundary})+({\rm BFKL})=-S$ (its closed value).
\item[4.] Minor: the $\delta(1-\xi)$ integration needs no $\vee\to\infty$ (the $\delta$ sits at
$\xi=1$ inside $[\frac{k^{+}}{\vee},1]$, picking up the LO phase); and the LHS of the
running-coupling category is relabeled $d^{3}N_{\rm RC}$.
\end{itemize}}

\subsection{Closed forms: the $y,y'$ integral of the subtraction $S$ and the $w,w'$ integral of the boundary term}
Here $S$ denotes the $W(z,z)$ piece of the DGLAP soft combination, with its own sign:
$S=-\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}\int_{k^{+}/\vee}^{1}
\frac{d\xi}{\xi^{2}}\frac{1-\xi}{\xi}\int e^{-ik(y'-y)\frac1\xi}
\frac{(y-z)\cdot(y'-z)}{(y-z)^{2}(y'-z)^{2}}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}\,W(z,z)$.
Both integrals are elementary, because the parent-leg kernel and the phase factorize,
\begin{align}
\frac{(y-z)\cdot(y'-z)}{(y-z)^{2}(y'-z)^{2}}
=\frac{(y-z)^{m}}{(y-z)^{2}}\,\frac{(y'-z)^{m}}{(y'-z)^{2}}\,,\qquad
e^{-ik\cdot(y'-y)\frac{1}{\xi}}=e^{+i\frac{k}{\xi}\cdot y}\;e^{-i\frac{k}{\xi}\cdot y'}\,,
\end{align}
and each factor is a two-dimensional Fourier transform:
\begin{align}\label{eq:ft}
\int d^{2}u\;e^{iq\cdot u}\,\frac{u^{m}}{u^{2}}=2\pi i\,\frac{q^{m}}{q^{2}}
\end{align}
(angular integral $\to iJ_{1}$, radial integral $\int_{0}^{\infty}J_{1}=1$; conditionally
convergent, no regulator and no $\mu$ needed). Shifting $y=z+u$, $y'=z+u'$,
\begin{align}
\int_{y}e^{+i\frac{k}{\xi}\cdot y}\,\frac{(y-z)^{m}}{(y-z)^{2}}
=e^{+i\frac{k}{\xi}\cdot z}\;2\pi i\,\frac{\xi\,k^{m}}{k^{2}}\,,\qquad
\int_{y'}e^{-i\frac{k}{\xi}\cdot y'}\,\frac{(y'-z)^{m}}{(y'-z)^{2}}
=-\,e^{-i\frac{k}{\xi}\cdot z}\;2\pi i\,\frac{\xi\,k^{m}}{k^{2}}\,,
\end{align}
so the $z$-phases cancel and
\begin{align}\label{eq:yypint}
\int_{y,y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\;\frac{(y-z)\cdot(y'-z)}{(y-z)^{2}(y'-z)^{2}}
=(2\pi i)(-2\pi i)\,\frac{\xi^{2}\,k\cdot k}{k^{4}}
=\frac{4\pi^{2}\,\xi^{2}}{k^{2}}\,,
\end{align}
independent of $z$. In $S$ the $\xi^{2}$ cancels the measure's $\frac{1}{\xi^{2}}$,
leaving $\int_{\bar\xi}^{1}d\xi\,\frac{1-\xi}{\xi}=\log\frac{\vee}{k^{+}}-(1-\frac{k^{+}}{\vee})$,
and with $\frac{g^{4}N_{c}}{8\pi^{5}}\cdot4\pi^{2}=\frac{g^{4}N_{c}}{2\pi^{3}}$:
\begin{align}\label{eq:Sint}
S
=-\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{2\pi^{3}}\,\frac{1}{k^{+}\,k^{2}}\,
\Big[\log\frac{\vee}{k^{+}}-\Big(1-\frac{k^{+}}{\vee}\Big)\Big]
\int_{x,x',z}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}
\left[U^{ce'}(z)-U^{ce'}(x')\right]\left[U^{ce}(z)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\,.
\end{align}
The boundary entry of the finite terms is the same computation at $\xi=1$ (phase $e^{-ik\cdot(w'-w)}$, kernel
$\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}(w'-z)^{2}}$), so $\int_{w,w'}(\cdots)=\frac{4\pi^{2}}{k^{2}}$ and
\begin{align}\label{eq:bint}
(\text{boundary})
=-\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{2\pi^{3}}\,\frac{1}{k^{+}\,k^{2}}\,
\Big(1-\frac{k^{+}}{\vee}\Big)
\int_{x,x',z}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}
\left[U^{ce'}(z)-U^{ce'}(x')\right]\left[U^{ce}(z)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\,.
\end{align}
The $(1-\frac{k^{+}}{\vee})$ pieces cancel between \eqref{eq:Sint} and \eqref{eq:bint}:
\begin{align}\label{eq:SplusB}
\eqref{eq:Sint}+\eqref{eq:bint}
=-\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{2\pi^{3}}\,\frac{1}{k^{+}\,k^{2}}\,
\log\frac{\vee}{k^{+}}
\int_{x,x',z}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}
\left[U^{ce'}(z)-U^{ce'}(x')\right]\left[U^{ce}(z)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\,,
\end{align}
which is exactly \emph{minus} the BFKL category after the same $w,w'$ integration
($\int_{w,w'}e^{-ik(w'-w)}\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}(w'-z)^{2}}=\frac{4\pi^{2}}{k^{2}}$
there too) --- the identity $S+\text{boundary}+\text{BFKL}=0$ made explicit at the level of
elementary integrals.
