\noindent The $z$ integral of $d^{3}N_{2b\,\rm UV}/d^{3}k$ is done exactly, in $d_{\perp}=2-\epsilon$ ($\overline{MS}$, $\bar\mu^{2}=4\pi e^{-\gamma_{E}}\mu^{2}$). It uses a Feynman parameter $\alpha$ and the Fourier transform $\int d^{d}l\,e^{iq\cdot l}(l^{2}+\Delta)^{-n}$. $K_{0},K_{1},K_{2}$ are modified Bessel functions; all of them have the same argument
$|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}$, written out in full below. Here $|k|=\sqrt{k^{2}}$ and $|y-x|=\sqrt{(y-x)^{2}}$ (transverse).

\begin{align}
\frac{d^{3}N_{2b\,\rm UV}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{\Lambda/k^{+}}^{1-\Lambda/k^{+}}\frac{d\xi}{2\pi}\int_{x,y,x',w'}e^{-ik\cdot(w'-x)}\,\Bigg\{\notag\\[4pt]
&\pi\,\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}\,(y-x)^{2}}\Bigg[\Big(\xi(1-\xi)+\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\Big)\Big(\frac{2}{\epsilon}+2\gamma_{E}+\log\frac{(y-x)^{2}\,\bar\mu^{2}}{4\,(1-\xi)}\Big)+\xi(1-\xi)\Bigg]\notag\\[6pt]
&+\Big(\xi(1-\xi)+\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\Big)\,\pi\,\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}}\notag\\&\qquad\times\int_{0}^{1}\frac{d\alpha}{\alpha}\Bigg[\frac{|k|\sqrt{\alpha(1-\xi)}\;e^{i\alpha(1-\xi)\,k\cdot(y-x)}}{|y-x|\sqrt{1-\alpha(1-\xi)}}\,K_{1}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)-\frac{1}{(y-x)^{2}}\Bigg]\notag\\[6pt]
&+\frac{2\,\xi(1-\xi)}{(x'-w')^{2}}\int_{0}^{1}d\alpha\;e^{i\alpha(1-\xi)\,k\cdot(y-x)}\Bigg\{\notag\\&\qquad\frac{\pi(1-\alpha)}{2}\Bigg[\frac{\sqrt{1-\xi}\,|k|}{|y-x|\sqrt{\alpha\big(1-\alpha(1-\xi)\big)}}\Bigg((x'-w')\cdot(y-x)\,K_{1}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)\notag\\
&\qquad\qquad-\frac{\big(k\cdot(x'-w')\big)\big(k\cdot(y-x)\big)}{k^{2}}\;|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\;K_{0}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)\Bigg)\notag\\
&\qquad+\frac{i\,\alpha\,(1-\xi)^{3/2}\,|k|}{|y-x|\sqrt{\alpha\big(1-\alpha(1-\xi)\big)}}\Big((x'-w')\cdot(y-x)\;k\cdot(y-x)+(y-x)^{2}\;k\cdot(x'-w')\Big)\notag\\&\qquad\qquad\times K_{1}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)\notag\\
&\qquad+\frac{\alpha\,(1-\xi)^{2}\,k^{2}\,(x'-w')\cdot(y-x)}{1-\alpha(1-\xi)}\,K_{2}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)\Bigg]\notag\\
&\qquad-\frac{\pi\,(x'-w')\cdot(y-x)\,|k|\sqrt{1-\xi}}{2\,|y-x|\sqrt{\alpha\big(1-\alpha(1-\xi)\big)}}\,K_{1}\Big(|k|\,|y-x|\sqrt{\alpha(1-\xi)\big(1-\alpha(1-\xi)\big)}\Big)\Bigg\}\Bigg\}\notag\\[6pt]
&\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y).
\end{align}

\subsection*{Remarks}
\begin{itemize}
\item In the convention $\int d^{2}p/p^{2}=0$ (all $1/\epsilon$ dropped), delete the $\frac{2}{\epsilon}$ in the first line. Nothing else changes.
\item The only $\bar\mu$ dependence is $\log\bar\mu^{2}$ in the first line. After the $\xi$ integral, $\int_{\Lambda/k^{+}}^{1-\Lambda/k^{+}}d\xi\,\big(\xi(1-\xi)+\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\big)=2\log\frac{k^{+}-\Lambda}{\Lambda}-\frac{11}{6}+O(\Lambda/k^{+})$, so its coefficient is $\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{4}}\frac{1}{k^{+}}\big[2\log\frac{k^{+}}{\Lambda}-\frac{11}{6}\big]$.
\item The second and third lines are finite. Both $\alpha$ integrals converge at $\alpha\to0$, and their integrands behave as $\alpha^{-1/2}$ or better.
\item The $-\frac{1}{(y-x)^{2}}$ inside the second line and the $+\xi(1-\xi)$ in the first line come from the $\epsilon$-expansion of the $d$-dimensional Feynman-parameter integral; they are not added by hand.
\item The two pieces were checked numerically against a direct two-dimensional $z$ integration with a small-$|z-x|$ cutoff $\delta$ (the cutoff replaces $\frac{2}{\epsilon}+2\gamma_{E}+\log\frac{\bar\mu^{2}}{4}$ by $\log\frac{1}{\delta^{2}}$). They agree to $5\times10^{-5}$, which is the accuracy of the grid.
\item The $\log\frac{1}{1-\xi}$ in the first line gives $\frac12\log^{2}\frac{k^{+}}{\Lambda}$ after the $\xi$ integral. This is the large-$|z|$ (infrared) behaviour at $\xi\to1$, and it has to cancel against the rest of $N_{2b}$ and the other JIMWLK-type terms.
\end{itemize}
