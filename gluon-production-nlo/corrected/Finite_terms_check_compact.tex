\section{Two fixes required}

\textbf{(a) Restore the absolute values (PV) in Group I Row III.}
This revision replaced the earlier $\log|\cdot|$ by plain $\log[\cdot]$. The arguments
\[
\frac{(x-w)^{2}-(x-z)^{2}}{(x-w)^{2}},\qquad
\frac{\frac{k^{+}}{\vee}(x-w)^{2}-(x-z)^{2}}{(x-w)^{2}-(x-z)^{2}},\qquad
\frac{(x-z)^{2}-\frac{k^{+}}{\vee}(x-w)^{2}}{(x-z)^{2}-(x-w)^{2}}
\]
change sign on half of the $(x,z,w)$ phase space, so as written these terms are complex.
Numerically: with $\log|\cdot|$ the Row-III identity against the $p^{+}$ masters holds to
$10^{-6}$; with plain logs it fails at $\mathcal O(0.1\div0.4)$ with a spurious imaginary
part. The $|\cdot|$ is exactly the PV reading of the $p^{+}$ poles, and after the collapse
of \S4.2 only \emph{one} absolute value survives, in
$\log\big|(x-z)^{2}-\xib\,(x-w)^{2}\big|$.

\textbf{(b) Restore the bookkeeping statements that were deleted.}
(i) definitions of $[\,1/p^{+}]_{+}$, $[k^{+}/p^{+}]_{+}$,
$[k^{+}/(p^{+}(p^{+}+k^{+}))]_{+}$ (subtraction at $p^{+}=0$ on the stated interval) and of
$\mathcal P_{gg}(\xi)$;
(ii) a sentence stating that the $\frac{11N_{c}}{3}\big(\frac2\epsilon+2\gamma_E+\log\frac{(x-w)^{2}\mu^{2}}{4}\big)$
piece of G-I Row II $N_{1}$ and the $\frac{11}{6}$ UV block of G-I Row V $N_{2b}$ (commented
out) are moved to the renormalization class of \S4.4, where together with the
$\beta_{0}$-term of G-II Row IV they build the running coupling, $\beta_{0}=\tfrac{11N_{c}}{3}$.

\section{The compact relation}

Let $\xib\equiv k^{+}/\vee$, and
\[
\ell\equiv\log(1-\xib),\qquad
u\equiv\frac{(x-z)^{2}}{(x-w)^{2}},\qquad
\mathbb L(A,B)\equiv\log\Big[\frac{(1-\xib)A+\xib B}{B}\Big],\qquad
\mathcal L\equiv\log\big|u-\xib\big| ,
\]
\[
\Azero=\frac{\theta_{w}}{\big|(x-w)\wedge(x-z)\big|},\qquad
\theta_{w}=\angle\big(x-w,\;z-w\big).
\]
Then the complete NLO cross-section organizes as
\begin{equation}
\boxed{\;
\frac{d^{3}N_{\rm NLO}}{d^{3}k}
=\LB\,\Big[K_{\rm BFKL}\otimes\frac{d^{3}N_{\rm LO}}{d^{3}k}\Big]
-\LJ\,\Big[H_{\rm JIMWLK}\,\frac{d^{3}N_{\rm LO}}{d^{3}k}\Big]
+\frac{d^{3}N_{\delta}}{d^{3}k}\;,}
\end{equation}
(the first two brackets verified earlier at operator level: the three- and four-$\rho$
logarithms cancel identically and the two-$\rho$ $\LJ$ set equals $-H_{\rm JIMWLK}N_{\rm LO}$
exactly), and the finite part falls into \emph{five} classes,
\begin{equation}
\boxed{\;
\frac{d^{3}N_{\delta}}{d^{3}k}
=\underbrace{N^{\rm UV}_{\delta}}_{\beta_{0},\,\mathcal P_{gg}}
+\underbrace{N^{\partial}_{\delta}}_{\propto\,\ell}
+\underbrace{N^{\rm coll}_{\delta}}_{\propto\,\mathbb L}
+\underbrace{N^{\rm geo}_{\delta}}_{\mathcal L,\;\Azero,\;\log\frac{(z-w)^{2}}{(x-w)^{2}}}
+\underbrace{N^{\rm rec}_{\delta}}_{[\;]_{+}\text{-convolutions}}\;}

\section{The simplified (collapsed) forms}

\subsection{The universal $\mathbb L$-family (G-I IV; G-II I; G-III I, II)}
All these rows are built from \emph{two} universal rank-2 kernels,
\begin{align}
T^{ij}(x;w,z)&=\frac{(x-z)^{j}(z-w)^{i}}{(z-w)^{2}(x-z)^{2}}
-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}
-\frac{(x-z)^{j}(x-w)^{i}}{(x-z)^{2}(x-w)^{2}}
+\frac{\delta^{ij}}{2(z-w)^{2}},\\
S^{ij}(x;w,z)&=\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}
+\frac{(x-w)^{i}(x-z)^{j}}{2(x-w)^{2}(x-z)^{2}},
\end{align}
and each such row reads, with its own legs and colour bracket $\mathfrak B_{r}$,
\begin{equation}
\boxed{\;
N^{r}_{\delta}=c_{r}\,\K^{i}(a_{r})\,\K^{j}(b_{r})\,
\Big[\,\mathbb L\big((x-z)^{2},(x-w_{r})^{2}\big)\,T^{ij}(x;w_{r},z)
+\ell\,S^{ij}(x;w_{r},z)\Big]\otimes\mathfrak B_{r}\;}
\end{equation}
($r=$ G-I~IV: $c=\tfrac{ig^{4}}{8\pi^{5}k^{+}}$, legs $(x'-w'),(y-z)$;
G-II~I$a$: $c=-\tfrac{ig^{4}}{16\pi^{5}k^{+}}$, same legs;
G-II~I$b$: $c=+\tfrac{ig^{4}}{16\pi^{5}k^{+}}$, legs $(x'-w),(y-z)$, $w\!\to\!w'$ inside;
G-III~I, II: $c=-\tfrac{ig^{4}}{8\pi^{5}k^{+}}$). The G-II Row I second part
($N_{2a}$--$N_{2f}$) is one single colour structure
\[
\mathfrak G=
f^{c'bd'}f^{cbd}U^{ac'}(w')U^{ac}(w)\rho^{d'}(x')\rho^{d}(x)
-f^{c'bd'}f^{acd}U^{ac'}(w')U^{cb}(z)U^{de}(x)\rho^{d'}(x')\rho^{e}(x)
-(\,'\leftrightarrow\,)+N_{c}U^{de'}(x')U^{de}(x)\rho^{e'}(x')\rho^{e}(x)
\]
multiplied by rational coefficients times only
$\mathbb L\big((x-z)^{2},(x-w)^{2}\big)$, $\mathbb L\big((x'-z)^{2},(x'-w')^{2}\big)$,
$\ell$, $\big[\tfrac1\vee-\tfrac1{k^{+}}\big]$, and the entirely finite
$\log\frac{(1-\xib)(x-z)^{2}+\xib(x-w)^{2}}{\xib(x'-w')^{2}+(1-\xib)(x'-z)^{2}}-\log\frac{(x-w)^{2}}{(x'-w')^{2}}$
of $N_{2a}$ (in which $\vee$ cancels identically).

\subsection{Group I Row III collapses to four structures}
Write the common bracket
${\rm Br}=\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')
\big[U^{de}(x)U^{bc}(z)-U^{de}(x)U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)$ and
$E'=(x'-w')^{2},\,F=(y-z)^{2},\,A=(x-z)^{2},\,B=(x-w)^{2},\,C=(z-w)^{2},\,G=(x-w)\!\cdot\!(x-z)$.
Using $\log\big|\tfrac{u-\xib}{u-1}\big|+\log|1-u|=\mathcal L$ in $N_{1}$, and the fact that the six
$\log\big|\tfrac{\xib-u}{1-u}\big|$ kernels of $N_{2}$ and the six $\log|1-u|$ kernels of
$N_{3a}$ are \emph{the same set} $\Sigma$ (so all $\log|1-u|$ cancels), the 23 finite terms
of Row III reduce \emph{exactly} (checked numerically to $10^{-24}$) to
\begin{equation}
\boxed{\;
N^{\rm RIII}_{\delta}
=\frac{1}{(2\pi)^{3}}\frac{ig^{4}f^{abd}}{16\pi^{5}k^{+}}
\int e^{-ik(w'-w)}\Big\{
\big(Y-2K_{2}+2\Sigma\big)\,\mathcal L
-\big(Y+2X_{12}\big)\,\ell
+2\,\Azero\,W
+\Xi\,\log\tfrac{C}{B}
\Big\}\otimes{\rm Br}\;}
\end{equation}
with $\mathcal L=\log\big|u-\xib\big|=\log\Big|\frac{(x-z)^{2}-\xib\,(x-w)^{2}}{(x-w)^{2}}\Big|$ and
\begin{align*}
Y&=\frac{(x'-w')\!\cdot\!(y-z)}{E'F}\,\frac{A+B}{(A-B)^{2}},\qquad
K_{2}=\frac{(x'-w')\!\cdot\!(x-w)\,(x-z)\!\cdot\!(y-z)}{E'BAF},\\
X_{12}&=\frac{(x'-w')\!\cdot\!(y-z)\;(z-w)\!\cdot\!\big[B(x-z)-A(x-w)\big]}{E'FC\,(B-A)^{2}},\\
\Sigma&=X_{12}
+\frac{1}{E'FC(B-A)}\Big[
\frac{A}{B}(x'-w')\!\cdot\!(x-w)\,(y-z)\!\cdot\!(z-w)
+\frac{B}{A}(x'-w')\!\cdot\!(z-w)\,(y-z)\!\cdot\!(x-z)\\
&\hspace{4.1cm}-(x'-w')\!\cdot\!(z-w)\,(y-z)\!\cdot\!(x-w)
-(x'-w')\!\cdot\!(x-z)\,(y-z)\!\cdot\!(z-w)\Big],\\
\Xi&=\frac{1}{E'FC}\Big[\frac{(y-z)\!\cdot\!(z-w)\,(x'-w')\!\cdot\!(x-w)}{B}
-\frac{(x'-w')\!\cdot\!(z-w)\,(y-z)\!\cdot\!(x-z)}{A}
-(x'-w')\!\cdot\!(y-z)\Big].
\end{align*}
The $\Azero$ coefficient has the manifestly finite wedge form (verified exactly):
\begin{equation}
\Azero\,W=\frac{\theta_{w}\;{\rm sgn}\big[(x-z)\wedge(x-w)\big]}{E'F\,C}
\left[\frac{(x'-w')\!\cdot\!(z-w)\,\big[(x-z)\wedge(y-z)\big]}{(x-z)^{2}}
-\frac{(y-z)\!\cdot\!(z-w)\,\big[(x-w)\wedge(x'-w')\big]}{(x-w)^{2}}\right],
\end{equation}
i.e.\ the arctangent $\Azero$ enters only through the bounded angle $\theta_{w}$ --- no
denominator $\sqrt{AB-G^{2}}$ survives. Moreover the coefficient $Y-2K_{2}+2\Sigma$ of
$\mathcal L$ is \emph{regular} at $u=1$ (the $1/(A-B)^{2}$ and $1/(A-B)$ poles cancel
identically between $Y$, $2X_{12}$ and the rest of $\Sigma$), so after the collapse the only
non-analyticity of Row III is the single PV log $\mathcal L$, whose argument vanishes only
in the shrinking region $(x-z)^{2}<\xib\,(x-w)^{2}$.

\subsection{The recoil class (G-I V; G-III III, V, VI)}
These rows keep one exact $p^{+}$ convolution with the recoil phase
$e^{-ik\cdot(z-x)\,p^{+}/k^{+}}$; they are finite at $\vee\to\infty$ by oscillation. Two
exact identities fix their bookkeeping against the BFKL file:
\begin{align}
\int_{0}^{P}\!\!dp^{+}\Big[\tfrac1{p^{+}}\Big]_{+}g
&=\int_{0}^{P}\!\!dp^{+}\Big(\Big[\tfrac{k^{+}}{p^{+}(p^{+}+k^{+})}\Big]_{+}+\tfrac1{p^{+}+k^{+}}\Big)g
\;-\;g(0)\,\LB ,\qquad P=\vee-k^{+},\\
\int_{0}^{\vee-k^{+}}\frac{dp^{+}}{p^{+}+k^{+}}&=\LB\qquad(\text{the }y'\!\to\!z\text{ subtraction term}),
\end{align}
and $\int_{y'}e^{-ik(y'-w)}\,\K^{m}(y'-z)\big|_{\rm FT}=\int_{w'}e^{-ik(w'-w)}\K^{m}(w'-z)$
reproduces exactly the $U(z)$-terms with $(w'-z)$ kernels in the BFKL file's Group-III
entries. The two induced $\LB$ pieces appear there with opposite signs against the source's
blue log --- the net BFKL content of these rows is the recoil-subtraction term alone, and
nothing is double counted.

\subsection{The UV / collinear class}
Collecting every $\big(\tfrac2\epsilon+2\gamma_{E}+\log\tfrac{r^{2}\bar\mu^{2}}{4}\big)$:
the $\mathcal P_{gg}$ convolution of G-II Row IV (collinear splitting of the measured
gluon, with its $(1-\xi)/\xi$ real-emission subtraction and the $(1-\xib)$ boundary term),
the $\beta_{0}$ term of G-II Row IV, the $\tfrac{11N_{c}}{3}$ piece of G-I Row II and the
$2\times\tfrac{11}{6}N_{c}$ piece of G-I Row V combine with $\beta_{0}=\tfrac{11N_{c}}{3}$
into the one-loop renormalization of $g^{2}$ in $d^{3}N_{\rm LO}$ plus the standard
$\mathcal P_{gg}\otimes\log$ collinear remainder; the $\ell\times(\tfrac2\epsilon+\dots)$
piece of G-I Row II is the $\int_{\xib}^{1}d\xi\,\tfrac{2N_{c}\xi}{1-\xi}$ endpoint of the
same convolution. (This assembly is structural; the final coefficient check should be done
once the $\mathcal P_{gg}$ and $[\;]_{+}$ conventions are restored in the file.)

\section{The $\vee\to\infty$ limit}
As $\xib\to0$: $\ell\to0$, $\mathbb L(A,B)\to\log\tfrac{A}{B}$, $\mathcal L\to\log u$, and the
entire closed-form finite sector is generated by just
\[
\log\frac{(x-z)^{2}}{(x-w)^{2}},\qquad
\theta_{w}\;(\text{via }\Azero),\qquad
\log\frac{(z-w)^{2}}{(x-w)^{2}},
\]
plus the $\vee$-finite recoil convolutions and the $\mathcal P_{gg}$/$\beta_{0}$ class ---
every $\vee$-dependence of $d^{3}N_{\delta}$ is a positive power of $\xib=k^{+}/\vee$.
