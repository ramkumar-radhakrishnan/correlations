\section{Group II, Row IV}\label{sec:R4}
We start from Eq.~\eqref{eq:R4start}. Throughout we use $d_{\perp}=2-\epsilon$, so that
\begin{align}
    \left[-\frac{d_{\perp}}{\epsilon}-d_{\perp}\,\gamma_{E}-\frac{d_{\perp}}{2}\,\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)}-1\right] & = -\frac{2}{\epsilon}+1-2\,\gamma_{E}-\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)} -1 \notag \\ & =  -\frac{2}{\epsilon}-2\,\gamma_{E}-\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)}\;\equiv\;-L_{\epsilon}\,,
\end{align}
and the leading order cross section is
\begin{align}
    \frac{d^{3}N_{\rm LO}}{d^{3}k} & = \frac{1}{(2\pi)^{3}}\, \frac{g^{2}}{2\pi^{2}}\, \frac{1}{k^{+}}\,\int_{x,y,x',y'}\, e^{-ik\cdot (y'-y)}\, \frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\times \left[U^{ce'}(y') - U^{ce'}(x')\right] \times\, \left[U^{ce}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\,,
\end{align}
with its $\xi$-rescaled version $\frac{d^{3}N_{\xi\,\rm LO}}{d^{3}k}$ carrying the phase $e^{-ik\cdot(y'-y)/\xi}$.

\subsection{Extraction of the JIMWLK term}
Splitting the blue logarithm of the second term of Eq.~\eqref{eq:R4start},
${\color{blue}\log\frac{(\vee-k^{+})\,k^{+}}{\vee\,\Lambda}}=\log\frac{k^{+}}{\Lambda}+\log\big(1-\frac{k^{+}}{\vee}\big)$,
its $\log\frac{k^{+}}{\Lambda}$ part is the JIMWLK contribution of this row,
\begin{align}
    \frac{d^{3}N^{\rm JIMWLK}_{\rm G\text{-}II,\,R\,IV}}{d^{3}k} & =  \frac{g^{2}}{4\pi^{2}}\, \frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\, \frac{N_{c}}{k^{+}}\,{\color{blue}\log{\left(\frac{k^{+}}{\Lambda}\right)}}\,\int_{x,y,x',y'}\,e^{-ik\cdot (y'-y)}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\left[-L_{\epsilon}\right]\notag \\ & \left[U^{ce'}(y') - U^{ce'}(x')\right] \times\, \left[U^{ce}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\,,
\end{align}
which is moved to the two-$\rho$ evolution (JIMWLK) file, where it appears verbatim with the bracket
$\big[-\frac{d_{\perp}}{\epsilon}-d_{\perp}\gamma_{E}-\frac{d_{\perp}}{2}\log\frac{(y'-y)^{2}\mu^{2}}{4}-1\big]=-L_{\epsilon}$.
The $\log\big(1-\frac{k^{+}}{\vee}\big)$ part of the blue logarithm stays here. In the third term of
Eq.~\eqref{eq:R4start} we may now set $\Lambda\to0$ in the upper limit, $\frac{k^{+}}{k^{+}+\Lambda}\to1$
(the integrand is regular at $\xi=1$), and $d_{\perp}\to2$ (no pole multiplies it).

\subsection{Assembling the splitting function}
With the \emph{standard} plus prescription on $[0,1]$,
\begin{align}
\int_{a}^{1}d\xi\,\frac{h(\xi)}{(1-\xi)_{+}}=\int_{a}^{1}d\xi\,\frac{h(\xi)-h(1)}{1-\xi}+h(1)\,\log(1-a)\,,\qquad a=\frac{k^{+}}{\vee}\,,
\end{align}
the first term of Eq.~\eqref{eq:R4start} and the leftover
$\log\big(1-\frac{k^{+}}{\vee}\big)\big[-L_{\epsilon}\big]$ endpoint term combine into a single
$\xi$-integral with $\frac{\xi}{(1-\xi)_{+}}$ in the standard convention. Denoting
\begin{align}
\mathcal{P}_{gg,+}(\xi) = 2N_{c}\,\left[\xi\,(1-\xi)+\frac{(1-\xi)}{\xi}+\frac{\xi}{(1-\xi)_{+}}\right],\qquad
    \mathcal{P}_{gg}(\xi) = \mathcal{P}_{gg,+}(\xi) + \frac{\beta_{0}}{2}\, \delta(1-\xi)\,,\qquad \beta_{0}=\frac{11\,N_{c}}{3}\ \ (\text{pure glue})\,,
\end{align}
and adding and subtracting the $\frac{\beta_{0}}{2}\delta(1-\xi)$ piece, we obtain
\begin{align}\label{eq:R4afterPgg}
    \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & =
   {\color{violet} -\,\frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\,\frac{g^{2}}{8\pi^{2}}\,\frac{1}{k^{+}}\,\int_{k^{+}/\vee}^{1}\, \frac{d\xi}{\xi^{2}}\,\mathcal{P}_{gg}(\xi)\,\int_{x,y,x',y'}\, e^{-ik\cdot (y'-y)\,\frac{1}{\xi}}\, \frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,L_{\epsilon}\,\times}\notag \\ &  {\color{violet}\left[U^{ce'}(y') - U^{ce'}(x')\right] \times\, \left[U^{ce}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)} \notag \\ & {\color{green}+\frac{1}{(2\pi)^{3}}\, \frac{g^{2}}{2\pi^{2}}\,\frac{g^{2}\,\beta_{0}}{16\pi^{2}}\, \frac{1}{k^{+}}\,\int_{x,y,x',y'}\, e^{-ik\cdot (y'-y)}\, \frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,L_{\epsilon}\,\times}\notag \\ &  {\color{green}\left[U^{ce'}(y') - U^{ce'}(x')\right] \times\, \left[U^{ce}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)}\notag \\

renormalization explicit.

\subsection{The soft subtraction and the extraction of the BFKL term}
The $2N_{c}\frac{1-\xi}{\xi}$ part of $\mathcal{P}_{gg}$ in the violet term produces a hidden
$\log\frac{\vee}{k^{+}}$ from $\xi\to0$: there the parent positions collapse onto the sibling,
$y=\xi w+(1-\xi)z\to z$, $y'=\xi w'+(1-\xi)z\to z$, so a rapidity logarithm with an $L_{\epsilon}$
coefficient would appear, which is not physical (the coefficient of $\log\vee$ must be the
$\mu$-independent BFKL kernel). Its $\xi\to0$ limit is subtracted \emph{exactly} with
\begin{align}
\frac{d^{3}N^{\rm sub}_{\xi}}{d^{3}k} & = \frac{1}{(2\pi)^{3}}\, \frac{g^{2}}{2\pi^{2}}\, \frac{1}{k^{+}}\,\int_{x,y,x',y',z}\, e^{-ik\cdot (y'-y)\,\frac{1}{\xi}}\;\frac{1}{\pi}\,\frac{(y-z)\cdot (y'-z)}{(y-z)^{2}\,(y'-z)^{2}}\;\frac{(x-z)\cdot (x'-z)}{(x-z)^{2}\,(x'-z)^{2}}\notag\\
&\times\left[U^{ce'}(z) - U^{ce'}(x')\right]\left[U^{ce}(z) - U^{ce}(x)\right]\rho^{e'}(x')\,\rho^{e}(x)\,,
\end{align}
which is IR and UV finite in $d=2$, contains no $\mu$, and is symmetric under
$(x,y)\leftrightarrow(x',y')$. Writing $W(u,u')=[U^{ce'}(u')-U^{ce'}(x')][U^{ce}(u)-U^{ce}(x)]\rho^{e'}(x')\rho^{e}(x)$
and $K(a,b,z)=\frac{(a-z)\cdot(b-z)}{(a-z)^{2}(b-z)^{2}}$, and using
$\int_{z}^{(d=2-\epsilon)}K(y,y',z)=-\pi L_{\epsilon}$, the combination
\begin{align}
2N_c\frac{1-\xi}{\xi}\Big[L_\epsilon\,\frac{d^{3}N_{\xi\,\rm LO}}{d^{3}k}+\frac{d^{3}N^{\rm sub}_{\xi}}{d^{3}k}\Big]
=-\frac{2N_c(1-\xi)}{\xi}\,\frac{1}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{1}{k^{+}}
\int_{x,y,x',y',z}e^{-ik\cdot(y'-y)\frac1\xi}\,\frac1\pi\,K(y,y',z)\Big[\mathcal K(x-y)\cdot\mathcal K(x'-y')\,W(y,y')-K(x,x',z)\,W(z,z)\Big]
\end{align}
contains no $\frac{2}{\epsilon}$ and no $\mu$, and at fixed $w,w'$ the square bracket is
$O(\xi)$ as $\xi\to0$ (since $y-z=\xi(w-z)$), so no $\log\vee$ survives with an $L_{\epsilon}$
coefficient. The exact value of the subtraction follows from the change of variables
$y=z+\xi(w-z)$, $y'=z+\xi(w'-z)$ (Jacobian $\xi^{4}$, $K(y,y',z)=K(w,w',z)/\xi^{2}$, phase
$\to e^{-ik(w'-w)}$) and $\int_{k^{+}/\vee}^{1}d\xi\,\frac{1-\xi}{\xi}=\log\frac{\vee}{k^{+}}-\big(1-\frac{k^{+}}{\vee}\big)$:
\begin{align}\label{eq:subvalue}
-\frac{g^{2}}{8\pi^{2}}\int_{k^+/\vee}^{1}\frac{d\xi}{\xi^{2}}\,2N_c\,\frac{1-\xi}{\xi}\,\frac{d^{3}N^{\rm sub}_{\xi}}{d^{3}k} &=-\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}\Big[\log\frac{\vee}{k^{+}}-\Big(1-\frac{k^{+}}{\vee}\Big)\Big]\int_{x,x',z,w,w'}e^{-ik\cdot(w'-w)}K(w,w',z)\,K(x,x',z)\notag\\
&\quad\times\left[U^{ce'}(z) - U^{ce'}(x')\right]\left[U^{ce}(z) - U^{ce}(x)\right]\rho^{e'}(x')\,\rho^{e}(x)\,.
\end{align}
We therefore add and subtract Eq.~\eqref{eq:subvalue}: the unintegrated subtraction joins the
violet term (rendering its soft sector $\mu$-free), while of its exact value the
$\log\frac{\vee}{k^{+}}$ part is the BFKL contribution of this row,
\begin{align}
    \frac{d^{3}N^{\rm BFKL}_{\rm G\text{-}II,\,R\,IV}}{d^{3}k} =      \frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,N_{c}}{8\pi^{5}}\,\frac{1}{k^{+}}\,{\color{blue}\log{\left(\frac{\vee}{k^{+}}\right)}}\,\int_{x,x',z,w,w'}\,e^{-ik\cdot (w'-w)}\,\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}\,(w'-z)^{2}}\,\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}\,(x'-z)^{2}}\,
    \left[U^{ce'}(z) - U^{ce'}(x')\right]\left[U^{ce}(z) - U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\,,
\end{align}
moved to the BFKL file (it appears there verbatim, G-II Row IV), and the
$-\big(1-\frac{k^{+}}{\vee}\big)$ part remains below as the boundary term. Note the coefficient of
$\log\frac{\vee}{k^{+}}$ is manifestly UV finite and $\mu$-independent, as a BFKL kernel must be.
