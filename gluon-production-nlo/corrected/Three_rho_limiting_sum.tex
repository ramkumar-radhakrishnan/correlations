\begin{equation}
X^{a}\equiv\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x'),\qquad
B^{ab}\equiv f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{de}(x)\rho^{e}(x).

\section{Step 1: the five rows with kernel $\log\frac{(x-z)^{2}}{(x-w)^{2}}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\,\mathcal K^{ij}_{\rm IV}$}
Group I Row III (its first term), Group I Row IV, Group II $N_{1a}$, and Group III Rows I and II all have this kernel and the phase $e^{-ik\cdot(w'-w)}$. In units of $\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{8\pi^{5}}\frac{1}{k^{+}}\int e^{-ik\cdot(w'-w)}\times$kernel, their colour factors are:
\begin{align}
\text{GI R III}&:\ f^{abd}X^{a}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)\equiv(A)+(B),\notag\\
\text{GI R IV}&:\ X^{a}\big[f^{cdc'}U^{ac}(w)U^{bd}(z)U^{be}(y)\rho^{e}(y)\rho^{c'}(x)-f^{abc}U^{be}(y)U^{cd}(x)\rho^{e}(y)\rho^{d}(x)\big]\equiv(C)+(D),\notag\\
\text{GII}+\text{GIII R I}+\text{GIII R II}&:\ -\big[(g_{1}-g_{2})+(g_{2}-g_{3})+(g_{2}-g_{4})\big]B^{ab}=+X^{a}\big[\rho^{b}(y)-U^{c'b}(z)U^{c'e}(y)\rho^{e}(y)\big]B^{ab},
\end{align}
where $g_{1}=U^{ac'}(w')\rho^{c'}(x')\rho^{b}(y)$, $g_{2}=U^{ab'}(x')U^{c'b}(z)U^{c'e}(y)\rho^{b'}(x')\rho^{e}(y)$, $g_{3}=U^{ac'}(w')U^{d'b}(z)U^{d'e}(y)\rho^{c'}(x')\rho^{e}(y)$ and $g_{4}=U^{ab'}(x')\rho^{b'}(x')\rho^{b}(y)$. Expanding the last line gives
\begin{align}
(1)&=f^{cbd}X^{a}U^{ac}(w)\rho^{b}(y)\rho^{d}(x), & (2)&=-f^{acd}X^{a}U^{cb}(z)U^{de}(x)\rho^{b}(y)\rho^{e}(x),\notag\\
(3)&=-f^{cbd}X^{a}U^{ac}(w)U^{c'b}(z)U^{c'e}(y)\rho^{e}(y)\rho^{d}(x), & (4)&=f^{acd}X^{a}U^{ce}(y)U^{de'}(x)\rho^{e}(y)\rho^{e'}(x).
\end{align}
In (4) we used $U^{cb}(z)U^{c'b}(z)=\delta^{cc'}$. Relabelling dummy indices gives $(A)+(2)=0$, $(C)+(3)=0$ and $(D)+(4)=0$: every structure with $U(z)$ cancels. What remains is
\begin{equation}
(1)+(B)=-\tfrac12\,X^{a}\Big[f^{hbc}U^{ah}(w)\{\rho^{b}(x),\rho^{c}(y)\}-f^{abc}U^{bd}(x)U^{ce}(y)\{\rho^{d}(x),\rho^{e}(y)\}\Big],
\end{equation}
which is the colour factor of Group I Row II.

\section{Step 2: Group III Rows III, V, VI}
These rows have identical kernels. With $\{\rho,\rho\}\to2\rho\rho$ in Row III, all three have the prefactor $\frac{ig^{4}}{4\pi^{4}}$. Write $F^{c'}=[U^{c'e'}(y')-U^{c'e'}(x')]\rho^{e'}(x')$; for the $\frac{(x'-z)^{k'}}{(x'-z)^{2}}$ pieces, $y'\to z$.
\begin{itemize}
\item In Rows III $+$ VI, the $U^{cb}(z)U^{ad}(y)U^{ce}(x)$ terms cancel, which leaves $f^{c'd'a}F^{c'}U^{d'b}(z)\rho^{b}(x)\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)$.
\item Row V is $-f^{c'd'a}F^{c'}U^{d'e}(x)\rho^{e}(x)\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)$.
\end{itemize}
Sum:
\begin{equation}
\text{III}+\text{V}+\text{VI}:\quad f^{c'd'a}\,F^{c'}\,\big[U^{d'b}(z)-U^{d'b}(x)\big]\rho^{b}(x)\,\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y).
\end{equation}
In the $(x'-z)$ subtraction pieces the kernel is $\frac{(x'-z)\cdot(x-z)}{(x'-z)^{2}(x-z)^{2}}$, which is symmetric under $x\leftrightarrow x'$. The colour factor $f^{c'd'a}[U(z)-U(x')]^{c'e'}\rho^{e'}(x')[U(z)-U(x)]^{d'b}\rho^{b}(x)$ is antisymmetric under $x\leftrightarrow x'$ together with $c'\leftrightarrow d'$. So these pieces vanish after $\int_{x,x'}$.

\noindent Both colour identities were checked with exact SU(3) adjoint Wilson lines to $10^{-15}$.

\section*{What each term is (terms are counted by their $\pm\frac{1}{(2\pi)^{3}}$ prefactors)}
\begin{itemize}
\item \textbf{Terms 1--2 (Row II and the Step 1 remainder).} Group I Row II is unchanged. Below it is everything left of the five $\log\frac{(x-z)^{2}}{(x-w)^{2}}\mathcal K_{\rm IV}$ rows: Group I Rows III and IV, Group II, Group III Rows I and II. The two terms share one colour factor, but they do not cancel: the second decays at large $|x-w|$, while $R^{i}\to-2q^{i}\log(p^{2}/q^{2})$ as $p\to0$ does not.
\item \textbf{Terms 3--5.} The rest of Group I Row III: the $\arccos$ and $\log\frac{(z-w)^{2}}{(x-w)^{2}}$ terms, the amended term with $U^{bc}(w)$, and $N_{i\pi}$. $N_{i\pi}$ is the only term with a real coefficient, so it gives $\cos$; its $z\to w$ piece is the Coulomb phase, which cancels against its c.c.
\item \textbf{Terms 6--7.} Group I Row V, unchanged: no other term has its kernel.
\item \textbf{Term 8.} Group III Rows III, V and VI combined into one term. It is $f^{c'd'a}$ times the three LO emission brackets at $y'$, $z$ and $w$. All $x'\to z$ subtraction pieces have dropped out.
\end{itemize}
\textbf{Reality.} Every term except $N_{i\pi}$ has coefficient $i\times$real ($R^{i}$ is odd, so term 1 is too). With the c.c.\ these give $\sin(k\cdot\Delta)$, and $N_{i\pi}$ gives $\cos(k\cdot(w'-w))$. The sum is real, but it is not zero: the finite three-$\rho$ terms do not cancel completely. For a Gaussian (MV) projectile, $\langle\rho\rho\rho\rangle=0$, and the whole sector drops out of the averaged cross section.
