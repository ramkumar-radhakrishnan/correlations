\noindent The decomposition (following Balitsky--Chirilli, eqs.~(59), (62), (63)) is
\begin{equation}
\frac{d^{3}N_{2\,\rm NLO}}{d^{3}k}=\underbrace{\Big(\frac{d^{3}N_{2\,\rm NLO}}{d^{3}k}-\frac{d^{3}N^{\rm sub}_{2\,\rm NLO}}{d^{3}k}\Big)}_{\text{(I) finite}}
+\underbrace{\frac{d^{3}N^{\rm sub,\,UV}_{2\,\rm NLO}}{d^{3}k}}_{\text{(II) UV}}
+\underbrace{\frac{d^{3}N^{\rm sub,\,fin}_{2\,\rm NLO}}{d^{3}k}}_{\text{(III) finite}} .
\end{equation}

\subsection*{(I) $N_{2}-N_{2}^{\rm sub}$: UV finite (colour vanishes at $z=x$), $d_{\perp}=2$}
\begin{align}
\frac{d^{3}N_{2\,\rm NLO}}{d^{3}k}-\frac{d^{3}N^{\rm sub}_{2\,\rm NLO}}{d^{3}k}  & =  \frac{1}{(2\pi)^{3}}\, \frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\,\int_{\Lambda}^{k^{+}-\Lambda}\, \frac{dp^{+}}{2\pi} \, \int_{x,y,z,x',w'}\,e^{-ik\cdot(w'-z)}\notag\\ &\times e^{-ik\cdot (z-x)\,\frac{p^{+}}{k^{+}}}\,\frac{1}{\left[p^{+}\,(y-x)^{2}+(k^{+}-p^{+})\,(y-z)^{2}\right]}\, \frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}} \notag \\ & \times\Bigg[\frac{\delta^{im}\,d_{\perp}}{2\,(x-z)^{2}}\,\left[(y-x)^{2}-(y-z)^{2}\right]\,\frac{p^{+}\,(k^{+}-p^{+})}{k^{+\,2}}\notag\\ &-\frac{p^{+}}{(k^{+}-p^{+})}\,\left[\frac{(y-x)^{i}\,(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}\,(y-z)^{m}}{2\,(y-z)^{2}}\right]\notag \\ & +\frac{(k^{+}-p^{+})\,\delta_{im}}{k^{+}}\,\left[\frac{(y-z)\cdot (x-z)}{(x-z)^{2}}+\frac{(y-x)\cdot (y-z)}{2\,(y-x)^{2}}-\frac{\left[(y-x)^{2}-(y-z)^{2}\right]}{2\,(x-z)^{2}}\right]\notag \\ & -\frac{(k^{+}-p^{+})}{p^{+}}\,\left[\frac{(y-z)^{i}\,(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}\,(y-z)^{i}}{2\,(y-x)^{2}}\right]\notag \\ & +\frac{p^{+}\,\delta_{im}}{k^{+}}\,\left[\frac{(y-x)\cdot (x-z)}{(x-z)^{2}}-\frac{(y-x)\cdot (y-z)}{2\,(y-z)^{2}}-\frac{\left[(y-x)^{2}-(y-z)^{2}\right]}{2\,(x-z)^{2}}\right]\notag \\ &-\left[\frac{(y-x)^{m}\,(x-z)^{i}}{(x-z)^{2}}-\frac{(y-x)^{m}\,(y-z)^{i}}{2\,(y-z)^{2}}+\frac{(y-z)^{m}\,(x-z)^{i}}{(x-z)^{2}}+\frac{(y-x)^{i}\,(y-z)^{m}}{2\,(y-x)^{2}}\right]\Bigg] \notag \\ & \times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\,\Big[-f^{adc}\,f^{fbe}\,U^{db}(x)\,U^{ce}(z)+N_{c}\,U^{af}(x)\Big]\,\rho^{b'}(x')\,\rho^{f}(y)
\end{align}

\subsection*{(II) UV part of $N_{2}^{\rm sub}$ (dim.\ reg., $d_{\perp}=2-\epsilon$, $\overline{MS}$)}
\begin{align}
\frac{d^{3}N^{\rm sub,\,UV}_{2\,\rm NLO}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\left[2\log\Big(\frac{k^{+}-\Lambda}{\Lambda}\Big)-\frac{11}{6}+\frac{4\Lambda}{k^{+}}-\frac{\Lambda^{2}}{k^{+2}}+\frac{2\Lambda^{3}}{3k^{+3}}\right]\int_{x,y,x',w'}e^{-ik\cdot(w'-x)}\notag\\&\qquad\times\left[\frac{2}{\epsilon}+2\gamma_{E}+\log\frac{(y-x)^{2}\bar\mu^{2}}{4}\right]\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}\,(y-x)^{2}}\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[U^{af}(y)-U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)
\end{align}

\subsection*{(III) Finite remainder of $N_{2}^{\rm sub}$, $d_{\perp}=2$}
\begin{align}
\frac{d^{3}N^{\rm sub,\,fin}_{2\,\rm NLO}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{\Lambda}^{k^{+}-\Lambda}\frac{dp^{+}}{2\pi}\int_{x,y,z,x',w'}\Bigg\{\notag\\&\quad e^{-ik\cdot(w'-z)}\,e^{-ik\cdot(z-x)\frac{p^{+}}{k^{+}}}\,\frac{1}{\left[p^{+}(y-x)^{2}+(k^{+}-p^{+})(y-z)^{2}\right]}\,\frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\notag\\
&\times\Bigg[\frac{\delta^{im}\,d_{\perp}}{2\,(x-z)^{2}}\,\left[(y-x)^{2}-(y-z)^{2}\right]\,\frac{p^{+}\,(k^{+}-p^{+})}{k^{+\,2}}\notag\\&-\frac{p^{+}}{(k^{+}-p^{+})}\,\left[\frac{(y-x)^{i}\,(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}\,(y-z)^{m}}{2\,(y-z)^{2}}\right]\notag\\
&+\frac{(k^{+}-p^{+})\,\delta_{im}}{k^{+}}\,\left[\frac{(y-z)\cdot (x-z)}{(x-z)^{2}}+\frac{(y-x)\cdot (y-z)}{2\,(y-x)^{2}}-\frac{\left[(y-x)^{2}-(y-z)^{2}\right]}{2\,(x-z)^{2}}\right]\notag\\
&-\frac{(k^{+}-p^{+})}{p^{+}}\,\left[\frac{(y-z)^{i}\,(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}\,(y-z)^{i}}{2\,(y-x)^{2}}\right]\notag\\
&+\frac{p^{+}\,\delta_{im}}{k^{+}}\,\left[\frac{(y-x)\cdot (x-z)}{(x-z)^{2}}-\frac{(y-x)\cdot (y-z)}{2\,(y-z)^{2}}-\frac{\left[(y-x)^{2}-(y-z)^{2}\right]}{2\,(x-z)^{2}}\right]\notag\\
&-\left[\frac{(y-x)^{m}\,(x-z)^{i}}{(x-z)^{2}}-\frac{(y-x)^{m}\,(y-z)^{i}}{2\,(y-z)^{2}}+\frac{(y-z)^{m}\,(x-z)^{i}}{(x-z)^{2}}+\frac{(y-x)^{i}\,(y-z)^{m}}{2\,(y-x)^{2}}\right]\Bigg]\notag\\
&-e^{-ik\cdot(w'-x)}\,\frac{1}{k^{+}}\left[\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}+\frac{p^{+}}{k^{+}-p^{+}}+\frac{k^{+}-p^{+}}{p^{+}}\right]\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}\,(y-x)^{2}}\,\frac{(y-z)\cdot(y-x)}{(y-z)^{2}\,(x-z)^{2}}\Bigg\}\notag\\
&\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\,N_{c}\Big[U^{af}(y)-U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)
\end{align}

\subsection*{Notes}
\begin{itemize}
\item (II) is the $d$-dimensional $z$ integral of the $z\to x$ limit of the subtraction kernel,
\[\frac{1}{k^{+}}P(\xi)\,e^{-ik\cdot(w'-x)}\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}(y-x)^{2}}\frac{(y-z)\cdot(y-x)}{(y-z)^{2}(x-z)^{2}},\] with
$\mu^{-\epsilon}\!\int d^{2-\epsilon}z\,\frac{(y-z)\cdot(y-x)}{(y-z)^{2}(z-x)^{2}}=\pi\big[\frac{2}{\epsilon}+2\gamma_{E}+\log\frac{(y-x)^{2}\bar\mu^{2}}{4}\big]$, $\bar\mu^{2}=4\pi e^{-\gamma_{E}}\mu^{2}$, and $\frac{1}{k^{+}}\int_{\Lambda}^{k^{+}-\Lambda}dp^{+}P=2\log\frac{k^{+}-\Lambda}{\Lambda}-\frac{11}{6}+O(\Lambda)$.
\item (III) is finite: the last line cancels the $1/(x-z)^{2}$ of the lines above it. They must stay together under one $\int_{z}$; integrating the last line alone over $z$ in dim.~reg.\ would just give $-$(II).
\item (I)+(II)+(III) $=N_{2}$ exactly. The only $\log\mu$ is in (II).
\end{itemize}
