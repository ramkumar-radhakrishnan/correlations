\noindent\textbf{Notation.} $\mathcal{K}(u)\equiv\dfrac{u}{u^{2}}$ is a transverse vector, so that $\mathcal{K}(u)\cdot\mathcal{K}(v)=\dfrac{u\cdot v}{u^{2}\,v^{2}}$. $U$ is the adjoint Wilson line, a real matrix, so $U^{\dagger}=U^{T}$. $(T^{e})_{ab}=-if^{eab}$, $(D^{e})_{ab}=d^{eab}$, ${\rm Tr}$ is the adjoint trace, and the index $e$ in the traces is summed. The $\rho$'s are operators and keep their order. This term belongs to Group III and comes with $+{\rm c.c.}$

\section{Colour-singlet form}
For a colour-singlet projectile the ordered three-$\rho$ average is an invariant tensor:
\begin{equation}
\big\langle\rho^{e^{\prime}}(x^{\prime})\rho^{d}(y)\rho^{e}(x)\big\rangle=\frac{f^{e^{\prime}de}}{N_{c}(N_{c}^{2}-1)}\big\langle f^{abc}\rho^{a}(x^{\prime})\rho^{b}(y)\rho^{c}(x)\big\rangle+\frac{N_{c}\,d^{e^{\prime}de}}{(N_{c}^{2}-4)(N_{c}^{2}-1)}\big\langle d^{abc}\rho^{a}(x^{\prime})\rho^{b}(y)\rho^{c}(x)\big\rangle.
\end{equation}
The order $x'$, $y$, $x$ of the operators is kept: no symmetrisation. Contracting with the colour tensor, with $M=U(q)$ a group element and $q=y'$, $z$ or $x'$, gives
\begin{align}
f^{c^{\prime}d^{\prime}a}f^{e^{\prime}de}\,M^{c^{\prime}e^{\prime}}U^{ad}(y)\big[U^{d^{\prime}e}(x)-U^{d^{\prime}e}(z)\big]&=-{\rm Tr}\Big\{T^{e}U^{\dagger}(y)\,M\,T^{e}M^{\dagger}\big(U(x)-U(z)\big)\Big\},\\ f^{c^{\prime}d^{\prime}a}d^{e^{\prime}de}\,M^{c^{\prime}e^{\prime}}U^{ad}(y)\big[U^{d^{\prime}e}(x)-U^{d^{\prime}e}(z)\big]&=i\,{\rm Tr}\Big\{M\,T^{e}M^{\dagger}\,U(y)\,D^{e}\big(U^{\dagger}(x)-U^{\dagger}(z)\big)\Big\}.
\end{align}
