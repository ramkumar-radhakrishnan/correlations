\section{Adding the two-$\rho$ terms ($\rho$'s do not commute)}
\subsection{Conventions}
\begin{itemize}
\item The $\rho$'s are operators; their order is kept. In every term below $\rho(x')$ stands to the left. The Wilson lines commute with the $\rho$'s. Groups I and III come with ``$+$c.c.'', which for operators is the hermitian conjugate (it reverses the order of the $\rho$'s). Group II has no c.c.; each of its rows is hermitian by itself.
\item Limiting case: $\Lambda\to0$, $\vee\to\infty$. So $\log\big(1-\frac{k^{+}}{\vee}\big)\to0$, and the upper limit $\vee-k^{+}\to\infty$. In the finite terms $d_{\perp}=2$.
\item \textbf{Colours.} Within one step, terms of the same colour cancel; they are struck through when they cancel exactly. \Gr{Grey} marks a commutator that is left over.
\item Every colour identity below was checked with exact SU(3) adjoint Wilson lines and non-commuting random matrices for $\rho$ (residual $10^{-12}$). Every kernel identity was checked numerically.
\end{itemize}

\noindent\textbf{Step 2: the three terms of $N_{2f}$ combine.} Write $u=\frac{(x-z)^{2}}{(x-w)^{2}}$ and $u'=\frac{(x'-z)^{2}}{(x'-w')^{2}}$; these are names for the two ratios only. \end{equation}
The function $\frac{s\log s}{s-1}$ equals $1$ at $s=1$. Writing $\frac{s\log s}{s-1}=1+(s-1)\frac{s\log s-s+1}{(s-1)^{2}}$:
\begin{equation}
\frac{(u'-1)\frac{u\log u}{u-1}-(u-1)\frac{u'\log u'}{u'-1}}{u-u'}=\R{-1}+(u-1)(u'-1)\frac{1}{u-u'}\Big[\frac{u\log u-u+1}{(u-1)^{2}}-\frac{u'\log u'-u'+1}{(u'-1)^{2}}\Big].
\end{equation}

\begin{align}

\subsection{Group II, Row IV: the $\xi$ integral}
\textbf{Step 1.} Only the phase depends on $\xi$. Put $\xi=1/t$, so $d\xi=-dt/t^{2}$ and $\frac{1-\xi}{\xi}=t-1$:
\begin{equation}

\end{equation}
Here $E_{1}$ is the exponential integral, and $\int_{1}^{\infty}\frac{dt}{t^{2}}e^{-iat}=e^{-ia}-iaE_{1}(ia)$ was used.

\noindent\textbf{Step 2.} With $E_{1}(ia)=-\mathrm{Ci}(|a|)+i\,\mathrm{sgn}(a)\big[\mathrm{Si}(|a|)-\frac{\pi}{2}\big]$, the real and imaginary parts are
\begin{align}

{\rm Im}&=\Big[\mathrm{sgn}\big(k\cdot(y'-y)\big)\Big(\mathrm{Si}\big(|k\cdot(y'-y)|\big)-\frac{\pi}{2}\Big)-k\cdot(y'-y)\,\mathrm{Ci}\big(|k\cdot(y'-y)|\big)+\sin\big(k\cdot(y'-y)\big)\Big].
\end{align}
These were checked against direct numerical integration to 10 digits. The real part is even in $k\cdot(y'-y)$ and the imaginary part is odd.

\subsection{Is there any cancellation between the groups?}
None. The five pieces have different kernels, or different phases, or both:
\begin{itemize}
\item Group I Row II and Group II have static kernels with the phase $e^{-ik\cdot(w'-w)}$, but different colour factors;
\item Group II Row IV has the phase $e^{-ik\cdot(y'-y)/\xi}$;
\item Group I Row V and Group III Row III have $p^{+}$-dependent phases.
\end{itemize}
The two-$\rho$ finite terms therefore add up to the sum below.
