Both rapidity-sensitive regions are removed at the level of the master expression, with the
$z$-integral still undone; every counterterm is subtracted once and added back once, so no
regrouping or relocation of terms is needed afterwards. The sequence is: master $\to$ per-$\xi$
form $\to$ two endpoint subtractions $\to$ (JIMWLK $+$ endpoint) and (BFKL $+$ boundary) from
the counterterms $\to$ $z$-integration of the doubly subtracted remainder $\to$ DGLAP $+$ RC
$+$ azimuthal.

\section{From the master to the per-$\xi$ form (no $z$-integral yet)}
The starting point is the master expression
\begin{align} \label{eq:first}
     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{4}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\int_{z,w,w^{\prime},x'} e^{-ik(w^{\prime}-w)}\,  \int_{x,y,y^{\prime}}\,\frac{(w-z)^{m}\,(x-y)^{k}\,(w'-z)^{m'}\,(x'-y')^{k'}}{(w-z)^{2}\,(x-y)^{2}\,(w'-z)^{2}\,(x'-y')^{2}}\,\frac{p^{+}\,k^{+}}{(p^{+}+k^{+})^{4}}\notag \\ & \delta^{(2)}\left[y-w+\frac{p^{+}}{(p^{+}+k^{+})}\,(w-z)\right]\,\times\, \delta^{(2)}\left[y'-w'+\frac{p^{+}}{(p^{+}+k^{+})}\,(w'-z)\right]\times\notag \\ &  \left[\delta_{km}\,\delta_{ij}-\frac{(p^{+}+k^{+})}{p^{+}}\,\delta_{jm}\,\delta_{ik}-\frac{(p^{+}+k^{+})}{k^{+}}\,\delta_{im}\,\delta_{kj}\right]\times\, \left[\delta_{k'm'}\,\delta_{ij}-\frac{(p^{+}+k^{+})}{p^{+}}\,\delta_{jm'}\,\delta_{ik'}-\frac{(p^{+}+k^{+})}{k^{+}}\,\delta_{im'}\,\delta_{k'j}\right]\notag \\ & \left[f^{c'd'a}\,U^{c'e'}(y')\,U^{d'b}(z)\,\rho^{e'}(x')-f^{c'd'a}\,
    U^{d'b}(z)\,U^{c'e'}(x')\,\rho^{e'}(x')\right]\times \left[f^{cda}\,U^{ce}(y)\,U^{db}(z)\,\rho^{e}(x)-f^{cda}\,
    U^{db}(z)\,U^{ce}(x)\,\rho^{e}(x)\right]
\end{align}
(obtained from the operator setup). Perform, exactly as before: the colour algebra
($U^{d'b}(z)U^{db}(z)=\delta^{d'd}$, $f^{c'da}f^{cda}=N_{c}\delta^{c'c}$, giving
$N_{c}\,\Wyy$); the $\delta$-function integration over $w,w'$ (Jacobian
$\big[\frac{p^{+}+k^{+}}{k^{+}}\big]^{2}$ per leg, $w-z=\frac{p^{+}+k^{+}}{p^{+}}(y-z)$
\dots\ equivalently $y-z=\frac{p^{+}}{p^{+}+k^{+}}(w-z)$, and phase
$e^{-ik(y'-y)\frac{p^{+}+k^{+}}{k^{+}}}$); the contraction of the two tensor brackets over
$i,j$; and the substitution $\xi=\frac{k^{+}}{p^{+}+k^{+}}$,
$dp^{+}=-\frac{k^{+}d\xi}{\xi^{2}}$ (the descending limits absorb the sign). The antisymmetric
$\frac{2}{k^{+}}\left(\delta_{mk'}\delta_{m'k}-\delta_{m'k'}\delta_{km}\right)$ channel is
dropped at once: for every term whose colour factor is $z$-independent, the $z$-integral
$\int_{z}\mathcal K^{m}(y-z)\mathcal K^{m'}(y'-z)$ is symmetric in $(m,m')$, so the
antisymmetric contraction vanishes identically at each $\xi$. The result, with the
\emph{$z$-integral still undone}, is
\begin{align}\label{eq:xiform}
\frac{d^{3}N_{\rm NLO}}{d^{3}k}
=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{8\pi^{5}}\,\frac{1}{k^{+}}
\int_{\bar\xi}^{\xi_{\max}}\frac{d\xi}{\xi^{2}}
\int_{x,x',y,y',z} e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,
\Big\{&\,d_{\perp}\,\xi(1-\xi)\,\big[\mathcal K(y-z)\!\cdot\!\mathcal K(x-y)\big]\big[\mathcal K(y'-z)\!\cdot\!\mathcal K(x'-y')\big]\notag\\
&+\Big[\frac{1-\xi}{\xi}+\frac{\xi}{1-\xi}\Big]\,K(y,y',z)\,\KK\,\Big\}\;\Wyy\,,
\end{align}
with $\bar\xi=\frac{k^{+}}{\vee}$, $\xi_{\max}=\frac{k^{+}}{k^{+}+\Lambda}$,
$\mathcal K^{m}(u)=\frac{u^{m}}{u^{2}}$, $K(a,b,z)=\mathcal K(a-z)\!\cdot\!\mathcal K(b-z)$.
(Check: carrying out $\int_{z}\mathcal K^{m}(y-z)\mathcal K^{m'}(y'-z)
=-\frac{\pi}{2}\Leps\delta^{mm'}-\pi\frac{r^{m}r^{m'}}{r^{2}}$, $r=y'-y$, on
\eqref{eq:xiform} reproduces verbatim the $d_{\perp}$-form used previously, so
\eqref{eq:xiform} is the same cross-section, one step earlier.)

\section{The two poles and their counterterms}
The curly bracket of \eqref{eq:xiform} is singular at the two ends of the $\xi$ range, and
nowhere else:
\begin{itemize}
\item $\xi\to1$ ($p^{+}\to0$): only the channel $\frac{\xi}{1-\xi}$ blows up. The unmeasured
gluon becomes soft on the \emph{target} side --- this is the JIMWLK region, regulated by
$\Lambda$ through $\xi_{\max}$. At the endpoint the integrand is smooth in the positions: the
phase $\to e^{-ik\cdot(y'-y)}$ and $\frac{1}{\xi}\to1$.
\item $\xi\to0$ ($p^{+}\to\vee$): the channel $\frac{1-\xi}{\xi}$, together with the measure
$\frac{d\xi}{\xi^{2}}$ and the phase, hides a logarithm of $\frac{\vee}{k^{+}}$. The unmeasured
gluon carries almost all of the \emph{projectile} momentum --- the BFKL region. In the
positions this is the collapse $y,y'\to z$ (from the $\delta$-functions of \eqref{eq:first},
$y-z=\xi(w-z)$, $y'-z=\xi(w'-z)$ at fixed $w,w'$).
\end{itemize}
Define one counterterm per pole, each equal to the exact limit of the corresponding channel:
\begin{align}
T_{J}(\xi)&=\frac{1}{1-\xi}\;e^{-ik\cdot(y'-y)}\,K(y,y',z)\,\KK\;\Wyy\,,\label{eq:TJ}\\
T_{B}(\xi)&=\frac{1}{\xi^{2}}\,\frac{1-\xi}{\xi}\;e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,K(y,y',z)\,K(x,x',z)\;\Wzz\,.\label{eq:TB}
\end{align}
$T_{J}$ is the $\frac{\xi}{1-\xi}$ channel with every regular factor set to its $\xi=1$ value
(the full integrand there is $\frac{1}{\xi(1-\xi)}e^{-ik(y'-y)/\xi}K\,\KK\,W$, so the
difference carries $\frac{e^{-ik(y'-y)/\xi}/\xi-e^{-ik(y'-y)}}{1-\xi}$, finite at $\xi=1$).
$T_{B}$ is the $\frac{1-\xi}{\xi}$ channel with the slow factors frozen at the collapse point,
$\KK\,W(y,y')\to K(x,x',z)\,W(z,z)$; their difference is the single bracket
\begin{align}\label{eq:combo}
\frac{1}{\xi^{2}}\,\frac{1-\xi}{\xi}\,e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,K(y,y',z)
\Big[\KK\,W(y,y')-K(x,x',z)\,W(z,z)\Big]\,,
\end{align}
whose square bracket vanishes linearly in $\xi$ at fixed $(w,w',z)$, because
$\mathcal K(x-z)\!\cdot\!\mathcal K(x'-z)=K(x,x',z)$ identically. With
$\mathcal N\equiv\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}$ the exact,
one-line decomposition of \eqref{eq:xiform} is
\begin{align}\label{eq:split}
\boxed{\;\frac{d^{3}N_{\rm NLO}}{d^{3}k}
=\underbrace{\mathcal N\!\int\! d\xi\!\int\Big[\,\{\text{integrand of \eqref{eq:xiform}}\}-T_{J}(\xi)-T_{B}(\xi)\Big]}_{\textstyle d^{3}N_{\rm reg}:\ \text{finite at both endpoints}}
\;+\;\underbrace{\mathcal N\!\int\! d\xi\!\int T_{J}}_{\textstyle J}
\;+\;\underbrace{\mathcal N\!\int\! d\xi\!\int T_{B}}_{\textstyle B}\;}
\end{align}
--- each counterterm subtracted once and added back once; nothing is moved twice. In
$d^{3}N_{\rm reg}$ the limits can be taken to $[\bar\xi,1]$ at a cost $O(\Lambda)$, and the
$\bar\xi$ end produces no large logarithm.

\section{The $\xi\to1$ counterterm: JIMWLK $+$ endpoint term}
$T_{J}$ depends on $\xi$ only through $\frac{1}{1-\xi}$, so
\begin{align}
J=\mathcal N\,\Big[\int_{\bar\xi}^{\xi_{\max}}\frac{d\xi}{1-\xi}\Big]
\int_{x,x',y,y',z} e^{-ik\cdot(y'-y)}\,K(y,y',z)\,\KK\,\Wyy\,,\qquad
\int_{\bar\xi}^{\xi_{\max}}\frac{d\xi}{1-\xi}
={\color{blue}\log\frac{k^{+}}{\Lambda}}+\log\Big(1-\frac{k^{+}}{\vee}\Big)+O(\Lambda)\,.
\end{align}
The $\log\frac{k^{+}}{\Lambda}$ part is the \textbf{JIMWLK term}, already in its
manifestly-$\mu$-free $z$-form,
\begin{align}\label{eq:jimwlk}
\boxed{\;\frac{d^{3}N_{\rm JIMWLK}}{d^{3}k}
=+\,\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}\,{\color{blue}\log\frac{k^{+}}{\Lambda}}
\int_{x,y,z,x',y'}e^{-ik\cdot(y'-y)}\,K(y,y',z)\,\KK\,\Wyy\;}
\end{align}
(equivalently, with $\frac{1}{\pi}\int_{z}K(y,y',z)=-\frac{2}{\epsilon}-2\gamma_{E}
-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}}{4}\equiv-\Leps$, the $\Leps$-form with the
all-negative bracket). The $\log(1-\frac{k^{+}}{\vee})$ part is the \textbf{endpoint
(plus-prescription convention) term}; $z$-integrated it reads
\begin{align}\label{eq:endpoint}
-\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{8\pi^{4}}\,\frac{1}{k^{+}}\,\log\Big(1-\frac{k^{+}}{\vee}\Big)
\int_{x,y,x',y'}e^{-ik\cdot(y'-y)}\,\KK\;\Leps\;\Wyy\,.
\end{align}

\section{The $\xi\to0$ counterterm: BFKL $+$ boundary term}
$T_{B}$ depends on $\xi$ only through the phase and explicit powers. The exact substitution
$y=z+\xi(w-z)$, $y'=z+\xi(w'-z)$ gives $d^{2}y\,d^{2}y'=\xi^{4}d^{2}w\,d^{2}w'$,
$K(y,y',z)=K(w,w',z)/\xi^{2}$ and $e^{-ik(y'-y)/\xi}=e^{-ik(w'-w)}$ exactly, so the transverse
integral of $T_{B}$ is $\xi^{2}\times(\xi$-independent$)$, the $\xi^{2}$ cancels the measure,
and
\begin{align}
B=\mathcal N\,\Big[\int_{\bar\xi}^{1}d\xi\,\frac{1-\xi}{\xi}\Big]\,\mathcal I
=\mathcal N\Big[{\color{blue}\log\frac{\vee}{k^{+}}}-\Big(1-\frac{k^{+}}{\vee}\Big)\Big]\mathcal I\,,
\qquad
\mathcal I=\int_{x,x',z,w,w'}e^{-ik\cdot(w'-w)}K(w,w',z)\,K(x,x',z)\,\Wzz\,.
\end{align}
The $\log\frac{\vee}{k^{+}}$ part is the \textbf{BFKL term},
\begin{align}\label{eq:bfkl}
\boxed{\;\frac{d^{3}N_{\rm BFKL}}{d^{3}k}
=+\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{8\pi^{5}}\,\frac{1}{k^{+}}\,{\color{blue}\log\frac{\vee}{k^{+}}}\;\mathcal I\;}
\end{align}
(UV finite, $\mu$-independent), and the remainder is the \textbf{boundary term}
$-\mathcal N\,(1-\frac{k^{+}}{\vee})\,\mathcal I$.

\section{The doubly subtracted remainder: DGLAP $+$ RC $+$ azimuthal}
$d^{3}N_{\rm reg}$ contains three structures: the $d_{\perp}\xi(1-\xi)$ channel, the locally
subtracted soft channel $\frac{\xi}{(1-\xi)_{+}}$ (the $\frac{\xi}{1-\xi}$ channel minus
$T_{J}$), and the $\mu$-free pair (the $\frac{1-\xi}{\xi}$ channel minus $T_{B}$,
eq.~\eqref{eq:combo}). The pair is finite in $d=2$ and is \emph{kept as it is}. In the other
two structures the colour factor is $z$-independent, so the $z$-integral is carried out in
dimensional regularization:
\begin{align}
\int_{z}\mathcal K^{m}(y-z)\,\mathcal K^{m'}(y'-z)
=-\frac{\pi}{2}\,\Leps\,\delta^{mm'}-\pi\,\frac{r^{m}r^{m'}}{r^{2}}\,,\qquad r=y'-y\,.
\end{align}
Channel by channel (prefactor $\mathcal N$, measure $\frac{d\xi}{\xi^{2}}$):
\begin{itemize}
\item[(a)] $d_{\perp}\,\xi(1-\xi)\times\delta^{mm'}$ part:
$\frac{d_{\perp}}{2}\Leps=\Leps-1+O(\epsilon)$, so it yields
$-\,\xi(1-\xi)\,(\Leps-1)$ on $\KK$;
\item[(b)] $d_{\perp}\,\xi(1-\xi)\times\frac{r^{m}r^{m'}}{r^{2}}$ part: the r-tensor
(azimuthal) structure, coefficient $-d_{\perp}\,\xi(1-\xi)$;
\item[(c)] $\frac{\xi}{(1-\xi)_{+}}$ channel: $\int_{z}K=-\pi\Leps$ exactly (the
$d_{\perp}$-trace and the $\frac{r r}{r^{2}}$-trace combine into $\Leps$ with no leftover),
giving $-\,\frac{\xi}{(1-\xi)_{+}}\,\Leps$ on $\KK$.
\end{itemize}
Using $\frac{\xi(1-\xi)}{\xi^{2}}=\frac{1-\xi}{\xi}$, the ``$-1$'' of (a) and the r-tensor of
(b) combine into the azimuthal term, and the $\Leps$ pieces of (a) and (c) assemble into the
soft-channel-removed splitting function
$\mathcal P_{gg,+}(\xi)-2N_{c}\frac{1-\xi}{\xi}
=2N_{c}\big[\xi(1-\xi)+\frac{\xi}{(1-\xi)_{+}}\big]$. Finally adding and subtracting
$\frac{\beta_{0}}{2}\delta(1-\xi)$ ($\beta_{0}=\frac{11N_{c}}{3}$; the $\delta$ sits at
$\xi=1$, where $\frac{1}{\xi^{2}}=1$ and the phase is the LO phase) splits off the
running-coupling term:
\begin{align}\label{eq:reg}
\frac{d^{3}N_{\rm reg}}{d^{3}k}
=&\underbrace{-\,\frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\,\frac{g^{2}}{8\pi^{2}}\,\frac{1}{k^{+}}
\int_{\bar\xi}^{1}\frac{d\xi}{\xi^{2}}\Big[\mathcal P_{gg}(\xi)-2N_{c}\frac{1-\xi}{\xi}\Big]
\int_{x,y,x',y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,\KK\;\Leps\;\Wyy}_{\textstyle\rm DGLAP\ (non\text{-}soft)}\notag\\
&\underbrace{+\,\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}
\int_{\bar\xi}^{1}\frac{d\xi}{\xi^{2}}\,\frac{1-\xi}{\xi}
\int_{x,y,x',y',z}e^{-ik\cdot(y'-y)\frac{1}{\xi}}K(y,y',z)
\Big[\KK\,W(y,y')-K(x,x',z)\,W(z,z)\Big]\rho\rho}_{\textstyle\rm DGLAP\ (soft\ pair;\ free\ of\ \frac{2}{\epsilon},\,\mu\ and\ \log\frac{\vee}{k^{+}})}\notag\\
&\underbrace{+\,\frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\,\frac{g^{2}\beta_{0}}{16\pi^{2}}\,\frac{1}{k^{+}}
\int_{x,y,x',y'}e^{-ik\cdot(y'-y)}\,\KK\;\Leps\;\Wyy}_{\textstyle\rm RC}\notag\\
&\underbrace{+\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{8\pi^{4}}\,\frac{1}{k^{+}}
\int_{\bar\xi}^{1}d\xi\,\frac{1-\xi}{\xi}
\int_{x,y,x',y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}
\Big[\KK-2\,\frac{(\mathcal K(x-y)\!\cdot\!r)(\mathcal K(x'-y')\!\cdot\!r)}{r^{2}}\Big]\Wyy}_{\textstyle\rm azimuthal}
\end{align}
(in the soft-pair line the colour brackets are those written in \eqref{eq:combo}, abbreviated;
$d_{\perp}\to2$ in the azimuthal term, which multiplies no pole).

\section{Result}
\begin{align}
\frac{d^{3}N_{\rm NLO}}{d^{3}k}
=\underbrace{\eqref{eq:jimwlk}}_{\rm JIMWLK}
+\underbrace{\eqref{eq:bfkl}}_{\rm BFKL}
+\underbrace{{\rm DGLAP\ (non\text{-}soft\ +\ soft\ pair)}+{\rm RC}}_{\eqref{eq:reg}}
+\underbrace{{\rm endpoint\ \eqref{eq:endpoint}}+{\rm boundary}+{\rm azimuthal}}_{\rm finite\ terms}\,,
\end{align}
with the boundary term $-\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}
(1-\frac{k^{+}}{\vee})\,\mathcal I$. This coincides term by term with the categories of the
corrected Row-IV note; the difference is only the order of operations:
\begin{itemize}
\item[(i)] both poles are removed \emph{before} any dimensional regularization, so the two
rapidity logarithms are produced directly by the two counterterm integrals ---
$\log\frac{k^{+}}{\Lambda}$ with the (z-form) LO$\,\otimes\,$soft-kernel coefficient (JIMWLK)
and $\log\frac{\vee}{k^{+}}$ with the UV-finite, $\mu$-free production kernel (BFKL) --- and
neither ever multiplies $\Leps$;
\item[(ii)] the subtraction of $T_{B}$ automatically leaves the soft channel in its $\mu$-free
paired form inside DGLAP: no term is ever counted twice and no term has to be moved between
categories afterwards;
\item[(iii)] the only ultraviolet logarithms in the final result are the collinear $\Leps$ of
the non-soft DGLAP kernel, the RC term, and the endpoint term; the finite terms (endpoint,
boundary, azimuthal) carry no large longitudinal logarithm.
\end{itemize}
