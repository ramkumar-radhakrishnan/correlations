\section{Starting expression}
The expression below has the cutoffs $\Lambda$ on the longitudinal integrals:
\begin{align}
    \frac{d^{3}N_{2b\,\rm finite\, NLO}^{\rm}}{d^{3}k} & = \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{\Lambda}^{k^{+}-\Lambda}\frac{dp^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\,\Bigg\{\notag\\
&e^{-ik\cdot(z-x)\frac{p^{+}}{k^{+}}}\,\frac{1}{\left[p^{+}(y-x)^{2}+(k^{+}-p^{+})(y-z)^{2}\right]}\,\frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\notag\\
&\times\Bigg[\frac{\delta^{im}\,d_{\perp}}{2\,(x-z)^{2}}\left[(y-x)^{2}-(y-z)^{2}\right]\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}
\notag \\ & +\frac{(k^{+}-p^{+})\,\delta_{im}}{k^{+}}\left[\frac{(y-z)\cdot(x-z)}{(x-z)^{2}}+\frac{(y-x)\cdot(y-z)}{2(y-x)^{2}}-\frac{(y-x)^{2}-(y-z)^{2}}{2(x-z)^{2}}\right]\notag\\
&-\frac{k^{+}-p^{+}}{p^{+}}\left[\frac{(y-z)^{i}(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}(y-z)^{i}}{2(y-x)^{2}}\right]\notag\\
&+\frac{p^{+}\,\delta_{im}}{k^{+}}\left[\frac{(y-x)\cdot(x-z)}{(x-z)^{2}}-\frac{(y-x)\cdot(y-z)}{2(y-z)^{2}}-\frac{(y-x)^{2}-(y-z)^{2}}{2(x-z)^{2}}\right]\notag\\
&-\left[\frac{(y-x)^{m}(x-z)^{i}}{(x-z)^{2}}-\frac{(y-x)^{m}(y-z)^{i}}{2(y-z)^{2}}+\frac{(y-z)^{m}(x-z)^{i}}{(x-z)^{2}}+\frac{(y-x)^{i}(y-z)^{m}}{2(y-x)^{2}}\right]\Bigg]\notag\\
&\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag\\
&-\frac{1}{k^{+}}\left[\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}+\frac{p^{+}}{k^{+}-p^{+}}+\frac{k^{+}-p^{+}}{p^{+}}\right]
\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}\,(y-z)^{2}}\,\frac{(y-x)\cdot(y-z)}{(y-x)^{2}\,(x-z)^{2}}\notag\\
&\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(z)\Big]\rho^{b'}(x')\,\rho^{f}(y)\Bigg\}\notag \\ &  + \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{k^{+}-\Lambda}^{\Lambda}\frac{dq^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\ e^{-ik\cdot(z-x)\frac{(k^{+}-q^{+})}{k^{+}}}\,\frac{1}{\left[(k^{+}-q^{+})(y-x)^{2}+q^{+}\,(y-z)^{2}\right]}\notag \\ & \frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\,\left[\frac{(y-x)^{i}(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}(y-z)^{m}}{2(y-z)^{2}}\right]\,\frac{k^{+}}{q^{+}}\times\notag \\ & \Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag \\ & - \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{k^{+}-\Lambda}^{\Lambda}\frac{dq^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\ e^{-ik\cdot(z-x)\frac{(k^{+}-q^{+})}{k^{+}}}\,\frac{1}{\left[(k^{+}-q^{+})(y-x)^{2}+q^{+}\,(y-z)^{2}\right]}\notag \\ & \frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\,\left[\frac{(y-x)^{i}(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}(y-z)^{m}}{2(y-z)^{2}}\right]\notag \\ & \Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)
\label{eq:start}
\end{align}

\section{Step 0: shorthand}
\begin{align}
&\mathcal{N}\equiv\frac{1}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{4}}\frac{1}{k^{+}},\qquad \frac{\mathcal{N}}{2\pi}=\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}},\\
&\Phi(p^{+})=e^{-ik\cdot(z-x)\frac{p^{+}}{k^{+}}},\qquad D(p^{+})=p^{+}(y-x)^{2}+(k^{+}-p^{+})(y-z)^{2},\qquad T^{im}=\frac{(x'-w')^{i}(z-x)^{m}}{(x'-w')^{2}(z-x)^{2}},\\
&\Psi(q^{+})=e^{-ik\cdot(z-x)\frac{k^{+}-q^{+}}{k^{+}}},\qquad \widetilde D(q^{+})=(k^{+}-q^{+})(y-x)^{2}+q^{+}(y-z)^{2},\\
&B^{im}=\frac{(y-z)^{i}(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}(y-z)^{i}}{2(y-x)^{2}},\qquad
A^{im}=\frac{(y-x)^{i}(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}(y-z)^{m}}{2(y-z)^{2}},\\
&\mathcal{C}_{x}=\big[U^{ab'}(x')-U^{ab'}(w')\big]\big[N_{c}U^{af}(y)-N_{c}U^{af}(x)\big]\rho^{b'}(x')\rho^{f}(y),\qquad \mathcal{C}_{z}=\mathcal{C}_{x}\big|_{U(x)\to U(z)},\\
&K_{S}=\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(y-z)^{2}}\,\frac{(y-x)\cdot(y-z)}{(y-x)^{2}(x-z)^{2}}.
\end{align}

\section{Step 1: which terms are singular at the endpoints}
In the $p^{+}$ block, $\Phi/D$ is finite at $p^{+}=0$ (where $D=k^{+}(y-z)^{2}$) and at $p^{+}=k^{+}$ (where $D=k^{+}(y-x)^{2}$). The coefficients $\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}$, $\frac{k^{+}-p^{+}}{k^{+}}$, $\frac{p^{+}}{k^{+}}$ and $1$ are finite. Only three structures are singular:
\begin{itemize}
\item $-\frac{k^{+}-p^{+}}{p^{+}}B^{im}$ in the $p^{+}$ block, at $p^{+}\to0$;
\item $\frac{k^{+}}{q^{+}}A^{im}$ in the $q^{+}$ block, at $q^{+}\to0$ (that is, $p^{+}\to k^{+}$);
\item $\frac{p^{+}}{k^{+}-p^{+}}$ and $\frac{k^{+}-p^{+}}{p^{+}}$ in the subtraction term.
\end{itemize}
In every other term $\Lambda\to0$ is taken directly, with an error of $O(\Lambda/k^{+})$.

\section{Step 2: split the singular coefficient}
\begin{equation}
-\frac{k^{+}-p^{+}}{p^{+}}\,B^{im}=-\frac{k^{+}}{p^{+}}\,B^{im}+B^{im}.
\end{equation}
The $+B^{im}$ piece is finite, so $\Lambda\to0$ can be taken in it.

\section{Step 3: plus prescription at $p^{+}\to0$}
Definition:
\begin{equation}
\int_{0}^{k^{+}}dp^{+}\Big[\frac{k^{+}}{p^{+}}\Big]_{+}f(p^{+})\equiv\int_{0}^{k^{+}}dp^{+}\,\frac{k^{+}}{p^{+}}\big[f(p^{+})-f(0)\big].
\end{equation}
Add and subtract $f(0)$ (this is exact):
\begin{align}
\int_{\Lambda}^{k^{+}-\Lambda}dp^{+}\frac{k^{+}}{p^{+}}f(p^{+})&=\int_{\Lambda}^{k^{+}-\Lambda}dp^{+}\frac{k^{+}}{p^{+}}\big[f(p^{+})-f(0)\big]+k^{+}f(0)\log\frac{k^{+}-\Lambda}{\Lambda}\notag\\
&\xrightarrow{\Lambda\to0}\int_{0}^{k^{+}}dp^{+}\Big[\frac{k^{+}}{p^{+}}\Big]_{+}f(p^{+})+k^{+}f(0)\log\frac{k^{+}}{\Lambda},
\end{align}
with $f(p^{+})=\Phi(p^{+})\,T^{im}B^{im}/D(p^{+})$.

\section{Step 4: evaluate $k^{+}f(0)$}
At $p^{+}=0$: $\Phi=1$ and $D=k^{+}(y-z)^{2}$, so $k^{+}f(0)=T^{im}B^{im}/(y-z)^{2}$. Contract the two terms of $B$ with $T$:
\begin{align}
\frac{(x'-w')^{i}(z-x)^{m}}{(x'-w')^{2}(z-x)^{2}}\frac{(y-z)^{i}(x-z)^{m}}{(x-z)^{2}}&=-\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(x-z)^{2}},\\
\frac{(x'-w')^{i}(z-x)^{m}}{(x'-w')^{2}(z-x)^{2}}\frac{(y-x)^{m}(y-z)^{i}}{2(y-x)^{2}}&=-\frac{(x'-w')\cdot(y-z)\,(x-z)\cdot(y-x)}{2(x'-w')^{2}(x-z)^{2}(y-x)^{2}}.
\end{align}
Hence
\begin{equation}
k^{+}f(0)=-\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(y-z)^{2}}\left[\frac{1}{(x-z)^{2}}+\frac{(x-z)\cdot(y-x)}{2(x-z)^{2}(y-x)^{2}}\right].
\end{equation}
Multiplying by the $-1$ of Step 2 and by $\frac{1}{2\pi}$ from $\frac{dp^{+}}{2\pi}$ gives the endpoint term of the $p^{+}$ block:
\begin{equation}
E_{p}=\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\log\frac{k^{+}}{\Lambda}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(y-z)^{2}}\left[\frac{1}{(x-z)^{2}}+\frac{(x-z)\cdot(y-x)}{2(x-z)^{2}(y-x)^{2}}\right]\mathcal{C}_{x}.
\end{equation}

\section{Step 5: the $q^{+}$ block}
The two $q^{+}$ integrals of \eqref{eq:start} together carry $\big(\frac{k^{+}}{q^{+}}-1\big)A^{im}$.
\begin{itemize}
\item The $-1$ piece is finite, so $\Lambda\to0$ gives $-\int_{k^{+}}^{0}dq^{+}(\cdots)A^{im}$.
\item For the $\frac{k^{+}}{q^{+}}$ piece, use $\int_{k^{+}-\Lambda}^{\Lambda}=-\int_{\Lambda}^{k^{+}-\Lambda}$ and Step 3 with $h(q^{+})=\Psi(q^{+})\,T^{im}A^{im}/\widetilde D(q^{+})$:
\end{itemize}
\begin{equation}
\int_{k^{+}-\Lambda}^{\Lambda}dq^{+}\frac{k^{+}}{q^{+}}h(q^{+})=\int_{k^{+}}^{0}dq^{+}\Big[\frac{k^{+}}{q^{+}}\Big]_{+}h(q^{+})-k^{+}h(0)\log\frac{k^{+}}{\Lambda}.
\end{equation}
At $q^{+}=0$: $\widetilde D=k^{+}(y-x)^{2}$ and $e^{-ik\cdot(w'-z)}\Psi(0)=e^{-ik\cdot(w'-z)}e^{-ik\cdot(z-x)}=e^{-ik\cdot(w'-x)}$. The contraction is
\begin{equation}
T^{im}A^{im}=\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}}\left[-\frac{1}{(x-z)^{2}}+\frac{(x-z)\cdot(y-z)}{2(x-z)^{2}(y-z)^{2}}\right],
\end{equation}
so the endpoint term of the $q^{+}$ block is
\begin{equation}
E_{q}=\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\log\frac{k^{+}}{\Lambda}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-x)}\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}(y-x)^{2}}\left[\frac{1}{(x-z)^{2}}-\frac{(x-z)\cdot(y-z)}{2(x-z)^{2}(y-z)^{2}}\right]\mathcal{C}_{x}.
\end{equation}

\section{Step 6: the subtraction term, integrated exactly}
The kernel $K_{S}$ is independent of $p^{+}$, so the $p^{+}$ integral is done exactly. With $\xi=p^{+}/k^{+}$ and $\lambda=\Lambda/k^{+}$:
\begin{align}
\frac{1}{k^{+}}\int_{\Lambda}^{k^{+}-\Lambda}dp^{+}\Big[\tfrac{p^{+}(k^{+}-p^{+})}{k^{+2}}+\tfrac{p^{+}}{k^{+}-p^{+}}+\tfrac{k^{+}-p^{+}}{p^{+}}\Big]
&=\int_{\lambda}^{1-\lambda}d\xi\Big[\xi(1-\xi)+\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\Big],\\
\int_{\lambda}^{1-\lambda}d\xi\,\xi(1-\xi)&=\frac{1}{6}+O(\lambda^{2}),\\
\int_{\lambda}^{1-\lambda}d\xi\,\frac{\xi}{1-\xi}=\int_{\lambda}^{1-\lambda}d\xi\Big[\frac{1}{1-\xi}-1\Big]&=\log\frac{1-\lambda}{\lambda}-(1-2\lambda),\\
\int_{\lambda}^{1-\lambda}d\xi\,\frac{1-\xi}{\xi}=\int_{\lambda}^{1-\lambda}d\xi\Big[\frac{1}{\xi}-1\Big]&=\log\frac{1-\lambda}{\lambda}-(1-2\lambda),\\
\text{sum}&=2\log\frac{k^{+}-\Lambda}{\Lambda}-\frac{11}{6}+O(\lambda)\ \to\ 2\log\frac{k^{+}}{\Lambda}-\frac{11}{6}.
\end{align}
The plus-prescription form gives the same result. Write $\frac{\xi}{1-\xi}=\big[\frac{1}{1-\xi}\big]_{+}-1$ and $\frac{1-\xi}{\xi}=\big[\frac{1}{\xi}\big]_{+}-1$. The kernel is $\xi$-independent and $\int_{0}^{1}[\cdot]_{+}=0$, so only $\frac{1}{6}-1-1=-\frac{11}{6}$ survives; the $\log$'s are the endpoint pieces.

Including $\frac{1}{2\pi}$, the subtraction term becomes
\begin{align}
&-\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\Big[2\log\frac{k^{+}}{\Lambda}-\frac{11}{6}\Big]\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}K_{S}\,\mathcal{C}_{z}\notag\\
&=\underbrace{+\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\frac{11}{6}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}K_{S}\,\mathcal{C}_{z}}_{\text{the }11/6\text{ term}}
\ \underbrace{-\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\,2\log\frac{k^{+}}{\Lambda}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}K_{S}\,\mathcal{C}_{z}}_{E_{S}}.
\end{align}

\section{Step 7: result}
\begin{align}
    \frac{d^{3}N_{2b\,\rm finite\, NLO}^{\rm}}{d^{3}k} & = \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{0}^{k^{+}}\frac{dp^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\,e^{-ik\cdot(z-x)\frac{p^{+}}{k^{+}}}\,\frac{1}{\left[p^{+}(y-x)^{2}+(k^{+}-p^{+})(y-z)^{2}\right]}\notag \\ & \frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\,\times\Bigg[\frac{\delta^{im}\,d_{\perp}}{2\,(x-z)^{2}}\left[(y-x)^{2}-(y-z)^{2}\right]\frac{p^{+}(k^{+}-p^{+})}{k^{+2}}
\notag \\ & +\frac{(k^{+}-p^{+})\,\delta_{im}}{k^{+}}\left[\frac{(y-z)\cdot(x-z)}{(x-z)^{2}}+\frac{(y-x)\cdot(y-z)}{2(y-x)^{2}}-\frac{(y-x)^{2}-(y-z)^{2}}{2(x-z)^{2}}\right]\notag\\
&-\left[\frac{k^{+}}{p^{+}}\right]_{+}\,\left[\frac{(y-z)^{i}(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}(y-z)^{i}}{2(y-x)^{2}}\right]+\left[\frac{(y-z)^{i}(x-z)^{m}}{(x-z)^{2}}+\frac{(y-x)^{m}(y-z)^{i}}{2(y-x)^{2}}\right]\notag\\
&+\frac{p^{+}\,\delta_{im}}{k^{+}}\left[\frac{(y-x)\cdot(x-z)}{(x-z)^{2}}-\frac{(y-x)\cdot(y-z)}{2(y-z)^{2}}-\frac{(y-x)^{2}-(y-z)^{2}}{2(x-z)^{2}}\right]\notag\\
&-\left[\frac{(y-x)^{m}(x-z)^{i}}{(x-z)^{2}}-\frac{(y-x)^{m}(y-z)^{i}}{2(y-z)^{2}}+\frac{(y-z)^{m}(x-z)^{i}}{(x-z)^{2}}+\frac{(y-x)^{i}(y-z)^{m}}{2(y-x)^{2}}\right]\Bigg]\notag\\
&\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag \\ & + \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{k^{+}}^{0}\frac{dq^{+}}{2\pi}\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\ e^{-ik\cdot(z-x)\frac{(k^{+}-q^{+})}{k^{+}}}\,\frac{1}{\left[(k^{+}-q^{+})(y-x)^{2}+q^{+}\,(y-z)^{2}\right]}\notag \\ & \frac{(x'-w')^{i}\,(z-x)^{m}}{(x'-w')^{2}\,(z-x)^{2}}\,\left[\frac{(y-x)^{i}(x-z)^{m}}{(x-z)^{2}}-\frac{(y-x)^{i}(y-z)^{m}}{2(y-z)^{2}}\right]\,\left\{\left[\frac{k^{+}}{q^{+}}\right]_{+}-1\right\}\notag \\ & \times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(x)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag \\ & +\, \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\,\frac{11}{6}\,\int_{x,y,z,x',w'}e^{-ik\cdot(w'-z)}\, \frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}\,(y-z)^{2}}\,\frac{(y-x)\cdot(y-z)}{(y-x)^{2}\,(x-z)^{2}}\notag \\ & \times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[N_{c}\,U^{af}(y)-N_{c}\,U^{af}(z)\Big]\rho^{b'}(x')\,\rho^{f}(y)\notag\\
&+\,{\color{blue}E_{p}+E_{q}+E_{S}},
\label{eq:result}
\end{align}
where
\begin{align}
{\color{blue}E_{p}+E_{q}+E_{S}}&=\frac{1}{(2\pi)^{3}}\frac{g^{4}}{16\pi^{5}}\frac{1}{k^{+}}\,{\color{blue}\log\frac{k^{+}}{\Lambda}}\int_{x,y,z,x',w'}\Bigg\{
e^{-ik\cdot(w'-z)}\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(y-z)^{2}}\left[\frac{1}{(x-z)^{2}}+\frac{(x-z)\cdot(y-x)}{2(x-z)^{2}(y-x)^{2}}\right]\mathcal{C}_{x}\notag\\
&+e^{-ik\cdot(w'-x)}\frac{(x'-w')\cdot(y-x)}{(x'-w')^{2}(y-x)^{2}}\left[\frac{1}{(x-z)^{2}}-\frac{(x-z)\cdot(y-z)}{2(x-z)^{2}(y-z)^{2}}\right]\mathcal{C}_{x}
-2\,e^{-ik\cdot(w'-z)}K_{S}\,\mathcal{C}_{z}\Bigg\}.
\end{align}

\section{Where each piece goes}
\begin{itemize}
\item The $\Lambda$-independent part of \eqref{eq:result} (everything except the blue terms) is exactly the $N_{2b\,\rm finite}$ equation of the file \path{Other_terms_updated.tex}, including the $\frac{11}{6}$ term.
\item The blue $\log(k^{+}/\Lambda)$ terms $E_{p}+E_{q}+E_{S}$ are the ``$N_{2b\,\rm finite}$'' entries of the JIMWLK file. They do not belong in the finite file.
\item $E_{S}=-2\log\frac{k^{+}}{\Lambda}\,K_{S}\,\mathcal{C}_{z}$ cancels the $+2\log\frac{k^{+}}{\Lambda}\,K_{S}\,\mathcal{C}_{z}$ of $N_{2b\,\rm UV}$. In the JIMWLK file these appear as the $+2N_{c}$ and $-2N_{c}$ blocks, which are equal and opposite.
\item The $\frac{11}{6}$ term is kept unintegrated over $x$. It diverges at $x\to z$ and is the subtraction partner of the remaining terms, which diverge oppositely there; their sum is UV finite. Its partner in $N_{2b\,\rm UV}$ is the $-\frac{11}{6}$ that cancels the $\frac{11}{3}$ of Row II.
\end{itemize}
