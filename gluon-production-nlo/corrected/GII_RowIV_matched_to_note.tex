\section*{What was changed relative to the previous version}
\begin{enumerate}
\item $\log\frac{(y'-y)^2\mu^2}{4}\to\log\frac{(y'-y)^2\mu^2}{4\xi^2}$ at $\xi\neq1$ (the note continues the
relative momentum $\ell$ of the pair to $d$ dimensions, Eq.~(50)). This is the only change that alters
the sum, by $+\frac{1}{(2\pi)^3}\frac{g^4N_c}{8\pi^4}\frac{1}{k^+}\int_0^1d\xi\,\hat P(\xi)\frac{\log\xi^2}{\xi^2}\int e^{-ik\cdot(y'-y)/\xi}(\dots)$.
\item DGLAP and RC now contain only the poles; their finite parts moved into ``finite''. The finite
parts of the $\frac{\beta_0}{2}\delta(1-\xi)$ term and of the old RC term were equal and opposite, so
nothing is lost.
\item $k^+/\vee\to0$ in the lower limits. The old ``$(1-\xi)/\xi$'' subtraction line and the boundary term
$\propto[\frac1\vee-\frac1{k^+}]$ combine exactly into the single $-\mathcal G(0^+)/\xi$ subtraction
inside ``finite'' (old S line $+$ boundary $=-\frac{\bar C}{k^+}\mathcal B\int_{k^+/\vee}^1\frac{d\xi}{\xi}$).
\item The old $\int_{k^+/\vee}^1d\xi\,\frac{\xi}{(1-\xi)_+}$ becomes the $\int_0^1$ plus distribution of the note,
Eq.~(33); the difference is $O(k^+/\vee)$.
\item BFKL and JIMWLK are unchanged.
\end{enumerate}

\section*{Remarks (as in the note)}
\textbf{Net-colour IR pole.} With $Q^c\equiv\int_x[\delta^{ce}-U^{ce}(x)]\rho^e(x)\neq0$, the JIMWLK, BFKL and finite
terms contain the large-$|y|$ pole
\begin{equation}
-\frac{1}{(2\pi)^3}\frac{g^4N_c}{2\pi^2}\frac{1}{k^+k^2}\,\frac{2}{\epsilon}\,Q^cQ^c\Big[\log\frac{k^+}{\Lambda}+\log\frac{\vee}{k^+}-\frac{11}{6}\Big],
\end{equation}
which is the note's Eq.~(63). It can be pulled out explicitly as in Eq.~(65); it vanishes iff $Q^c=0$.

\textbf{After counterterms} (note, Sec.~7.5). The $\overline{\rm MS}$ fragmentation counterterm cancels the DGLAP
pole and the coupling counterterm cancels the RC pole. The $\frac{11}{12}$ share of the $\log(y'-y)^2$
in $\mathcal G(1)$ then sets the scale of one power of $\alpha_s$ inside the LO integral,
$\alpha_s\big(4e^{-2\gamma_E}/|y-y'|^2\big)$ (note, Eq.~(55)).
