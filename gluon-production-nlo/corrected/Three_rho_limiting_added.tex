\section{Adding all the three-$\rho$ terms ($\rho$'s do not commute)}
\subsection{Conventions}
\begin{itemize}
\item The $\rho$'s are operators and do not commute; their order is kept everywhere. The Wilson lines $U$ commute with the $\rho$'s and with each other. Only two identities are used: $\{A,B\}=2AB-[A,B]=2BA+[A,B]$, and $U^{cb}(z)U^{gb}(z)=\delta^{cg}$.
\item For operators, ``c.c.'' means the hermitian conjugate (it reverses the order of the $\rho$'s). For Group II the conjugate term is $N_{1b}=N_{1a}^{\dagger}$, so every three-$\rho$ term has the form ``term $+$ c.c.''. The c.c.\ is not written in the steps below.
\item \textbf{Colours.} Within one step, terms of the same colour cancel; they are struck through when they cancel exactly. \Gr{Grey} terms are commutators $[\rho,\rho]$, which are left over. Colours are reused from step to step.
\item Every colour identity below was checked with exact SU(3) adjoint Wilson lines and with \emph{non-commuting} random matrices for $\rho$ (residual $10^{-11}$ or better).
\end{itemize}

\subsection{Cancellation I: the six terms with the kernel $\log\frac{(x-z)^{2}}{(x-w)^{2}}$}
\subsubsection*{The common factor}
Six contributions have exactly the form
\begin{align}
&\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\log\frac{(x-z)^{2}}{(x-w)^{2}}\notag\\&\quad\times\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\notag\\
&\quad\times\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\times\big[\text{colour factor}\big].\label{eq:common}
\end{align}
Below, only the colour factors are written. Everything in front of them is the same for all six.

\subsubsection*{Step 0: Group I Row II in the form (\ref{eq:common})}
The Row II integrand splits as
\begin{align}
\frac{R^{i}}{p^{2}q^{2}}&=\log\frac{q^{2}}{(p-q)^{2}}\Big[\frac{p^{i}}{p^{2}(p-q)^{2}}-\frac{q^{i}}{q^{2}(p-q)^{2}}+\frac{q^{i}}{p^{2}q^{2}}\Big]\notag\\
&\quad+\Gr{\frac{2q^{i}}{q^{2}}\Big[\frac{1}{(p-q)^{2}}\log\frac{q^{2}}{(p-q)^{2}}-\frac{1}{p^{2}}\log\frac{p^{2}}{q^{2}}\Big]}.
\end{align}

\subsubsection*{Step 4: change of variable $r\to w$ (from your note)}
\textbf{First integral.} Put $w=(1-\xi)z+\xi r$, i.e.\ $r=z+\frac{w-z}{\xi}$ and $d^{2}r=\frac{d^{2}w}{\xi^{2}}$, and let $t=1-\xi$. Then $\frac{(z-r)^{i}}{(z-r)^{2}}=\xi\frac{(z-w)^{i}}{(z-w)^{2}}$ and $\frac{(x-r)^{j}}{(x-r)^{2}}=\xi\frac{[(x-w)-t(x-z)]^{j}}{[(x-w)-t(x-z)]^{2}}$. The two factors of $\xi$ cancel the Jacobian:
\begin{align}

&=\int_{w}e^{ik\cdot w}\int_{0}^{1}dt\,\frac{(x'-w')\cdot(z-w)\,(y-z)\cdot[(x-w)-t(x-z)]}{(z-w)^{2}[(x-w)-t(x-z)]^{2}}.
\end{align}
\textbf{Second integral.} Put $w=(1-\xi)r+\xi z$; the same formulas hold with $\xi\to1-\xi$. At $\xi=0$, $w=r$, so the plus prescription subtracts at fixed $w$:
\begin{align}

\end{align}

\subsection{Group III, Rows III, V, VI}
The three rows have identical kernels and phases. Work in units of $\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{4\pi^{4}}\frac{1}{k^{+}}$; Row III has $\frac{ig^{4}}{8\pi^{4}}$, hence the factor $\frac12$ in front of its anticommutators. In the first part the bracket containing $\rho(x')$ is $\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')$; in the $(x'-z)$ subtraction it is the same with $y'\to z$. The algebra below is identical for both, so the bracket is written as $\big[U^{c'e'}(\cdot)-U^{c'e'}(x')\big]\rho^{e'}(x')$; it is always leftmost. The colour factors are
\begin{align}

\end{align}

\noindent\textbf{Step 1: bring every product to a fixed order.} Use $\{\rho^{c}(y),\rho^{b}(x)\}=2\rho^{b}(x)\rho^{c}(y)+[\rho^{c}(y),\rho^{b}(x)]$, $\{\rho^{d}(y),\rho^{e}(x)\}=2\rho^{d}(y)\rho^{e}(x)-[\rho^{d}(y),\rho^{e}(x)]$ and $\rho^{d}(y)\rho^{b}(x)=\rho^{b}(x)\rho^{d}(y)+[\rho^{d}(y),\rho^{b}(x)]$. \end{align}
\textbf{Step 2.} The \R{red} pair cancels exactly. Rename $c\to d$ in the first green term and $e\to b$ in R V; the \G{green} and \B{blue} terms then combine:
\begin{align}

\begin{equation}

The $x'\to z$ subtraction is kept explicitly: for operators it does not vanish.
