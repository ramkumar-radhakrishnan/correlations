\section*{Three-$\rho$ terms, limiting case ($\Lambda\to0$, $\vee\to\infty$), $\rho$'s non-commuting}
Plus prescriptions (all acting on everything to their right, including the phase):
$\int_{0}^{1}d\xi\big[\frac1\xi\big]_{+}F(\xi)=\int_{0}^{1}\frac{d\xi}{\xi}\big[F(\xi)-F(0)\big]$,
$\int_{0}^{k^{+}}dp^{+}\big[\frac{k^{+}}{p^{+}}\big]_{+}f(p^{+})=\int_{0}^{k^{+}}dp^{+}\frac{k^{+}}{p^{+}}\big[f(p^{+})-f(0)\big]$,
$\int_{0}^{\infty}dp^{+}\big[\frac{k^{+}}{p^{+}(p^{+}+k^{+})}\big]_{+}f(p^{+})=\int_{0}^{\infty}dp^{+}\frac{k^{+}}{p^{+}(p^{+}+k^{+})}\big[f(p^{+})-f(0)\big]$,
and $\big[\frac{1}{\xi(1+\xi)}\big]_{+}$ likewise with $\xi=p^{+}/k^{+}$. With these, $\big[\frac{k^{+}}{p^{+}}\big]_{+}dp^{+}=k^{+}\big[\frac{1}{\xi}\big]_{+}d\xi$ and $\big[\frac{k^{+}}{p^{+}(p^{+}+k^{+})}\big]_{+}dp^{+}=\big[\frac{1}{\xi(1+\xi)}\big]_{+}d\xi$.
For operators, c.c.\ means the hermitian conjugate.

\subsection*{Copy 1: every longitudinal integral in $\xi=p^{+}/k^{+}$}
\begin{align}
\frac{d^{3}N^{3\rho}_{\rm NLO}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\log\frac{(x-z)^{2}}{(x-w)^{2}}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\notag\\
&\times\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\times\bigg\{\tfrac12U^{ab'}(x')U^{gb}(z)U^{ge}(y)\big[\rho^{b'}(x'),\rho^{e}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]\notag\\
&+\Big(\tfrac12U^{ab'}(w')-U^{ab'}(x')\Big)\big[\rho^{b'}(x'),\rho^{b}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]\notag\\
&-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{bc}(z)U^{de}(x)\big[\rho^{c}(y),\rho^{e}(x)\big]\notag\\
&+\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big(f^{hbc}U^{ah}(w)\big[\rho^{b}(x),\rho^{c}(y)\big]+f^{abc}U^{bd}(x)U^{ce}(y)\big[\rho^{d}(x),\rho^{e}(y)\big]\Big)\bigg\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}\frac{e^{-ik\cdot(w'-w)}}{(x'-w')^{2}(y-z)^{2}}\notag\\
&\times\Bigg\{\frac{\mathrm{arc\,cos}\left(\frac{(x-w)^{2}-(x-w)\cdot (x-z)}{\sqrt{(x-w)^{2}\,(z-w)^{2}}}\right)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\frac{1}{(z-w)^{2}}\notag\\
&\times\Big[{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-w)}{-\frac{(x-w)\cdot(x-z)}{(x-z)^{2}}(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}\notag\\
&{+(x'-w')\cdot(x-z)\,(y-z)\cdot(z-w)}{-\frac{(x-w)\cdot(x-z)}{(x-w)^{2}}(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}\Big]\notag\\
&+\frac{1}{2(z-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}\Big[{\frac{(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}{(x-w)^{2}}}{-\frac{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}{(x-z)^{2}}}\Big]\Bigg\}\notag\\
&\times\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{adc}U^{db}(x)\big[U^{ce}(z)-U^{ce}(y)\big]\big[\rho^{e}(y),\rho^{b}(x)\big]\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,x',w'}e^{-ik\cdot w'}\int_{r}\int_{0}^{1}d\xi\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(y-z)\,(x-r)\cdot(z-r)}{(x'-w')^{2}(y-z)^{2}(x-r)^{2}(z-r)^{2}}\notag\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big\{f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)-\tfrac12f^{adc}U^{db}(x)\big[U^{ce}(z)-U^{ce}(y)\big]\big[\rho^{e}(y),\rho^{b}(x)\big]\Big\}\notag\\[6pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}f^{abd}}{16\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\log\frac{(z-w)^{2}}{(x-w)^{2}}\frac{(x'-w')\cdot(y-w)\,(x-z)\cdot(w-z)}{(x'-w')^{2}(y-w)^{2}(x-z)^{2}(w-z)^{2}}\notag\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\big[U^{de}(x)U^{bc}(w)-U^{de}(x)U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)\notag\\[6pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}f^{abd}}{8\pi^{4}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\notag\\
&\times\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\big[U^{de}(x)U^{bc}(z)-U^{de}(x)U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{32\pi^{5}}\,\frac{1}{k^{+}}\int_{0}^{1}d\xi\int_{x,z,y,x',y',w'}e^{-ik\cdot(w'-z)}\,e^{-ik\cdot(z-x)\,\xi}\notag\\
&\times\bigg(\frac{(x'-w')\cdot(z-x)\,(y-z)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}-2\Big[\frac{1}{\xi}\Big]_{+}\frac{(x'-w')\cdot(y-z)\,(z-x)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}\bigg)\notag\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{adc}\big[U^{db}(x)-U^{db}(y')\big]\big[U^{ce}(z)-U^{ce}(y)\big]\{\rho^{e}(y),\rho^{b}(y')\}\notag\\[6pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{0}^{\infty}d\xi\int_{w,x',y',z,y,x}e^{-ik\cdot(y'-w)}\,e^{-ik\cdot(y'-z)\,\xi}\frac{(y'-z)^{m}\,(y-w)^{i}\,(x-z)^{j}}{(y'-z)^{2}\,(y-w)^{2}\,(x-z)^{2}}\notag\\
&\times\Bigg\{\bigg[\frac{\delta_{k'm}\delta_{ij}}{1+\xi}-\Big(\Big[\frac{1}{\xi(1+\xi)}\Big]_{+}+\frac{1}{1+\xi}\Big)\delta_{jm}\delta_{ik'}-\delta_{im}\delta_{jk'}\bigg]\times\frac{(x'-y')^{k'}}{(x'-y')^{2}}\,f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\notag\\
&+\delta_{im}\delta_{jk'}\frac{(x'-z)^{k'}}{(x'-z)^{2}}\,f^{c'd'a}\big[U^{c'e'}(z)-U^{c'e'}(x')\big]\rho^{e'}(x')\Bigg\}\times\bigg\{\big[U^{d'b}(z)-U^{d'b}(x)\big]\rho^{b}(x)\,\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)\notag\\
&+\Big[\tfrac12U^{d'b}(z)U^{ad}(w)-U^{d'b}(z)U^{ad}(y)+\tfrac12U^{d'b}(x)U^{ad}(y)\Big]\big[\rho^{d}(y),\rho^{b}(x)\big]\bigg\}+{\rm c.c.}
\end{align}
