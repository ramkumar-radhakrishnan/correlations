\begin{center}{\large\bf Cancellation of the three-$\rho$ terms in the limiting case ($\Lambda\to0$, $\vee\to\infty$), written out in full}\end{center}

\noindent\textbf{Colours.} Within each step, terms of the same colour cancel each other; they are struck through when they cancel exactly. \Gr{Grey} terms are what is left over: commutators $[\rho,\rho]$, which are two-$\rho$ terms. Colours are reused from step to step and only mean something within one step. Every term comes with its complex conjugate. For Group II, the conjugate is $N_{1b}=N_{1a}^{\dagger}$.

\section{Cancellation I: the six terms with the common kernel of eq.\ (1.1)}

\subsection*{The common factor}
Each of the six contributions has the form
\begin{align}
\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\log\frac{(x-z)^{2}}{(x-w)^{2}}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\notag\\
\times\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\times\big[\text{colour factor}\big]+{\rm c.c.}
\end{align}
Everything in front of $[\text{colour factor}]$ is identical for all six. In Steps 1--4 below only the colour factors change, and they are written out in full.

\subsection*{Step 0: bring Group I Row II to the same form}
The Row II integrand splits as
\begin{equation}
\frac{R^{i}}{p^{2}q^{2}}=\underbrace{\log\frac{q^{2}}{(p-q)^{2}}\Big[\frac{p^{i}}{p^{2}(p-q)^{2}}-\frac{q^{i}}{q^{2}(p-q)^{2}}+\frac{q^{i}}{p^{2}q^{2}}\Big]}_{\text{antisymmetric part}}
+\Gr{\underbrace{\frac{2q^{i}}{q^{2}}\Big[\frac{1}{(p-q)^{2}}\log\frac{q^{2}}{(p-q)^{2}}-\frac{1}{p^{2}}\log\frac{p^{2}}{q^{2}}\Big]}_{\text{symmetric under }p^{2}\leftrightarrow(p-q)^{2}}},
\end{equation}
with $R^{i}=-2q^{i}\log\frac{p^{2}}{q^{2}}+\frac{(2p^{2}-2q\cdot p+q^{2})q^{i}+q^{2}p^{i}}{(q-p)^{2}}\log\frac{q^{2}}{(q-p)^{2}}$. I checked this split numerically to machine precision.

The exchange $x\leftrightarrow y$ is $p\to q-p$, i.e.\ $p^{2}\leftrightarrow(p-q)^{2}$. The colour factor of Row II, $f^{hbc}U^{ah}(w)\{\rho^{b}(x),\rho^{c}(y)\}-f^{abc}U^{bd}(x)U^{ce}(y)\{\rho^{d}(x),\rho^{e}(y)\}$, is antisymmetric under $x\leftrightarrow y$, so the \Gr{grey} part drops out. The remaining part is the Fourier transform of the $z$ kernel:
\begin{equation}
\int d^{2}p\,d^{2}q\,e^{iq\cdot(w-y)}e^{ip\cdot(y-x)}\log\frac{q^{2}}{(p-q)^{2}}\Big[\frac{p^{i}}{p^{2}(p-q)^{2}}-\frac{q^{i}}{q^{2}(p-q)^{2}}+\frac{q^{i}}{p^{2}q^{2}}\Big]
=-4\pi i\int_{z}\log\frac{(x-z)^{2}}{(x-w)^{2}}\,\frac{(y-z)^{j}}{(y-z)^{2}}\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg].
\end{equation}
So Row II, $-\frac{1}{(2\pi)^{3}}\frac{g^{4}}{64\pi^{6}}\frac{1}{k^{+}}\times(-4\pi i)=+\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{16\pi^{5}}\frac{1}{k^{+}}$, is of the form (1.1) with colour factor $\frac12\times$ its own colour factor.

\subsection*{The six colour factors in the form (1.1)}
\begin{align}
\text{I.II}:\ &+\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big[f^{hbc}U^{ah}(w)\{\rho^{b}(x),\rho^{c}(y)\}-f^{abc}U^{bd}(x)U^{ce}(y)\{\rho^{d}(x),\rho^{e}(y)\}\Big],\\[3pt]
\text{I.III}:\ &+\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y),\\[3pt]
\text{I.IV}:\ &+\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big[f^{cdc'}U^{ac}(w)U^{bd}(z)U^{be}(y)\rho^{e}(y)\rho^{c'}(x)-f^{abc}U^{be}(y)U^{cd}(x)\rho^{e}(y)\rho^{d}(x)\Big]\notag\\
&=+\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,U^{gb}(z)U^{ge}(y)\rho^{e}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big],\\[3pt]
\text{II.I}:\ &-\tfrac12\Big[U^{ab'}(w')\{\rho^{b'}(x'),\rho^{b}(y)\}-U^{ab'}(x')U^{gb}(z)U^{ge}(y)\{\rho^{b'}(x'),\rho^{e}(y)\}\Big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big],\\[3pt]
\text{III.I+III.II}:\ &-\Big[U^{ab'}(x')U^{gb}(z)U^{ge}(y)\{\rho^{b'}(x'),\rho^{e}(y)\}-U^{ab'}(w')U^{gb}(z)U^{ge}(y)\rho^{b'}(x')\rho^{e}(y)-U^{ab'}(x')\rho^{b}(y)\rho^{b'}(x')\Big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big].
\end{align}
The second form of I.IV follows from $U^{gb}(z)U^{cb}(z)=\delta^{gc}$ in its second term, and from relabelling $d\to b$, $b\to g$, $c'\to d$ in its first term. In III.I $+$ III.II the two terms of the form $U^{ab'}(x')U^{gb}(z)U^{ge}(y)\rho\rho$ were added into one anticommutator.

\subsection*{Step 1: the terms with $U^{gb}(z)U^{ge}(y)$}
\begin{align}
\text{I.IV}:\ &\R{+U^{ab'}(x')U^{gb}(z)U^{ge}(y)\rho^{b'}(x')\rho^{e}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\;\B{\cancel{-U^{ab'}(w')U^{gb}(z)U^{ge}(y)\rho^{b'}(x')\rho^{e}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}}\notag\\
\text{II.I}:\ &\R{+\tfrac12U^{ab'}(x')U^{gb}(z)U^{ge}(y)\{\rho^{b'}(x'),\rho^{e}(y)\}\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\notag\\
\text{III}:\ &\R{-U^{ab'}(x')U^{gb}(z)U^{ge}(y)\{\rho^{b'}(x'),\rho^{e}(y)\}\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\;\B{\cancel{+U^{ab'}(w')U^{gb}(z)U^{ge}(y)\rho^{b'}(x')\rho^{e}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}}
\end{align}
\B{Blue} cancels exactly. For \R{red}, $\rho^{b'}(x')\rho^{e}(y)+\tfrac12\{\rho^{b'}(x'),\rho^{e}(y)\}-\{\rho^{b'}(x'),\rho^{e}(y)\}=\Gr{\tfrac12[\rho^{b'}(x'),\rho^{e}(y)]}$, so
\begin{equation}
\R{\text{red}}=\Gr{\tfrac12U^{ab'}(x')U^{gb}(z)U^{ge}(y)\big[\rho^{b'}(x'),\rho^{e}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}.
\end{equation}

\subsection*{Step 2: the terms without $U(z)U(y)$ from II.I and III}
\begin{align}
&\G{-\tfrac12U^{ab'}(w')\{\rho^{b'}(x'),\rho^{b}(y)\}\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\;\Or{+U^{ab'}(x')\rho^{b}(y)\rho^{b'}(x')\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\notag\\
&=\G{-U^{ab'}(w')\rho^{b'}(x')\rho^{b}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\;\Or{+U^{ab'}(x')\rho^{b'}(x')\rho^{b}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\notag\\
&\quad+\Gr{\Big(\tfrac12U^{ab'}(w')-U^{ab'}(x')\Big)\big[\rho^{b'}(x'),\rho^{b}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\notag\\
&=\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,\rho^{b}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]+\Gr{\Big(\tfrac12U^{ab'}(w')-U^{ab'}(x')\Big)\big[\rho^{b'}(x'),\rho^{b}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}.
\end{align}

\subsection*{Step 3: add Group I Row III}
Expand the result of Step 2 and the colour factor of I.III:
\begin{align}
\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,\rho^{b}(y)\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]
&=\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{cbd}U^{ac}(w)\rho^{b}(y)\rho^{d}(x)\;\Ma{-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{acd}U^{cb}(z)U^{dm}(x)\rho^{b}(y)\rho^{m}(x)},\notag\\
\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)&=\Ma{\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)U^{bc}(z)\rho^{e}(x)\rho^{c}(y)}-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)U^{bc}(y)\rho^{e}(x)\rho^{c}(y).
\end{align}
In the first \Ma{magenta} term, relabel $c\to b$, $b\to c$, $m\to e$, and move $\rho(y)$ to the right:
\begin{align}
&\Ma{-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{bc}(z)U^{de}(x)\rho^{c}(y)\rho^{e}(x)}\notag\\
&=\Ma{\cancel{-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)U^{bc}(z)\rho^{e}(x)\rho^{c}(y)}}\;\Gr{-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{bc}(z)U^{de}(x)\big[\rho^{c}(y),\rho^{e}(x)\big]},
\end{align}
which cancels the \Ma{magenta} term of I.III. \textbf{All $U(z)$ dependence has gone}, apart from the grey commutator. What is left is
\begin{equation}
\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big\{f^{cbd}U^{ac}(w)\rho^{b}(y)\rho^{d}(x)-f^{abd}U^{de}(x)U^{bc}(y)\rho^{e}(x)\rho^{c}(y)\Big\}.
\end{equation}
Relabel $(c,b,d)\to(h,c,b)$ in the first term and $(b,d,c,e)\to(c,b,e,d)$ in the second:
\begin{equation}
\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big\{\Vi{-f^{hbc}U^{ah}(w)\rho^{c}(y)\rho^{b}(x)}\;\Br{+f^{abc}U^{bd}(x)U^{ce}(y)\rho^{d}(x)\rho^{e}(y)}\Big\}.
\end{equation}

\subsection*{Step 4: add Group I Row II}
\begin{align}
&\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big\{\Vi{-f^{hbc}U^{ah}(w)\rho^{c}(y)\rho^{b}(x)}\;\Br{+f^{abc}U^{bd}(x)U^{ce}(y)\rho^{d}(x)\rho^{e}(y)}\notag\\
&\qquad\Vi{+\tfrac12f^{hbc}U^{ah}(w)\big(\rho^{b}(x)\rho^{c}(y)+\rho^{c}(y)\rho^{b}(x)\big)}\;\Br{-\tfrac12f^{abc}U^{bd}(x)U^{ce}(y)\big(\rho^{d}(x)\rho^{e}(y)+\rho^{e}(y)\rho^{d}(x)\big)}\Big\}\notag\\
&=\Gr{\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big\{f^{hbc}U^{ah}(w)\big[\rho^{b}(x),\rho^{c}(y)\big]+f^{abc}U^{bd}(x)U^{ce}(y)\big[\rho^{d}(x),\rho^{e}(y)\big]\Big\}}.
\end{align}

\subsection*{Result of Cancellation I}
\begin{align}
&\text{I.II}+\text{I.III}+\text{I.IV}+\text{II.I}+\text{III.I}+\text{III.II}\notag\\
&=\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\,\log\frac{(x-z)^{2}}{(x-w)^{2}}\,\frac{(x'-w')^{i}(y-z)^{j}}{(x'-w')^{2}(y-z)^{2}}\,\bigg[\frac{\delta^{ij}}{2(z-w)^{2}}+\frac{(z-w)^{i}(x-z)^{j}}{(z-w)^{2}(x-z)^{2}}-\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}(x-w)^{2}}-\frac{(x-w)^{i}(x-z)^{j}}{(x-w)^{2}(x-z)^{2}}\bigg]\notag\\
&\times\Gr{\bigg\{\tfrac12U^{ab'}(x')U^{gb}(z)U^{ge}(y)\big[\rho^{b'}(x'),\rho^{e}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]+\Big(\tfrac12U^{ab'}(w')-U^{ab'}(x')\Big)\big[\rho^{b'}(x'),\rho^{b}(y)\big]\big[f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{dm}(x)\rho^{m}(x)\big]}\notag\\
&\qquad\Gr{-\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{bc}(z)U^{de}(x)\big[\rho^{c}(y),\rho^{e}(x)\big]}\notag\\
&\qquad\Gr{+\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big(f^{hbc}U^{ah}(w)\big[\rho^{b}(x),\rho^{c}(y)\big]+f^{abc}U^{bd}(x)U^{ce}(y)\big[\rho^{d}(x),\rho^{e}(y)\big]\Big)\bigg\}}+{\rm c.c.}
\end{align}
\textbf{Every three-$\rho$ term of these six contributions has cancelled.} Only commutators remain; for commuting $\rho$ the right-hand side is zero. This was checked with SU(3) Wilson lines to $10^{-15}$.

\section{Cancellation II: the arccos and $\log\frac{(z-w)^{2}}{(x-w)^{2}}$ terms of Group I Row III against Group I Row V, first part}

\subsection*{The Row III terms}
With the prefactor $\frac{1}{(2\pi)^{3}}\frac{ig^{4}f^{abd}}{16\pi^{5}}\frac{1}{k^{+}}\times2$ written as $\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{8\pi^{5}}\frac{1}{k^{+}}$, the arccos term and the $\log\frac{(z-w)^{2}}{(x-w)^{2}}$ bracket of Row III are
\begin{align}
&\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,w,x',w'}e^{-ik\cdot(w'-w)}\frac{1}{(x'-w')^{2}(y-z)^{2}}\Bigg\{\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\frac{1}{(z-w)^{2}}\Big[\R{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-w)}\notag\\
&\qquad\G{-\frac{(x-w)\cdot(x-z)}{(x-z)^{2}}(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}\;\B{+(x'-w')\cdot(x-z)\,(y-z)\cdot(z-w)}\;\Or{-\frac{(x-w)\cdot(x-z)}{(x-w)^{2}}(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}\Big]\notag\\
&\qquad+\frac{1}{2(z-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}\Big[\Br{\frac{(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}{(x-w)^{2}}}\;\Vi{-\frac{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}{(x-z)^{2}}}\Big]\Bigg\}\notag\\
&\qquad\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)+{\rm c.c.}
\end{align}
The arccos bracket was obtained by writing out $(y-z)\cdot X_{\perp}$ and $(x'-w')\cdot Z_{\perp}$. Each of the six terms has its own colour.

\subsection*{The Row V terms}
With $\xi=p^{+}/k^{+}$ (resp.\ $q^{+}/k^{+}$), $\big[\frac{k^{+}}{p^{+}}\big]_{+}\frac{dp^{+}}{k^{+}}=\big[\frac1\xi\big]_{+}d\xi$, and $\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{16\pi^{4}}\frac{1}{2\pi k^{+}}=\frac14\cdot\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{8\pi^{5}}\frac{1}{k^{+}}$, Row V (first part) is
\begin{align}
&\frac14\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,z,y,x',y',w'}\int_{0}^{1}d\xi\,e^{-ik\cdot w'}\notag\\
&\times\bigg\{e^{ik\cdot[(1-\xi)z+\xi x]}\bigg(\frac{(x'-w')\cdot(z-x)\,(y-z)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}\R{-\Big[\frac1\xi\Big]_{+}\frac{(x'-w')\cdot(y-z)\,(z-x)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}}\bigg)\notag\\
&\qquad\R{-\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)x+\xi z]}\frac{(x'-w')\cdot(y'-x)\,(z-x)\cdot(y-z)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}}\bigg\}\notag\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big[f^{adc}U^{db}(x)U^{ce}(z)\{\rho^{e}(y),\rho^{b}(y')\}-f^{abc}U^{cd}(y)U^{be}(y')\{\rho^{d}(y),\rho^{e}(y')\}\Big]+{\rm c.c.}
\end{align}

\subsection*{Step 1: the two \R{red} terms are equal}
Exchange the integration variables $(x,y')\leftrightarrow(z,y)$ in the second red term:
\begin{itemize}
\item its phase $e^{ik\cdot[(1-\xi)x+\xi z]}$ becomes $e^{ik\cdot[(1-\xi)z+\xi x]}$;
\item its kernel $(x'-w')\cdot(y'-x)\,(z-x)\cdot(y-z)$ becomes $(x'-w')\cdot(y-z)\,(x-z)\cdot(y'-x)=-(x'-w')\cdot(y-z)\,(z-x)\cdot(y'-x)$;
\item the colour bracket changes sign (relabel $d\leftrightarrow c$, $b\leftrightarrow e$ in its first term and $b\leftrightarrow c$, $d\leftrightarrow e$ in its second).
\end{itemize}
So the second red term equals the first red term, and Row V (first part) becomes
\begin{align}
&\frac14\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,z,y,x',y',w'}\int_{0}^{1}d\xi\,e^{-ik\cdot w'}e^{ik\cdot[(1-\xi)z+\xi x]}\bigg(\frac{(x'-w')\cdot(z-x)\,(y-z)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}\notag\\
&\qquad-2\Big[\frac1\xi\Big]_{+}\frac{(x'-w')\cdot(y-z)\,(z-x)\cdot(y'-x)}{(x'-w')^{2}(z-x)^{2}(y-z)^{2}(y'-x)^{2}}\bigg)\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\Big[f^{adc}U^{db}(x)U^{ce}(z)\{\rho^{e}(y),\rho^{b}(y')\}-f^{abc}U^{cd}(y)U^{be}(y')\{\rho^{d}(y),\rho^{e}(y')\}\Big]+{\rm c.c.}
\end{align}

\subsection*{Step 2: split the colour bracket}
Insert $U^{db}(x)=[U^{db}(x)-U^{db}(y')]+U^{db}(y')$ and $U^{ce}(z)=[U^{ce}(z)-U^{ce}(y)]+U^{ce}(y)$ into the first term:
\begin{align}
&f^{adc}U^{db}(x)U^{ce}(z)\{\rho^{e}(y),\rho^{b}(y')\}\B{\cancel{-f^{abc}U^{cd}(y)U^{be}(y')\{\rho^{d}(y),\rho^{e}(y')\}}}\notag\\
&=f^{adc}\Big\{\underbrace{U^{db}(y')[U^{ce}(z)-U^{ce}(y)]}_{\rm(i)}+\underbrace{[U^{db}(x)-U^{db}(y')]U^{ce}(y)}_{\rm(ii)}+\underbrace{[U^{db}(x)-U^{db}(y')][U^{ce}(z)-U^{ce}(y)]}_{\rm(iii)}+\B{\cancel{U^{db}(y')U^{ce}(y)}}\Big\}\{\rho^{e}(y),\rho^{b}(y')\}.
\end{align}
The two \B{blue} terms are equal up to relabelling $b\to d$, $d\to e$, $e\to b$, so they cancel. In (i) only the gluon from $y$ scatters, in (ii) only the gluon from $y'$, and in (iii) both.

\subsection*{Step 3: pieces (i) and (ii) have the Row III colour factor}
\textbf{Piece (i).} Rename $y'\to x$, and call the old $x$ (which now appears only in the kernel and the phase) $r$. Using $\{\rho^{e}(y),\rho^{b}(x)\}=2\rho^{b}(x)\rho^{e}(y)+[\rho^{e}(y),\rho^{b}(x)]$ and relabelling, the colour factor becomes
\begin{equation}
\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')f^{adc}U^{db}(x)[U^{ce}(z)-U^{ce}(y)]\{\rho^{e}(y),\rho^{b}(x)\}
=-2\,\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)+\Gr{\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{adc}U^{db}(x)\big[U^{ce}(z)-U^{ce}(y)\big]\big[\rho^{e}(y),\rho^{b}(x)\big]}.
\end{equation}
The phase becomes $e^{ik\cdot[(1-\xi)z+\xi r]}$, and the kernels become
\begin{equation}
\frac{(x'-w')\cdot(z-r)\,(y-z)\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}},\qquad
\frac{(x'-w')\cdot(y-z)\,(z-r)\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}}.
\end{equation}
\textbf{Piece (ii).} Rename $y\to x$, $x\to z$, $y'\to y$, and call the old $z$ $r$. Relabelling $d\leftrightarrow c$, $b\leftrightarrow e$, the colour factor becomes
\begin{equation}
+2\,\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)-\Gr{\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{adc}U^{db}(x)\big[U^{ce}(z)-U^{ce}(y)\big]\big[\rho^{e}(y),\rho^{b}(x)\big]}.
\end{equation}
The phase becomes $e^{ik\cdot[(1-\xi)r+\xi z]}$, and the two kernels become
\begin{equation}
-\frac{(x'-w')\cdot(z-r)\,(y-z)\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}},\qquad
-\frac{(y-z)\cdot(z-r)\,(x'-w')\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}}.
\end{equation}

\noindent\textbf{Sum of (i) and (ii).} Substitute $\xi\to1-\xi$ in the first kernel of (ii). Then (i) $+$ (ii) of Row V is
\begin{align}
&\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\int_{x,y,z,x',w'}e^{-ik\cdot w'}\Big\{\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{abd}U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)-\Gr{\tfrac12\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,f^{adc}U^{db}(x)\big[U^{ce}(z)-U^{ce}(y)\big]\big[\rho^{e}(y),\rho^{b}(x)\big]}\Big\}\notag\\
&\times\int_{r}\int_{0}^{1}d\xi\,\bigg\{-e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(z-r)\,(y-z)\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}}
+\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)r+\xi z]}\frac{(y-z)\cdot(z-r)\,(x'-w')\cdot(x-r)}{(x'-w')^{2}(z-r)^{2}(y-z)^{2}(x-r)^{2}}\notag\\
&\qquad+\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(y-z)\,(x-r)\cdot(z-r)}{(x'-w')^{2}(y-z)^{2}(x-r)^{2}(z-r)^{2}}\bigg\}+{\rm c.c.}
\end{align}
The last line is the end-point term; it is not part of the cancellation.

\subsection*{Step 4: change of variable $r\to w$}
\textbf{First integral.} Put $w=(1-\xi)z+\xi r$, so that $r=z+\frac{w-z}{\xi}$ and $d^{2}r=\frac{d^{2}w}{\xi^{2}}$. With $t=1-\xi$:
\begin{equation}
z-r=\frac{z-w}{\xi},\qquad x-r=\frac{(x-w)-t\,(x-z)}{\xi},\qquad
\frac{(z-r)^{i}}{(z-r)^{2}}=\xi\frac{(z-w)^{i}}{(z-w)^{2}},\qquad\frac{(x-r)^{j}}{(x-r)^{2}}=\xi\frac{[(x-w)-t(x-z)]^{j}}{[(x-w)-t(x-z)]^{2}}.
\end{equation}
The two factors of $\xi$ cancel the Jacobian, and the phase becomes $e^{ik\cdot w}$:
\begin{equation}
\int_{r}\int_{0}^{1}d\xi\,e^{ik\cdot[(1-\xi)z+\xi r]}\frac{(x'-w')\cdot(z-r)\,(y-z)\cdot(x-r)}{(z-r)^{2}(x-r)^{2}}
=\int_{w}e^{ik\cdot w}\int_{0}^{1}dt\,\frac{(x'-w')\cdot(z-w)\;(y-z)\cdot[(x-w)-t(x-z)]}{(z-w)^{2}\,[(x-w)-t(x-z)]^{2}}.
\end{equation}
\textbf{Second integral.} Put $w=(1-\xi)r+\xi z$. The same formulas hold with $\xi\to1-\xi$. At $\xi=0$ one has $w=r$, so the plus prescription subtracts the $\xi=0$ value at fixed $w$:
\begin{equation}
\int_{r}\int_{0}^{1}d\xi\Big[\frac1\xi\Big]_{+}e^{ik\cdot[(1-\xi)r+\xi z]}\frac{(y-z)\cdot(z-r)\,(x'-w')\cdot(x-r)}{(z-r)^{2}(x-r)^{2}}
=\int_{w}e^{ik\cdot w}\frac{(y-z)\cdot(z-w)}{(z-w)^{2}}\int_{0}^{1}\frac{d\xi}{\xi}\bigg[\frac{(x'-w')\cdot[(x-w)-\xi(x-z)]}{[(x-w)-\xi(x-z)]^{2}}-\frac{(x'-w')\cdot(x-w)}{(x-w)^{2}}\bigg].
\end{equation}

\subsection*{Step 5: the one-parameter integrals}
With $[(x-w)-t(x-z)]^{2}=(x-w)^{2}-2t\,(x-w)\cdot(x-z)+t^{2}(x-z)^{2}$:
\begin{align}
\int_{0}^{1}\frac{dt}{[(x-w)-t(x-z)]^{2}}&=\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}},\notag\\
\int_{0}^{1}\frac{t\,dt}{[(x-w)-t(x-z)]^{2}}&=\frac{1}{2(x-z)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}+\frac{(x-w)\cdot(x-z)}{(x-z)^{2}}\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}},\notag\\
\int_{0}^{1}\frac{d\xi}{\xi}\bigg[\frac{[(x-w)-\xi(x-z)]^{i}}{[(x-w)-\xi(x-z)]^{2}}-\frac{(x-w)^{i}}{(x-w)^{2}}\bigg]&=\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\Big[-(x-z)^{i}+\frac{(x-w)\cdot(x-z)}{(x-w)^{2}}(x-w)^{i}\Big]-\frac{(x-w)^{i}}{2(x-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}.
\end{align}
All three were checked numerically. Inserting them, the first integral (with its minus sign) and the second integral give, as the coefficient of $e^{ik\cdot w}/[(x'-w')^{2}(y-z)^{2}]$:
\begin{align}
-\text{first}:&\quad-\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\frac{1}{(z-w)^{2}}\Big[\R{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-w)}\G{-\frac{(x-w)\cdot(x-z)}{(x-z)^{2}}(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}\Big]\notag\\
&\quad\Vi{+\frac{1}{2(z-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}\frac{(x'-w')\cdot(z-w)\,(y-z)\cdot(x-z)}{(x-z)^{2}}},\notag\\
\text{second}:&\quad-\frac{\arccos\Big(\frac{(x-w)\cdot(z-w)}{|x-w|\,|z-w|}\Big)}{\sqrt{(x-w)^{2}(x-z)^{2}-[(x-w)\cdot(x-z)]^{2}}}\frac{1}{(z-w)^{2}}\Big[\B{(x'-w')\cdot(x-z)\,(y-z)\cdot(z-w)}\Or{-\frac{(x-w)\cdot(x-z)}{(x-w)^{2}}(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}\Big]\notag\\
&\quad\Br{-\frac{1}{2(z-w)^{2}}\log\frac{(z-w)^{2}}{(x-w)^{2}}\frac{(x'-w')\cdot(x-w)\,(y-z)\cdot(z-w)}{(x-w)^{2}}}.
\end{align}
These are exactly the six Row III terms of Section 2, each with the opposite sign and the same colour.

\section{Group III, Rows III (first part), V, VI}
The three rows have identical kernels and phases. With $\{\rho,\rho\}\to2\rho\rho$ in Row III, all three have the prefactor $\frac{1}{(2\pi)^{3}}\frac{ig^{4}}{4\pi^{4}}\frac{1}{k^{+}}$. The colour factors are
\begin{align}
\text{III}:\ &f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,U^{d'b}(z)\Big[\G{U^{ac}(w)\rho^{c}(y)\rho^{b}(x)}\;\R{\cancel{-U^{cb}(z)U^{ad}(y)U^{ce}(x)\rho^{d}(y)\rho^{e}(x)}}\Big],\notag\\
\text{VI}:\ &f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,U^{d'b}(z)\Big[\R{\cancel{U^{cb}(z)U^{ad}(y)U^{ce}(x)\rho^{d}(y)\rho^{e}(x)}}\;\G{-U^{ad}(y)\rho^{d}(y)\rho^{b}(x)}\Big],\notag\\
\text{V}:\ &f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\Big[\B{U^{d'e}(x)U^{ad}(y)}\;\B{-U^{ad}(w)U^{d'e}(x)}\Big]\rho^{e}(x)\rho^{d}(y).
\end{align}
The \R{red} pair cancels exactly. The \G{green} and \B{blue} terms combine:
\begin{align}
&\G{f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,U^{d'b}(z)\rho^{b}(x)\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)}\notag\\
&\B{-f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,U^{d'b}(x)\rho^{b}(x)\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)}\notag\\
&=f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,\big[U^{d'b}(z)-U^{d'b}(x)\big]\rho^{b}(x)\,\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y).
\end{align}
This was checked with SU(3) Wilson lines to $10^{-15}$.

In the subtraction pieces, $U^{c'e'}(y')\to U^{c'e'}(z)$ and the kernel is $\frac{(x'-z)\cdot(x-z)}{(x'-z)^{2}(x-z)^{2}}$, which is symmetric under $x\leftrightarrow x'$. The colour factor $f^{c'd'a}[U^{c'e'}(z)-U^{c'e'}(x')]\rho^{e'}(x')[U^{d'b}(z)-U^{d'b}(x)]\rho^{b}(x)$ is antisymmetric under $x\leftrightarrow x'$ together with $c'\leftrightarrow d'$. So these pieces drop after $\int_{x,x'}$. The sum of the three rows is
\begin{align}
&\frac{1}{(2\pi)^{3}}\,\frac{ig^{4}}{4\pi^{4}}\,\frac{1}{k^{+}}\int_{0}^{\infty}\frac{dp^{+}}{2\pi}\int_{w,x',y',z,y,x}e^{-ik\cdot(y'-w)}\,e^{-ik\cdot\frac{p^{+}}{k^{+}}(y'-z)}\,\frac{(y'-z)^{m}\,(x'-y')^{k'}\,(y-w)^{i}\,(x-z)^{j}}{(y'-z)^{2}\,(x'-y')^{2}\,(y-w)^{2}\,(x-z)^{2}}\notag\\
&\times\bigg[\frac{\delta_{k'm}\delta_{ij}}{p^{+}+k^{+}}-\Big(\Big[\frac{k^{+}}{p^{+}(p^{+}+k^{+})}\Big]_{+}+\frac{1}{p^{+}+k^{+}}\Big)\delta_{jm}\delta_{ik'}-\frac{1}{k^{+}}\delta_{im}\delta_{jk'}\bigg]\notag\\
&\times f^{c'd'a}\big[U^{c'e'}(y')-U^{c'e'}(x')\big]\rho^{e'}(x')\,\big[U^{d'b}(z)-U^{d'b}(x)\big]\rho^{b}(x)\,\big[U^{ad}(w)-U^{ad}(y)\big]\rho^{d}(y)+{\rm c.c.}
\end{align}

\section{What is left of the three-$\rho$ terms}
\begin{enumerate}
