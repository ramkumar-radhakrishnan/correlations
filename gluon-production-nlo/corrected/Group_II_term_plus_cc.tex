\section*{Group II two-$\rho$ terms written as term $+$ c.c.}
\noindent\textbf{Why this is allowed.} Group II has no c.c.\ of its own, but it is hermitian, $T^{\dagger}=T$. Therefore $T=\tfrac12 T+\tfrac12 T^{\dagger}=\big(\tfrac12 T\big)+{\rm c.c.}$: every coefficient is halved and ``$+$c.c.'' is added. Proof of $T^{\dagger}=T$:
\begin{itemize}
\item \textbf{Row I.} Rename the dummy indices $c^{\prime}\to c$, $d^{\prime}\to d$, $e^{\prime}\to e$ in the first colour bracket. It is then the second bracket evaluated at $(x^{\prime},w^{\prime})$:
\begin{equation*}
B^{ab}(x,w)\equiv f^{cbd}U^{ac}(w)\rho^{d}(x)-f^{acd}U^{cb}(z)U^{de}(x)\rho^{e}(x),\qquad\text{colour factor}=B^{ab}(x^{\prime},w^{\prime})\,B^{ab}(x,w).
\end{equation*}
\begin{enumerate}
\item The coefficients and $U$ are real and $\rho$ is hermitian, so the hermitian conjugate is $e^{+ik\cdot(w^{\prime}-w)}\,K\,B^{ab}(x,w)B^{ab}(x^{\prime},w^{\prime})$, with the kernel $K$ unchanged.
\item Relabel the integration variables $(x,w)\leftrightarrow(x^{\prime},w^{\prime})$ (with $z$ fixed). The phase returns to $e^{-ik\cdot(w^{\prime}-w)}$ and the colour factor returns to $B^{ab}(x^{\prime},w^{\prime})B^{ab}(x,w)$.
\item The kernel is symmetric under this relabelling. This holds for each of $N_{2a},\dots,N_{2f}$ separately (checked numerically at random points to machine precision; for example, the first and third $d_{\perp}$ terms map into each other).
\end{enumerate}
Hence $T^{\dagger}=T$.
\item \textbf{Row IV.} The same argument applies with $(x,y)\leftrightarrow(x^{\prime},y^{\prime})$. The phase $e^{-ik\cdot(y^{\prime}-y)(1+p^{+}/k^{+})}$ returns to itself, the kernel is symmetric, and $[U(y^{\prime})-U(x^{\prime})]^{ce^{\prime}}[U(y)-U(x)]^{ce}\rho^{e}(x)\rho^{e^{\prime}}(x^{\prime})$ goes back to the original order.
\end{itemize}
\noindent The only changes from the original are:
\begin{itemize}
\item $\frac{g^{4}}{8\pi^{5}}\to\frac{g^{4}}{16\pi^{5}}$;
\item $\frac{g^{4}N_{c}}{4\pi^{3}}\to\frac{g^{4}N_{c}}{8\pi^{3}}$;
\item ``$+$c.c.'' is added.
\end{itemize}
