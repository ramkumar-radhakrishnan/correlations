\section*{Three-$\rho$ terms: adding the c.c.\ and showing the result is $2\times{\rm Re}$}
\noindent Starting point: the trace form of the three-$\rho$ result (\texttt{ThreeRho\_final\_traces.tex}), limiting case, $\rho$'s non-commuting,
\begin{equation}
\frac{d^{3}N^{3\rho}_{\rm NLO}}{d^{3}k}=\frac{d^{3}N^{(3)}}{d^{3}k}+\frac{d^{3}N^{(2)}}{d^{3}k}+{\rm c.c.},
\end{equation}
where c.c.\ is the hermitian conjugate of \emph{everything} before it. Conventions are those of the trace file: $U$ is the real adjoint Wilson line ($U^{\dagger}=U^{T}$), $(T^{e})_{ab}=-if^{eab}$, $(D^{e})_{ab}=d^{eab}$, $[\rho^{a}(u),\rho^{b}(v)]=i\kappa f^{abc}\rho^{c}(u)\,\delta^{(2)}(u-v)$, and $\langle\cdots\rangle$ is the projectile average with the $\rho$'s in the written operator order. The target average over $U$ is taken last and is an average of real numbers.

\section{What has to be shown}
Every term has the form
\begin{equation}
\mathcal{T}=c\int e^{-i\varphi}\,K\;\mathcal{C}[U]\;\big\langle P[\rho]\big\rangle ,\label{eq:T}
\end{equation}
with $c$ a constant ($c=\pm i\times$real or real), $\varphi$ a real phase linear in $k$, $K$ a kernel (rational functions of the transverse coordinates, logarithms, $\arccos$, and $\xi$-distributions), $\mathcal{C}[U]$ a colour bracket of adjoint traces, and $P[\rho]$ an ordered product of two or three $\rho$'s. We show
\begin{equation}
\mathcal{T}+\mathcal{T}^{\dagger}=2\,{\rm Re}\,\mathcal{T}
\end{equation}
term by term, and then write $2\,{\rm Re}\,\mathcal{T}$ out explicitly.

\section{Four lemmas}
\noindent\textbf{Lemma 0 (what $\dagger$ does).} The projectile operators $\rho$ and the target Wilson lines $U$ act on different systems, so they commute, and $U$, $K$, $\varphi$ are c-numbers. Hence the hermitian conjugate of (\ref{eq:T}) is
\begin{align}
&\text{(R1)}\quad c\to c^{*},\qquad \text{(R2)}\quad e^{-i\varphi}\to e^{+i\varphi},\qquad \text{(R3)}\quad K\to K^{*},\ \ \mathcal{C}[U]\to\mathcal{C}[U]^{*},\notag\\
&\text{(R4)}\quad \big(\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big)^{\dagger}=\rho^{c}(t)\rho^{b}(v)\rho^{a}(u),\qquad\big(\rho^{a}(u)\rho^{b}(v)\big)^{\dagger}=\rho^{b}(v)\rho^{a}(u)\qquad(\rho^{a\dagger}=\rho^{a}).
\end{align}
For any operator $O$ and any state (or density matrix) of the projectile, $\langle O^{\dagger}\rangle=\langle O\rangle^{*}$. Therefore
\begin{equation}
\mathcal{T}^{\dagger}=c^{*}\int e^{+i\varphi}K^{*}\,\mathcal{C}[U]^{*}\,\big\langle P[\rho]^{\dagger}\big\rangle=\Big(c\int e^{-i\varphi}K\,\mathcal{C}[U]\,\big\langle P[\rho]\big\rangle\Big)^{*}=\mathcal{T}^{*},\qquad\boxed{\ \mathcal{T}+\mathcal{T}^{\dagger}=2\,{\rm Re}\,\mathcal{T}\ }\label{eq:2Re}
\end{equation}
This holds for non-commuting $\rho$'s, with no approximation. The remaining lemmas are needed only to write ${\rm Re}\,\mathcal{T}$ in explicit form.

\medskip\noindent\textbf{Lemma 1 (kernels and colour brackets are real).}
\begin{itemize}
\item Every kernel is real: products of $(u-v)^{i}/(u-v)^{2}$, $\delta^{ij}$, $\log$ of ratios of squared distances, $\arccos$ of a number in $[-1,1]$ divided by a positive square root, and the real distributions $[1/\xi]_{+}$, $[1/(\xi(1+\xi))]_{+}$, $1/(1+\xi)$ on real $\xi$ intervals. So $K^{*}=K$.
\item $U(x)$ is real, $T^{e}$ is purely imaginary ($T^{e*}=-T^{e}$), $D^{e}$ is real. For real matrices $A,B$:
\begin{align}
\Big({\rm Tr}\{T^{e}A\,T^{e}B\}\Big)^{*}&={\rm Tr}\{(-T^{e})A(-T^{e})B\}={\rm Tr}\{T^{e}A\,T^{e}B\},\\
\Big(i\,{\rm Tr}\{U(p)T^{e}U^{\dagger}(p)\,A\,D^{e}B\}\Big)^{*}&=(-i)\,{\rm Tr}\{U(p)(-T^{e})U^{\dagger}(p)\,A\,D^{e}B\}=i\,{\rm Tr}\{U(p)T^{e}U^{\dagger}(p)\,A\,D^{e}B\},\\
\Big({\rm Tr}\{U^{\dagger}(p)U(q)\}\Big)^{*}&={\rm Tr}\{U^{\dagger}(p)U(q)\}.
\end{align}
Every bracket in (A) and (B) is a sum of such traces with real coefficients ($\pm1,\pm\tfrac12,\pm N_{c},\pm\tfrac{N_{c}}{2}$). So $\mathcal{C}[U]^{*}=\mathcal{C}[U]$: each bracket is a real number for every target configuration, and so is its target average.
\end{itemize}

\medskip\noindent\textbf{Lemma 2 (reordering the three $\rho$'s).} The operator identity (checked by expanding the commutators)
\begin{equation}
\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)=\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)+\rho^{a}(u)\big[\rho^{c}(t),\rho^{b}(v)\big]+\big[\rho^{c}(t),\rho^{a}(u)\big]\rho^{b}(v)+\rho^{c}(t)\big[\rho^{b}(v),\rho^{a}(u)\big]
\end{equation}
is contracted with $f^{abc}$ and with $d^{abc}$. With $[\rho^{c}(t),\rho^{b}(v)]=i\kappa f^{cbg}\rho^{g}(t)\delta^{(2)}(t-v)$ etc.\ and
\begin{equation}
f^{abc}f^{cbg}=-N_{c}\delta^{ag},\qquad f^{abc}f^{cag}=N_{c}\delta^{bg},\qquad f^{abc}f^{bag}=-N_{c}\delta^{cg},\qquad d^{abc}f^{cbg}=d^{abc}f^{cag}=d^{abc}f^{bag}=0,
\end{equation}
the three commutator terms give
\begin{align}
f^{abc}\rho^{a}(u)\big[\rho^{c}(t),\rho^{b}(v)\big]&=-i\kappa N_{c}\,\delta^{(2)}(v-t)\,\rho^{e}(u)\rho^{e}(v),\\
f^{abc}\big[\rho^{c}(t),\rho^{a}(u)\big]\rho^{b}(v)&=+i\kappa N_{c}\,\delta^{(2)}(t-u)\,\rho^{e}(u)\rho^{e}(v),\\
f^{abc}\rho^{c}(t)\big[\rho^{b}(v),\rho^{a}(u)\big]&=-i\kappa N_{c}\,\delta^{(2)}(u-v)\,\rho^{e}(t)\rho^{e}(u),
\end{align}
and hence
\begin{align}
f^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)&=f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)-i\kappa N_{c}\Big[\delta^{(2)}(v-t)\rho^{e}(u)\rho^{e}(v)-\delta^{(2)}(t-u)\rho^{e}(u)\rho^{e}(v)+\delta^{(2)}(u-v)\rho^{e}(t)\rho^{e}(u)\Big],\label{eq:frev}\\
d^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)&=d^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t).\label{eq:drev}
\end{align}

\medskip\noindent\textbf{Lemma 3 (real and imaginary parts of the projectile correlators).}
\begin{itemize}
\item Two $\rho$'s: $\langle\rho^{e}(u)\rho^{e}(v)\rangle^{*}=\langle\rho^{e}(v)\rho^{e}(u)\rangle=\langle\rho^{e}(u)\rho^{e}(v)\rangle+i\kappa f^{eeg}\langle\rho^{g}(v)\rangle\delta^{(2)}(u-v)=\langle\rho^{e}(u)\rho^{e}(v)\rangle$. So $\langle\rho^{e}(u)\rho^{e}(v)\rangle$ is \textbf{real and symmetric} in $u\leftrightarrow v$.
\item $d$-correlator: by (\ref{eq:drev}), $\langle d^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\rangle^{*}=\langle d^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)\rangle=\langle d^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\rangle$: \textbf{real}.
\item $f$-correlator: by (\ref{eq:frev}) and the previous item,
\begin{align}
\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle^{*}&=\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle-i\kappa N_{c}\Big[\delta^{(2)}(v-t)\langle\rho^{e}(u)\rho^{e}(v)\rangle-\delta^{(2)}(t-u)\langle\rho^{e}(u)\rho^{e}(v)\rangle+\delta^{(2)}(u-v)\langle\rho^{e}(u)\rho^{e}(t)\rangle\Big],\notag\\
{\rm Im}\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle&=\frac{\kappa N_{c}}{2}\Big[\delta^{(2)}(v-t)\langle\rho^{e}(u)\rho^{e}(v)\rangle-\delta^{(2)}(t-u)\langle\rho^{e}(u)\rho^{e}(v)\rangle+\delta^{(2)}(u-v)\langle\rho^{e}(u)\rho^{e}(t)\rangle\Big],\label{eq:Imf}\\
{\rm Re}\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle&=\tfrac12\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)+f^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)\big\rangle .\notag
\end{align}
The ordered $f$-correlator is real at pairwise distinct points, but as a distribution it has an imaginary part supported at coincident points. That part contributes to the integrals, since none of the kernels vanishes at $x=y$, $x'=x$, $x'=y$ or $x'=y'$, and it has to be kept.
\item The difference that appears in $\mathcal{A}_{4}$ follows from (\ref{eq:Imf}) (the $\delta^{(2)}(v-t)$ pieces cancel by the symmetry of $\langle\rho\rho\rangle$):
\begin{equation}
{\rm Im}\Big[\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle-\big\langle f^{abc}\rho^{a}(u)\rho^{b}(t)\rho^{c}(v)\big\rangle\Big]=\kappa N_{c}\Big[\delta^{(2)}(u-v)\langle\rho^{e}(u)\rho^{e}(t)\rangle-\delta^{(2)}(u-t)\langle\rho^{e}(u)\rho^{e}(v)\rangle\Big].\label{eq:Imf4}
\end{equation}
\end{itemize}

\section{The hermitian-conjugate terms, written out}
Write the result of the trace file as $\frac{d^{3}N^{3\rho}_{\rm NLO}}{d^{3}k}=\sum_{n=1}^{5}\big(\mathcal{A}_{n}+\mathcal{A}_{n}^{\dagger}\big)+\sum_{n=1}^{4}\big(\mathcal{B}_{n}+\mathcal{B}_{n}^{\dagger}\big)$, where $\mathcal{A}_{n}$ are the five genuine three-$\rho$ terms (A) and $\mathcal{B}_{n}$ the four two-$\rho$ commutator terms (B), each exactly as in \texttt{ThreeRho\_final\_traces.tex}. For every term the conjugate is obtained with rules (R1)--(R4): $i\to-i$ in the coefficient, $e^{-i\varphi}\to e^{+i\varphi}$, kernels and trace brackets unchanged (they are real, Lemma 1), and the order of the three $\rho$'s reversed (Lemma 2).

\section{Adding each term to its conjugate}
Compare each $\mathcal{A}_{n}^{\dagger}$ with $\mathcal{A}_{n}$. The coefficient, the phase and the reversed correlator are exactly the complex conjugates of those in $\mathcal{A}_{n}$:
\begin{equation}
\big\langle f^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)\big\rangle=\big\langle f^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle^{*},\qquad\big\langle d^{abc}\rho^{c}(t)\rho^{b}(v)\rho^{a}(u)\big\rangle=\big\langle d^{abc}\rho^{a}(u)\rho^{b}(v)\rho^{c}(t)\big\rangle^{*}.
\end{equation}
The kernels and brackets are real. So $\mathcal{A}_{n}^{\dagger}=\mathcal{A}_{n}^{*}$, and in the same way $\mathcal{B}_{n}^{\dagger}=\mathcal{B}_{n}^{*}$. Therefore
\begin{equation}
\frac{d^{3}N^{3\rho}_{\rm NLO}}{d^{3}k}=\sum_{n=1}^{5}\big(\mathcal{A}_{n}+\mathcal{A}_{n}^{*}\big)+\sum_{n=1}^{4}\big(\mathcal{B}_{n}+\mathcal{B}_{n}^{*}\big)=2\,{\rm Re}\Big[\sum_{n=1}^{5}\mathcal{A}_{n}+\sum_{n=1}^{4}\mathcal{B}_{n}\Big].
\end{equation}
To write the real part explicitly, decompose the colour-projected correlator of a term as $X=R+iI$, with $R$ and $I$ real; $a$ and $b$ are the real constants of the coefficients. Then
\begin{align}
i\,a\,e^{-i\varphi}X+\big(i\,a\,e^{-i\varphi}X\big)^{\dagger}&=i\,a\,e^{-i\varphi}(R+iI)-i\,a\,e^{+i\varphi}(R-iI)=2a\big[\sin\varphi\;R-\cos\varphi\;I\big],\label{eq:rule-i}\\
b\,e^{-i\varphi}X+\big(b\,e^{-i\varphi}X\big)^{\dagger}&=b\,e^{-i\varphi}(R+iI)+b\,e^{+i\varphi}(R-iI)=2b\big[\cos\varphi\;R+\sin\varphi\;I\big].\label{eq:rule-r}
\end{align}
\begin{itemize}
\item $\mathcal{A}_{1},\mathcal{A}_{2},\mathcal{A}_{4},\mathcal{A}_{5}$ have $c=\pm i\times$real. The genuine three-$\rho$ part ($R$) goes with $\sin\varphi$. The imaginary part of the ordered $f$-correlator ($I$, eqs.~(\ref{eq:Imf}), (\ref{eq:Imf4})) goes with $-\cos\varphi$ and is a two-$\rho$ term. The $d$-correlator has $I=0$.
\item $\mathcal{A}_{3}$ (the $i\pi$ term) has a real coefficient: $R$ goes with $\cos\varphi$, and $I$ with $+\sin\varphi$.
\item $\mathcal{B}_{1},\dots,\mathcal{B}_{4}$ have real coefficients ($i\cdot i\kappa=-\kappa$ was already used) and $I=0$ (Lemma 3). They go with $\cos\varphi$.
\end{itemize}

\section{Final result, manifestly real}
Applying (\ref{eq:rule-i}) to $\mathcal{A}_{1,2,4,5}$, (\ref{eq:rule-r}) to $\mathcal{A}_{3}$ and $\mathcal{B}_{1,\dots,4}$, and Lemma 3 for the real and imaginary parts of the correlators, every exponential is replaced by a sine or a cosine and every $i$ disappears. Nothing is dropped: the $\delta^{(2)}$ terms are the imaginary parts of the ordered $f$-correlators (Lemma 3).

\begin{itemize}
\item \textbf{Exact statement.} $d^{3}N^{3\rho}/d^{3}k=2\,{\rm Re}[\,\cdots]$ follows from $\langle O^{\dagger}\rangle=\langle O\rangle^{*}$ alone. The manifestly real form above uses in addition Lemma 1, which was checked with exact SU(3) adjoint Wilson lines (the traces come out real to machine precision), and Lemma 3.
\item \textbf{Lemma 3 checked numerically.} $\rho^{a}(s)$ was realised on four sites carrying $\mathbf{3},\mathbf{3},\mathbf{3},\mathbf{8}$ (so $\kappa=1$ and $\delta^{(2)}\to\delta_{ss'}$), in a mixed state of two random colour singlets. \begin{itemize}
\item ${\rm Im}\langle\rho\rho\rangle=0$ and $\langle\rho(u)\rho(v)\rangle=\langle\rho(v)\rho(u)\rangle$;
\item ${\rm Im}\langle d\rho\rho\rho\rangle=0$;

\item (\ref{eq:Imf4}) holds to $4\times10^{-16}$.
\end{itemize}
\item \textbf{Equivalent bookkeeping.} If (A) is written with the hermitian (symmetrised) correlator ${\rm Re}\langle f\rho\rho\rho\rangle=\tfrac12\langle f\rho\rho\rho+{\rm h.c.}\rangle$ from the start, the imaginary parts are removed from (A). They then appear as the extra $\delta^{(2)}$ two-$\rho$ terms above, which add to (B). The total is the same.
\item \textbf{Parity in $k$.} The genuine $\langle f\rho\rho\rho\rangle$ and $\langle d\rho\rho\rho\rangle$ terms of $\mathcal{A}_{1,2,4,5}$ multiply $\sin\varphi$, which is odd under $k\to-k$: they are the odd (odderon-like) part. For a Gaussian (MV) projectile they vanish, and only the $\cos\varphi$ two-$\rho$ terms remain.
\item \textbf{Open points, as before.}
\begin{itemize}
\item The sign of $\mathcal{A}_{3}$ depends on the $-i0$ prescription.
\item The end-point term diverges like $\log^{2}$ at large $r$.
\item The amended $N_{3b}$ term needs the upstream $B_{3}$ re-derivation.
\item Cancellation I relies on the Fourier-transform identity, which has only been checked numerically.
\end{itemize}
\end{itemize}
