Energy denominators that vanish inside the integration region are principal values (author's answer of 2026-11-20).
\item \textbf{Plus prescriptions} act on the whole function to their right, including the phase and the energy denominators: with $t=p^{+}/k^{+}$,
\begin{align}\label{eq:plus}
\int_{0}^{1}dt\Big[\frac{1}{t}\Big]_{+}h(t)=\int_{0}^{1}\frac{dt}{t}\big[h(t)-h(0)\big]\,,\notag\\
\int_{0}^{1}dt\Big[\frac{1}{1-t}\Big]_{+}h(t)=\int_{0}^{1}\frac{dt}{1-t}\big[h(t)-h(1)\big]\,,\notag\\
\int_{0}^{\infty}dt\Big[\frac{1}{t(1+t)}\Big]_{+}h(t)=\int_{0}^{\infty}\frac{dt}{t(1+t)}\big[h(t)-h(0)\big].
\end{align}
\item \textbf{Functions of one real variable} (defined once here; ${\rm Ci}$, ${\rm Si}$, $E_{1}$ are the cosine, sine and exponential integrals):
\begin{align}
\mathcal{E}(s)&\equiv\int_{0}^{1}\frac{dt}{t}\big(e^{-ist}-1\big)={\rm Ci}(|s|)-\gamma_{E}-\log|s|-i\,{\rm Si}(s)\,,\label{eq:Ecal}\\
\mathcal{J}(s)&\equiv\int_{0}^{1}dt\,e^{-ist}=\frac{1-e^{-is}}{is}\,,\label{eq:Jcal}\\
\widehat{\mathcal{E}}(s)&\equiv\int_{0}^{\infty}\frac{dt}{1+t}\,e^{-ist}=e^{is}E_{1}(is)\,,\label{eq:Ehat}\\
\Psi(s)&\equiv\int_{0}^{\infty}\frac{dt}{t(1+t)}\big(e^{-ist}-1\big)=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\widehat{\mathcal{E}}(s)\,,\label{eq:Psi}\\
\int_{0}^{\infty}dt\,e^{-ist}&=\pi\delta(s)-i\,\mathcal{P}\frac{1}{s}\qquad(\mathcal{P}:\ \text{principal value; complex conjugate }\pi\delta(s)+i\mathcal{P}\tfrac1s)\,.\label{eq:distr}
\end{align}
For $|s|\to\infty$: $\mathcal{E}(s)=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\frac{e^{-is}}{is}+O(s^{-2})$, $\Psi(s)=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\frac{1}{is}+O(s^{-2})$, $\widehat{\mathcal{E}}(s)\simeq\frac{1}{is}$; for $s\to0$: $\mathcal{E}(s)\simeq-is$, $\Psi(s)=O(s\log|s|)$. All satisfy $f(-s)=f(s)^{*}$. $\mathcal{E}$ is the recoil function of Group I (soft integral up to $k^{+}$), $\Psi$ that of Group III (soft integral to infinity); at large $|s|$ both tend to the same function, including the odd part $-\frac{i\pi}{2}{\rm sgn}(s)$.
\item \textbf{Integrable} means: integrable singularities at all coincidence points, and decay faster than $1/z^{2}$ at large position $z$ of a soft gluon (in practice $\log|z|/|z|^{3}$); a term $\propto e^{-ik\cdot z}/z^{2}$ (the measured momentum flowing through $z$) converges as a Fourier integral, as $\int_{w}e^{-ik\cdot w}\mathcal{K}(x-w)$ does at LO, and is not counted as a tail.
\end{itemize}
