\section{Conventions of this file}\label{sec:conv}
\begin{itemize}
\item \textbf{Phase.} ${\color{red}e^{-ik\cdot(\text{unprimed}-\text{primed})}}$ in all terms: $e^{-ik\cdot(w-w')}$ (Groups I, II), $e^{-ik\cdot(w-w')}$ with the recoil phase $e^{-ik\cdot(z-w)p^{+}/k^{+}}$ (Group I Row III), $e^{-ik\cdot(z-w')}$ with $e^{-ik\cdot(x-z)p^{+}/k^{+}}$ (Group I Row V), $e^{-ik\cdot(w-y')}$ with $e^{-ik\cdot(z-y')p^{+}/k^{+}}$ (Group III Rows III, V, VI), $e^{-ik\cdot(y-y')}$ (LO, RC, Group II Row IV), $e^{-ik\cdot(y-y')/\xi}$ (DGLAP). Every phase that differs from the author's file is printed in red. Relative to the author's file this is $k\to-k$ in every term: the parts of the terms that are even in $k$ are unchanged, the parts that are odd in $k$ (the symmetrised three-$\rho$ parts, Secs.~\ref{sec:D}, \ref{sec:E}) change sign.
\item \textbf{Complex conjugate} implied in Groups I and III, not in Group II. The charges $\rho$ are operators in the order written. $\int_{x}\equiv\int d^{2}x$. $\mathcal{K}(x)\equiv\frac{x}{x^{2}}$ (vector), $K(x,y,z)=\frac{(x-z)\cdot(y-z)}{(x-z)^{2}(y-z)^{2}}=\mathcal{K}(x-z)\cdot\mathcal{K}(y-z)$, adjoint Wilson lines $U^{ab}$ (real, $U^{T}U=1$), $(T^{e})_{bc}=-if^{ebc}$, $(D^{e})_{bc}=d^{ebc}$.
\item \textbf{Limits.} $\Lambda\to0$, $\vee\to\infty$ after the logarithmic (BFKL, JIMWLK) terms are taken out, $\vee\to\infty$ before $\epsilon\to0$ (author's answer of 2026-11-20); terms $O(\Lambda/k^{+})$, $O(k^{+}/\vee)$ dropped. Energy denominators that vanish inside the integration region (only those of $\mathbb{B}_{3}$, which enter only Group I Row III and its complex conjugate) are given in \emph{both} prescriptions (author's instruction of 2026-11-20, ``Do it in two ways with $i\epsilon$ and PV''): principal value (standing-wave dressing of the final state; all equations unless stated otherwise) and $i\epsilon$ (out-state dressing, $\eta=+1$; $\eta=-1$ would be an in-state dressing of the final state, PV is the average), which adds the term $\Delta N$ of Eq.~\eqref{eq:B-DN}. \textbf{Pure glue}: $N_{f}=0$ throughout ($\beta_{0}=\frac{11N_{c}}{3}$).
\item \textbf{Longitudinal integrals and plus prescriptions} (in place of the closed-form functions of the earlier version of this file). All $p^{+}$ integrals are written as in the author's file, $\int\frac{dp^{+}}{2\pi}$ with the author's plus prescriptions; inside a bracket that also contains terms without a $p^{+}$ integral (Group I Row III, Secs.~\ref{sec:C4}, \ref{sec:D}, \ref{sec:E}, and the identity~\eqref{eq:C-pairform}) the measure is written $dp^{+}=2\pi\frac{dp^{+}}{2\pi}$ and the prefactor is that of the row. Upper limits: $k^{+}$ in Group I (the cut-offs $\Lambda$ removed, $\Lambda\to0$, after the $\log\frac{k^{+}}{\Lambda}$ terms have been taken out into the JIMWLK list), $\infty$ in Group III (the author's $\vee-k^{+}$, $\vee\to\infty$, Sec.~\ref{sec:A}, item A4). The plus prescriptions are the author's, $\big[\frac{1}{p^{+}}\big]_{+}f(p^{+})=\frac{f(p^{+})-f(0)}{p^{+}}$ under the integral, and act on the whole function $f$ they multiply in the term, i.e.\ on the recoil phase $e^{-ik\cdot(\dots)p^{+}/k^{+}}$ (wherever it is written in the term) together with the energy denominators:
\begin{align}\label{eq:plus}
\int_{0}^{k^{+}}dp^{+}\Big[\frac{1}{p^{+}}\Big]_{+}f(p^{+})&=\int_{0}^{k^{+}}\frac{dp^{+}}{p^{+}}\big[f(p^{+})-f(0)\big]\,,\qquad\Big[\frac{k^{+}}{p^{+}}\Big]_{+}=k^{+}\Big[\frac{1}{p^{+}}\Big]_{+}\,,\notag\\
\int_{0}^{k^{+}}dp^{+}\Big[\frac{k^{+}}{k^{+}-p^{+}}\Big]_{+}f(p^{+})&=\int_{0}^{k^{+}}dp^{+}\,\frac{k^{+}}{k^{+}-p^{+}}\big[f(p^{+})-f(k^{+})\big]\,,\notag\\
\int_{0}^{\infty}dp^{+}\Big[\frac{k^{+}}{p^{+}\,(p^{+}+k^{+})}\Big]_{+}f(p^{+})&=\int_{0}^{\infty}dp^{+}\,\frac{k^{+}}{p^{+}\,(p^{+}+k^{+})}\big[f(p^{+})-f(0)\big]\,.
\end{align}
In the soft remainders the only $p^{+}$-dependent factor is the recoil phase (the energy denominators are taken at the subtraction point), e.g.\ $\int_{0}^{k^{+}}dp^{+}\big[\frac{1}{p^{+}}\big]_{+}e^{-ik\cdot(z-w)p^{+}/k^{+}}=\int_{0}^{k^{+}}\frac{dp^{+}}{p^{+}}\big[e^{-ik\cdot(z-w)p^{+}/k^{+}}-1\big]$.
\item \textbf{Instantaneous terms (Group III).} The $p^{+}$ integral of the phase alone, $\int_{0}^{\infty}dp^{+}\,e^{-ik\cdot(z-y')p^{+}/k^{+}}$, exists only as a distribution in $s=k\cdot(z-y')$, namely $k^{+}\big[\pi\delta(s)-i\,\mathcal{P}\frac{1}{s}\big]$ ($\mathcal{P}$: principal value; complex conjugate $k^{+}[\pi\delta(s)+i\mathcal{P}\frac{1}{s}]$), i.e.\ it is defined by integrating against the transverse kernels, with the two terms of the instantaneous bracket under one $y'$ integral done before the $p^{+}$ integral (Sec.~\ref{sec:A}, item A4).
\item \textbf{Large and small recoil arguments} (used for the tail statements of Sec.~\ref{sec:C}); with $s$ the coefficient in the phase $e^{-isp^{+}/k^{+}}$ and $|s|\to\infty$:
\begin{align}\label{eq:asym}
\int_{0}^{k^{+}}dp^{+}\Big[\frac{1}{p^{+}}\Big]_{+}e^{-is\frac{p^{+}}{k^{+}}}&=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\frac{e^{-is}}{is}+O(s^{-2})\,,\notag\\
\int_{0}^{\infty}dp^{+}\Big[\frac{k^{+}}{p^{+}\,(p^{+}+k^{+})}\Big]_{+}e^{-is\frac{p^{+}}{k^{+}}}&=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\frac{1}{is}+O(s^{-2})\,,\qquad\int_{0}^{\infty}dp^{+}\,\frac{e^{-is\frac{p^{+}}{k^{+}}}}{p^{+}+k^{+}}\simeq\frac{1}{is}\,;
\end{align}
for $s\to0$ the first is $\simeq-is$, the second $O(s\log|s|)$. Under $s\to-s$ each of these integrals goes into its complex conjugate. The Group I soft integral (upper limit $k^{+}$) and the Group III one (upper limit $\infty$) tend to the same function at large $|s|$, including the odd part $-\frac{i\pi}{2}{\rm sgn}(s)$. Equivalently, in closed form (used only in the numerical checks, Secs.~\ref{sec:A}, \ref{sec:F}; ${\rm Ci}$, ${\rm Si}$, $E_{1}$ the cosine, sine and exponential integrals): $\int_{0}^{k^{+}}dp^{+}\big[\frac{1}{p^{+}}\big]_{+}e^{-isp^{+}/k^{+}}={\rm Ci}(|s|)-\gamma_{E}-\log|s|-i\,{\rm Si}(s)$, $\frac{1}{k^{+}}\int_{0}^{k^{+}}dp^{+}e^{-isp^{+}/k^{+}}=\frac{1-e^{-is}}{is}$, $\int_{0}^{\infty}dp^{+}\frac{e^{-isp^{+}/k^{+}}}{p^{+}+k^{+}}=e^{is}E_{1}(is)$, $\int_{0}^{\infty}dp^{+}\big[\frac{k^{+}}{p^{+}(p^{+}+k^{+})}\big]_{+}e^{-isp^{+}/k^{+}}=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-e^{is}E_{1}(is)$.
\item \textbf{Integrable} means: integrable singularities at all coincidence points, and decay faster than $1/z^{2}$ at large position $z$ of a soft gluon (in practice $\log|z|/|z|^{3}$); a term $\propto e^{-ik\cdot z}/z^{2}$ (the measured momentum flowing through $z$) converges as a Fourier integral, as $\int_{w}e^{-ik\cdot w}\mathcal{K}(x-w)$ does at LO, and is not counted as a tail.
\end{itemize}
