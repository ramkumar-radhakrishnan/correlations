\section{Soft-recoil block $\mathcal{S}$ (Group I Rows III, V and Group III Rows III, V, VI; c.c.\ implied)}\label{sec:fx-soft}
\noindent[\textbf{New section; fix P1, P2} (fix\_soft Steps 2--4, 6, 7).] The soft remainders of the five rows cancel in pairs: each Group I term (soft gluon absorbed by the measured gluon in the amplitude: virtual, recoil on $w$) has one Group III partner (soft gluon emitted by the measured gluon in the conjugate amplitude: real, recoil on $w'$) with \emph{the same colour structure and the same kernel, except that $w$ is replaced by $w'$}:\\[2pt]
\hspace*{1em}(A) Group I Row III, last term of $N_{3b}$ (recoil form) $\;\leftrightarrow\;$ Group III Row V;\\
\hspace*{1em}(B) Group I Row V, first part $\;\leftrightarrow\;$ Group III Row III, first part;\\
\hspace*{1em}(C) Group I Row V, second part ($N_{2a}+N_{2b\,\rm rest}$) $\;\leftrightarrow\;$ Group III Row III, second part;\\
\hspace*{1em}(D) Group I Row III, third term of $N_{3b}$ (recoil part $\mathcal{E}$) $\;\leftrightarrow\;$ Group III Row VI.

\subsection*{Colour structures}
\begin{align}
C_{A}&\equiv f^{abd}\,\mathcal{L}^{a}\,U^{de}(x)\big[U^{bc}(w)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)\,,\qquad
C_{D}\equiv f^{abd}\,\mathcal{L}^{a}\,U^{de}(x)\big[U^{bc}(z)-U^{bc}(y)\big]\rho^{e}(x)\rho^{c}(y)\,,\notag\\
C_{B}&\equiv \mathcal{L}^{a}\Big[f^{adc}\,U^{db}(z)\,U^{ce}(w)\,\{\rho^{e}(y),\rho^{b}(x)\}-f^{abc}\,U^{cd}(y)\,U^{be}(x)\,\{\rho^{d}(y),\rho^{e}(x)\}\Big]\,,\label{eq:fx-col}\\
C_{C}&\equiv\big[U^{ab'}(x')-U^{ab'}(w')\big]\Big[N_{c}\,U^{ah}(x)-f^{adc}f^{hbe}\,U^{db}(z)\,U^{ce}(w)\Big]\rho^{b'}(x')\,\rho^{h}(x)\,.\notag
\end{align}
$C_{D}$ is the colour structure of Group I Row III, $C_{A}$ that of its last term (with $U^{bc}(w)$). The Group III structures are those of the rows with $y'\to w'$: $\mathcal{L}^{c'd'b}(w')$, $\mathcal{R}_{1}^{ab}$, $\mathcal{R}_{2}^{ab}$, $\mathcal{L}_{\rm V}^{c'}(w')=-\Delta^{c'e'}\rho^{e'}(x')$, $\mathcal{R}_{\rm V}^{ac}$, $\mathcal{R}_{\rm VI}^{ab}$ (Group III section).

\subsection*{The soft remainders (units $\mathcal{N}_{\rm f}$, under $\int_{w,w'}e^{-ik\cdot(w-w')}\int_{x,y,x',z}$; $\mathcal{E}\equiv\mathcal{E}(k\cdot(z-w))$, $\Psi\equiv\Psi(k\cdot(z-w'))$)}
\noindent Each piece is (coefficient)$\times$(recoil function)$\times$(kernel)$\times$(colour), first as it comes from the row, then (``$=$'') in the common variables (soft $z$, hard $w$ resp.\ $w'$; Group I Row V relabelled from its row variables, in pair B also $y\leftrightarrow y'$ and then $y'\to x$; Row VI written with $x\leftrightarrow y$):
\begin{align}
\text{A}^{\rm I}\ &\text{(I-III, last term of $N_{3b}$):}\quad -2i\,\mathcal{E}\,K(x,w,z)\,P\!\cdot\!K(y-w)\,C_{A}\,,\notag\\
\text{D}^{\rm I}\ &\text{(I-III, third term of $N_{3b}$, recoil part):}\quad +2i\,\mathcal{E}\,K(y,w,z)\,P\!\cdot\!K(x-w)\,C_{D}\,,\notag\\
\text{B}^{\rm I}\ &\text{(I-V first part, $s=k\cdot(x-z)$, phase $e^{-ik\cdot(z-w')}$):}\quad-\frac{i}{2}\Big[\mathcal{E}(s)\,P\!\cdot\!K(y-z)\,K(z-x)\!\cdot\!K(y'-x)\notag\\&\hspace{12em}+e^{-is}\mathcal{E}(-s)\,P\!\cdot\!K(y'-x)\,K(z-x)\!\cdot\!K(y-z)\Big]\,[\text{V}]\notag\\
&\hspace{22em}=\;-i\,\mathcal{E}\,K(x,w,z)\,P\!\cdot\!K(y-w)\,C_{B}\,,\notag\\
\text{C}^{\rm I}\ &\text{(I-V, $N_{2a}+N_{2b\,\rm rest}$):}\quad-\Big[\mathcal{E}(s)\,\frac{K^{i}K^{m}B_{2}^{im}}{(y-z)^{2}}+e^{-is}\mathcal{E}(-s)\,\frac{K^{i}K^{m}B_{1}^{im}}{(y-x)^{2}}\Big]\,[\dots]_{2a,2b}\notag\\&\hspace{22em}\;=\;-\mathcal{E}\,P\!\cdot\!K(x-w)\big[K(x,w,z)-2K(w,w,z)\big]\,C_{C}\,,\notag\\
\text{A}^{\rm III}\ &\text{(III-V):}\quad-2i\,\Psi\,K(w',x,z)\,P\!\cdot\!K(y-w)\,f^{c'ca}\mathcal{L}_{\rm V}^{c'}\mathcal{R}_{\rm V}^{ac}\;=\;+2i\,\Psi\,K(x,w',z)\,P\!\cdot\!K(y-w)\,C_{A}\,,\label{eq:fx-pieces}\\
\text{B}^{\rm III}\ &\text{(III-III first part):}\quad-i\,\Psi\,K(w',x,z)\,P\!\cdot\!K(y-w)\,f^{c'd'a}\mathcal{L}^{c'd'b}\mathcal{R}_{1}^{ab}\;=\;+i\,\Psi\,K(x,w',z)\,P\!\cdot\!K(y-w)\,C_{B}\,,\notag\\
\text{C}^{\rm III}\ &\text{(III-III second part):}\quad-2\,\Psi\,P\!\cdot\!K(x-w)\Big[\frac12K(x,w',z)-K(w,w',z)\Big]f^{c'd'a}\mathcal{L}^{c'd'b}\mathcal{R}_{2}^{ab}\notag\\&\hspace{22em}\;=\;+\Psi\,P\!\cdot\!K(x-w)\big[K(x,w',z)-2K(w,w',z)\big]\,C_{C}\,,\notag\\
\text{D}^{\rm III}\ &\text{(III-VI):}\quad-2i\,\Psi\,K(w',x,z)\,P\!\cdot\!K(y-w)\,f^{c'd'a}\mathcal{L}^{c'd'b}\mathcal{R}_{\rm VI}^{ab}\;\overset{x\leftrightarrow y}{=}\;-2i\,\Psi\,K(y,w',z)\,P\!\cdot\!K(x-w)\,C_{D}\,.\notag
\end{align}
Here $[\text{V}]=\mathcal{L}^{a}[f^{adc}U^{db}(x)U^{ce}(z)\{\rho^{e}(y),\rho^{b}(y')\}-f^{abc}U^{cd}(y)U^{be}(y')\{\rho^{d}(y),\rho^{e}(y')\}]$; $K^{i}K^{m}\equiv K^{i}(x'-w')K^{m}(z-x)$; $[\dots]_{2a,2b}$: the colour factors of $N_{2a}$ and $N_{2b\,\rm rest}$. The last column uses $K(z-w)\cdot K(y-z)=-K(w,y,z)$, $K(z-w)\cdot K(w'-z)=-K(w,w',z)$, $f^{adc}f^{hbe}U^{db}(w)U^{ce}(z)=f^{adc}f^{hbe}U^{db}(z)U^{ce}(w)$ (relabel $d\leftrightarrow c$, $b\leftrightarrow e$), $N_{c}[U(z)+U(w)]+N_{c}[2U(x)-U(z)-U(w)]=2N_{c}U(x)$, and the colour identities (with $U^{d'b}(z)U^{cb}(z)=\delta^{d'c}$, $f^{c'd'a}f^{dd'a}=N_{c}\delta^{c'd}$, relabelling of colour indices only, no reordering of $\rho$'s)
\begin{align}
f^{c'ca}\,\mathcal{L}_{\rm V}^{c'}\,\mathcal{R}_{\rm V}^{ac}=-C_{A}\,,\qquad f^{c'd'a}\,\mathcal{L}^{c'd'b}(w')\,\mathcal{R}_{1}^{ab}=-C_{B}\,,\qquad f^{c'd'a}\,\mathcal{L}^{c'd'b}(w')\,\mathcal{R}_{2}^{ab}=-C_{C}\,,\qquad f^{c'd'a}\,\mathcal{L}^{c'd'b}(w')\,\mathcal{R}_{\rm VI}^{ab}\big|_{x\leftrightarrow y}=+C_{D}\,.\label{eq:fx-colid}
\end{align}
[fix\_soft: every entry of the last column equals the row's own coefficient $f(0)$ at arbitrary $z$ (two- and three-$\rho$ sectors, $SU(2)$, $SU(3)$, differences $\le2\times10^{-15}$); Eq.~\eqref{eq:fx-colid} to $\le2\times10^{-15}$.]

\subsection*{Pair structure and cancellation of the tails}
Every pair has the form
\begin{align}
\text{pair }X:\qquad \alpha_{X}\,V_{X}\,C_{X}\Big[\kappa_{X}(w)\,\mathcal{E}\big(k\cdot(z-w)\big)-\kappa_{X}(w')\,\Psi\big(k\cdot(z-w')\big)\Big]\,,\label{eq:fx-pair}
\end{align}
\begin{center}
\renewcommand{\arraystretch}{1.3}
\begin{tabular}{|c|c|c|c|l|}\hline
pair & $\alpha_{X}$ & $V_{X}$ & $\kappa_{X}(u)$ & $T_{X}\equiv\lim_{|z|\to\infty} z^{2}\alpha_{X}V_{X}\kappa_{X}(w)C_{X}$\\\hline
A & $-2i$ & $P\cdot K(y-w)$ & $K(x,u,z)$ & $T_{A}=-2i\,P\!\cdot\!K(y-w)\,C_{A}$\\
B & $-i$ & $P\cdot K(y-w)$ & $K(x,u,z)$ & $T_{B}=-i\,P\!\cdot\!K(y-w)\,C_{B}$\\
C & $-1$ & $P\cdot K(x-w)$ & $K(x,u,z)-2K(w,u,z)$ & $T_{C}=+P\!\cdot\!K(x-w)\,C_{C}$\\
D & $+2i$ & $P\cdot K(x-w)$ & $K(y,u,z)$ & $T_{D}=+2i\,P\!\cdot\!K(x-w)\,C_{D}$\\\hline
\end{tabular}
\end{center}
At large $|z|$, by Eq.~\eqref{eq:fx-fun} (the two recoil functions have the same large-distance form, including the odd part $-\frac{i\pi}{2}{\rm sgn}$), for every value of $U(z)$:
\begin{align}
\alpha_{X}V_{X}C_{X}\Big[\kappa_{X}(w)\,\mathcal{E}-\kappa_{X}(w')\,\Psi\Big]\;\longrightarrow\;\frac{T_{X}}{z^{2}}\Big[{\color{green!50!black}\cancel{\mathcal{E}_{\infty}(k\cdot z)}}-{\color{green!50!black}\cancel{\mathcal{E}_{\infty}(k\cdot z)}}\Big]+{\cal O}\Big(\frac{\log|z|}{|z|^{3}}\Big)+(\text{terms}\propto e^{-ik\cdot z})\,.\notag
\end{align}
More precisely (fix\_soft Step 4),
\begin{align}
\kappa(w)\,\mathcal{E}\big(k\cdot(z-w)\big)-\kappa(w')\,\Psi\big(k\cdot(z-w')\big)=\underbrace{\big[\kappa(w)-\kappa(w')\big]\,\mathcal{E}\big(k\cdot(z-w)\big)}_{\text{real minus virtual}\,\times\,\text{kinematic log}}+\underbrace{\kappa(w')\Big[\mathcal{E}\big(k\cdot(z-w)\big)-\Psi\big(k\cdot(z-w')\big)\Big]}_{\text{mismatch of the recoil on $w$ and on $w'$}}\,:\label{eq:fx-rmv}
\end{align}
the first term is ${\cal O}(|w-w'|\log|z|/|z|^{3})$; in the second, $\mathcal{E}(k\cdot(z-w))-\Psi(k\cdot(z-w'))=\log\frac{|k\cdot(z-w')|}{|k\cdot(z-w)|}+{\cal O}(1/|k\cdot z|)$, the odd parts cancel, and the remainder is ${\cal O}(1/|z|)$ outside the strip $|k\cdot z|\lesssim|k||w-w'|$ and bounded by logarithms inside it (the strip contributes $\int dz_{\perp}/z_{\perp}^{2}$, convergent). The colour structure $C_{X}$, with all its dependence on $U(z)$, is a common factor. Hence: (i) the $U(z)$-linear tails cancel between the $w$ and the $w'$ recoil sides without any assumption on $U(z)$ at large distance (beyond $|U^{ab}|\le1$); (ii) for a finite target ($U\to1$) the same holds; (iii) the c.c.\ half cancels pair by pair with $\mathcal{E}^{*}$, $\Psi^{*}$. [\textbf{Correction} of \texttt{Finite\_terms\_final} (OPEN I-2(b), II/III-3): there the $w$-side terms of Group I were paired with the c.c.\ terms of Group III, which carry $\Psi^{*}$, and a $-i\pi\,{\rm sgn}(k\cdot z)\,T^{(1)}/z^{2}$ seemed to be left; with the pairing \eqref{eq:fx-pair} it is absent.] [fix\_soft numbers: $z^{2}|f^{\rm I}_{X}+f^{\rm III}_{X}|$ falls like $1/|z|$ from $|z|=10^{2}$ to $10^{4}$ for all four pairs, two- and three-$\rho$ sectors, $SU(2)$, $SU(3)$, also with $U(z)=1$ and with a different random $U(z)$ in every direction; single members $0.4$--$15$.]

\subsection*{The block}
\begin{align}
\frac{d^{3}N^{\rm soft}}{d^{3}k}&=\mathcal{N}_{\rm f}\int_{w,w'}e^{-ik\cdot(w-w')}\int_{x,y,x',z}\Bigg\{-i\,K(x'-w')\cdot K(y-w)\,\big[2C_{A}+C_{B}\big]\;\mathfrak{F}\big[K(x,\cdot,z)\big]\notag\\
&\qquad-K(x'-w')\cdot K(x-w)\;C_{C}\;\mathfrak{F}\big[K(x,\cdot,z)-2K(w,\cdot,z)\big]+2i\,K(x'-w')\cdot K(x-w)\;C_{D}\;\mathfrak{F}\big[K(y,\cdot,z)\big]\Bigg\}\,,\label{eq:fx-S}\\
\mathfrak{F}[\kappa]&\equiv\kappa(w)\,\mathcal{E}\big(k\cdot(z-w)\big)-\kappa(w')\,\Psi\big(k\cdot(z-w')\big)=\big[\kappa(w)-\kappa(w')\big]\,\mathcal{E}\big(k\cdot(z-w)\big)+\kappa(w')\Big[\mathcal{E}\big(k\cdot(z-w)\big)-\Psi\big(k\cdot(z-w')\big)\Big]\,,\notag
\end{align}
where $\kappa(\cdot)$ is evaluated at $u=w$ (Group I member) and $u=w'$ (Group III member); $C_{C}$ has $U(w)$ also in its second term ($K(w,u,z)$ is $1/(z-w)^{2}$ at $u=w$ and $K(w,w',z)$ at $u=w'$). Coincidences: $z\to w$: $\mathcal{E}={\cal O}(|z-w|)$ against at most $(z-w)^{-2}$; $z\to w'$: $\Psi\to0$, $\kappa(w')\sim|z-w'|^{-1}$; $z\to x,y$: $|\cdot|^{-1}$. The rows of Group I (Rows III, V) and Group III (Rows III, V, VI) as written above, together with Eq.~\eqref{eq:fx-S}, are exactly the finite terms of these five rows (fix\_soft Step 6: the regular terms are those of \texttt{Finite\_terms\_final}, checked there pointwise against the rows with $\Lambda$, $\vee$ kept; the soft remainders are their recoil terms, rewritten with Eq.~\eqref{eq:fx-colid}).
