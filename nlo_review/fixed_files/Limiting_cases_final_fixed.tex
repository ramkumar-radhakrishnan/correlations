\subsection*{Notation and conventions of the fixed finite terms}
\noindent[Added; fix\_soft Sec.~``Notation'' and Step 1; fix\_poles\_dglap ``Notation''.] Phase $e^{-ik\cdot(w-w')}$ (the conjugated bracket, with the primed coordinates, stands to the left; the amplitude is at the unprimed point); c.c.\ implied for Groups I and III; the $\rho$'s are operators in the order written; energy denominators are principal values, so logarithms whose argument can change sign are $\log|\cdot|$; $\Lambda\to0$, $\vee\to\infty$; terms with $\log\big(1-\frac{k^{+}}{\vee}\big)$ dropped; no $1/\epsilon$ written. Abbreviations:
\begin{align}
&\mathcal{N}_{\rm f}\equiv\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{16\pi^{5}}\,\frac{1}{k^{+}}\,,\qquad K^{i}(v)\equiv\mathcal{K}^{i}(v)=\frac{v^{i}}{v^{2}}\,,\qquad K(a,b,z)\equiv K(a-z)\cdot K(b-z)\quad\Big(K(w,w,z)=\frac{1}{(z-w)^{2}}\Big)\,,\notag\\&P\equiv K(x'-w')\,,\qquad \mathcal{L}^{a}\equiv\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,,\qquad \Delta^{c'e'}\equiv U^{c'e'}(w')-U^{c'e'}(x')\,.\notag
\end{align}
($\mathcal{N}_{\rm f}$ is called $c_{0}$ in fix\_soft; it is renamed here because $c_{0}\equiv2e^{-\gamma_{E}}$ is used in the DGLAP and RC logarithms, as in fix\_poles\_dglap.) Recoil functions, with $t=p^{+}/k^{+}$ and ${\rm Ci}$, ${\rm Si}$, $E_{1}$ the cosine, sine and exponential integrals:
\begin{align}
\mathcal{E}(s)&\equiv\int_{0}^{1}\frac{dt}{t}\big(e^{-ist}-1\big)=\mathrm{Ci}(|s|)-\gamma_{E}-\log|s|-i\,\mathrm{Si}(s)\,,\qquad
\Psi(s)\equiv\int_{0}^{\infty}\frac{dt}{t(1+t)}\big(e^{-ist}-1\big)=-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)-\widehat{\mathcal{E}}(s)\,,\notag\\
\widehat{\mathcal{E}}(s)&\equiv\int_{0}^{\infty}\frac{dt}{1+t}\,e^{-ist}=e^{is}E_{1}(is)\,,\qquad \mathcal{J}(s)\equiv\int_{0}^{1}dt\,e^{-ist}=\frac{1-e^{-is}}{is}\,,\qquad
\mathcal{E}_{\infty}(s)\equiv-\gamma_{E}-\log|s|-\frac{i\pi}{2}{\rm sgn}(s)\,,\tag{N.1}\label{eq:fx-fun}\\
\mathcal{E}(s)&=\mathcal{E}_{\infty}(s)-\frac{e^{-is}}{is}+{\cal O}(s^{-2})\,,\quad \Psi(s)=\mathcal{E}_{\infty}(s)-\frac{1}{is}+{\cal O}(s^{-2})\quad(|s|\to\infty)\,;\qquad \mathcal{E}(s)\simeq-is\,,\quad \Psi(s)={\cal O}(s\log|s|)\quad(s\to0)\,.\notag
\end{align}
All these functions satisfy $f(-s)=f(s)^{*}$. \textbf{``Integrable''} means: integrable singularities at all coincidence points and, at large soft position $z$, decay faster than $1/z^{2}$ (in practice $\log|z|/|z|^{3}$); a term $\propto e^{-ik\cdot z}/z^{2}$ (the measured momentum flowing through $z$) converges as a Fourier integral, as $\int_{w}e^{-ik\cdot w}K(x-w)$ at LO does, and is not counted as a tail.

\noindent\textbf{Origin of the soft remainders} (fix\_soft Step 1). In the recoil-carrying rows the measured gluon is a hard gluon at $w$ (Group I, amplitude) or $w'$ (Group III, conjugate amplitude) together with a soft gluon at $z$ with $p^{+}=tk^{+}$; the phase is $e^{-itk\cdot(z-w)}$, resp.\ $e^{-itk\cdot(z-w')}$. The plus-prescription integral of a row term is split exactly as
\begin{align}
\int_{\Lambda}^{k^{+}}\frac{dp^{+}}{p^{+}}\,e^{-i\frac{p^{+}}{k^{+}}s}f(p^{+})=\underbrace{\log\frac{k^{+}}{\Lambda}\,f(0)}_{\text{JIMWLK list}}+\underbrace{f(0)\,\mathcal{E}(s)}_{\text{soft remainder}\ \to\ \mathcal{S}}+\underbrace{\int_{0}^{k^{+}}\frac{dp^{+}}{p^{+}}e^{-i\frac{p^{+}}{k^{+}}s}\big[f(p^{+})-f(0)\big]}_{\text{regular term}\ \to\ \text{row}}+{\cal O}\Big(\frac{\Lambda}{k^{+}}\Big)\,,\tag{N.2}\label{eq:fx-split}
\end{align}
and in Group III, with $\frac{k^{+}dp^{+}}{p^{+}(p^{+}+k^{+})}$ and $\int_{\Lambda}^{\infty}$, the same with $\Psi(s)$ in place of $\mathcal{E}(s)$. The regular terms stay in the rows below; the soft remainders, which fall only like $\frac{\log|z|}{z^{2}}$ one by one, are collected in the block $\mathcal{S}$ of Sec.~\ref{sec:fx-soft}.}\par

\subsection{Projectile average: what is assumed}
\noindent The projectile enters every term as an ordered product of two or three colour-charge operators, $C^{a_{1}\dots a_{n}}\,\rho^{a_{1}}(p_{1})\cdots\rho^{a_{n}}(p_{n})$, with a $c$-number tensor $C$ made of adjoint Wilson lines, $f^{abc}$ and $N_{c}$, and $[\rho^{a}(x),\rho^{b}(y)]=if^{abc}\rho^{c}(x)\,\delta^{(2)}(x-y)$. We assume
\begin{itemize}
\item[(A1)] \emph{Global colour invariance} (colour-singlet projectile): the projectile density matrix commutes with the total charge $Q^{a}=\int_{x}\rho^{a}(x)$. Then $\langle\rho^{a}(x)\rangle=0$ (there is no invariant vector in the adjoint representation) and
\begin{align}
\langle\rho^{a}(x)\,\rho^{b}(y)\rangle=\frac{\delta^{ab}}{N_{c}^{2}-1}\,n(x,y)\,,\qquad n(x,y)\equiv\langle\rho^{f}(x)\rho^{f}(y)\rangle=n(y,x)=n(x,y)^{*}\,;\label{eq:s-A1}
\end{align}
the symmetry follows from $\langle[\rho^{a}(x),\rho^{b}(y)]\rangle=if^{abc}\langle\rho^{c}\rangle\delta^{(2)}(x-y)=0$, the reality from the hermiticity of $\rho^{f}$.
\item[(A2)] \emph{Symmetrised three-point function of $d$-type} [\textbf{changed}; author's answer (date 2026-11-20), verbatim: ``This is symmetrized form: $\langle\rho^{a}\rho^{b}\rho^{c}\rangle_{\rm sym}=d^{abc}\langle d^{efg}\rho^{e}\rho^{f}\rho^{g}\rangle_{\rm sym}$'', normalisation confirmed (``Yes thats my indended''); \texttt{notes/fix\_dabc.tex}]: with $S^{abc}(x,y,z)\equiv\langle\{\rho^{a}(x)\rho^{b}(y)\rho^{c}(z)\}_{\rm sym}\rangle$, $\{\dots\}_{\rm sym}$ the average over the six orderings,

\end{align}

\subsection{Symmetrised part of the cubic rows ($d^{abc}$)}\label{sec:s-dabc}
\noindent[\textbf{New} (\texttt{notes/fix\_dabc.tex}), following from the author's (A2).] Contracting the three $\rho$ indices of a cubic structure with $d^{abc}$ gives a network with one $f$, one $d$ and Wilson-line chains; chains joining two legs of $f$ to two legs of $d$ give $f^{abe}d^{abe'}=0$, the others reduce, with $U^{T}U=1$ and $(D^{e})_{bc}\equiv d^{ebc}$, to
\begin{align}
\mf{G}(a,b,c)&\equiv f^{ijk}\,d^{lmn}\,U^{il}(a)\,U^{jm}(b)\,U^{kn}(c)\,,\qquad
\mf{Q}_{d}(a,b;c,e)\equiv i\,{\rm Tr}\{T^{h}U^{\dagger}(a)U(b)D^{h}U^{\dagger}(c)U(e)\}\,.\label{eq:s-GQ}
\end{align}
$\mf{G}$ is real and totally antisymmetric ($\mf{G}(a,a,c)=0$); $\mf{Q}_{d}$ is real, $\mf{Q}_{d}(a,b;c,e)=-\mf{Q}_{d}(e,c;b,a)$, $\mf{Q}_{d}(a,a;c,e)=\mf{Q}_{d}(a,b;c,c)=0$, $\mf{Q}_{d}(a,b;c,a)=-\mf{G}(a,b,c)$, $\mf{Q}_{d}(a,b;b,c)=\mf{G}(a,b,c)|_{U\to U^{T}}$. Both are invariant under $U(p)\to RU(p)R'$, C-odd ($V\to V^{*}$ changes their sign; $\mf{D},\mf{F},\mf{Q}$ are C-even), $O(\epsilon^{3})$ for a weak target $V=e^{i\epsilon\alpha^{a}t^{a}}$, and zero for $SU(2)$. With $U^{ab}=2{\rm tr}(t^{a}Vt^{b}V^{\dagger})$ (Fierz; exact)
\begin{align}
\mf{G}(a,b,c)&={\rm Im}\big[{\rm tr}(V_{a}^{\dagger}V_{b})\,{\rm tr}(V_{b}^{\dagger}V_{c})\,{\rm tr}(V_{c}^{\dagger}V_{a})\big]+{\rm Im}\,{\rm tr}\big(V_{a}^{\dagger}V_{b}V_{c}^{\dagger}V_{a}V_{b}^{\dagger}V_{c}\big)\,,\notag\\
\mf{Q}_{d}(a,b;c,e)&=-{\rm Im}\big[{\rm tr}A\,{\rm tr}B\,{\rm tr}(A^{\dagger}B^{\dagger})\big]-{\rm Im}\,{\rm tr}\big(ABA^{\dagger}B^{\dagger}\big)\,,\qquad A=V_{a}^{\dagger}V_{b},\ B=V_{c}^{\dagger}V_{e}\,.\label{eq:s-Gfund}
\end{align}
