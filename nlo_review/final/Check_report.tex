\section{Summary}
\begin{itemize}
\item The BFKL and JIMWLK terms of \texttt{Final\_terms.tex} are the correct evolution of the leading-order cross section, and the derivation files that lead to them are consistent list by list.
\item The chain \texttt{Group\_I/II/III} $\to$ \texttt{Other\_terms} $\to$ \texttt{Limiting\_Cases} $\to$ \texttt{Final\_terms} is faithful row by row, apart from the items in Sec.~\ref{sec:fixed}.
\item Two finite terms were wrong or ill defined and are corrected in the files of the folder \texttt{final/}: the last term of Group I Row III, and the soft term of Group III Row V.
\item Three points are open (Sec.~\ref{sec:open}): one three-$\rho$ term of Group I Row II that differs between \texttt{Group\_I.tex} and the evolution files by $\log\frac{\vee}{\Lambda}$; the soft plus-prescription terms that contain $U(z)$; and part of the four-$\rho$ step for $\LV$.
\end{itemize}

\paragraph{Reordering convention found in the files.} Full symmetrisation of the three $\rho$'s, row by row: $\rho_{1}\rho_{2}\rho_{3}\to S(\rho_{1}\rho_{2}\rho_{3})+\frac12\sum_{i<j}[\rho_{i},\rho_{j}]\rho_{k}$, with $[\rho^{a}(x),\rho^{b}(y)]=if^{abc}\rho^{c}(x)\delta(x-y)$. One-$\rho$ terms are dropped and the two remaining $\rho$'s are treated as commuting. For Groups I and III the complex conjugate is included before reducing.

\section{Corrections made}\label{sec:fixed}
Changes are in red in the files of \texttt{final/}; notes are in teal.
\begin{enumerate}
\item \textbf{Group I Row III, last term} (in \texttt{Other\_terms}, \texttt{Limiting\_Cases} and \texttt{Final\_terms}). The term containing the factor $\log\frac{(z-w)^{2}}{(x-w)^{2}}\,\K(x'-w')\!\cdot\!\K(y-w)\,\K(x-z)\!\cdot\!\K(w-z)$ does not follow from \texttt{Group\_I.tex}, and its $z$ integral diverges at large $z$. In \texttt{Group\_I.tex} two terms multiply $\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}(y-z)^{2}(z-w)^{2}}$; only their sum, with $\Lambda$ kept, is integrable. With $\mathcal{F}^{i}(z)=\K^{i}(y-z)[U^{bc}(z)-U^{bc}(y)]\rho^{c}(y)$ the $z$ integral of the pair is, up to $O(\Lambda)$,
\begin{align*}
2\pi\ln^{2}\!\frac{k^{+}}{\Lambda}\,\mathcal{F}^{i}(w)
&+2\ln\frac{k^{+}}{\Lambda}\int_{z}\frac{1}{(z-w)^{2}}\Big[\mathcal{F}^{i}(z)-\tfrac{(x-w)^{2}}{(x-w)^{2}+(z-w)^{2}}\mathcal{F}^{i}(w)\Big]\\
&+\int_{z}\frac{1}{(z-w)^{2}}\ln\frac{(z-w)^{2}}{(x-w)^{2}}\Big[\mathcal{F}^{i}(z)-\tfrac{(x-w)^{2}}{(x-w)^{2}+(z-w)^{2}}\mathcal{F}^{i}(w)\Big]+\frac{\pi^{3}}{6}\mathcal{F}^{i}(w),
\end{align*}
with prefactor $-\frac{ig^{4}f^{abd}}{16\pi^{5}k^{+}}$ (checked numerically to $10^{-5}$ at $\Lambda/k^{+}=10^{-6}$). The second line is what is kept as the finite term.
\item \textbf{Group III Row V, the $-1/p^{+}$ term.} Nothing else in this row depends on $z$, the subtraction term of the plus prescription is infinite, and the split into ``log term $+$ plus term'' is not defined. Doing the $z$ integral first gives exactly
\[
\int_{\Lambda}^{\infty}\frac{dp^{+}}{p^{+}}\int_{z}e^{-i\frac{p^{+}}{k^{+}}k\cdot(y'-z)}\,\K(y'-z)\!\cdot\!\K(x-z)=\pi\ln^{2}\!\Big[\frac{2e^{-\gamma_{E}}k^{+}}{\Lambda|k||x-y'|}\Big]+O(\Lambda),
\]
with no further constant (checked numerically; assumes $\vee\gg k^{+}$). After adding the complex conjugate this squared logarithm cancels completely against item~1, including the $\ln^{2}$ of the transverse scales: under $\int_{w}e^{ik\cdot w}$ the second line of item~1 equals $\big[2\pi\ln^{2}\frac{2e^{-\gamma_{E}}}{|k||x-w|}-\mathcal{D}(k,x-w)\big]\mathcal{F}^{i}(w)$ with $\mathcal{D}(k,b)=\int d^{2}v\,e^{-ik\cdot v}\frac{1}{v^{2}}\ln\frac{(b+v)^{2}}{b^{2}}$. The two rows together therefore leave the single term $+\frac{ig^{4}f^{abd}}{16\pi^{5}k^{+}}\mathcal{D}(k,x-w)\,\mathcal{F}^{i}(w)$ (times the same structure), which is what the corrected files list in Group I Row III; nothing is listed in Group III Row V. $\mathcal{D}\neq0$; it comes from the denominator $[k^{+}(x-w)-\Lambda(x-z)]^{2}$ of the logarithm of $N_{3b}$.
\item \texttt{Limiting\_Cases.tex}: $R^{i}$ started with $-2k^{i}$ instead of $-2q^{i}$; the last Row III term had $U(z)$ in one place and $U(w)$ in another; one stray closing brace.
\item \texttt{Final\_terms.tex}: after the JIMWLK formula the kernel is defined as $\mathcal{K}(x,y,z)$ but used as $K(x,y,z)$.
\item \texttt{BFKL.tex}: five missing closing brackets in lines 41--44.
\item \texttt{4rho.tex}, eq.~(4rho): four Wilson lines in the second group are transposed. Only with the corrected orientation does the symmetric four-$\rho$ coefficient vanish and the three-$\rho$ part equal $N_{a}+N_{b}$.
\item \texttt{Evolution\_three\_rho.tex}: transposed Wilson lines in several bracketed forms (the monomial lists and the Group files have the right orientation); the header kernels of the $\beta$ and $\beta'$ cancellation blocks; $U(z)\to U(w)$ in the Row III entry.
\item \texttt{Group\_II.tex} line 80 (missing $U$), \texttt{Group\_III.tex} line 51 (misplaced $U$), and, from the earlier rounds, the sign of the $p^{+}>k^{+}$ exchange term in \texttt{Calculation\_main.tex}.
\end{enumerate}
Points noticed but not changed: the pole term $\frac{11}{3}[\ldots]$ of Group I Row II and the pole term $-\frac{11}{6}[\ldots]$ of Group I Row V cancel each other exactly and appear in none of \texttt{Other\_terms}, \texttt{Limiting\_Cases}, \texttt{Final\_terms}; this should be stated. \texttt{Group\_II.tex} line 219 has $2N_{c}/\xi$ where line 253 has $2N_{c}(1-\xi)/\xi$. In \texttt{Final\_terms.tex} the plus-prescription rows keep the upper limit $\vee-k^{+}$ although the section says $\vee\to\infty$.

\section{Open points}\label{sec:open}
\begin{enumerate}
\item \textbf{Group I Row II, one three-$\rho$ structure.} Let
\begin{align*}
X=\frac{1}{(2\pi)^{3}}\frac{ig^{4}f^{hbc}}{32\pi^{5}k^{+}}\int e^{-ik\cdot(w'-w)}\,&\K^{i}(x'-w')\big[\K^{i}(x-w)-\K^{i}(y-w)\big]K(x,y,z)\\
&\times\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x')\,U^{ah}(w)\{\rho^{b}(x),\rho^{c}(y)\}.
\end{align*}
\texttt{Group\_I.tex} ($N_{2a}$, using $\int_{z}[K(x,y,z)-K(w,y,z)]=\pi\log\frac{(w-y)^{2}}{(x-y)^{2}}$) contains $+\LL\,X$ and no $\LV$ at all in this row. The evolution files contain instead $-\LV\,X$ (\texttt{BFKL.tex}, Row II) and no $\LL\,X$ (\texttt{Evolution\_three\_rho.tex}, Row II). The difference is $\log\frac{\vee}{\Lambda}\,X$. With the version of the evolution files all three-$\rho$ logarithms cancel and the BFKL and JIMWLK terms come out as the correct evolution; with the version of \texttt{Group\_I.tex} a three-$\rho$ term $\log\frac{\vee}{\Lambda}X$ would be left. So the evolution files are very likely right for the total, but the term is not derived from Row II as written. Its source has to be identified; a candidate is the ordering of the $\rho^{b}\rho^{b}\rho^{a}$ term of $\mathbb{A}^{(3)}$, which carries $\log\frac{\vee}{\Lambda}$.
\item \textbf{Soft plus-prescription terms with $U(z)$} (Group III Rows III and VI, Group I Row V). The subtraction term and the plus-prescription integral are not separately convergent at large $z$. They are meaningful only when the rows with the same soft gluon are added before the $z$ integration. The pieces of Group III Rows III and VI without $U(z)$ add up to a commutator, i.e.\ a two-$\rho$ term with the same $\ln^{2}$ as in Row V. Which rows pair up was not verified; these rows are left as written, with a note. The projectile is not assumed colour neutral and nothing is assumed about $U(z)$ at large $z$, so these tails do not drop out.
\item \textbf{Four-$\rho$ step for $\LV$.} For $\LL$ the reduction of eq.~(4rho) to three- and two-$\rho$ terms is confirmed (with the corrected Wilson lines). For $\LV$ the first two-$\rho$ block of \texttt{BFKL.tex} is confirmed as following from $N_{a}+N_{b}$; the second block (line 69) was not reproduced.
\item \textbf{Not checked.} The $p^{+}$ integrations of Group II Row I second part from the unintegrated form; the derivation of Group I Row II from $\mathbb{A}^{(3)}$; the intermediate $\xi$ steps of Group II Row IV; one-$\rho$ terms, which are dropped throughout.
\end{enumerate}
