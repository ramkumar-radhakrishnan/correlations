\section{Result}\label{sec:result}
\paragraph{As written, it does not work.}
\begin{itemize}
\item The symmetrized three-$\rho$ terms do not cancel (Sec.~\ref{sec:three}).
\item The two-$\rho$ terms do not add up to your Eq.~(LOBFKL). They do not add up to BFKL$\otimes$LO$-$JIMWLK$\otimes$LO either.
\item Two signs of this that need no numerics:
\begin{itemize}
\item a $\frac1\epsilon$ pole is left over (Sec.~\ref{sec:poles});
\item the kernel of the second line of (LOBFKL), $\KK(z-w)\cdot\KK(z-w')K(x,y,z)$, appears in none of your terms (Sec.~\ref{sec:two}).
\end{itemize}
\end{itemize}

\paragraph{The reason.} Your list contains only the \emph{explicit} factors $\log\frac{\vee}{k^+}$, $\log\frac{k^+}{\vee}$, $\log\frac{\vee}{k^++\Lambda}$. In your rows $\log\frac{\vee}{k^+}$ also sits inside other logarithms that have $\vee$ in their argument, for example
\begin{equation*}
\log\frac{(\vee-k^+)(x-z)^2+k^+(x-w)^2}{\Lambda(x-z)^2+k^+(x-w)^2}=\log\frac{\vee}{k^+}+\log\Big[\Big(1-\frac{k^+}{\vee}\Big)\frac{(x-z)^2}{(x-w)^2}+\frac{k^+}{\vee}\Big]+O\Big(\frac{\Lambda}{k^+}\Big),
\end{equation*}
and in $\log(\frac{\vee}{k^+}-1)$ and at the lower end $\xi\to\frac{k^+}{\vee}$ of $\xi$ integrals. Row by row, against the exact coefficient of $\log\frac{\vee}{k^+}$ of the same row of \texttt{Finite\_terms\_1.tex} (Sec.~\ref{sec:rows}):
\begin{itemize}
\item \textbf{5 rows are exactly right:} Group I Row II 2a, Row V parts 1, 2a, 2b UV, 2b finite.
\item \textbf{6 rows are wrong:} Group I Rows III, IV; Group II Row I parts 1 and 2; Group III Rows I, II.
\item \textbf{5 rows that do contain $\log\frac{\vee}{k^+}$ are marked NA:} Group II Row IV; Group III Row III (both parts), Rows V, VI.
\item \textbf{Group I Row II part 1:} the coefficient is $1$; it must be $2$ ($\log\frac{\vee}{k^+}+\log(\frac{\vee}{k^+}-1)=2\log\frac{\vee}{k^+}+\log(1-\frac{k^+}{\vee})$).
\end{itemize}

\paragraph{With the exact coefficients.} The complete list is in Sec.~\ref{sec:complete}.
\begin{itemize}
\item The symmetrized three-$\rho$ terms cancel.
\item The two-$\rho$ terms add up to
\begin{equation}
\boxed{\ \text{all }\log\frac{\vee}{k^+}\text{ terms}=\log\frac{\vee}{k^+}\Big(\BB-\JJ\Big),\qquad\BB={\rm BFKL}\otimes{\rm LO}\ \text{(your Eq.~(LOBFKL))},\quad\JJ={\rm JIMWLK}\otimes{\rm LO}.\ }
\label{eq:main}
\end{equation}
This was verified exactly before (\texttt{LogVeeKplus\_BFKL.pdf}: all 39 three-$\rho$ structures cancel, all 37 two-$\rho$ structures agree, pointwise in $z$). Here it is re-checked for the colour structure of Sec.~\ref{sec:three}, to $10^{-14}$.
\item It is $\BB-\JJ$, not $\BB$, because of your rule $\log\frac{k^+}{\Lambda}=\log\frac\vee\Lambda-\log\frac{\vee}{k^+}$. In every row the $\log\frac{\vee}{k^+}$ terms are $P-T$:
\begin{itemize}
\item $T$ is the row's $\log\frac\vee\Lambda$ coefficient (your evolution list, which sums to $\JJ$);
\item $P$ is its large-$p^+$ limit (which sums to $\BB$).
\end{itemize}
\item BFKL multiplies $\log\frac{\vee}{k^+}$ once you split $\log\frac\vee\Lambda\,\JJ=\log\frac{k^+}{\Lambda}\JJ+\log\frac{\vee}{k^+}\JJ$:
\begin{equation}
\log\frac{\vee}{\Lambda}\,\JJ+\log\frac{\vee}{k^+}\big(\BB-\JJ\big)=\log\frac{k^+}{\Lambda}\,\JJ+\log\frac{\vee}{k^+}\,\BB.
\end{equation}
\item Your Eq.~(LOBFKL) itself is correct: it is $\BB$, your App.~E, checked before to machine precision.
\end{itemize}

\paragraph{A correction to my previous notes} (Sec.~\ref{sec:fix}). The hidden $\log\frac{\vee}{k^+}$ at the lower end of the $\xi$ integral of Group II Row IV was not in the table of \texttt{Finite\_terms\_simplified.pdf}, Sec.~6. As a result:
\begin{itemize}
\item its DGLAP term (5.5) still contains $\log\frac{\vee}{k^+}\,H_4$;
\item the final results FS~(11.1) and \texttt{Finite\_terms\_singlet.pdf}~(7.1) counted $\frac12\log\frac\vee\Lambda\,H_4$ twice.
\end{itemize}
Both notes are corrected and re-sent.

\section{Three \texorpdfstring{$\rho$}{rho}'s}\label{sec:three}
The symmetrized three-$\rho$ colour of your rows is that of FS Secs.~8, 9. One of its structures does not depend on $z$:
\begin{equation}
C_w\equiv f^{abc}\,\CL^a\,\big(U(w)\rho(x)\big)^b\big(U(w)\rho(y)\big)^c,\qquad \CL^a=\big[U^{ab'}(x')-U^{ab'}(w')\big]\rho^{b'}(x').
\end{equation}
\begin{itemize}
\item It comes from two places (checked numerically to $10^{-15}$):
\begin{itemize}
\item Group I Row II 2a, whose symmetric colour contains $2C_w$;
\item the $\KK_{\rm IV}$ family (G1 IV, G3 I, II, G2 I(1) block A), whose combined symmetric colour is $-C_w+f^{acd}\CL^a(U(z)\rho(y))^c(U(x)\rho(x))^d$ (FS Eq.~(8.2)).
\end{itemize}
\item Since $C_w$ does not depend on $z$, its total coefficient is the $z$ integral of the kernels, antisymmetrized in $x\leftrightarrow y$. It must vanish:
\begin{equation}
-\frac{ig^4}{32\pi^4k^+}\,2\,X'\!\cdot\mathfrak F(w;x,y)\;-\;\frac{ig^4}{8\pi^5k^+}\int_z\big[\text{kernel}\big]_{x\leftrightarrow y\text{ antisym.}}=0,
\end{equation}
\begin{equation*}
\mathfrak F^i(w;x,y)=\KK^i(x-w)\log\frac{(w-y)^2}{(x-y)^2}-\KK^i(y-w)\log\frac{(w-x)^2}{(x-y)^2}\quad\text{(your Row II 2a)}.
\end{equation*}
\item \textbf{Exact kernel} $X'^iY^j\mathbb T^{ij}-\frac{X'\cdot Y}{2(z-w)^2}$: numerically, $\int_z[\dots]_A=-\frac\pi2X'\!\cdot\mathfrak F$ to @@CWREL@@. The structure cancels.
\item \textbf{Your kernel} $-\frac{X'\cdot Y}{2(z-w)^2}$: the integral diverges at $z\to w$. With a cut $|z-w|>\varepsilon$ it grows like $\log\frac1\varepsilon$ (@@CWGROW@@ for $\varepsilon=10^{-2},10^{-3},10^{-4}$). Nothing else in your list carries $C_w$, so your three-$\rho$ terms cannot cancel.
\end{itemize}

\begin{itemize}

\begin{equation}
\int d^2\Delta\,e^{-i\mathbf q\cdot\Delta}\log(\Delta^2)\,h(\Delta)\ \xrightarrow{q\to\infty}\ -\frac{4\pi}{q^2}h(0)
\end{equation}
