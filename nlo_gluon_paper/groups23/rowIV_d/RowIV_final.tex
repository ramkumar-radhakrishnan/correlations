\section*{Group II, Row IV: final result ($\delta^{mm}=d_\perp$, exact limits)}
\begin{align}
\frac{d^{3}N_{\rm IV}}{d^{3}k}={}&\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,d_{\perp}\,\bigg[\log\Big(\frac{\vee}{\Lambda}\Big)-\log\Big(\frac{\vee}{k^{+}}\Big)+\log\Big(1-\frac{k^{+}}{\vee}\Big)+\log\Big(1+\frac{\Lambda}{k^{+}}\Big)\bigg]\notag\\
&\qquad\times\int_{x,x',y,y'}e^{-ik\cdot(y'-y)}\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\bigg[\frac{1}{\bar\epsilon}-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}\,e^{2\gamma_{E}}}{4}\bigg]\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,2\,\bigg[\log\Big(\frac{\vee}{\Lambda}\Big)-\log\Big(\frac{\vee}{k^{+}}\Big)+\log\Big(1-\frac{k^{+}}{\vee}\Big)+\log\Big(1+\frac{\Lambda}{k^{+}}\Big)\bigg]\notag\\
&\qquad\times\int_{x,x',y,y'}e^{-ik\cdot(y'-y)}\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,d_{\perp}\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}d\xi\int_{x,x',y,y'}\frac{1}{1-\xi}\bigg[\frac{1}{\xi^{2}}\,e^{-ik\cdot(y'-y)/\xi}-e^{-ik\cdot(y'-y)}\bigg]\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\bigg[\frac{1}{\bar\epsilon}-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}\,e^{2\gamma_{E}}}{4}\bigg]\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,d_{\perp}\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,\xi(1-\xi)\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\bigg[\frac{1}{\bar\epsilon}-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}\,e^{2\gamma_{E}}}{4}\bigg]\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,d_{\perp}\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,\frac{1-\xi}{\xi}\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\bigg[\frac{1}{\bar\epsilon}-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}\,e^{2\gamma_{E}}}{4}\bigg]\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,d_{\perp}\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\bigg[\frac{1}{\bar\epsilon}-\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}\,e^{2\gamma_{E}}}{4}\bigg]\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,2\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}d\xi\int_{x,x',y,y'}\frac{1}{1-\xi}\bigg[\frac{1}{\xi^{2}}\,e^{-ik\cdot(y'-y)/\xi}-e^{-ik\cdot(y'-y)}\bigg]\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,2\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,2\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,\frac{1-\xi}{\xi}\notag\\
&\qquad\times\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)\notag\\[8pt]
&-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,2\,d_{\perp}\int_{\frac{k^{+}}{\vee}}^{\frac{k^{+}}{k^{+}+\Lambda}}\frac{d\xi}{\xi^{2}}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,\xi(1-\xi)\notag\\
&\qquad\times\frac{(x-y)\cdot(y'-y)\,(x'-y')\cdot(y'-y)}{(x-y)^{2}\,(x'-y')^{2}\,(y'-y)^{2}}\notag\\
&\qquad\times\Big[U^{ce'}(y')-U^{ce'}(x')\Big]\rho^{e'}(x')\,\Big[U^{ce}(y)-U^{ce}(x)\Big]\rho^{e}(x)
\end{align}
