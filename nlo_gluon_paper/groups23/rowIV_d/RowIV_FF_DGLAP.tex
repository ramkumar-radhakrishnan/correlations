\noindent In this version the $\frac1\xi$ of the phase is never absorbed. No ``LO at $(\frac{k^+}{\xi},\frac{\mathbf k}{\xi})$'' is used. Every equation shows explicitly:
\begin{itemize}
\item the phase $e^{-ik\cdot(y'-y)/\xi}$;
\item the measure $\frac{d\xi}{\xi^2}$;
\item the prefactor $\frac{1}{k^+}$.
\end{itemize}
Where a parton spectrum is needed at another momentum, the substitution is written out.

\section{Your equation}\label{sec:check}
Your $\xi$ form is correct. One change is needed, in blue: the measure is $\int_{x,x',y,y'}$ in both terms. The charge $x'$ of the conjugate amplitude is integrated, as at LO.
\begin{align}
\frac{d^{3}N_{\rm NLO}}{d^{3}k}={}&\frac{1}{(2\pi)^{3}}\,\frac{g^{4}d_{\perp}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\int_{\frac{k^{+}}{k^{+}+\Lambda}}^{\frac{k^{+}}{\vee}}\frac{d\xi}{2\pi}\int_{x,\new{x'},y,y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\Big[\xi(1-\xi)+\frac{1-\xi}{\xi}+\frac{\xi}{1-\xi}\Big]\frac{1}{\xi^{2}}\notag\\
&\times\Big[\frac{1}{\epsilon}+\gamma_{E}+\frac{1}{2}\log\frac{(y'-y)^{2}\mu^{2}_{\overline{MS}}}{4}\Big]\big[U^{e'c}(y')-U^{e'c}(x')\big]\big[U^{ec}(y)-U^{ce}(x)\big]\rho^{e'}(x')\rho^{e}(x)\notag\\
&+\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\int_{\frac{k^{+}}{k^{+}+\Lambda}}^{\frac{k^{+}}{\vee}}\frac{d\xi}{2\pi}\int_{x,\new{x'},y,y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\,\frac{1}{\xi^{2}}\Bigg[\frac{(x-y)\cdot(y'-y)\,(x'-y')\cdot(y'-y)}{(x-y)^{2}(x'-y')^{2}(y'-y)^{2}}\,d_{\perp}\,\xi(1-\xi)\notag\\
&+\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\Big[\frac{1-\xi}{\xi}+\frac{\xi}{1-\xi}\Big]\Bigg]\big[U^{e'c}(y')-U^{e'c}(x')\big]\big[U^{ec}(y)-U^{ce}(x)\big]\rho^{e'}(x')\rho^{e}(x).
\label{eq:user}
\end{align}
It is Eq.~(1.5) of \texttt{RowIV\_dperp.pdf}, term by term (\texttt{sympy}: difference $0$). The reasons:
\begin{itemize}
\item The limits are reversed, $\int_{k^+/(k^++\Lambda)}^{k^+/\vee}=-\int_{k^+/\vee}^{k^+/(k^++\Lambda)}$. This removes the overall minus sign of the $E$ form of (1.5).
\item $\frac{1}{\epsilon}+\gamma_E+\frac12\log\frac{(y'-y)^2\mu^2}{4}=\frac{E}{2}$, and $\frac{g^4d_\perp}{4\pi^3}\cdot\frac{1}{2\pi}\cdot\frac{E}{2}=\frac{g^4}{16\pi^4}\,d_\perp E$.
\item In the second term, $\frac{g^4}{4\pi^3}\cdot\frac{1}{2\pi}=\frac{g^4}{16\pi^4}\cdot2$.
\item The $\frac{1}{\xi^2}$ and the phase are right. The $\frac{1}{\xi^2}$ comes from $dp^+=-k^+\frac{d\xi}{\xi^2}$. The phase $\frac1\xi$ comes from $w'-w=\frac{p^++k^+}{k^+}(y'-y)=\frac{y'-y}{\xi}$, after the $w$ and $w'$ integrations ($z$ drops out of the difference).
\end{itemize}

\paragraph{Notation used below.}
\begin{itemize}
\item $\Dh\equiv\frac{y'-y}{|y'-y|}$, $X\equiv\frac{x-y}{(x-y)^2}$, $X'\equiv\frac{x'-y'}{(x'-y')^2}$, and the colour factor (with $U^{bc}=U^{cb}$)
\begin{equation}
\OO\equiv\big[U^{ce'}(y')-U^{ce'}(x')\big]\rho^{e'}(x')\,\big[U^{ce}(y)-U^{ce}(x)\big]\rho^{e}(x).
\end{equation}
None of $X$, $X'$, $\Dh$, $\OO$ depends on $\xi$ or $\mathbf k$; only the phase does.
\item $\xi_0\equiv\frac{k^+}{\vee}$, $\xi_1\equiv\frac{k^+}{k^++\Lambda}$, $\delta Y=\log\frac{\vee}{\Lambda}$ and $\delta\equiv\frac{\Lambda}{k^+}$.
\item $\frac1{\bar\epsilon}\equiv-\frac2\epsilon$, which is $-\frac{1}{\epsilon_{\rm std}}$ in the usual $D=4-2\epsilon_{\rm std}$. Then $\frac{E}{2}=-\frac{\bar E}{2}$, with
\begin{equation}
\bar E\equiv\frac1{\bar\epsilon}-\ell,\qquad \ell\equiv\log\frac{(y'-y)^2\mu^2_{\overline{MS}}e^{2\gamma_E}}{4}=\log\frac{(y'-y)^2\mu^2_{\overline{MS}}}{c_0^2},\qquad c_0\equiv2e^{-\gamma_E}.
\end{equation}
The pole is collinear (infrared): $\epsilon<0$, $\frac1{\bar\epsilon}>0$. $\bar E$ depends on $y'-y$ only, \emph{not} on $\xi$.
\item $P^{(r)}_{gg}(\xi)\equiv2N_c\big[\xi(1-\xi)+\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\big]=2N_c\hat\Phi(\xi)$.
\end{itemize}

\section{Row IV in fragmentation form}\label{sec:conv}
Start from Eq.~(1.5) of \texttt{RowIV\_dperp.pdf}:
\begin{align}
\frac{d^3N_{\rm IV}}{d^3k}={}&\frac{1}{(2\pi)^3}\frac{g^4N_c}{16\pi^4}\frac{1}{k^+}\int_{\xi_0}^{\xi_1}\frac{d\xi}{\xi^2}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\notag\\
&\times\Big\{X\cdot X'\Big[d_\perp\hat\Phi(\xi)\bar E-2\Big(\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\Big)\Big]-2d_\perp\xi(1-\xi)(X\cdot\Dh)(X'\cdot\Dh)\Big\}\OO .
\label{eq:start}
\end{align}
Two identities, neither of which touches the phase or the $\frac{1}{\xi^2}$:
\begin{itemize}
\item $\frac{g^4}{16\pi^4}=\frac{\alpha_s}{2\pi}\cdot\frac{g^2}{2\pi^2}$, with $\alpha_s=\frac{g^2}{4\pi}$ (\texttt{sympy}: $0$);
\item $N_c\big[d_\perp\hat\Phi\bar E-2\big(\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}\big)\big]=P^{(r)}_{gg}(\xi)\frac{d_\perp\bar E-2}{2}+2N_c\xi(1-\xi)$, because $\hat\Phi-\xi(1-\xi)=\frac{\xi}{1-\xi}+\frac{1-\xi}{\xi}$.
\end{itemize}
Hence, exactly and for any $d_\perp$,
\begin{equation}
\boxed{\begin{aligned}
\frac{d^3N_{\rm IV}}{d^3k}={}&\frac{\alpha_s}{2\pi}\pre\int_{\xi_0}^{\xi_1}\frac{d\xi}{\xi^2}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\\
&\times\Big\{P^{(r)}_{gg}(\xi)\,\frac{d_\perp\bar E-2}{2}\,X\cdot X'+2N_c\,\xi(1-\xi)\Big[X\cdot X'-d_\perp(X\cdot\Dh)(X'\cdot\Dh)\Big]\Big\}\OO
\end{aligned}}
\label{eq:conv}
\end{equation}
(\texttt{sympy}: braces minus those of \eqref{eq:start} $=$ $0$; minus the compact form of \texttt{RowIV\_compact.pdf} $=$ $0$).
\begin{itemize}
\item \textbf{The pole.} It sits in the first term only:
\begin{equation}
\frac{d^3N_{\rm IV}}{d^3k}\bigg|_{\rm pole}=\frac{\alpha_s}{2\pi}\,\frac{d_\perp}{2}\,\frac{1}{\bar\epsilon}\,\pre\int_{\xi_0}^{\xi_1}\frac{d\xi}{\xi^2}\,P^{(r)}_{gg}(\xi)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO .
\label{eq:pole}
\end{equation}
\item \textbf{The second term has no pole.} $\delta^{ij}-d_\perp\Dh^i\Dh^j$ is traceless in $d_\perp$ dimensions. It is the linearly polarized part of $g\to gg$.
\item \textbf{The two choices of $d_\perp$.}
\begin{itemize}
\item $d_\perp=2-\epsilon$: $\frac{d_\perp\bar E-2}{2}=\frac1{\bar\epsilon}-\ell+O(\epsilon)$ (\texttt{sympy}: $0$). The ``$+1$'' of $\frac{d_\perp}{2}\frac{1}{\bar\epsilon}$ cancels the $-1$, so the residue is $P^{(r)}_{gg}$ exactly.
\item $d_\perp=2$: $\frac{d_\perp\bar E-2}{2}=\frac1{\bar\epsilon}-\ell-1$. The residue is the same; the finite part differs by a scheme change.
\end{itemize}
\end{itemize}

\section{The endpoint \texorpdfstring{$\xi\to1$}{xi to 1}: plus prescription with the phase explicit}\label{sec:plus}
Under the $\xi$ integral, $P^{(r)}_{gg}(\xi)$ multiplies $\frac{1}{\xi^2}e^{-ik\cdot(y'-y)/\xi}$. At $\xi=1$ this is $e^{-ik\cdot(y'-y)}$, so the subtraction term of the plus prescription carries the phase $e^{-ik\cdot(y'-y)}$, without $\frac1\xi$. The plus distribution on $[0,1]$, cut at $\xi_0$, is $\int_{\xi_0}^1d\xi\big[\frac{1}{1-\xi}\big]_+f(\xi)\equiv\int_{\xi_0}^1d\xi\frac{f(\xi)-f(1)}{1-\xi}+f(1)\log(1-\xi_0)$. With $P^{(r)}_{gg,+}(\xi)\equiv2N_c\big[\frac{\xi}{(1-\xi)_+}+\frac{1-\xi}{\xi}+\xi(1-\xi)\big]$, written out:
\begin{align}
\int_{\xi_0}^1\frac{d\xi}{\xi^2}P^{(r)}_{gg,+}(\xi)\,e^{-ik\cdot(y'-y)/\xi}\equiv{}&2N_c\int_{\xi_0}^1d\xi\bigg[\frac{\frac{1}{\xi}e^{-ik\cdot(y'-y)/\xi}-e^{-ik\cdot(y'-y)}}{1-\xi}+\Big(\frac{1-\xi}{\xi^3}+\frac{1-\xi}{\xi}\Big)e^{-ik\cdot(y'-y)/\xi}\bigg]\notag\\
&+2N_c\log(1-\xi_0)\,e^{-ik\cdot(y'-y)}.
\label{eq:plusdef}
\end{align}
Then, exactly (numerically, rel.\ $7\times10^{-14}$),
\begin{equation}
\int_{\xi_0}^{\xi_1}\frac{d\xi}{\xi^2}P^{(r)}_{gg}(\xi)\,e^{-ik\cdot(y'-y)/\xi}=\int_{\xi_0}^1\frac{d\xi}{\xi^2}P^{(r)}_{gg,+}(\xi)\,e^{-ik\cdot(y'-y)/\xi}+2N_c\log\frac{k^++\Lambda}{\Lambda}\,e^{-ik\cdot(y'-y)}-R_\Lambda(y'-y),
\label{eq:plusid}
\end{equation}
\begin{equation}
R_\Lambda(y'-y)\equiv\int_{\xi_1}^1d\xi\Big[\frac{P^{(r)}_{gg}(\xi)}{\xi^2}e^{-ik\cdot(y'-y)/\xi}-\frac{2N_c}{1-\xi}e^{-ik\cdot(y'-y)}\Big]=2N_c\frac{\Lambda}{k^+}\big[1-ik\cdot(y'-y)\big]e^{-ik\cdot(y'-y)}+O\Big(\frac{\Lambda^2}{k^{+2}}\Big).
\label{eq:RL}
\end{equation}
For $k\cdot(y'-y)=1.3$, numerically $R_\Lambda/\delta=$ $-1.97000-2.62408\,i$, $-1.97021-2.62276\,i$, $-1.97025-2.62263\,i$ for $\delta=10^{-2},10^{-3},10^{-4}$, vs.\ $-1.97025-2.62261\,i$ $=2(1-ia)e^{-ia}$. Here
\begin{equation}
\log\frac{k^++\Lambda}{\Lambda}=\log\frac{\vee}{\Lambda}-\log\frac{\vee}{k^+}+\log\Big(1+\frac{\Lambda}{k^+}\Big).
\end{equation}

\paragraph{The pole of Row IV, exactly.}
\begin{align}
\frac{d^3N_{\rm IV}}{d^3k}\bigg|_{\rm pole}={}&\frac{\alpha_s}{2\pi}\frac{d_\perp}{2}\frac{1}{\bar\epsilon}\pre\int_{x,x',y,y'}X\cdot X'\,\OO\,\bigg\{\int_{\xi_0}^1\frac{d\xi}{\xi^2}P^{(r)}_{gg,+}(\xi)\,e^{-ik\cdot(y'-y)/\xi}\notag\\
&+2N_c\Big[\log\frac{\vee}{\Lambda}-\log\frac{\vee}{k^+}+\log\Big(1+\frac{\Lambda}{k^+}\Big)\Big]e^{-ik\cdot(y'-y)}-R_\Lambda(y'-y)\bigg\}.
\label{eq:poleplus}
\end{align}

\paragraph{The full Row IV, exactly.} $\bar E$ does not depend on $\xi$, so it comes out of the $\xi$ integral:
\begin{align}
\frac{d^3N_{\rm IV}}{d^3k}={}&\frac{\alpha_s}{2\pi}\pre\int_{x,x',y,y'}\OO\,\bigg\{\frac{d_\perp\bar E-2}{2}\,X\cdot X'\bigg[\int_{\xi_0}^1\frac{d\xi}{\xi^2}P^{(r)}_{gg,+}(\xi)\,e^{-ik\cdot(y'-y)/\xi}\notag\\
&+2N_c\log\frac{k^++\Lambda}{\Lambda}\,e^{-ik\cdot(y'-y)}-R_\Lambda(y'-y)\bigg]+2N_c\int_{\xi_0}^{\xi_1}d\xi\,\frac{1-\xi}{\xi}\,e^{-ik\cdot(y'-y)/\xi}\Big[X\cdot X'-d_\perp(X\cdot\Dh)(X'\cdot\Dh)\Big]\bigg\}
\label{eq:full}
\end{align}
(numerically against \eqref{eq:conv}, random $d_\perp$, $\bar E$, $X\cdot X'$, $(X\cdot\Dh)(X'\cdot\Dh)$, $k\cdot(y'-y)$, $\xi_0$, $\Lambda/k^+$: rel.\ $1\times10^{-13}$).
\begin{itemize}
\item The terms with $e^{-ik\cdot(y'-y)/\xi}$ under $\int d\xi$ are the real ($\xi<1$) part of the splitting.
\item The terms with $e^{-ik\cdot(y'-y)}$, i.e.\ without $\frac1\xi$, are the $\delta(1-\xi)$ terms.
\item $R_\Lambda$ contains $ik\cdot(y'-y)\,e^{-ik\cdot(y'-y)}$. In $\xi$ language this is a $\delta'(1-\xi)$, because $\frac{d}{d\xi}\big[\frac{1}{\xi^2}e^{-ik\cdot(y'-y)/\xi}\big]_{\xi=1}=\big[-2+ik\cdot(y'-y)\big]e^{-ik\cdot(y'-y)}$.
\end{itemize}

\section{The \texorpdfstring{$\overline{\rm MS}$}{MS-bar} fragmentation function}\label{sec:FF}
\paragraph{Factorization, written out.} A final state $h$ with $(k^+,\mathbf k)$ comes from a gluon with $(q^+,\mathbf q)$ and momentum fraction $\xi$:
\begin{equation}
\frac{d^3N_h}{d^3k}(k^+,\mathbf k)=\int_{\xi_0}^1d\xi\,D^{(0)}_{h/g}(\xi)\int dq^+d^2\mathbf q\;\delta(k^+-\xi q^+)\,\delta^{(2)}(\mathbf k-\xi\mathbf q)\,\frac{d^3N_g}{d^3q}(q^+,\mathbf q).
\label{eq:factdef}
\end{equation}
The $\delta$ functions give $\frac{1}{\xi^3}$ and set $q^+=\frac{k^+}{\xi}$, $\mathbf q=\frac{\mathbf k}{\xi}$. For the LO gluon spectrum
\begin{equation}
\frac{d^3N^{\rm LO}_g}{d^3q}(q^+,\mathbf q)=\frac{1}{(2\pi)^3}\frac{g^2}{2\pi^2q^+}\int_{x,x',y,y'}e^{-i\mathbf q\cdot(y'-y)}\,X\cdot X'\,\OO ,
\end{equation}
the substitution $q^+=\frac{k^+}{\xi}$, $\mathbf q=\frac{\mathbf k}{\xi}$ is
\begin{itemize}
\item $\frac{1}{q^+}=\frac{\xi}{k^+}$;
\item $e^{-i\mathbf q\cdot(y'-y)}=e^{-ik\cdot(y'-y)/\xi}$.
\end{itemize}
So, with $\frac{1}{\xi^3}\cdot\xi=\frac{1}{\xi^2}$,
\begin{equation}
\frac{d^3N^{\rm LO}_h}{d^3k}=\pre\int_{\xi_0}^1\frac{d\xi}{\xi^2}\,D^{(0)}_{h/g}(\xi)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO .
\label{eq:fact}
\end{equation}
This has the same structure as the pole of Row IV, Eq.~\eqref{eq:pole}:
\begin{itemize}
\item the same $\frac{d\xi}{\xi^2}$;
\item the same phase $e^{-ik\cdot(y'-y)/\xi}$;
\item the same $\frac{1}{k^+}$.
\end{itemize}
The only difference is $D^{(0)}_{h/g}(\xi)\leftrightarrow\frac{\alpha_s}{2\pi}\frac{d_\perp}{2}\frac{1}{\bar\epsilon}P^{(r)}_{gg}(\xi)$. That is why the pole can be absorbed into the fragmentation function.

\paragraph{The pole of the gluon spectrum.} Sec.~\ref{sec:delta} adds the poles of Row IV, of the virtual rows of Group I, and of the charge renormalization of LO. Up to terms of order $\frac{\Lambda}{k^+}$ and $\frac{k^+}{\vee}$, the sum is
\begin{equation}
\frac{d^3N_g}{d^3k}\bigg|_{\rm pole}(k^+,\mathbf k)=\frac{\alpha_s}{2\pi}\frac{1}{\bar\epsilon}\pre\int_{\xi_0}^1\frac{d\xi}{\xi^2}\,P_{gg}(\xi)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO ,
\label{eq:totpole}
\end{equation}
\begin{equation}
P_{gg}(\xi)=2N_c\Big[\frac{\xi}{(1-\xi)_+}+\frac{1-\xi}{\xi}+\xi(1-\xi)\Big]+\frac{11N_c}{6}\,\delta(1-\xi)\qquad(n_f=0).
\label{eq:Pgg}
\end{equation}
\begin{itemize}
\item The $\delta(1-\xi)$ term gives $\frac{11N_c}{6}e^{-ik\cdot(y'-y)}$.
\item Its coefficient is also fixed by momentum conservation. $\int_0^1d\xi\,\xi\,P^{(r)}_{gg,+}=$ $-\frac{11N_c}{6}$ (\texttt{sympy}), so $\int_0^1d\xi\,\xi P_{gg}=0$ requires $\frac{11N_c}{6}\delta(1-\xi)$.
\end{itemize}
To use Eq.~\eqref{eq:totpole} inside Eq.~\eqref{eq:factdef}, it is needed at the gluon momentum $(q^+,\mathbf q)=(\frac{k^+}{\zeta},\frac{\mathbf k}{\zeta})$. Rename its integration variable $\xi\to z$. Then $\frac{1}{k^+}\to\frac{\zeta}{k^+}$, the lower limit $\frac{k^+}{\vee}\to\frac{\xi_0}{\zeta}$, and the phase is
\begin{equation}
e^{-i\mathbf q\cdot(y'-y)/z}=e^{-ik\cdot(y'-y)/(\zeta z)}:
\end{equation}
\begin{equation}
\frac{d^3N_g}{d^3q}\bigg|_{\rm pole}\Big(\frac{k^+}{\zeta},\frac{\mathbf k}{\zeta}\Big)=\frac{\alpha_s}{2\pi}\frac{1}{\bar\epsilon}\frac{1}{(2\pi)^3}\frac{g^2\zeta}{2\pi^2k^+}\int_{\xi_0/\zeta}^1\frac{dz}{z^2}\,P_{gg}(z)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/(\zeta z)}\,X\cdot X'\,\OO .
\label{eq:polezeta}
\end{equation}

\paragraph{The counterterm.}
\begin{equation}
\boxed{\Gamma_{gg}(z)\equiv\delta(1-z)-\frac{\alpha_s}{2\pi}\frac{1}{\bar\epsilon}P_{gg}(z)=\delta(1-z)+\frac{\alpha_s}{2\pi}\frac{2}{\epsilon}P_{gg}(z),\qquad
D^{(0)}_{h/g}(\xi)=\int_\xi^1\frac{dz}{z}\,\Gamma_{gg}(z)\,D_{h/g}\Big(\frac{\xi}{z},\mu\Big).}
\label{eq:Gamma}
\end{equation}
The second form of $\Gamma_{gg}$ is in your $\epsilon$; $\overline{\rm MS}$ subtracts only $\frac{2}{\epsilon}$. Insert $D^{(0)}$ into Eq.~\eqref{eq:factdef}. At $O(\alpha_s)$ there are two pole terms.
\begin{itemize}
\item \textbf{A: the counterterm, from LO, Eq.~\eqref{eq:fact}:}
\begin{equation*}
A=-\frac{\alpha_s}{2\pi}\frac{1}{\bar\epsilon}\pre\int_{\xi_0}^1\frac{d\xi}{\xi^2}\Big[\int_\xi^1\frac{dz}{z}P_{gg}(z)D_{h/g}\Big(\frac{\xi}{z},\mu\Big)\Big]\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO .
\end{equation*}
\item \textbf{B: the pole of the gluon spectrum, Eq.~\eqref{eq:polezeta}, fragmented with $\int_{\xi_0}^1d\zeta\,\frac{1}{\zeta^3}D_{h/g}(\zeta,\mu)$} ($\frac{1}{\zeta^3}\cdot\zeta=\frac{1}{\zeta^2}$):
\begin{equation*}
B=\frac{\alpha_s}{2\pi}\frac{1}{\bar\epsilon}\pre\int_{\xi_0}^1\frac{d\zeta}{\zeta^2}D_{h/g}(\zeta,\mu)\int_{\xi_0/\zeta}^1\frac{dz}{z^2}P_{gg}(z)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/(\zeta z)}\,X\cdot X'\,\OO .
\end{equation*}
\end{itemize}
In $B$, change variables $(\zeta,z)\to(\xi,z)$ with $\xi\equiv\zeta z$, at fixed $z$:
\begin{itemize}
\item the phase: $e^{-ik\cdot(y'-y)/(\zeta z)}=e^{-ik\cdot(y'-y)/\xi}$;
\item the measure: $d\zeta=\frac{d\xi}{z}$ and $\frac{1}{\zeta^2}=\frac{z^2}{\xi^2}$, so $\frac{d\zeta}{\zeta^2}\frac{dz}{z^2}=\frac{d\xi}{\xi^2}\frac{dz}{z}$;
\item the fragmentation function: $D_{h/g}(\zeta)=D_{h/g}(\frac{\xi}{z})$;
\item the limits: $\xi_0\le\zeta\le1$, $\frac{\xi_0}{\zeta}\le z\le1$ become $\xi_0\le\xi\le1$, $\xi\le z\le1$.
\end{itemize}
So $B=-A$ (numerically, with the phase kept: rel.\ $2\times10^{-15}$): the counterterm removes the pole, with the same phase $e^{-ik\cdot(y'-y)/\xi}$ on both sides. The kernel $\int_\xi^1\frac{dz}{z}$ arises exactly because $\frac{1}{\zeta z}=\frac1\xi$ in the phase.

\paragraph{Row IV after the subtraction ($d_\perp=2-\epsilon$).} The part of Row IV that remains is Eq.~\eqref{eq:conv} without its $\frac{1}{\bar\epsilon}$:
\begin{align}
\frac{d^3N^{\overline{MS}}_{\rm IV}}{d^3k}={}&\frac{\alpha_s}{2\pi}\pre\int_{\xi_0}^{\xi_1}\frac{d\xi}{\xi^2}\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\notag\\
&\times\Big\{-P^{(r)}_{gg}(\xi)\log\frac{(y'-y)^2\mu^2}{c_0^2}\,X\cdot X'+2N_c\,\xi(1-\xi)\Big[X\cdot X'-2(X\cdot\Dh)(X'\cdot\Dh)\Big]\Big\}\OO .
\end{align}
The logarithm vanishes at $\mu=c_0/|y'-y|$, i.e.\ $\mu\sim|\mathbf k|$.

\section{DGLAP}\label{sec:dglap}
\paragraph{The cross section through $O(\alpha_s)$, written out.}
\begin{align}
\frac{d^3N_h}{d^3k}={}&\frac{1}{(2\pi)^3}\frac{g^2(\mu)}{2\pi^2k^+}\int_{\xi_0}^1\frac{d\xi}{\xi^2}\,D_{h/g}(\xi,\mu)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO\notag\\
&+\int_{\xi_0}^1\frac{d\zeta}{\zeta^3}\,D_{h/g}(\zeta,\mu)\,\frac{d^3N^{\overline{MS}}_{g,\rm NLO}}{d^3q}\Big(\frac{k^+}{\zeta},\frac{\mathbf k}{\zeta};\mu\Big).
\end{align}
\begin{itemize}
\item In $\frac{d^3N^{\overline{MS}}_{g,\rm NLO}}{d^3q}$, every pole came with its logarithm, $\bar E=\frac1{\bar\epsilon}-\log\frac{L_\perp^2\mu^2}{c_0^2}$, with $L_\perp=y'-y$ (Row IV), $x-w$ (Row II), $y-z$ (Row V 2b). So $\frac{\partial}{\partial\log\mu^2}$ gives $-1$ times the residues. These are $P^{(r)}_{gg,+}$ plus the $\delta(1-\xi)$ terms of Sec.~\ref{sec:delta} without the charge renormalization; the soft logarithms and the $\frac{11}{3}$ cancel there.
\item $g^2(\mu)$ in the LO term runs, $\frac{d\alpha_s}{d\log\mu^2}=-\frac{\beta_0}{4\pi}\alpha_s^2$. This supplies the $-\frac{\alpha_s}{2\pi}\frac{\beta_0}{2}\delta(1-\xi)$.
\end{itemize}
Together, at the gluon momentum $(\frac{k^+}{\zeta},\frac{\mathbf k}{\zeta})$:
\begin{equation*}
-\frac{\alpha_s}{2\pi}\frac{1}{(2\pi)^3}\frac{g^2\zeta}{2\pi^2k^+}\int_{\xi_0/\zeta}^1\frac{dz}{z^2}\,P_{gg}(z)\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/(\zeta z)}\,X\cdot X'\,\OO .
\end{equation*}
Fragmented with $\int\frac{d\zeta}{\zeta^3}D_{h/g}(\zeta,\mu)$ and with $\xi=\zeta z$ as above, $\frac{\partial}{\partial\log\mu^2}\frac{d^3N_h}{d^3k}=0$ reads
\begin{equation}
\pre\int_{\xi_0}^1\frac{d\xi}{\xi^2}\Big[\frac{\partial D_{h/g}(\xi,\mu)}{\partial\log\mu^2}-\frac{\alpha_s}{2\pi}\int_\xi^1\frac{dz}{z}P_{gg}(z)D_{h/g}\Big(\frac{\xi}{z},\mu\Big)\Big]\int_{x,x',y,y'}e^{-ik\cdot(y'-y)/\xi}\,X\cdot X'\,\OO=0 .
\end{equation}
This holds for every $k^+$, every $\mathbf k$, and every projectile and target. Through the phase $e^{-ik\cdot(y'-y)/\xi}$, different $\xi$ are different transverse momenta $\mathbf k/\xi$ of the parent. So the bracket vanishes for each $\xi$:
\begin{equation}
\boxed{\mu^2\frac{\partial}{\partial\mu^2}D_{h/g}(\xi,\mu)=\frac{\alpha_s(\mu)}{2\pi}\int_\xi^1\frac{dz}{z}\,P_{gg}(z)\,D_{h/g}\Big(\frac{\xi}{z},\mu\Big)}
\label{eq:DGLAP}
\end{equation}
with $P_{gg}$ of Eq.~\eqref{eq:Pgg}: the time-like DGLAP equation for gluon fragmentation ($n_f=0$).

\paragraph{The same from $D^{(0)}$.} In $d_\perp=2-\epsilon$, $g^2$ has mass dimension $\epsilon$, so $\frac{d\alpha_s}{d\log\mu^2}=-\frac\epsilon2\alpha_s+O(\alpha_s^2)$. Then
\begin{equation}
\frac{\partial}{\partial\log\mu^2}\Big[\frac{\alpha_s}{2\pi}\frac1{\bar\epsilon}\Big]=\frac{\alpha_s}{2\pi}\Big(-\frac\epsilon2\Big)\Big(-\frac2\epsilon\Big)=\frac{\alpha_s}{2\pi}
\end{equation}
(\texttt{sympy}: ratio $1$), and $\frac{dD^{(0)}}{d\log\mu^2}=0$ in Eq.~\eqref{eq:Gamma} gives Eq.~\eqref{eq:DGLAP}.

\paragraph{Solution to first order.}
\begin{equation}
D_{h/g}(\xi,\mu)=D_{h/g}(\xi,\mu_0)+\frac{\alpha_s}{2\pi}\log\frac{\mu^2}{\mu_0^2}\int_\xi^1\frac{dz}{z}P_{gg}(z)\,D_{h/g}\Big(\frac{\xi}{z},\mu_0\Big)+O(\alpha_s^2).
\end{equation}

\section{The \texorpdfstring{$\delta(1-\xi)$}{delta(1-xi)} part: Row IV and Group I}\label{sec:delta}
A $\delta(1-\xi)$ term is one with the phase $e^{-ik\cdot(y'-y)}$, without $\frac1\xi$. Besides the soft term of Row IV, Eq.~\eqref{eq:poleplus}, the poles of this kind are in Group I, Row II and Row V (2b, UV). Their phases are $e^{-ik\cdot(w'-w)}$ and $e^{-ik\cdot(w'-z)}$. Neither has a $\frac1\xi$; after renaming they are $e^{-ik\cdot(y'-y)}$. With the exact brackets of \texttt{Group\_I\_corrected.pdf} (G5a, G5d):
\begin{itemize}
\item \textbf{Row II} (per ordering):
\begin{equation*}
\frac{1}{(2\pi)^3}\frac{g^4N_c}{32\pi^4}\frac1{k^+}\int_{w,w',x,x'}e^{-ik\cdot(w'-w)}\frac{(x'-w')\cdot(x-w)}{(x'-w')^2(x-w)^2}\,\bar E_{x-w}\,B_{\rm II}\,[U(x')-U(w')][U(w)-U(x)]\rho\rho,
\end{equation*}
\begin{equation*}
B_{\rm II}=\frac{11}{3}+\log\frac{\Lambda^2}{k^{+2}-\Lambda^2}+\log\Big(1-\frac{k^+}{\vee}\Big)-8\delta+2\delta^2-\frac43\delta^3 .
\end{equation*}
\begin{itemize}
\item With $w\to y$, $w'\to y'$ the colour factor is $-\OO$ (numerically: ratio $-1$).
\item The prefactor is $\frac{\alpha_s}{2\pi}\cdot$$\frac{N_c}{2}$$\cdot\pre$.
\end{itemize}
\item \textbf{Row V, 2b UV} (per ordering):
\begin{equation*}
\frac{1}{(2\pi)^3}\frac{g^4N_c}{16\pi^4}\frac1{k^+}\,B_{\rm V}\int_{y,z,x',w'}e^{-ik\cdot(w'-z)}\,E_{y-z}\,\frac{(x'-w')\cdot(y-z)}{(x'-w')^2(y-z)^2}[U(x')-U(w')][U(y)-U(z)]\rho\rho,
\end{equation*}
\begin{equation*}
B_{\rm V}=2\log\frac{\vee}{\Lambda}-2\log\frac{\vee}{k^+}+2\log\Big(1-\frac{\Lambda}{k^+}\Big)-\frac{11}{6}+4\delta-\delta^2+\frac23\delta^3 .
\end{equation*}
\begin{itemize}
\item With $y\to x$, $z\to y$, $w'\to y'$ the colour factor is $+\OO$ (ratio $+1$).
\item The prefactor is $\frac{\alpha_s}{2\pi}\cdot$$N_c$$\cdot\pre$, and $E=-\bar E$.
\end{itemize}
\item \textbf{Complex conjugate.} Group I is $\mathcal C\cdot\mathcal V+{\rm c.c.}$, so both rows count twice. Row IV is $4\mathfrak C'^\dagger\mathfrak C'$ and counts once.
\item \textbf{Charge renormalization.} $\alpha_0=\alpha_s(\mu)\big[1+\frac{\alpha_s\beta_0}{4\pi}\frac1{\bar\epsilon}\big]$ multiplies the LO term by $1+\frac{\alpha_s}{2\pi}\frac{\beta_0}{2}\frac1{\bar\epsilon}$.
\end{itemize}
