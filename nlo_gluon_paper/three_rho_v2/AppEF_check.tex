\section*{Appendices E and F of \texttt{main.tex} compared with JIMWLK$\otimes$LO and BFKL$\otimes$LO}

\paragraph{Result.}
\begin{itemize}
\item \textbf{Appendix E (BFKL), Eq.~(LOBFKL): correct as written.} In the colour-singlet projection $\langle\rho^a(x)\rho^b(y)\rangle=\delta^{ab}\langle\rho^f(x)\rho^f(y)\rangle/(N_c^2-1)$ it equals BFKL$\otimes$LO of \texttt{Evolution\_from\_final.pdf}, Eq.~(6.2), exactly. This holds for symmetric and for general adjoint Wilson lines.
\item \textbf{Appendix F (JIMWLK): two corrections needed}. They are the same two as reported before, and are still in the pasted version:
\begin{enumerate}
\item In $dN^{(1)}$--$dN^{(4)}$ the matrix $(U^\dagger(x)-U^\dagger(t))(U(y)-U(t))$ (or its transpose) is written in every line. It must be $(U^\dagger(u)-U^\dagger(t))(U(v)-U(t))$ at the points $u,v$ where the two $J_R$ act, i.e.\ the points in ${\cal K}(u,v,t)$ of that line. In Eq.~(HLOfinal) this matrix sits next to $J_R^c(u)J_R^d(v)$. After the commutators, the $\delta$-functions set $u$ and $v$ to $x,w,y,w'$; the matrix must follow them.
\item The overall sign of $dN^{(5)}$ is $+$. Eq.~(HLOfinal) is right: $[(U^\dagger(u)-U^\dagger(t))U(u)T^c]_{cd}=-{\rm tr}[(U^\dagger(u)-U^\dagger(t))U(u)T^d]$. Together with $d/dY=-H$ the two minus signs give $+$.
\end{enumerate}
With both corrections, Appendix F equals JIMWLK$\otimes$LO of \texttt{Evolution\_from\_final.pdf}, Eq.~(3.2), exactly. With only one of them it does not.
\end{itemize}
The corrected Appendix F follows; changed factors are blue. The prefactor is
\begin{equation*}
P\equiv\frac{\alpha_s^2\,\delta Y}{8\pi^6(N_c^2-1)}\int_{x,y,t}{\cal K}(x-w)\cdot{\cal K}(y-w')\,\langle\rho^f(x)\rho^f(y)\rangle .
\end{equation*}

\begin{align}
\frac{dN^{(1)}}{d^2w\,d^2w'\,dy}={}&P\,\Big[
{\cal K}(y,x,t)\,{\rm Tr}\big\{T^eU^\dagger(x)U(y)T^e\,\textcolor{blue}{(U^\dagger(y)-U^\dagger(t))(U(x)-U(t))}\big\}\notag\\
&-{\cal K}(w',x,t)\,{\rm Tr}\big\{T^eU^\dagger(x)U(w')T^e\,\textcolor{blue}{(U^\dagger(w')-U^\dagger(t))(U(x)-U(t))}\big\}\notag\\
&-{\cal K}(y,w,t)\,{\rm Tr}\big\{T^eU^\dagger(w)U(y)T^e\,\textcolor{blue}{(U^\dagger(y)-U^\dagger(t))(U(w)-U(t))}\big\}\notag\\
&+{\cal K}(w',w,t)\,{\rm Tr}\big\{T^eU^\dagger(w)U(w')T^e\,\textcolor{blue}{(U^\dagger(w')-U^\dagger(t))(U(w)-U(t))}\big\}\Big]\\[4pt]
\frac{dN^{(2)}}{d^2w\,d^2w'\,dy}={}&P\,\Big[
{\cal K}(x,y,t)\,{\rm Tr}\big\{T^eU^\dagger(x)U(y)T^e\,(U^\dagger(y)-U^\dagger(t))(U(x)-U(t))\big\}\notag\\
&-{\cal K}(x,w',t)\,{\rm Tr}\big\{T^eU^\dagger(x)U(w')T^e\,\textcolor{blue}{(U^\dagger(w')-U^\dagger(t))(U(x)-U(t))}\big\}\notag\\
&-{\cal K}(w,y,t)\,{\rm Tr}\big\{T^eU^\dagger(w)U(y)T^e\,\textcolor{blue}{(U^\dagger(y)-U^\dagger(t))(U(w)-U(t))}\big\}\notag\\
&+{\cal K}(w,w',t)\,{\rm Tr}\big\{T^eU^\dagger(w)U(w')T^e\,\textcolor{blue}{(U^\dagger(w')-U^\dagger(t))(U(w)-U(t))}\big\}\Big]\\[4pt]
\frac{dN^{(3)}}{d^2w\,d^2w'\,dy}={}&-P\,\Big[
{\cal K}(y,y,t)\,{\rm Tr}\big\{T^e(U^\dagger(x)-U^\dagger(w))U(y)T^e\,\textcolor{blue}{(U^\dagger(y)-U^\dagger(t))(U(y)-U(t))}\big\}\notag\\
&-{\cal K}(w',w',t)\,{\rm Tr}\big\{T^e(U^\dagger(x)-U^\dagger(w))U(w')T^e\,\textcolor{blue}{(U^\dagger(w')-U^\dagger(t))(U(w')-U(t))}\big\}\Big]\\[4pt]
\frac{dN^{(4)}}{d^2w\,d^2w'\,dy}={}&-P\,\Big[
{\cal K}(x,x,t)\,{\rm Tr}\big\{T^eU^\dagger(x)(U(y)-U(w'))T^e\,\textcolor{blue}{(U^\dagger(x)-U^\dagger(t))(U(x)-U(t))}\big\}\notag\\
&-{\cal K}(w,w,t)\,{\rm Tr}\big\{T^eU^\dagger(w)(U(y)-U(w'))T^e\,\textcolor{blue}{(U^\dagger(w)-U^\dagger(t))(U(w)-U(t))}\big\}\Big]\\[4pt]
\frac{dN^{(5)}}{d^2w\,d^2w'\,dy}={}&\textcolor{blue}{+}P\,\Big[
{\cal K}(y,y,t)\,{\rm Tr}\big\{(U^\dagger(y)-U^\dagger(t))U(y)T^d\big\}\,{\rm Tr}\big\{(U^\dagger(x)-U^\dagger(w))U(y)T^d\big\}\notag\\
&-{\cal K}(w',w',t)\,{\rm Tr}\big\{(U^\dagger(w')-U^\dagger(t))U(w')T^d\big\}\,{\rm Tr}\big\{(U^\dagger(x)-U^\dagger(w))U(w')T^d\big\}\notag\\
&-{\cal K}(x,x,t)\,{\rm Tr}\big\{(U^\dagger(x)-U^\dagger(t))U(x)T^d\big\}\,{\rm Tr}\big\{U^\dagger(x)(U(y)-U(w'))T^d\big\}\notag\\
&+{\cal K}(w,w,t)\,{\rm Tr}\big\{(U^\dagger(w)-U^\dagger(t))U(w)T^d\big\}\,{\rm Tr}\big\{U^\dagger(w)(U(y)-U(w'))T^d\big\}\Big]
\end{align}
and $dN/d^2w\,d^2w'\,dy=\sum_{n=1}^5dN^{(n)}$.
\begin{itemize}
\item In $dN^{(1)}$ the matrix is $(U^\dagger(u)-U^\dagger(t))(U(v)-U(t))$ for ${\cal K}(u,v,t)$.
\item In $dN^{(2)}$ it is the transpose, $(U^\dagger(v)-U^\dagger(t))(U(u)-U(t))$ for ${\cal K}(u,v,t)$, because there ${\rm Tr}[WT^dT^c]M_{cd}={\rm Tr}[T^eWT^eM^T]$.
\item In $dN^{(3)}$, $dN^{(4)}$ it is $(U^\dagger(v)-U^\dagger(t))(U(v)-U(t))$ for ${\cal K}(v,v,t)$.
\item The first line of $dN^{(2)}$ was already right as printed.
\end{itemize}
