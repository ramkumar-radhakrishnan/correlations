\section{Summary}\label{sec:summary}
\paragraph{Input.} \texttt{Finite\_terms\_1.tex} is identical, byte for byte, to the \texttt{Finite\_terms.tex} of \texttt{Finite\_terms\_exact.pdf}. This note replaces that one. As there:
\begin{itemize}
\item $a\equiv\frac{k^+}{\vee}$ is kept exactly;
\item the only limit taken is $\Lambda\to0$, wherever it exists;
\item every expression is checked numerically against your file.
\end{itemize}

\paragraph{What is new.}
\begin{enumerate}
\item \textbf{Consistent with the corrected evolution terms.} The evolution is now $\NN\,\frac12\log\frac{\vee}{\Lambda}(\JJ+\BB)$ (Secs.~3--5 of \texttt{Evolution\_terms\_corrected.tex}).
\begin{itemize}
\item Every row is split exactly as $C_{\rm row}\log\frac{\vee}{k^+}+R_{\rm row}(a)$, with $R_{\rm row}$ finite for $a\to0$ (Sec.~\ref{sec:split}).
\item The coefficients add up to $\sum C_{\rm row}=\BB-\JJ$. So in every row $\log\frac{\vee}{k^+}$ in front of $C_{\rm row}$ is replaced by
\begin{equation*}
\Ls\equiv\log\frac{\sqrt{\vee\Lambda}}{k^+}=\log\frac{\vee}{k^+}-\frac12\log\frac{\vee}{\Lambda}.
\end{equation*}
This is finite and is zero at mid rapidity.
\item Summed over the rows, these terms are the remainder $\log\frac{k^+}{\sqrt{\vee\Lambda}}(\JJ-\BB)$ of Sec.~5 of the corrected evolution file (counted once).
\end{itemize}
\item \textbf{c.c.\ checked}, as you say: Group II rows are self-conjugate (no c.c.); Group I and Group III rows are not, and get $+$c.c.\ (Sec.~\ref{sec:cc}).
\item \textbf{Rows with the same kernel combined.} Their colour factors collapse:
\begin{itemize}
\item Group I Row IV, Group III Rows I, II and Group II Row I part 1 share the kernel $\KK_{\rm IV}$;
\item Group III Rows III(1), V, VI share the kernel $\KK_C$; their colour becomes one product of three dipole-like factors (Sec.~\ref{sec:families}).
\end{itemize}
\item \textbf{Three $\rho$'s reduced completely.} Every three-$\rho$ row is written as its fully symmetric three-$\rho$ part (compact) plus its two-$\rho$ part. The two-$\rho$ part is written out, merge pattern by merge pattern (Sec.~\ref{sec:three}).
\item Group II Row I part 2 is collected into one bracket (coefficients of $\lambda$, $\lambda'$, $\Ls$, $\log(1-a)$, $1$).
\end{enumerate}

\paragraph{Results.}
\begin{itemize}
\item \textbf{Running coupling} (Group I Row II part 1 $+$ Row V 2b UV, both with c.c.): the $\beta_0$ terms cancel exactly. Left over (Eq.~\eqref{eq:rc}):
\[\frac{\alpha_s}{2\pi}N_c\Big[\tfrac{67}{9}-\tfrac{\pi^2}{3}\Big]{\rm LO}\ \new{\text{(CMW constant)}},\qquad \new{-\frac{\alpha_s}{2\pi}N_c\log\Big(1-\frac{k^+}{\vee}\Big)\log\Big(\frac{\vee}{k^+}-1\Big){\rm LO}\ \text{(not running coupling)}}.\]
\new{The second term is $-\log^2(1-a)-\log\frac{\vee}{k^+}\log(1-a)$ written as one product. It comes from the limit $\xi\ge\frac{k^+}{\vee}$ of the dressed one-$\rho$ state, like the pole of Q1, and vanishes for $k^+\ll\vee$ (\texttt{Finite\_terms\_singlet.pdf}, Sec.~1).}
\item \textbf{DGLAP} (Group II Row IV, no c.c.): the pole is absorbed in $D_{h/g}$ with $\int_{k^+/\vee}^1\frac{d\xi}{\xi^2}P_{gg}(\xi)$. The finite part is Eq.~\eqref{eq:dglap}; the linear-polarization term is Eq.~\eqref{eq:lin}.
\item \textbf{Final result}: Eq.~\eqref{eq:final}.
\end{itemize}

\paragraph{Still open} (unchanged from \texttt{Finite\_terms\_exact.pdf}; your decision is needed, Sec.~\ref{sec:open}):
\begin{itemize}
\item \textbf{Q1:} the pole $\frac{\alpha_s}{2\pi}\frac2\epsilon N_c\log(1-\frac{k^+}{\vee}){\rm LO}$ is left over;
\item \textbf{Q2:} the four-$\rho$ finite term.
\end{itemize}

\subsection{The c.c.}\label{sec:cc}

\begin{equation*}
T+{\rm c.c.},\qquad {\rm c.c.}\equiv\Big[T\Big]^*_{w\leftrightarrow w'}\quad(\text{resp. }w\leftrightarrow y').
\end{equation*}
\begin{itemize}
\item $^*$ means $i\to-i$ in the prefactors, $E_1(i\beta)\to E_1(-i\beta)$, $\log(i\kappa)\to\log(-i\kappa)$, and the order of the $\rho$'s reversed.
\item The exchange of the two points of the measured gluon restores the phase.
\end{itemize}
