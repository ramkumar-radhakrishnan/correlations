\noindent Your equation (Group II Row IV, no c.c.). $a\equiv\frac{k^+}{\vee}$, $\alpha_s=\frac{g^2}{4\pi}$, $d_\perp=2-\epsilon$, $\beta_0=\frac{11}{3}N_c$ ($n_f=0$), $c_0=2e^{-\gamma_E}$, $\Delta\equiv y'-y$, $\ell_\Delta(\mu)\equiv\log\frac{\Delta^2\mu^2}{c_0^2}=2\gamma_E+\log\frac{(y'-y)^2\mu^2}{4}$.

\section{Answer}
\begin{center}\fbox{\begin{minipage}{0.97\textwidth}
\begin{itemize}
\item \textbf{DGLAP.} The pole of the row is $\frac{\alpha_s}{2\pi}\big(-\frac2\epsilon\big)P^{(r)}_{gg,+}\otimes\LO$, where $P^{(r)}_{gg,+}$ is the real part of $P_{gg}$. Write $P^{(r)}_{gg,+}=P_{gg}-\frac{\beta_0}{2}\delta(1-\xi)$.
\begin{itemize}
\item The $P_{gg}$ part is the DGLAP pole. It is absorbed in the fragmentation function $D_{h/g}(\zeta,\mu_F)$, which then obeys $\mu_F^2\partial_{\mu_F^2}D=\frac{\alpha_s}{2\pi}P_{gg}\otimes D$.
\item The $-\frac{\beta_0}{2}\delta(1-\xi)$ part is left in the row.
\end{itemize}
\item \textbf{Running coupling.} The $\frac{\beta_0}{2}$ term left in the row is $+\frac{\alpha_s\beta_0}{4\pi}\LO[\frac2\epsilon+\ell_\Delta(\mu)]$. Its pole is cancelled exactly by the $\overline{MS}$ counterterm of the coupling of LO. What remains turns the coupling of LO into a running one:
\begin{equation*}
\LO\big|_{\alpha_s(\mu_R)}+\frac{\alpha_s\beta_0}{4\pi}\LO\big[\ell_\Delta(\mu_R)\big]=\LO\Big[\alpha_s\Big(\frac{c_0}{|y'-y|}\Big)\Big]+O(\alpha_s^3).
\end{equation*}
The coupling runs at the transverse distance $|y'-y|$, which is conjugate to $\mathbf k$ through $e^{-ik\cdot(y'-y)}$. $\mu_R$ drops out.
\item \textbf{Finite terms of the row:}
\begin{itemize}
\item the DGLAP coefficient $-\frac{\alpha_s}{2\pi}\int_{k^+/\vee}^1\frac{d\xi}{\xi^2}P_{gg}(\xi)\,\LO_\xi[\ell_\Delta(\mu_F)]$;
\item $-\log\frac{\vee}{k^+}H_4$, from (L2)$+$(L3);
\item the linearly polarized line (L5) at $d_\perp=2$.
\end{itemize}
Eq.~\eqref{eq:final} gives all of them in your notation.
\end{itemize}
\textbf{Other $\beta_0$ terms.} Group I Row II part 1 ($\frac{11}{3}$) and Group I Row V $d^3N_{2b\,\rm UV}$ ($-\frac{11}{6}$) have $\beta_0$ terms of their own. These cancel each other exactly, including their poles and $\mu$ (\texttt{Other\_terms\_analysis.pdf}, Sec.~5). Keep both or drop both: in your latest \texttt{Other\_terms.tex}, Row II part 1 is commented out but Row V $2b$ UV is not.
\end{minipage}}\end{center}

\section{Notation}
\begin{align}
\LO[f]&\equiv\frac{1}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{1}{k^{+}}\int_{x,y,x',y'}e^{-ik\cdot(y'-y)}\,f\,\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\left[U^{e'c}(y')-U^{e'c}(x')\right]\left[U^{ec}(y)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x),\label{eq:LO}
\end{align}
$\LO\equiv\LO[1]$. $\LO_\xi[f]$ is the same with $e^{-ik\cdot(y'-y)}\to e^{-ik\cdot(y'-y)\frac1\xi}$, so $\LO_\xi=\frac1\xi\frac{d^3N_{\rm LO}}{d^3k}(\frac{k^+}{\xi},\frac{\mathbf k}{\xi})$ ($d^3k=dk^+d^2\mathbf k$).
\begin{equation}
P^{(r)}_{gg,+}(\xi)=2N_c\Big[\frac{\xi}{(1-\xi)_+}+\frac{1-\xi}{\xi}+\xi(1-\xi)\Big],\qquad P_{gg}(\xi)=P^{(r)}_{gg,+}(\xi)+\frac{\beta_0}{2}\,\delta(1-\xi).\label{eq:P}
\end{equation}
Here $\frac{1}{(1-\xi)_+}$ is the plus distribution of $[0,1]$. On $\xi\ge a$:
\begin{equation*}
\int_a^1d\xi\frac{g(\xi)}{(1-\xi)_+}=\int_a^1d\xi\frac{g(\xi)-g(1)}{1-\xi}+g(1)\log(1-a).
\end{equation*}
\begin{align}
H_4\equiv{}&\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}\int_{x,x',z,w,w'}e^{-ik\cdot(w'-w)}\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}(w'-z)^{2}}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}\left[U^{e'c}(z)-U^{e'c}(x')\right]\left[U^{ec}(z)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x).\label{eq:H4}
\end{align}

\section{Step by step}
\paragraph{Step 0: your row, line by line.}
\begin{align}
(L1)\quad &   \frac{g^{2}}{4\pi^{2}}\, \frac{N_{c}}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\, \frac{1}{k^{+}}\,\int_{x,y,x',y'}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\left[-\frac{d_{\perp}}{\epsilon}-d_{\perp}\,\gamma_{E}-\frac{d_{\perp}}{2}\,\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)}-1\right]\,\notag \\ & \Bigg[\int_{\frac{k^{+}}{\vee}}^{1}\,\frac{d\xi}{\xi^{2}}\,e^{-ik\cdot (y'-y)\,\frac{1}{\xi}}\,\left[\xi\,(1-\xi)+\frac{(1-\xi)}{\xi}+\xi+\frac{\xi^{2}}{(1-\xi)_{+}}\right]\Bigg]\times \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag\\
(L2)\quad&   -\,\frac{g^{2}}{4\pi^{2}}\, \frac{N_{c}}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\, \frac{1}{k^{+}}\,\int_{x,y,x',y'}\,\frac{(x-y)\cdot (x'-y)}{(x-y)^{2}\,(x'-y)^{2}}\,\left[-\frac{d_{\perp}}{2}\,\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)}\right]\,\notag \\ &  \Bigg[\int_{\frac{k^{+}}{\vee}}^{1}\,\frac{d\xi}{\xi^{2}}\,e^{-ik\cdot (y'-y)\,\frac{1}{\xi}}\,\left[\frac{(1-\xi)}{\xi}\right]\Bigg]\times \left[U^{e'c}(y) - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag\\
(L3)\quad&   -\frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,N_{c}}{8\pi^{5}}\,\frac{1}{k^{+}}\,\left(1-\frac{k^{+}}{\vee}\right)\,\int_{x,x',z,w,w'}\,e^{-ik\cdot (w'-w)}\,\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}\,(w'-z)^{2}}\,\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}\,(x'-z)^{2}}\notag\\
& \left[U^{e'c}(z) - U^{e'c}(x')\right] \times\, \left[U^{ec}(z) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag\\
(L4)\quad&  +\, \frac{g^{2}}{4\pi^{2}}\, \frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\, \frac{N_{c}}{k^{+}}\,\log{\left(1-\frac{k^{+}}{\vee}\right)}\,\int_{x,y,x',y'}\,e^{-ik\cdot (y'-y)}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\bigg[-\frac{d_{\perp}}{\epsilon}-d_{\perp}\,\gamma_{E}\notag \\ & -\frac{d_{\perp}}{2}\,\log{\left(\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right)}-1\bigg]\,\left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag\\
(L5)\quad& +\,\frac{g^{2}}{4\pi^{2}}\, \frac{1}{(2\pi)^{3}}\,\frac{g^{2}}{2\pi^{2}}\, \frac{N_{c}}{k^{+}}\,\int_{\frac{k^{+}}{\vee}}^{1}\,d\xi\, \frac{(1-\xi)}{\xi}\, \int_{x,y,x',y'}\,e^{-ik\cdot (y'-y)\frac{1}{\xi}}\,\Bigg[\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\, (x'-y')^{2}} -\frac{(x-y)\cdot (y'-y)\, (x'-y')\cdot (y'-y)}{(x-y)^{2}\,(x'-y')^{2}\,(y'-y)^{2}}\,d_{\perp}\Bigg]\times\notag \\ &  \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)
\end{align}

\paragraph{Step 1: units of LO.} $\frac{g^2}{4\pi^2}=\frac{\alpha_s}{\pi}$. The $x,y,x',y'$ integrands of (L1) and (L4) are those of \eqref{eq:LO}, (L1) with $e^{-ik\cdot\Delta}\to e^{-ik\cdot\Delta/\xi}$. With $\mathcal B_\Delta$ your bracket:
\begin{equation}
({\rm L1})=\frac{\alpha_sN_c}{\pi}\int_a^1\frac{d\xi}{\xi^2}\Big[\xi(1-\xi)+\frac{1-\xi}{\xi}+\xi+\frac{\xi^2}{(1-\xi)_+}\Big]_{\rm yours}\LO_\xi[\mathcal B_\Delta],\qquad
({\rm L4})=\frac{\alpha_sN_c}{\pi}\log(1-a)\,\LO[\mathcal B_\Delta].
\end{equation}

\paragraph{Step 2: the bracket.} With $d_\perp=2-\epsilon$,
\begin{equation}
\mathcal B_\Delta=-\frac{d_\perp}{\epsilon}-d_\perp\gamma_E-\frac{d_\perp}{2}\log\frac{\Delta^2\mu^2}{4}-1=-\frac2\epsilon+1-2\gamma_E-\log\frac{\Delta^2\mu^2}{4}-1+O(\epsilon)=-\frac{2}{\epsilon}-\ell_\Delta(\mu)+O(\epsilon).\label{eq:B}
\end{equation}
The $-1$ cancels the $+1$ of $-\frac{d_\perp}{\epsilon}$. The $O(\epsilon)$ multiplies a finite integral and drops out.

\paragraph{Step 3: (L4) completes the plus distribution.} Your prescription, $\int\frac{d\xi}{\xi^2}\frac{\xi^2}{(1-\xi)_+}G=\int\frac{d\xi}{1-\xi}[G(\xi)-G(1)]$, has no $\log(1-a)$. On $[a,1]$:
\begin{equation*}
\int_a^1\frac{d\xi}{\xi^2}\Big[\xi+\frac{\xi^2}{(1-\xi)_+}\Big]_{\rm yours}G=\int_a^1\frac{d\xi}{\xi^2}\frac{\xi}{(1-\xi)_+}G-\log(1-a)\,G(1),
\end{equation*}
with $\LO_{\xi=1}=\LO$ (\texttt{other\_check/checks.py}, item 3). (L4) adds the missing term:
\begin{equation}
({\rm L1})+({\rm L4})=\frac{\alpha_s}{2\pi}\int_a^1\frac{d\xi}{\xi^2}\,P^{(r)}_{gg,+}(\xi)\,\LO_\xi\Big[-\frac2\epsilon-\ell_\Delta(\mu)\Big].\label{eq:L14}
\end{equation}

\paragraph{Step 4: split $P^{(r)}_{gg,+}$.} By \eqref{eq:P}, $P^{(r)}_{gg,+}=P_{gg}-\frac{\beta_0}{2}\delta(1-\xi)$:
\begin{equation}
({\rm L1})+({\rm L4})=\underbrace{\frac{\alpha_s}{2\pi}\int_a^1\frac{d\xi}{\xi^2}\,P_{gg}(\xi)\,\LO_\xi\Big[-\frac2\epsilon-\ell_\Delta(\mu)\Big]}_{\rm DGLAP}\;+\;\underbrace{\frac{\alpha_s\beta_0}{4\pi}\,\LO\Big[\frac2\epsilon+\ell_\Delta(\mu)\Big]}_{\rm running\ coupling}.\label{eq:split}
\end{equation}

\paragraph{Step 5: DGLAP.} In the first term, $\int_a^1\frac{d\xi}{\xi^2}P\,\LO_\xi=\int_{k^+/\vee}^1\frac{d\xi}{\xi^3}P(\xi)\frac{d^3N_{\rm LO}}{d^3k}(\frac{k^+}{\xi},\frac{\mathbf k}{\xi})$: a gluon of momentum $\frac{k}{\xi}$, produced at LO, turns into one with the fraction $\xi$. For a hadron,
\begin{equation}
\frac{d^3N_h}{d^3k}=\int_{k^+/\vee}^1\frac{d\zeta}{\zeta^3}\,D^{\rm bare}_{h/g}(\zeta)\,\frac{d^3N_g}{d^3k}\Big(\frac{k^+}{\zeta},\frac{\mathbf k}{\zeta}\Big).
\end{equation}
\begin{itemize}
\item With $z=\zeta\xi$, the region $a\le\zeta\le1$, $\frac a\zeta\le\xi\le1$ is $a\le z\le\zeta\le1$. This needs the plus distribution of Step 3, i.e.\ (L4).
\item The pole becomes $\frac{\alpha_s}{2\pi}(-\frac2\epsilon)\int_a^1\frac{dz}{z^3}\frac{d^3N_{\rm LO}}{d^3k}(\frac kz)\,[P_{gg}\otimes D](z)$, with $[P\otimes D](z)=\int_z^1\frac{d\zeta}{\zeta}P(\frac z\zeta)D(\zeta)$ (\texttt{other\_check/checks.py}, item 5: $2\times10^{-12}$).
\item In $\overline{MS}$ at $\mu_F$ it is removed by
\begin{equation}
D^{\rm bare}_{h/g}(z)=D_{h/g}(z,\mu_F)+\frac{\alpha_s}{2\pi}\Big[\frac2\epsilon+\log\frac{\mu^2}{\mu_F^2}\Big][P_{gg}\otimes D_{h/g}](z,\mu_F),\qquad\mu_F^2\frac{\partial D_{h/g}}{\partial\mu_F^2}=\frac{\alpha_s}{2\pi}P_{gg}\otimes D_{h/g}.\label{eq:D}
\end{equation}
\end{itemize}
What the DGLAP term leaves in the gluon cross section ($\ell_\Delta(\mu)=\ell_\Delta(\mu_F)+\log\frac{\mu^2}{\mu_F^2}$):
\begin{equation}
\boxed{\frac{d^3N}{d^3k}\bigg|_{\rm DGLAP}=-\frac{\alpha_s}{2\pi}\int_{k^+/\vee}^1\frac{d\xi}{\xi^2}\,P_{gg}(\xi)\,\LO_\xi\big[\ell_\Delta(\mu_F)\big].}\label{eq:dglap}
\end{equation}
Its $\mu_F$ dependence cancels that of $D_{h/g}$ through \eqref{eq:D}.

\paragraph{Step 6: running coupling.} LO is $\propto g^2$. In $\overline{MS}$, $g^2\to g^2(\mu_R)\big[1-\frac{\alpha_s\beta_0}{4\pi}(\frac2\epsilon+\log\frac{\mu^2}{\mu_R^2})\big]$ adds $-\frac{\alpha_s\beta_0}{4\pi}\big(\frac2\epsilon+\log\frac{\mu^2}{\mu_R^2}\big)\LO$. Added to the second term of \eqref{eq:split}:
\begin{equation}
\frac{\alpha_s\beta_0}{4\pi}\LO\Big[\frac2\epsilon+\ell_\Delta(\mu)\Big]-\frac{\alpha_s\beta_0}{4\pi}\Big(\frac2\epsilon+\log\frac{\mu^2}{\mu_R^2}\Big)\LO=\boxed{\frac{\alpha_s\beta_0}{4\pi}\,\LO\Big[\log\frac{(y'-y)^2\mu_R^2}{c_0^2}\Big].}\label{eq:rc}
\end{equation}
The pole and $\mu$ cancel. With $\alpha_s(Q)=\alpha_s(\mu_R)\big[1-\frac{\alpha_s\beta_0}{4\pi}\log\frac{Q^2}{\mu_R^2}\big]+O(\alpha_s^3)$ and $Q=\frac{c_0}{|y'-y|}$:
\begin{equation}
\LO\big|_{\alpha_s(\mu_R)}+\frac{\alpha_s\beta_0}{4\pi}\LO\big[\ell_\Delta(\mu_R)\big]=\LO\Big[\alpha_s\Big(\frac{c_0}{|y'-y|}\Big)\Big]+O(\alpha_s^3).
\end{equation}
\begin{itemize}
\item This is the running of the coupling of LO. The scale is the separation $|y'-y|$ of the measured gluon in the amplitude and in the conjugate amplitude, which is conjugate to $\mathbf k$; $\mu_R$ has dropped out.
\item If you prefer a fixed scale: $\eqref{eq:rc}=\frac{\alpha_s\beta_0}{4\pi}\log\frac{\mu_R^2}{\mu_F^2}\LO+\frac{\alpha_s\beta_0}{4\pi}\LO[\ell_\Delta(\mu_F)]$. The first term is the renormalization-group term of \texttt{Other\_terms\_analysis.pdf}; the second combines with \eqref{eq:dglap} into $-\frac{\alpha_s}{2\pi}P^{(r)}_{gg,+}\otimes\LO[\ell_\Delta(\mu_F)]$. The two groupings are identical (sympy check of the pole and $\mu$ bookkeeping in \texttt{make\_note.py}).
\end{itemize}

\paragraph{Step 7: (L2)$+$(L3).} In (L2), $y'$ appears only in $e^{-ik\cdot\Delta/\xi}$ and the log, and $\frac{d_\perp}{2}\to1$ (no pole).
\begin{itemize}
\item $\int_{y'}e^{-ik\cdot(y'-y)\frac1\xi}\big[-\log\frac{(y'-y)^2\mu^2}{4}\big]=\frac{4\pi\xi^2}{\mathbf k^2}$ for $\mathbf k\ne0$; the $\mu$ term multiplies $\delta^{(2)}(\mathbf k)$ and drops out.
\item $\int_a^1\frac{d\xi}{\xi^2}\frac{1-\xi}{\xi}\xi^2=\log\frac{\vee}{k^+}-(1-\frac{k^+}{\vee})$.
\item $\int_{w,w'}e^{-ik\cdot(w'-w)}\frac{(w-z)\cdot(w'-z)}{(w-z)^2(w'-z)^2}=\frac{(2\pi)^2}{\mathbf k^2}$.
\end{itemize}
With $y\to z$ the prefactors agree, so
\begin{equation}
({\rm L2})=-\Big[\log\frac{\vee}{k^+}-\Big(1-\frac{k^+}{\vee}\Big)\Big]H_4,\qquad({\rm L3})=-\Big(1-\frac{k^+}{\vee}\Big)H_4,\qquad({\rm L2})+({\rm L3})=-\log\frac{\vee}{k^+}\,H_4.
\end{equation}
This cancels the $+\log\frac{\vee}{k^+}H_4$ of the BFKL list of this row and makes \eqref{eq:dglap} finite for $\vee\to\infty$.

\paragraph{Step 8: (L5).} The linear-polarization line has no pole, so $d_\perp=2$ in it.
\begin{itemize}
\item The $\Delta$ angular average of its bracket is zero, so it is not DGLAP; its Fourier transform carries $\frac{2k^ik^j}{\mathbf k^2}-\delta^{ij}$.
\item Its measure is $d\xi$, not $\frac{d\xi}{\xi^2}$, so it is finite for $\vee\to\infty$.
\end{itemize}

\section{Result: the row after DGLAP and running coupling}
After $D_{h/g}(\zeta,\mu_F)$ takes the $P_{gg}$ pole and the coupling counterterm takes the $\beta_0$ pole, your row is ($\epsilon\to0$ exact in what is finite)
\begin{align}
\frac{d^{3}N^{\rm G2\,IV}}{d^{3}k}={}&\new{-\frac{\alpha_s}{2\pi}\int_{k^+/\vee}^1\frac{d\xi}{\xi^2}\,P_{gg}(\xi)\,\LO_\xi\big[\ell_\Delta(\mu_F)\big]}\qquad\text{(DGLAP, finite part)}\notag\\
&\new{+\frac{\alpha_s\beta_0}{4\pi}\,\LO\big[\ell_\Delta(\mu_R)\big]}\qquad\text{(running coupling: $\alpha_s(\mu_R)\to\alpha_s(c_0/|y'-y|)$ in LO)}\notag\\
&-\log\frac{\vee}{k^+}\,H_4\;+\;({\rm L5})_{d_\perp=2}.\label{eq:final}
\end{align}
In your notation:
\begin{align}
\frac{d^{3}N^{\rm G2\,IV}}{d^{3}k}={}&\frac{g^{2}}{4\pi^{2}}\frac{N_{c}}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{1}{k^{+}}\int_{x,y,x',y'}\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\left[-2\gamma_{E}-\log\frac{(y'-y)^{2}\mu_{F}^{2}}{4}\right]\notag\\
&\quad\times\Bigg[\int_{\frac{k^{+}}{\vee}}^{1}\frac{d\xi}{\xi^{2}}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\left[\xi(1-\xi)+\frac{(1-\xi)}{\xi}+\xi+\frac{\xi^{2}}{(1-\xi)_{+}}\right]+\Big(\log\Big(1-\frac{k^{+}}{\vee}\Big)\new{+\frac{11}{12}}\Big)e^{-ik\cdot(y'-y)}\Bigg]\notag\\
&\quad\times\left[U^{e'c}(y')-U^{e'c}(x')\right]\left[U^{ec}(y)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\notag\\
&\new{+\frac{g^{2}}{4\pi^{2}}\frac{N_{c}}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{1}{k^{+}}\,\frac{11}{12}\int_{x,y,x',y'}e^{-ik\cdot(y'-y)}\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\left[2\gamma_{E}+\log\frac{(y'-y)^{2}\mu_{R}^{2}}{4}\right]}\notag\\
&\quad\new{\times\left[U^{e'c}(y')-U^{e'c}(x')\right]\left[U^{ec}(y)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)}\notag\\
&-\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{8\pi^{5}}\frac{1}{k^{+}}\log\Big(\frac{\vee}{k^{+}}\Big)\int_{x,x',z,w,w'}e^{-ik\cdot(w'-w)}\frac{(w-z)\cdot(w'-z)}{(w-z)^{2}(w'-z)^{2}}\frac{(x-z)\cdot(x'-z)}{(x-z)^{2}(x'-z)^{2}}\notag\\
&\quad\times\left[U^{e'c}(z)-U^{e'c}(x')\right]\left[U^{ec}(z)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x)\notag\\
&+\frac{g^{2}}{4\pi^{2}}\frac{1}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{N_{c}}{k^{+}}\int_{\frac{k^{+}}{\vee}}^{1}d\xi\frac{(1-\xi)}{\xi}\int_{x,y,x',y'}e^{-ik\cdot(y'-y)\frac{1}{\xi}}\Bigg[\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}-\new{2}\frac{(x-y)\cdot(y'-y)\,(x'-y')\cdot(y'-y)}{(x-y)^{2}(x'-y')^{2}(y'-y)^{2}}\Bigg]\notag\\
&\quad\times\left[U^{e'c}(y')-U^{e'c}(x')\right]\left[U^{ec}(y)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x).\label{eq:final2}
\end{align}
\begin{itemize}
\item Lines 1--3: the DGLAP term \eqref{eq:dglap}. With the prefactor $\frac{g^2N_c}{4\pi^2}=\frac{\alpha_s}{2\pi}2N_c$, the \new{$\frac{11}{12}$} is $\frac{\beta_0}{2}\cdot\frac{1}{2N_c}$, the $\delta(1-\xi)$ of $P_{gg}$.
\item Lines 4--5: the running coupling \eqref{eq:rc}, $\frac{\alpha_s\beta_0}{4\pi}=\frac{g^2N_c}{4\pi^2}\cdot\frac{11}{12}$.
\item Lines 6--7: (L2)$+$(L3).
\item Lines 8--9: (L5) with $d_\perp=2$.
\end{itemize}
The two \new{$\frac{11}{12}$} terms cancel when $\mu_R=\mu_F$; their sum is always $\frac{\alpha_s\beta_0}{4\pi}\log\frac{\mu_R^2}{\mu_F^2}\LO$.
