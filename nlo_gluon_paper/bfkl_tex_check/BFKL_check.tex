\paragraph{Method.} Every equation is evaluated pointwise: random transverse points, random classical charges, $U=U^T$, $\LV=1$, and $\epsilon=0.37$ kept finite in the pole brackets.
\begin{itemize}
\item Three-$\rho$ first forms are compared with the reference after symmetrization over the charges (with the c.c.\ for Groups I, III).
\item The ``This can be written as'' lists are compared with the two-$\rho$ (commutator) terms of the first form, computed directly. With the charges as operators, a product $ABC$ of three charges is its symmetric part plus $\tfrac12([A,B]C+[B,C]A+[A,C]B)$, with $[\rho^a(x),\rho^b(y)]=i f^{abc}\rho^c(x)\,\delta^{(2)}(x-y)$; anticommutators are kept ordered, $\{A,B\}=AB+BA$, and the c.c.\ reverses the order.
\item This rule was checked in two ways. It reproduces the reordering engine used before term by term (6 rows, $\le2\times10^{-16}$). It also reproduces the checked two-$\rho$ lists of \texttt{Evolution\_terms.tex} from the $\log\frac{k^+}{\Lambda}$ three-$\rho$ terms (Group II Row I, Group III Rows I, II: exact).
\item Index sums ($\delta^{ij}$, $(x-w)^i\ldots$) are done term by term. A term without the index is not summed over it.
\end{itemize}

\item[C1] \textbf{Group I, Row II, first form.} The equation as written is a copy of Group~I Row~III, $d^3N_1$ (identical text). The $\LV$ terms of Row~II are (\texttt{Checks\_JIMWLK\_BFKL.pdf}, eq.\ (B:G1II2a)):
\begin{align}
\frac{d^{3}N_{2\,a}}{d^{3}k} & = -\frac{1}{(2\pi)^{3}}\,\frac{i\,g^{4}\,f^{hbc}}{32\pi^{5}}\,\frac{1}{k^{+}}\,\log\Big(\frac{\vee}{k^{+}}\Big)\int_{w,w'} e^{-ik\cdot(w'-w)}\int_{x,y,x',z}\frac{(x'-w')^{i}}{(x'-w')^{2}}\,\Bigg[\frac{(x-w)^{i}}{(x-w)^{2}}\, \frac{(w-z)\cdot (y-z)}{(w-z)^{2}\,(y-z)^{2}}- \frac{(y-w)^{i}}{(y-w)^{2}}\, \frac{(w-z)\cdot (x-z)}{(w-z)^{2}\,(x-z)^{2}}\Bigg]\notag\\
&\qquad\times\Big[U^{ab'}(x')\,\rho^{b'}(x')-U^{ab'}(w')\,\rho^{b'}(x')\Big]\, \Big[U^{ah}(w)\,\{\rho^{b}(x),\rho^{c}(y)\}\Big]\notag \\
& -\frac{1}{(2\pi)^{3}}\,\frac{i\,g^{4}\,f^{hbc}}{32\pi^{4}}\,\frac{1}{k^{+}}\,\log\Big(\frac{\vee}{k^{+}}\Big)\int_{w,w'} e^{-ik\cdot(w'-w)}\int_{x,y,x'}\frac{(x'-w')^{i}}{(x'-w')^{2}}\,\Bigg[\frac{(x-w)^{i}}{(x-w)^{2}}\,\log\frac{(w-y)^{2}}{(x-y)^{2}}-\frac{(y-w)^{i}}{(y-w)^{2}}\,\log\frac{(w-x)^{2}}{(x-y)^{2}}\Bigg]\notag \\

\end{align}
The logarithmic term has no $z$ left, so its measure is $\int_{x,y,x'}$. Your ``This can be written as'' list for Row~II is exactly the two-$\rho$ part of these terms ($1.9\times10^{-15}$), so only the displayed first form changes.
\item[C2] \textbf{Group II, Row IV, third line} ($H_4$): the sign at the start of the line is missing; it is \new{$+$}. Without it LaTeX prints the line as a product. It was also missing in my \texttt{Checks\_JIMWLK\_BFKL.pdf}; corrected there.
\item[C3] \textbf{Colour-singlet form:} the measure is $\int_{x,\new{y},z,w,w'}$ ($y$ is used, there is no $x'$).
\end{description}
