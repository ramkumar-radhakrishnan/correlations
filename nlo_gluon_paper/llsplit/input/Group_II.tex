     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{4}}\,\frac{1}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\int_{z}\,e^{-ik\,\left(y'-y\right)\, \frac{(k^{+}+p^{+})}{k{+}}}\,\int_{x,y,y^{\prime},x'}\,\frac{(y-z)^{m}\,(x-y)^{k}\,(y'-z)^{m'}\,(x'-y')^{k'}}{(y-z)^{2}\,(x-y)^{2}\,(y'-z)^{2}\,(x'-y')^{2}}\notag \\ &  \bigg[\frac{\delta_{km}\delta_{k'm'}\,d_{\perp}\,p^{+}}{(k^{+}+p^{+})^{2}}+\frac{2}{k^{+}}\,\left(\delta_{mk'}\delta_{m'k}-\delta_{m'k'}\delta_{km}\right) + \frac{p^{+}}{k^{+\,2}}\,\delta_{mm'}\delta_{kk'}+\frac{1}{p^{+}}\,\delta_{mm'}\delta_{kk'}\bigg]\,\times\,\notag \\ & \left[f^{c'd'a}\,U^{e'c'}(y')\,U^{d'b}(z)\,\rho^{e'}(x')-f^{c'd'a}\,

    U^{bd}(z)\,U^{ce}(x)\,\rho^{e}(x)\right]

Reducing Wilson lines, we get
\begin{align}
     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = \frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{4}}\,\frac{N_{c}}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\int_{z}\,e^{-ik\,\left(y'-y\right)}\,e^{-ik\,\left(y'-y\right)\frac{p^{+}}{k^{+}}}\,\int_{x,y,y^{\prime},x'}\,\frac{(y-z)^{m}\,(x-y)^{k}\,(y'-z)^{m'}\,(x'-y')^{k'}}{(y-z)^{2}\,(x-y)^{2}\,(y'-z)^{2}\,(x'-y')^{2}}\notag \\ &  \bigg[\frac{\delta_{km}\delta_{k'm'}\,d_{\perp}\,p^{+}}{(k^{+}+p^{+})^{2}}+\frac{2}{k^{+}}\,\left(\delta_{mk'}\delta_{m'k}-\delta_{m'k'}\delta_{km}\right) + \frac{p^{+}}{k^{+\,2}}\,\delta_{mm'}\delta_{kk'}+\frac{1}{p^{+}}\,\delta_{mm'}\delta_{kk'}\bigg]\,\times\,\notag \\ & \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)
\end{align}
We are performing $z$ integral and using $\overline{MS}$ and $d_{\perp} = 2-\epsilon$, we get
\begin{align}
    \int_{z}\, \frac{(y-z)^{m}\,(y'-z)^{m'}}{(y-z)^{2}\,(y'-z)^{2}} = -\frac{\pi}{2}\, \left[\frac{2}{\epsilon} +2\gamma_{E} + \log{\left[\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right]}\right]\delta^{mm'}\, - \pi\, \frac{(y'-y)^{m}\,(y'-y)^{m'}}{(y'-y)^{2}}
\end{align}
This gives us
\begin{align}
     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = -\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,d_{\perp}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\,\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{p^{+}+k^{+}}{k^{+}}}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\notag \\ &  \bigg[\frac{p^{+}}{(k^{+}+p^{+})^{2}} + \frac{p^{+}}{k^{+\,2}}+\frac{1}{p^{+}}\bigg]\,\times\left[\frac{1}{\epsilon} +\gamma_{E} +\frac{1}{2}\,\log{\left[\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right]}\right]\notag \\ &  \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag \\ & -\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\,\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{p^{+}}{(p^{+}+k^{+})}}\,\frac{(x-y)^{k}\,(x'-y')^{k'}\,(y'-y)^{m}\,(y'-y)^{m'}}{(x-y)^{2}\,(x'-y')^{2}\,(y'-y)^{2}}\notag \\ &  \bigg[\frac{\delta_{km}\delta_{k'm'}\,d_{\perp}\,p^{+}}{(k^{+}+p^{+})^{2}}+\frac{2}{k^{+}}\,\left(\delta_{mk'}\delta_{m'k}-\delta_{m'k'}\delta_{km}\right) + \frac{p^{+}}{k^{+\,2}}\,\delta_{mm'}\delta_{kk'}+\frac{1}{p^{+}}\,\delta_{mm'}\delta_{kk'}\bigg]\,\times\notag \\ &  \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)
\end{align}
Simplifying, we get
\begin{align}
     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = -\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,d_{\perp}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{p^{+}+k^{+}}{k^{+}}}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\bigg[\frac{p^{+}}{(k^{+}+p^{+})^{2}} + \frac{p^{+}}{k^{+\,2}}+\frac{1}{p^{+}}\bigg]\,\times\notag \\ & \left[\frac{1}{\epsilon} +\gamma_{E} +\frac{1}{2}\,\log{\left[\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right]}\right]\,\left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag \\ & -\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\Lambda}^{\vee-k^{+}}\, \frac{dp^{+}}{2\pi}\,\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{p^{+}+k^{+}}{k^{+}}}\,\Bigg[\frac{(x-y)\cdot (y'-y)\, (x'-y')\cdot (y'-y)}{(x-y)^{2}\,(x'-y')^{2}\,(y'-y)^{2}}\,\frac{d_{\perp}\,p^{+}}{(k^{+}+p^{+})^{2}}\notag \\ & + \frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\left[\frac{p^{+}}{k^{+\,2}} + \frac{1}{p^{+}}\right]\Bigg]\,\times \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)
\end{align}
Using $\xi = \frac{k^{+}}{(p^{+}+k^{+})}$, we get
\begin{align}
     \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & =  \frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,d_{\perp}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\frac{k^{+}}{k^{+}+\Lambda}}^{\frac{k^{+}}{\vee}}\, \frac{d\xi}{2\pi}\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{1}{\xi}}\,\frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\bigg[\xi\,(1-\xi)+\frac{(1-\xi)}{\xi}+\frac{\xi}{(1-\xi)}\bigg]\,\frac{1}{\xi^{2}}\notag \\ & \left[\frac{1}{\epsilon} +\gamma_{E} +\frac{1}{2}\,\log{\left[\frac{(y'-y)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right]}\right]\,\left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)\notag \\ & +\,\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{4\pi^{3}}\,\frac{N_{c}}{k^{+}}\,\int_{\frac{k^{+}}{k^{+}+\Lambda}}^{\frac{k^{+}}{\vee}}\, \frac{d\xi}{2\pi}\,\int_{x,y,y^{\prime},x'}\,e^{-ik\,\left(y'-y\right)\,\frac{1}{\xi}}\,\frac{1}{\xi^{2}}\Bigg[\frac{(x-y)\cdot (y'-y)\, (x'-y')\cdot (y'-y)}{(x-y)^{2}\,(x'-y')^{2}\,(y'-y)^{2}}\,d_{\perp}\,\xi(1-\xi)\notag \\ & + \frac{(x-y)\cdot (x'-y')}{(x-y)^{2}\,(x'-y')^{2}}\,\left[\frac{(1-\xi)}{\xi}+\frac{\xi}{(1-\xi)}\right]\Bigg]\,\times \left[U^{e'c}(y') - U^{e'c}(x')\right] \times\, \left[U^{ec}(y) - U^{ce}(x)\right]\, \rho^{e'}(x') \rho^{e}(x)
\end{align}
