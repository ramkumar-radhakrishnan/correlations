\section{Terms proportional to $\log{\left(\frac{\vee}{\Lambda}\right)}$}
%The terms proportional to $\log{\left(\frac{\vee}{\Lambda}\right)}$ play a special role: their origin is due to small-x evolution of the target and the projectile. Neither the JIMWLK evolution of the target nor the BFKL evolution of the projectile can result in more than two $\rho$ contributions. Therefore the terms involving $\log{\left(\frac{\vee}{\Lambda}\right)}$ and four or three $\rho$ should identically cancel.  Below we will consider these terms.
\subsection{Four $\rho$ contributions}
\begin{align} \label{eq:4rho}
    \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = \frac{1}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{3}}\frac{\log{\left(\frac{\vee}{\Lambda}\right)}}{2\pi^{2}} \frac{1}{k^{+}}\int_{x,y,z,u,x^{\prime},w,w^{\prime}}e^{-ik(w^{\prime}-w)}\frac{(x^{\prime}-w^{\prime})\cdot (u-w)}{(x^{\prime}-w^{\prime})^{2} (u-w)^{2}} \frac{(x-z)\cdot (y-z)}{(x-z)^{2}(y-z)^{2}}\notag \\ & \Bigg\{-\bigg[U^{ab^{\prime}}(x^{\prime})\rho^{b^{\prime}}(x^{\prime})-\rho^{b^{\prime}}(x^{\prime})U^{ab^{\prime}}(w^{\prime})\bigg]\times\bigg[U^{ab}(u)\rho^{b}(u)\rho^{c}(x)\rho^{c}(y)-U^{ab}(w)U^{cd}(x)\rho^{d}(x)U^{ce}(y)\rho^{e}(y)\rho^{b}(u)\bigg]\notag \\ & + \bigg[U^{ab^{\prime}}(x^{\prime})\rho^{b^{\prime}}(x^{\prime})-\rho^{b^{\prime}}(x^{\prime})U^{ab^{\prime}}(w^{\prime})\bigg]\times \bigg[U^{ab}(w)\rho^{c}(x)\rho^{c}(y)\rho^{b}(u)-U^{cd}(x)\rho^{d}(x)U^{ce}(y)\rho^{e}(y)U^{ab}(u)\rho^{b}(u)\bigg]\notag \\ &

The terms with two Wilson lines are as follows
\begin{align}
    \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & = \frac{1}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{3}}\frac{\log{\left(\frac{\vee}{\Lambda}\right)}}{2\pi^{2}} \frac{1}{k^{+}}\int_{x,y,z,u,x^{\prime},w,w^{\prime}}e^{-ik(w^{\prime}-w)}\frac{(x^{\prime}-w^{\prime})\cdot (u-w)}{(x^{\prime}-w^{\prime})^{2} (u-w)^{2}} \frac{(x-z)\cdot (y-z)}{(x-z)^{2}(y-z)^{2}}\rho^{b^{\prime}}(x^{\prime})\rho^{b}(u)\rho^{c}(x)\rho^{c}(y)\notag \\ & \bigg[-U^{ab^{\prime}}(x^{\prime})U^{ab}(u) +U^{ab^{\prime}}(w^{\prime})U^{ab}(u) + U^{ab}(w)U^{ab^{\prime}}(x^{\prime}) - U^{ab}(u)U^{ab^{\prime}}(x^{\prime}) + U^{ab^{\prime}}(x^{\prime})U^{ab}(w) - U^{ab^{\prime}}(w^{\prime})U^{ab}(w) \notag \\ & + U^{ab}(u)U^{ab^{\prime}}(w^{\prime})  - U^{ab}(w)U^{ab^{\prime}}(w^{\prime})+2U^{ab}(w)U^{ab^{\prime}}(w^{\prime}) + 2 U^{ab^{\prime}}(x^{\prime})U^{ab}(u) \notag \\ & -2 U^{ab^{\prime}}(x^{\prime})U^{ab}(w) - 2U^{ab}(u)U^{ab^{\prime}}(w^{\prime})\bigg] = 0

    & +2\, U^{ac}(u)U^{bd}(x)U^{ab'}(x')U^{be}(z) + 2\,U^{ac}(w)U^{bd}(x)U^{ab'}(x')U^{be}(y)- 2\,U^{ac}(w)U^{bd}(x)U^{ab'}(x')U^{be}(z)\bigg]\Bigg\}\notag \\ & + \frac{1}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{3}}\frac{\log{\left(\frac{\vee}{\Lambda}\right)}}{2\pi^{2}} \frac{1}{k^{+}}\int_{x,y,z,u,x^{\prime},w,w^{\prime}}e^{-ik(w^{\prime}-w)}\frac{(x^{\prime}-w^{\prime})\cdot (u-w)}{(x^{\prime}-w^{\prime})^{2} (u-w)^{2}} \frac{(x-z)\cdot (y-z)}{(x-z)^{2}(y-z)^{2}}\,\rho^{b'}(x^{\prime})\rho^{d}(x)\rho^{c}(u)\rho^{e}(y)\notag \\ & \bigg[-2\,U^{ab'}(w')U^{bd}(z)U^{be}(y)U^{ac}(u) - Simplifying the terms with four Wilson lines, we get
\begin{align}
       \frac{d^{3}N_{\rm NLO}^{\rm}}{d^{3}k}  & =  \Bigg\{\frac{1}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{3}}\frac{\log{\left(\frac{\vee}{\Lambda}\right)}}{2\pi^{2}} \frac{1}{k^{+}}\int_{x,y,z,u,x^{\prime},w,w^{\prime}}e^{-ik(w^{\prime}-w)}\frac{(x^{\prime}-w^{\prime})\cdot (u-w)}{(x^{\prime}-w^{\prime})^{2} (u-w)^{2}} \frac{(x-z)\cdot (y-z)}{(x-z)^{2}(y-z)^{2}}\rho^{b'}(x') \rho^{d}(x) \rho^{c}(u) \rho^{e}(y)\notag \\ & \bigg[4\,U^{ab'}(w')U^{ac}(w)U^{bd}(z)U^{be}(y)+2\,U^{ab'}(x')U^{ac}(u)U^{bd}(x)U^{be}(z)-4\,U^{bd}(x)U^{be}(z)U^{ab'}(w')U^{ac}(w)\notag \\ & -2\,U^{ab'}(x')U^{ac}(u)U^{bd}(z)U^{be}(y)-2\,U^{ab'}(w')U^{ac}(u)U^{bd}(z)U^{be}(y)+2\,U^{ab'}(w')U^{ac}(u)U^{bd}(x)U^{be}(z)\bigg]

Since this equation is symmetric when $x\leftrightarrow y$ and by doing the algebra explicitly, we get all the terms that corresponds to four Wilson lines and $\rho$'s to be $0$.
\subsection{Three $\rho$ contributions }
The three $\rho$ contribution would arise from reordering the four $\rho$ contributions. For our convenience we will write it as three $\rho$ contribution with two, three and four Wilson lines. 
\subsubsection{Two Wilson lines}
\begin{align}
    \frac{d^{3}N_{\rm NLO}}{d^{3}k} & = -\,\frac{i}{(2\pi)^{3}}\frac{g^{4}}{8\pi^{3}}\frac{\log{\left(\frac{\vee}{\Lambda}\right)}}{2\pi^{2}} \frac{1}{k^{+}}\,\int_{x,y,z,x^{\prime},w,w^{\prime}}e^{-ik(w^{\prime}-w)}\, \Bigg[\notag \\ &  \frac{( - w' + x') \cdot ( - w + y) (x - z) \cdot (y - z)}{( - w' + x')^{2} ( - w + y)^{2} (x - z)^{2} (y - z)^{2}} f^{a b c} \rho^{d}(x') \rho^{a}(x) \rho^{c}(y) \bigg[U^{d e}(w') U^{e b}(w) - U^{e b}(w) U^{e d}(x')\bigg]\nonumber\\
    & + \frac{( - w' + x') \cdot ( - w + y) (x - z) \cdot (x' - z)}{( - w' + x')^{2} ( - w + y)^{2} (x - z)^{2} (x' - z)^{2}} f^{a b c} \rho^{c}(x') \rho^{a}(x) \rho^{d}(y) \bigg[U^{e b}(w') U^{e d}(y) - U^{b e}(w') U^{e d}(w) \bigg]\Bigg]
\end{align}
\subsubsection{Three Wilson lines}
\begin{align}

\end{align}
\subsubsection{Four Wilson lines}
\begin{align}
    \frac{d^{3}N_{\rm NLO}}{d^{3}k} & = 0 
\end{align}    
\subsection{Two $\rho$ contribution}

\subsubsection{First part}
\begin{align}

        &+ 2 \frac{( - w' + x') \cdot ( - w + y) (y - z) \cdot (y - z)}{( - w' + x')^{2} ( - w + y)^{2} (y - z)^{4}}\,\rho^{b}(x') \rho^{a}(y) U^{c a}(y) U^{c b}(w')\nonumber\\
    &-2\frac{( - w' + x') \cdot ( - w + y) (y - z) \cdot (y - z)}{( - w' + x')^{2} ( - w + y)^{2} (y - z)^{4}}\,\rho^{b}(x') \rho^{a}(y) U^{c a}(y) U^{c b}(x')\nonumber\\
    &+ 3 \frac{( - w' + x') \cdot ( - w + y) (x' - z) \cdot (y - z)}{2 ( - w' + x')^{2} ( - w + y)^{2} (x' - z)^{2} (y - z)^{2}}\,\rho^{b}(x') \rho^{a}(y) U^{b c}(x') U^{a c}(y) \nonumber\\
    &- 8 \frac{( - w' + x') \cdot ( - w + x') (y - z) \cdot (x' - z)}{( - w + x')^{2} ( - w' + x')^{2} (x' - z)^{2} (y - z)^{2}}\,\rho^{b}(x') \rho^{a}(y) U^{b c}(x') U^{c a}(y) \Bigg]
\end{align}
\subsubsection{Second part}
\begin{align}
