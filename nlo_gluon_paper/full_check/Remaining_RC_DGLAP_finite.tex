\section{What is done}\label{sec:what}
\begin{enumerate}
\item \textbf{Remaining terms} (Secs.~\ref{sec:FgI}--\ref{sec:FgIII}). Each row $=\log\frac{k^+}{\Lambda}\mathfrak T+\log\frac{\vee}{k^+}\mathfrak P+R$ exactly; here is $R$.
\begin{itemize}
\item Every logarithm is replaced by its remainder (Sec.~3 of \texttt{Checks\_JIMWLK\_BFKL.pdf}), with $\Lambda\to0$: $\big[\log\frac{k^+}{\Lambda}+\log(\frac{\vee}{k^+}-1)\big]\to\log(1-\frac{k^+}{\vee})$, $\log\frac{\vee}{\Lambda+k^+}\to0$, $\log\frac{(\vee-k^+)A+k^+B}{\Lambda A+k^+B}\to\log\frac{(1-\frac{k^+}{\vee})A+\frac{k^+}{\vee}B}{B}$, \dots
\item Line 1 of Group III Rows III, V, VI and Group II Row IV is rewritten exactly, (I2)--(I5): what it contains of $\log\frac{\vee}{k^+}$ is in the BFKL list.
\item $\Lambda\to0$ everywhere else: $\mathcal A\to\mathcal A_0$; the limits of the plus-prescribed integrals ($\int_\Lambda^{k^+-\Lambda}\to\int_0^{k^+}$, $\frac{k^+}{k^++\Lambda}\to1$); $\frac{1}{k^++\Lambda}\to\frac{1}{k^+}$; $2\log(1-\frac{\Lambda}{k^+})\to0$.
\item Removed (\old{red}): the $\frac{1}{\Lambda}$ lines of Group I Row III (they add up to zero), your two red lines of $d^3N_2$ (they cancel), and the edge-kernel line of $d^3N_{3b}$ (C4).
\end{itemize}
Each of the 79 pointwise-evaluable terms equals its three parts to $1.5\times10^{-14}$, and the remainder has the $\Lambda\to0$ limit written here ($3.5\times10^{-8}$ at $\Lambda=10^{-9}$; \texttt{ll\_checks.py split}). These terms stay finite for $\vee\to\infty$ and $\Lambda\to0$, apart from the open finite part of the $\mathcal D$ region (Q3).
\item \textbf{Running coupling} (Sec.~\ref{sec:rc}) and \textbf{DGLAP} (Sec.~\ref{sec:dglap}) are taken out of Group I Row II part 1, Group I Row V 2b UV and Group II Row IV, with the coupling counterterm of LO.
\item \textbf{The finite terms} are what is left (Sec.~\ref{sec:fin}); Sec.~\ref{sec:res} is the result.
\end{enumerate}

\section{Running coupling}\label{sec:rc}
\paragraph{Notation.} $a\equiv\frac{k^+}{\vee}$, $\beta_0=\frac{11}{3}N_c$ ($n_f=0$), $d_\perp=2-\epsilon$, $c_0\equiv2e^{-\gamma_E}$, $\ell_r\equiv\log\frac{r^2\mu^2}{c_0^2}=2\gamma_E+\log\frac{r^2\mu^2}{4}$, $\hat E_r\equiv\frac2\epsilon+\ell_r$; $X=x-y$, $X'=x'-y'$, $\Delta=y'-y$. LO in your notation, and LO with a weight $f$:
\begin{align*}
{\rm LO}[f]&\equiv\frac{1}{(2\pi)^{3}}\frac{g^{2}}{2\pi^{2}}\frac{1}{k^{+}}\int_{x,y,x',y'}e^{-ik\cdot(y'-y)}\,f\,\frac{(x-y)\cdot(x'-y')}{(x-y)^{2}(x'-y')^{2}}\left[U^{e'c}(y')-U^{e'c}(x')\right]\left[U^{ec}(y)-U^{ce}(x)\right]\rho^{e'}(x')\rho^{e}(x),
\end{align*}
${\rm LO}\equiv{\rm LO}[1]$; ${\rm LO}_\xi[f]$ is the same with $e^{-ik\cdot(y'-y)}\to e^{-ik\cdot(y'-y)\frac1\xi}$. $P^{(r)}_{gg,+}(\xi)=2N_c\big[\frac{\xi}{(1-\xi)_+}+\frac{1-\xi}{\xi}+\xi(1-\xi)\big]$ with the plus prescription of $[0,1]$ restricted to $[a,1]$ ($\int_a^1\frac{d\xi\,g(\xi)}{(1-\xi)_+}=\int_a^1\frac{g(\xi)-g(1)}{1-\xi}d\xi+g(1)\log(1-a)$), $P_{gg}=P^{(r)}_{gg,+}+\frac{\beta_0}{2}\delta(1-\xi)$. Only three rows have poles or $\beta_0$: Group I Row II part 1, Group I Row V 2b UV, Group II Row IV (lines 1, 2); all other remaining terms are finite. Their conversion to units of LO was checked against your rows before (\texttt{finite\_v2/final\_checks.json}); the bookkeeping is \texttt{dglap\_rc/checks.py}: $\beta_0$ sum $$0$$, poles $-\frac{\alpha_s}{2\pi}\big(-\frac{2}{\epsilon}\big)\big[P_{gg}\otimes{\rm LO}-N_c\log(1-a){\rm LO}\big]=0$, finite parts $-$(CMW $+V_\vee+$ Q1 $+$ DGLAP) $=0$.
\paragraph{The $\beta_0$ terms.} Two rows have $\beta_0$: the $\frac{11}{3}$ of Group I Row II part 1 and the $-\frac{11}{6}$ of Group I Row V 2b UV:
\begin{align}
\frac{d^{3}N^{\beta_0}_{1}}{d^{3}k}&=\frac{1}{(2\pi)^{3}}\,\frac{g^{4}\,N_{c}}{32\pi^{4}}\,\frac{1}{k^{+}}\,\frac{11}{3}\int_{w,w',x,x'}e^{-ik(w'-w)}\,\frac{(x'-w')\cdot (x-w)}{(x'-w')^{2}\, (x-w)^{2}}\left[-\frac{2}{\epsilon}-2\gamma-\log{\left[\frac{(x-w)^{2}\,\mu^{2}_{\overline{MS}}}{4}\right]}\right]\notag\\
&\quad\times\left[U^{ab'}(x')\,\rho^{b'}(x') - U^{ab'}(w')\,\rho^{b'}(x')\right]\times\,\left[U^{ab}(w)\,\rho^{b}(x) - U^{ab}(x)\,\rho^{b}(x)\right]\ \ +{\rm c.c.}\;=\;\frac{\alpha_s\beta_0}{4\pi}\,{\rm LO}\big[\hat E_X+\hat E_{X'}\big],\label{eq:b1}\\
\frac{d^{3}N^{\beta_0}_{2b\,\rm UV}}{d^{3}k}&=-\frac{1}{(2\pi)^{3}}\,\frac{g^{4}N_{c}}{16\pi^{4}}\,\frac{1}{k^{+}}\,\frac{11}{6}\int_{y,z,x',w'}e^{-ik\cdot(w'-z)}\left[\frac{2}{\epsilon}+2\gamma_{E}+\log\frac{(y-z)^{2}\mu^{2}_{\overline{MS}}}{4}\right]\frac{(x'-w')\cdot(y-z)}{(x'-w')^{2}\,(y-z)^{2}}\notag\\
&\quad\times\Big[U^{ab'}(x')-U^{ab'}(w')\Big]\Big[U^{af}(y)-U^{af}(z)\Big]\rho^{b'}(x')\,\rho^{f}(y)\ \ +{\rm c.c.}\;=\;-\frac{\alpha_s\beta_0}{4\pi}\,{\rm LO}\big[\hat E_X+\hat E_{X'}\big].\label{eq:b2}
\end{align}
With $y\to x$, $z\to w$ in \eqref{eq:b2}, the measure, the phase, the kernel and the bracket are those of \eqref{eq:b1}, the colour factor changes sign, and $\frac{g^4N_c}{16\pi^4}\frac{11}{6}=\frac{g^4N_c}{32\pi^4}\frac{11}{3}$. So
\begin{equation}
\boxed{\frac{d^{3}N^{\beta_0}_{1}}{d^{3}k}+\frac{d^{3}N^{\beta_0}_{2b\,\rm UV}}{d^{3}k}=0}
\end{equation}
for every $a$: pole, $\gamma_E$, $\mu$ and $\log(x-w)^2$ all cancel (\texttt{checks.json}). The virtual correction (Row II) and the real emission at short distance (Row V 2b UV) carry opposite $\beta_0$ terms, and no $\beta_0\log(x-w)^2$, i.e.\ no running of the coupling at the emission distance, is left.

\paragraph{The coupling of LO.} LO is $\propto g^2$. In $\overline{MS}$, $g^2\to g^2(\mu_R)\big[1-\frac{\alpha_s\beta_0}{4\pi}\frac{2}{\epsilon}\big]$ ($d_\perp=2-\epsilon$), which adds
\begin{equation}
\frac{d^3N}{d^3k}\bigg|_{\rm CT}=-\frac{\alpha_s\beta_0}{4\pi}\,\frac{2}{\epsilon}\,{\rm LO}=\frac{\alpha_s}{2\pi}\Big(-\frac{2}{\epsilon}\Big)\frac{\beta_0}{2}\,{\rm LO}.
\label{eq:ct}
\end{equation}
This pole is the $\frac{\beta_0}{2}\delta(1-\xi)$ part of the DGLAP pole (Sec.~\ref{sec:dglap}). After the coupling is renormalized at $\mu_R$ and the fragmentation function at $\mu_F$ (both $\overline{MS}$), the only $\beta_0$ term of the result is the renormalization-group term
\begin{equation}
\boxed{\frac{d^3N}{d^3k}\bigg|_{\beta_0}=\frac{\alpha_s\beta_0}{4\pi}\,\log\frac{\mu_R^2}{\mu_F^2}\;{\rm LO}\Big|_{\alpha_s(\mu_R)}}
\label{eq:rg}
\end{equation}
It makes the result independent of $\mu_R$: $\frac{d}{d\log\mu_R^2}{\rm LO}|_{\alpha_s(\mu_R)}=-\frac{\alpha_s\beta_0}{4\pi}{\rm LO}$. It vanishes for $\mu_R=\mu_F$; then LO is taken with $\alpha_s(\mu_F)$.

\paragraph{The constant.} The non-logarithmic part of the bracket of Row II part 1, $-\frac{67}{9}+\frac{\pi^2}{3}$, gives
\begin{align}
&\frac{1}{(2\pi)^{3}}\frac{g^{4}N_{c}}{32\pi^{4}}\frac{1}{k^{+}}\int_{w,w',x,x'}e^{-ik(w'-w)}\frac{(x'-w')\cdot (x-w)}{(x'-w')^{2}\, (x-w)^{2}}\Big[-\frac{67}{9}+\frac{\pi^{2}}{3}\Big]\notag\\
&\qquad\times\left[U^{ab'}(x')\,\rho^{b'}(x') - U^{ab'}(w')\,\rho^{b'}(x')\right]\times\,\left[U^{ab}(w)\,\rho^{b}(x) - U^{ab}(x)\,\rho^{b}(x)\right]+{\rm c.c.}\;=\;\boxed{\frac{\alpha_s}{2\pi}\,N_c\Big[\frac{67}{9}-\frac{\pi^2}{3}\Big]{\rm LO}}
\label{eq:cmw}
\end{align}
$N_c(\frac{67}{9}-\frac{\pi^2}{3})=2K$ with $K=N_c(\frac{67}{18}-\frac{\pi^2}{6})$, the constant of the two-loop cusp anomalous dimension that the CMW scheme absorbs into the coupling. It is the finite companion of $\beta_0$ in this row.
