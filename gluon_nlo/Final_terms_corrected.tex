\begin{equation}
{\rm Tr}\{T^{e}U^{\dagger}(a)U(b)T^{e}U^{\dagger}(b)U(z)\}={\rm Tr}\{T^{e}U^{\dagger}(a)U(b)T^{e}U^{\dagger}(z)U(a)\}=f^{cde}f^{c'd'e'}U^{cc'}(a)U^{dd'}(b)U^{ee'}(z),
\end{equation}
and the Jacobi identity
${\rm Tr}\{T^{e}XT^{e}Y\}={\rm Tr}\{T^{e}XT^{e}Y^{T}\}+{\rm Tr}\{T^{e}X\}{\rm Tr}\{T^{e}Y\}$ removes all double traces. With
\begin{equation}
\mathcal{D}(a,b;z)=N_{c}\,{\rm Tr}\{U^{\dagger}(a)U(b)\}-f^{cde}f^{c'd'e'}\,U^{cc'}(a)\,U^{dd'}(b)\,U^{ee'}(z)
\end{equation}
the expression above is identical to
\begin{align}
\frac{d^{3}N_{\mathrm{NLO}}}{d^{3}k} & = -\frac{1}{(2\pi)^{3}}\,\frac{g^{4}}{8\pi^{5}}\,\frac{1}{k^{+}}\,\frac{1}{N^{2}_{c}-1}\,\log{\left(\frac{k^{+}}{\Lambda}\right)}\,\int_{x,y,z,w,w'}e^{-ik(w'-w)}\,\langle\rho^{f}(x)\rho^{f}(y) \rangle\,\mathcal K(x-w)\cdot\mathcal K(y-w')\notag\\

\end{align}
i.e. the leading-order cross section with each adjoint dipole ${\rm Tr}\{U^{\dagger}(a)U(b)\}$ replaced by $\log(k^{+}/\Lambda)\,\partial_{Y}{\rm Tr}\{U^{\dagger}(a)U(b)\}$, with $\partial_{Y}{\rm Tr}\{U^{\dagger}(a)U(b)\}=-\frac{\alpha_{s}}{\pi^{2}}\int_{z}M(a,b;z)\,\mathcal{D}(a,b;z)$.

\end{align}
where:
$\widetilde{\mathcal{P}}_{gg}(\xi)$ is gluon-gluon splitting function.
$$\widetilde{\mathcal{P}}_{gg}(\xi) = 2\,N_{c}\left[\xi(1-\xi) + \frac{\xi}{(1-\xi)_{+}} + \Big[\frac{1-\xi}{\xi}\Big]_{+}\right]+\frac{\beta_{0}}{2}\,\delta(1-\xi),\qquad \beta_{0}=\frac{11N_{c}}{3}.$$
The two plus prescriptions are defined on the function of $\xi$ obtained after the rescaling $y'=y+\xi r$ (so that $\frac{1}{\xi^{2}}\int_{y'}e^{-ik\cdot(y'-y)/\xi}\to\int_{r}e^{-ik\cdot r}$ and the logarithm becomes $\log\frac{r^{2}\mu^{2}_{\overline{MS}}}{4}$): for $F(\xi)$ denoting everything that multiplies them,
\begin{equation}
\int_{0}^{1}d\xi\,\frac{\xi}{(1-\xi)_{+}}F(\xi)=\int_{0}^{1}d\xi\,\frac{\xi F(\xi)-F(1)}{1-\xi},\qquad
\int_{0}^{1}d\xi\,\Big[\frac{1-\xi}{\xi}\Big]_{+}F(\xi)=\int_{0}^{1}d\xi\,\frac{(1-\xi)F(\xi)-F(0)}{\xi}.
\end{equation}
The subtraction at $\xi=0$ (parent gluon at large rapidity) is the part already contained in the BFKL term; it is well defined only in the rescaled variable $r$.
The scale $\mu_{\overline{MS}}$ in this term is a collinear factorization scale of the final state: the corresponding pole is absorbed into the gluon fragmentation function (or jet function) that must accompany the produced gluon.

\begin{itemize}

\item In the first term the last line of the curly bracket, $\tfrac12\rho^{b'}(x')\{\rho^{e}(y),\rho^{n}(x)\}(\dots)$, is the ordered-product part of Group II, Group I Row IV and Group I Row III with the kernel $\log\frac{(x-z)^{2}}{(x-w)^{2}}$; the first four lines are its commutator parts.
