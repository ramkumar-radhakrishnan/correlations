\noindent\textbf{Conventions.} All terms below are the finite remainders left after the logarithms $\log(\vee/k^{+})$ (BFKL), $\log(k^{+}/\Lambda)$ (JIMWLK), the collinear logarithm (DGLAP) and the running-coupling logarithm have been removed. Throughout:
\begin{itemize}
\item energy denominators that can vanish are defined by the Cauchy principal value, so that every coordinate-space kernel is real;
\item scaleless transverse integrals are set to zero, $\int d^{2}p/p^{2}=0$; in this prescription
\begin{equation}
-\frac{1}{\pi}\int_{z}\frac{(y-z)\cdot(y-x)}{(y-z)^{2}\,(x-z)^{2}}=-\frac{2}{\epsilon}-2\gamma_{E}-\log\frac{(y-x)^{2}\mu^{2}_{\overline{MS}}}{4};\label{eq:Lid}
\end{equation}
\item the transverse polarization algebra is done in two dimensions, $\delta^{ii}=d_{\perp}=2$;
\item plus prescriptions act on everything that multiplies them, including the phases:
$\int_{0}^{k^{+}}dp^{+}\big[\tfrac{k^{+}}{p^{+}}\big]_{+}F(p^{+})=\int_{0}^{k^{+}}dp^{+}\tfrac{k^{+}}{p^{+}}\big[F(p^{+})-F(0)\big]$ and
$\int_{0}^{k^{+}}dp^{+}\big[\tfrac{k^{+}}{k^{+}-p^{+}}\big]_{+}F(p^{+})=\int_{0}^{k^{+}}dp^{+}\tfrac{k^{+}}{k^{+}-p^{+}}\big[F(p^{+})-F(k^{+})\big]$;
in Group III, $\big[\tfrac{k^{+}}{p^{+}(p^{+}+k^{+})}\big]_{+}$ subtracts at $p^{+}=0$;
\item the labels $N^{G,R}$ refer to Group $G$, Row $R$. Groups I and III are to be doubled by adding the complex conjugate; Group II is real by itself.
\end{itemize}
