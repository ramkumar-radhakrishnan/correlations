\subsection{Coefficients of the evolution operator}\label{subsec:coefficients}
We can now read off the coefficients of $\Omega$ by comparing Eqs.~\eqref{eq:Omegavac}--\eqref{Omegaongposzeta}
with the light-cone wave functions of Secs.~\ref{sec:vacin} and~\ref{sec:singleglue}. The dressed vacuum
fixes $\mf{N}$, $\mf{A}_{2}$ and $\mf{B}_{2}$. The dressed one-gluon state fixes $\mf{B}_{3}$ and $\mf{C}_{1}$,
and its vacuum component reproduces $\mf{A}^{(1)}_{1}$. We use the conventions of
Sec.~\ref{subsec:omegacoord}. In momentum space
\begin{align}\label{eq:matchrules}
(2\pi)^{-3/2}\mf{A}_{2}\,,\qquad (2\pi)^{-3}\mf{B}_{2}\,,\qquad (2\pi)^{-3/2}\mf{A}_{1}\,,\qquad
(2\pi)^{-3}\mf{B}_{3}\,,\qquad (2\pi)^{-3/2}\mf{C}_{1}
\end{align}
are the amplitudes of the corresponding states per unit $d^{3}p$ (per unit $d^{3}p\,d^{3}q$ for two
gluons), with $\mf{B}_{2}$ and $\mf{C}_{1}$ symmetrised as in Eq.~\eqref{eq:symm}. In transverse
coordinates the factors are $(2\pi)^{-1/2}$ for one gluon and $(2\pi)^{-1}$ for two, per unit $dp^{+}$ and
$d^{2}x$. When a wave function is written in terms of the parent momentum and the fraction $\xi$, the
Jacobian is $dp^{+}dq^{+}=k^{+}dk^{+}d\xi$ for two gluons and $dp^{+}=k^{+}d\xi$ for one. All longitudinal
momenta lie in the soft window, $\Lambda<p^{+}<\vee$; additional constraints are written as step
functions. We write $L\equiv\ln(\vee/\Lambda)$ and $\xi=p^{+}/k^{+}$ throughout.

\subsubsection{Coefficients from the dressed vacuum}\label{subsec:coeffvac}

\paragraph{Normalisation.}
Comparing the first term of Eq.~\eqref{eq:Omegavac} with Eq.~\eqref{eq:Nmom},
\begin{align}\label{Ncoeffmom}
\mf{N}=1+\mf{N}^{(2)}\,,\qquad
\mf{N}^{(2)}&=-\frac{g^{2}}{(2\pi)^{3}}\,L\int d^{2}\v p\,\frac{\rho^{b}(-\v p)\,\rho^{b}(\v p)}{\v p^{2}}\notag\\
&=-\frac{g^{2}}{(2\pi)^{3}}\,L\int_{\v x,\v y,\v z}\frac{(x-z)\cdot(y-z)}{(x-z)^{2}\,(y-z)^{2}}\,
\rho^{a}(\v x)\,\rho^{a}(\v y)\,.
\end{align}

\paragraph{One gluon at order $g$.}
Comparing the one-gluon term of Eq.~\eqref{eq:Omegavac} with Eq.~\eqref{lowav1mom},
\begin{align}\label{A1coeffM}
\mf{A}^{(1)a}_{2i}(k)=-\mf{A}^{(1)\dagger a}_{1i}(k)=-\frac{\sqrt{2}\,g\,\v k^{i}}{\sqrt{k^{+}}\,\v k^{2}}\,\rho^{a}(-\v k)\,,\qquad
\mf{A}^{(1)a}_{1i}(k)=\frac{\sqrt{2}\,g\,\v k^{i}}{\sqrt{k^{+}}\,\v k^{2}}\,\rho^{a}(\v k)\,.
\end{align}
The second relation is Eq.~\eqref{eq:A1rel}; it agrees with the absorption amplitude of
Eq.~\eqref{eq:psirho}. In transverse coordinates, from Eqs.~\eqref{lowav1} and~\eqref{eq:psirhoX},
\begin{align}\label{A1coeff}
\mf{A}^{(1)a}_{2i}(k^{+},-\v z)&=-\mf{A}^{(1)\dagger a}_{1i}(k^{+},\v z)
=\frac{ig}{\sqrt{2}\,\pi\sqrt{k^{+}}}\int_{\v x}\frac{(x-z)^{i}}{(x-z)^{2}}\,\rho^{a}(\v x)\,,\notag\\
\mf{A}^{(1)a}_{1i}(k^{+},\v z)&=\frac{ig}{\sqrt{2}\,\pi\sqrt{k^{+}}}\int_{\v x}\frac{(x-z)^{i}}{(x-z)^{2}}\,\rho^{a}(\v x)\,.
\end{align}

\paragraph{One gluon at order $g^{3}$.}
$\mf{A}^{(3)}_{2}$ collects the amplitudes of $\ket{\Psi_{g\rho}}$, $\ket{\Psi_{g\rho\rho}}$ and
$\ket{\Psi_{g\rho\rho\rho}}$. Before reading off the linear term, the two terms of Eq.~\eqref{combinedgrho}
must be combined, as in Eq.~\eqref{grhoposition}: in the second term the gluon carries $\xi k^{+}$, and we
use its momentum as the integration variable and carry out the $\xi$ integral. Then
\begin{align}\label{A3coeffM}
\mf{A}^{(3)a}_{2i}(k)=\ &\frac{g^{3}N_{c}\,\v k^{i}}{8\sqrt{2}\,\pi^{2}\sqrt{k^{+}}\,\v k^{2}}\,\rho^{a}(-\v k)\,
\bigg\{\Big[-\frac{2}{\epsilon}+\ln\frac{\v k^{2}}{\mu^{2}}\Big]
\Big[\frac{11}{3}+2\ln\frac{\Lambda}{k^{+}}+\ln\Big(1-\frac{k^{+}}{\vee}\Big)\Big]\notag\\
&\hspace{4.2cm}+\frac{\pi^{2}}{3}-\frac{67}{9}+\ln^{2}\Big(1-\frac{k^{+}}{\vee}\Big)
+\ln\Big(1-\frac{k^{+}}{\vee}\Big)\ln\frac{\vee}{k^{+}}\bigg\}\notag\\
&+\frac{ig^{3}f^{abc}}{8\sqrt{2}\,\pi^{3}\sqrt{k^{+}}}\int d^{2}\v p\,
\frac{\{\rho^{b}(-\v p),\rho^{c}(-\v k+\v p)\}}{\v p^{2}\,\v k^{2}}\,\mathcal{K}^{i}(\v k,\v p;k^{+})
+\mf{N}^{(2)}\,\mf{A}^{(1)a}_{2i}(k)\,,
\end{align}
with $\mathcal{K}^{i}$ of Eq.~\eqref{grhorhokernel}. The last term is the amplitude of
$\ket{\Psi_{g\rho\rho\rho}}$, Eq.~\eqref{3rhosmom},
\begin{align}
\mf{N}^{(2)}\,\mf{A}^{(1)a}_{2i}(k)=\frac{g^{3}\,L}{4\sqrt{2}\,\pi^{3}\sqrt{k^{+}}}\int d^{2}\v p\,
\frac{\v k^{i}}{\v k^{2}\,\v p^{2}}\,\rho^{b}(\v p)\,\rho^{b}(-\v p)\,\rho^{a}(-\v k)\,,
\end{align}
with $\mf{N}^{(2)}$ to the left of $\rho^{a}$. In transverse coordinates, from Eqs.~\eqref{grhoposition},
\eqref{grhorhopos} and~\eqref{3rhospos},
\begin{align}\label{A3coeffX}
\mf{A}^{(3)a}_{2i}(k^{+},-\v z)=\ &-\frac{ig^{3}N_{c}}{16\sqrt{2}\,\pi^{3}\sqrt{k^{+}}}\int_{\v x}
\frac{(x-z)^{i}}{(x-z)^{2}}\,\rho^{a}(\v x)\notag\\
&\hspace{1cm}\times\bigg\{\Big[-\frac{2}{\epsilon}-2\gamma_{E}-\ln\frac{(x-z)^{2}\mu^{2}}{4}\Big]
\Big[\frac{11}{3}+2\ln\frac{\Lambda}{k^{+}}+\ln\Big(1-\frac{k^{+}}{\vee}\Big)\Big]\notag\\
&\hspace{1.5cm}+\frac{\pi^{2}}{3}-\frac{67}{9}+\ln^{2}\Big(1-\frac{k^{+}}{\vee}\Big)
+\ln\Big(1-\frac{k^{+}}{\vee}\Big)\ln\frac{\vee}{k^{+}}\bigg\}\notag\\
&+\frac{ig^{3}f^{abc}}{32\sqrt{2}\,\pi^{5}\sqrt{k^{+}}}\int_{\v x,\v y}\{\rho^{b}(\v x),\rho^{c}(\v y)\}
\int d^{2}\v k\,d^{2}\v p\;\frac{e^{i\v k\cdot(\v z-\v y)}\,e^{i\v p\cdot(\v y-\v x)}}{\v p^{2}\,\v k^{2}}\,
\mathcal{K}^{i}(\v k,\v p;k^{+})\notag\\
&+\mf{N}^{(2)}\,\mf{A}^{(1)a}_{2i}(k^{+},-\v z)\,,
\end{align}
where
\begin{align}
\mf{N}^{(2)}\,\mf{A}^{(1)a}_{2i}(k^{+},-\v z)=-\frac{ig^{3}\,L}{8\sqrt{2}\,\pi^{4}\sqrt{k^{+}}}
\int_{\v w,\v x,\v y,\v u}\frac{(x-u)\cdot(y-u)\,(w-z)^{i}}{(x-u)^{2}\,(y-u)^{2}\,(w-z)^{2}}\,
\rho^{b}(\v x)\,\rho^{b}(\v y)\,\rho^{a}(\v w)\,.
\end{align}
Together with Eq.~\eqref{A1coeff}, Eq.~\eqref{A3coeffX} gives the full one-gluon coefficient
$\mf{A}^{(3)}_{2}-\mf{A}^{(1)\dagger}_{1}$ of Eq.~\eqref{Omegaonvacpos}.

\paragraph{Two gluons.}
Comparing the two-gluon term of Eq.~\eqref{eq:Omegavac} with Eqs.~\eqref{psiggrhomomre}
and~\eqref{ggrhorhomom}, and writing the amplitude in terms of the two gluon momenta $k$ and $p$,
\begin{align}\label{B2coeffM}
\mf{B}^{ba}_{2ji}(k,p)=\ &\bigg\{\frac{g^{2}\,\v k^{i}\,\v p^{j}}{2\sqrt{k^{+}p^{+}}\,\v k^{2}\,\v p^{2}}\,
\{\rho^{a}(-\v k),\rho^{b}(-\v p)\}
+ig^{2}f^{abc}\rho^{c}(-\v k-\v p)\bigg[\frac{\sqrt{k^{+}p^{+}}\;\v k^{i}\,\v p^{j}}{2\,(p^{+}\v k^{2}+k^{+}\v p^{2})}
\Big(\frac{1}{k^{+}\v p^{2}}-\frac{1}{p^{+}\v k^{2}}\Big)\notag\\
&-\frac{(\v k+\v p)^{l}\,(k^{+}\v p^{m}-p^{+}\v k^{m})}
{\sqrt{k^{+}p^{+}}\,(k^{+}+p^{+})\,(\v k+\v p)^{2}\big(\frac{\v k^{2}}{k^{+}}+\frac{\v p^{2}}{p^{+}}\big)}
\Big(\frac{\delta_{ij}\,\delta_{lm}}{k^{+}+p^{+}}-\frac{\delta_{im}\,\delta_{jl}}{k^{+}}-\frac{\delta_{il}\,\delta_{jm}}{p^{+}}\Big)\notag\\
&-\frac{(p^{+}-k^{+})\,\delta_{ij}}{2\sqrt{k^{+}p^{+}}\,(k^{+}+p^{+})^{2}\big(\frac{\v k^{2}}{k^{+}}+\frac{\v p^{2}}{p^{+}}\big)}\bigg]\bigg\}\,
\Theta(\vee-k^{+}-p^{+})\,,
\end{align}
where $(a,i)$ belong to $k$ and $(b,j)$ to $p$. This expression is already symmetric under
$(k,a,i)\leftrightarrow(p,b,j)$. The step function is the condition $k^{+}+p^{+}<\vee$ of the wave
functions. The first term is $\frac{1}{4}\{\mf{A}^{(1)a}_{2i}(k),\mf{A}^{(1)b}_{2j}(p)\}$: the two gluons are
emitted independently. In transverse coordinates, from Eqs.~\eqref{ggrhopos} and~\eqref{ggrhorhopos}, with
$(a,i)$ at $\v w$ and $(b,j)$ at $\v z$,
\begin{align}\label{B2coeffX}
\mf{B}^{ba}_{2ji}(k^{+},-\v w;p^{+},-\v z)=\ &\bigg\{\int_{\v x}
\frac{ig^{2}f^{abc}\,\rho^{c}(\v x)\,\sqrt{k^{+}p^{+}}}{4\pi^{2}\,(k^{+}+p^{+})\big[k^{+}(x-w)^{2}+p^{+}(x-z)^{2}\big]}
\bigg[\frac{\delta^{ij}\big((x-z)^{2}-(x-w)^{2}\big)}{2(z-w)^{2}}\notag\\
&\hspace{1cm}+\frac{k^{+}+p^{+}}{k^{+}}\Big(\frac{(x-z)^{j}(z-w)^{i}}{(z-w)^{2}}-\frac{(x-z)^{j}(x-w)^{i}}{2(x-w)^{2}}\Big)\notag\\
&\hspace{1cm}+\frac{k^{+}+p^{+}}{p^{+}}\Big(\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}}+\frac{(x-z)^{j}(x-w)^{i}}{2(x-z)^{2}}\Big)\bigg]\notag\\
&-\int_{\v x,\v y}\frac{g^{2}\,\{\rho^{a}(\v y),\rho^{b}(\v x)\}\,(y-w)^{i}\,(x-z)^{j}}{8\pi^{2}\sqrt{k^{+}p^{+}}\,(y-w)^{2}\,(x-z)^{2}}\bigg\}\,
\Theta(\vee-k^{+}-p^{+})\,.
\end{align}
In terms of the longitudinal momentum fraction, with $k^{+}$ now the total momentum of the pair and the
gluon at $\v w$ carrying $\xi k^{+}$,
\begin{align}\label{B2coeffXxi}
\mf{B}^{ba}_{2ji}\big(\xi k^{+},-\v w;(1-\xi)k^{+},-\v z\big)=\ &\int_{\v x}
\frac{ig^{2}f^{abc}\,\rho^{c}(\v x)\,\sqrt{\xi(1-\xi)}}{4\pi^{2}k^{+}\big[\xi(x-w)^{2}+(1-\xi)(x-z)^{2}\big]}
\bigg[\frac{\delta^{ij}\big((x-z)^{2}-(x-w)^{2}\big)}{2(z-w)^{2}}\notag\\
&\hspace{1cm}+\frac{1}{\xi}\Big(\frac{(x-z)^{j}(z-w)^{i}}{(z-w)^{2}}-\frac{(x-z)^{j}(x-w)^{i}}{2(x-w)^{2}}\Big)\notag\\
&\hspace{1cm}+\frac{1}{1-\xi}\Big(\frac{(x-w)^{i}(z-w)^{j}}{(z-w)^{2}}+\frac{(x-z)^{j}(x-w)^{i}}{2(x-z)^{2}}\Big)\bigg]\notag\\
&-\int_{\v x,\v y}\frac{g^{2}\,\{\rho^{a}(\v y),\rho^{b}(\v x)\}\,(y-w)^{i}\,(x-z)^{j}}
{8\pi^{2}k^{+}\sqrt{\xi(1-\xi)}\,(y-w)^{2}\,(x-z)^{2}}\,,
\end{align}
for $\Lambda<k^{+}<\vee$ and $\Lambda/k^{+}<\xi<1-\Lambda/k^{+}$.

\subsubsection{Coefficients from the dressed one-gluon state}\label{subsec:coeffglue}
The incoming gluon carries $(k,a,i)$ and sits at $\v z$; the outgoing gluon carries $(p,b,j)$ and sits at
$\v u$. Two step functions distinguish the direction of the plus-momentum flow:
\begin{align}
\Theta_{>}\equiv\Theta(p^{+}-k^{+}-\Lambda)\,,\qquad \Theta_{<}\equiv\Theta(k^{+}-p^{+}-\Lambda)\,.
\end{align}

\paragraph{One gluon to one gluon.}
Comparing the $\mf{B}_{3}$ term of Eq.~\eqref{eq:Omegag} with Eqs.~\eqref{oneginst11}, \eqref{oneginst2},
\eqref{onegrho12xi}, \eqref{onegrho22xi} and~\eqref{onegrho4mom},
\begin{align}\label{B3coeffM}
\mf{B}^{ba}_{3ji}(k,p)=\ &\frac{ig^{2}f^{abc}}{k^{+}}\,\frac{\sqrt{\xi}\,(1+\xi)}{(1-\xi)^{2}}\,
\frac{\rho^{c}(\v k-\v p)}{\xi\v k^{2}-\v p^{2}}\,\delta_{ij}\,\big(\Theta_{>}+\Theta_{<}\big)\notag\\
&+\frac{2ig^{2}f^{abc}}{k^{+}}\,\frac{\sqrt{\xi}}{1-\xi}\,
\frac{(\v k-\v p)^{m}\,(\v p-\xi\v k)^{l}}{\v p^{2}-\xi\v k^{2}}\,\rho^{c}(\v k-\v p)\,\mathcal{T}_{ijlm}(\xi)
\bigg[\frac{\Theta_{>}}{(\v k-\v p)^{2}}+\frac{\xi\,\Theta_{<}}{(\v p-\xi\v k)^{2}}\bigg]\notag\\
&-\frac{g^{2}}{k^{+}\sqrt{\xi}}\,\frac{\v k^{i}\,\v p^{j}}{\v k^{2}\,\v p^{2}}
\bigg[\{\rho^{a}(\v k),\rho^{b}(-\v p)\}-if^{abc}\rho^{c}(\v k-\v p)\,\frac{\v p^{2}+\xi\v k^{2}}{\v p^{2}-\xi\v k^{2}}\bigg]\,,
\end{align}
with $\xi=p^{+}/k^{+}$ and $\mathcal{T}_{ijlm}$ of Eq.~\eqref{eq:Ttensor}. The three lines are the
instantaneous, the triple-gluon and the two-$\rho$ contributions. In the last line
$k^{+}\sqrt{\xi}=\sqrt{k^{+}p^{+}}$, and the anticommutator term is
$\frac{1}{2}\{\mf{A}^{(1)a}_{1i}(k),\mf{A}^{(1)b}_{2j}(p)\}$. In transverse coordinates, from
Eqs.~\eqref{Eq:onegluoninstX1}, \eqref{Eq:onegluoninstX2}, \eqref{Eq:onegrhoX1}, \eqref{Eq:onegrhoX2},
\eqref{gtworhoX} and~\eqref{gtworhoX2},
\begin{align}\label{B3coeffX}
\mf{B}^{ba}_{3ji}(k^{+},\v z;p^{+},-\v u)=\ &\frac{ig^{2}f^{abc}}{4\pi^{2}k^{+}}\,\frac{\sqrt{\xi}\,(1+\xi)}{(1-\xi)^{2}}
\int_{\v x}\frac{\rho^{c}(\v x)\,\delta_{ij}}{\xi(x-u)^{2}-(x-z)^{2}}\,\big(\Theta_{>}+\Theta_{<}\big)\notag\\
&-\frac{ig^{2}f^{abc}}{2\pi^{2}k^{+}}\,\frac{\sqrt{\xi}}{1-\xi}\int_{\v x}
\frac{\big((x-z)-\xi(x-u)\big)^{m}\,(u-z)^{l}}{(x-z)^{2}-\xi(x-u)^{2}}\,\rho^{c}(\v x)\,\mathcal{T}_{ijlm}(\xi)\notag\\
&\hspace{1cm}\times\bigg[\frac{\Theta_{>}}{(u-z)^{2}}+\frac{\xi\,\Theta_{<}}{\big((x-z)-\xi(x-u)\big)^{2}}\bigg]\notag\\
&-\frac{g^{2}}{4\pi^{2}k^{+}\sqrt{\xi}}\int_{\v x,\v y}\frac{(x-z)^{i}\,(y-u)^{j}}{(x-z)^{2}\,(y-u)^{2}}\,
\{\rho^{a}(\v x),\rho^{b}(\v y)\}\notag\\
&+\frac{ig^{2}f^{abc}}{4\pi^{2}k^{+}\sqrt{\xi}}\int_{\v x}\frac{(x-z)^{i}\,(x-u)^{j}}{(x-z)^{2}\,(x-u)^{2}}\,
\frac{(x-z)^{2}+\xi(x-u)^{2}}{(x-z)^{2}-\xi(x-u)^{2}}\,\rho^{c}(\v x)\,,
\end{align}
again with $\xi=p^{+}/k^{+}$. Here $(x-z)$ and $(x-u)$ are the $X'$ and $X$ of Eq.~\eqref{eq:XXY}, and
$(x-z)-(x-u)=u-z$.

\paragraph{One gluon to two gluons.}
Comparing the $\mf{C}_{1}$ term of Eq.~\eqref{eq:Omegag} with Eq.~\eqref{rhoindep}, with $(b,j)$ belonging to
$p$ and $(c,k)$ to $k-p$,
\begin{align}\label{C1coeffM}
\mf{C}^{bca}_{1jki}(k,p)=\ &-\frac{igf^{abc}\sqrt{k^{+}p^{+}(k^{+}-p^{+})}}{\sqrt{2}}\,
\frac{k^{+}\v p^{m}-p^{+}\v k^{m}}{(k^{+}\v p-p^{+}\v k)^{2}}
\Big[\frac{\delta_{im}\,\delta_{jk}}{k^{+}}-\frac{\delta_{ik}\,\delta_{jm}}{p^{+}}-\frac{\delta_{ij}\,\delta_{km}}{k^{+}-p^{+}}\Big]\,
\Theta_{<}\notag\\
=\ &-\frac{igf^{abc}}{\sqrt{2k^{+}}}\,\sqrt{\xi(1-\xi)}\,\frac{\v p^{m}-\xi\v k^{m}}{(\v p-\xi\v k)^{2}}\,
\mathcal{T}_{ijmk}(\xi)\,\Theta_{<}\,.
\end{align}
It is independent of $\rho$. In transverse coordinates, with the arguments of
Eq.~\eqref{eq:OmegaCoord},
\begin{align}\label{C1coeffX}
\mf{C}^{bca}_{1jki}(k^{+},\v z;p^{+},\v x)=\ &-\frac{gf^{abc}}{2\sqrt{2}\,\pi\sqrt{k^{+}}}\,\sqrt{\xi(1-\xi)}\;
\delta^{(2)}(\v z+\xi\v x)\,\frac{x^{m}}{x^{2}}\,\mathcal{T}_{ijmk}(\xi)\,\Theta_{<}\,.
\end{align}
The delta function places the plus-momentum-weighted centre of the two outgoing gluons at the position
of the incoming one, as in Eq.~\eqref{gindeppos}.

The remaining coefficients follow from unitarity: $\mf{B}_{1}$ from Eq.~\eqref{eq:B1rel},
$\mf{C}_{2}=-\mf{C}^{\dagger}_{1}$ from Eq.~\eqref{eq:C2rel}, and $\mf{A}^{(3)}_{1}$ from Eq.~\eqref{eq:A3rel},
including the commutator~\eqref{eq:NAcomm}.
