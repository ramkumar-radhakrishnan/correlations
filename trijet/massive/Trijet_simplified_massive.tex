\noindent\textbf{What changed with respect to the massless note.} The quark and the antiquark of flavour $f$ carry the mass $m_{f}$ (the gluon stays massless). All matrix elements, energy denominators, Fourier transforms, wave functions and cross sections of the massless note are rewritten below with the mass kept everywhere; nothing is expanded in $m_{f}$. Setting $m_{f}=0$ in every formula gives back the massless note (after the corrections listed in Appendix~\ref{app:changes}). Because the new radial functions depend on $m_{f}$, the flavour sum $\sum_{f}e^{2}_{f}$ now sits inside the coordinate integrals, and the factor $Q^{2}$ that multiplied the massless cross sections is absorbed into the radial functions. The Wilson-line structures $\mathcal{W}$ do not depend on the quark mass and are unchanged.

\section{Setup}
The cross section for trijet production is given by
\begin{align}
    \frac{d^{3}N}{d^{3}k_{1}\,d^{3}k_{2}\, d^{3}k_{3}}\, (2\pi)\,\delta\left(q^{+}-k^{+}_{1}-k^{+}_{2}-k^{+}_{3}\right) & =  \frac{1}{2}\,\int_{w,w'}\, e^{-iq\cdot (w-w')}
\bra{\gamma^{i}_{T}(Q,q^{+},w')}\,\mathcal{N}_{q}(k_{1})\,\mathcal{N}_{\Bar{q}}(k_{2})\,\mathcal{N}_{g}(k_{3})\,\ket{\gamma^{i}_{T}(Q,q^{+},w)}_{q\Bar{q}g}
\end{align}
\subsection{Kinematics}
The virtual photon with the four momentum as $q^{\mu} = \left(q^{+},-\frac{Q^{2}}{2q^{+}},q\right)$. Let us define the kinematics in the following:
\begin{table}[h!]
\centering
\begin{tabular}{||c c c c||}
 \hline
 Component & Longitudinal momentum fraction & Transverse momentum & Mass \\ [0.5ex]
 \hline\hline
 Quark & $\zeta = \frac{k^{+}_{1}}{q^{+}}$  & $k_{1}$ & $m_{f}$ \\
 Anti Quark & $\eta = 1-\zeta-\xi$  &  $k_{2}$ & $m_{f}$ \\
 Gluon & $\xi = \frac{k^{+}_{3}}{q^{+}}$ & $k_{3}$ & $0$ \\
 \hline
\end{tabular}
\label{table:1}
\end{table}

\noindent The on-shell light-cone energies are $E_{q}(k)=\frac{k^{2}+m^{2}_{f}}{2k^{+}}$, $E_{\Bar{q}}(p)=\frac{p^{2}+m^{2}_{f}}{2p^{+}}$ and $E_{g}(p)=\frac{p^{2}}{2p^{+}}$; this is the only place where the mass enters the energy denominators.
\subsection{Matrix Elements}
The matrix elements that are used in this process are as follows. The mass enters the two-component vertices through $\sigma\cdot k\rightarrow \sigma\cdot k - m_{f}\,\sigma^{3}$ for a quark line and $\sigma\cdot p\rightarrow \sigma\cdot p + m_{f}\,\sigma^{3}$ for an antiquark line (the opposite sign of the mass term of the antiquark spinor); the instantaneous vertex does not change.
\begin{enumerate}
\item Virtual photon splitting into a quark and anti quark pair
\begin{align}
    \bra{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(p)}H_{\gamma\rightarrow q\Bar{q}}\ket{\gamma_{\lambda}(q)} & = (2\pi)^{3}\, \delta^{(3)}(q-k-p)\, \frac{e\,e_{f}\,\delta_{\alpha\beta}}{2\,\sqrt{2q^{+}}}\, \chi^{\dagger}_{\lambda_{1}}\,\left[\frac{2q^{\lambda}}{q^{+}}-\sigma^{\lambda}\,\frac{\sigma\cdot p+m_{f}\,\sigma^{3}}{p^{+}}-\frac{\sigma\cdot k-m_{f}\,\sigma^{3}}{k^{+}}\,\sigma^{\lambda}\right]\chi_{\lambda_{2}}
\end{align}
rewriting it using the delta function ($k^{+}=\zeta q^{+}$, $p^{+}=(1-\zeta)q^{+}$, $\Tilde{k}=k-\zeta q$), we get
$$\left[\frac{2q^{\lambda}}{q^{+}}-\sigma^{\lambda}\,\frac{\sigma\cdot (q-k)+m_{f}\,\sigma^{3}}{q^{+}-k^{+}}-\frac{\sigma\cdot k-m_{f}\,\sigma^{3}}{k^{+}}\,\sigma^{\lambda}\right] = \frac{\left[(2\zeta -1)\,\delta^{\lambda\,j} + i\epsilon^{\lambda\, j}\sigma^{3}\right]\Tilde{k}^{j}+m_{f}\,\sigma^{3}\sigma^{\lambda}}{\zeta\,(1-\zeta)\,q^{+}}$$
where we used $\sigma^{i}\sigma^{j}=\delta^{ij}+i\epsilon^{ij}\sigma^{3}$ and $\sigma^{\lambda}\sigma^{3}=-\sigma^{3}\sigma^{\lambda}$, and $$\chi^{\dagger}_{\lambda_{1}}\left[(2\zeta -1)\,\delta^{\lambda\,j} + i\epsilon^{\lambda\, j}\sigma^{3}\right]\chi_{\lambda_{2}} = \delta_{\lambda_{1}\,\lambda_{2}}\left[(2\zeta - 1)\delta^{\lambda\,j} +2i\epsilon^{\lambda\,j}\,\lambda_{1}\right],\hspace{2em} \chi^{\dagger}_{\lambda_{1}}\,\sigma^{3}\sigma^{\lambda}\,\chi_{\lambda_{2}} = \delta_{\lambda_{2},-\lambda_{1}}\left[2\lambda_{1}\,\delta^{\lambda 1}-i\,\delta^{\lambda 2}\right]\equiv \mu^{\lambda}_{\lambda_{1}\lambda_{2}}.$$
The mass term flips the helicity ($\lambda_{2}=-\lambda_{1}$). The matrix element defined above can be rewritten as follows:
\begin{align}
    \bra{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(p)}H_{\gamma\rightarrow q\Bar{q}}\ket{\gamma_{\lambda}(q)} & = (2\pi)^{3}\, \delta^{(3)}(q-k-p)\, \frac{e\,e_{f}\,\delta_{\alpha\beta}\,\left\{\left[(2\zeta -1)\,\delta^{\lambda\,j}+2i\epsilon^{\lambda\,j}\,\lambda_{1}\right]\,\delta_{\lambda_{1}\,\lambda_{2}}\,\Tilde{k}^{j}+m_{f}\,\mu^{\lambda}_{\lambda_{1}\lambda_{2}}\right\}}{2\,\sqrt{2\,q^{+\,3}}\, \zeta\,(1-\zeta)}
\end{align}
\item  Gluon emission from anti - quark
\begin{align}
    \bra{q^{\alpha'}_{\lambda'_{1}}(k')\,\Bar{q}^{\beta'}_{\lambda'_{2}}(p')\,g^{a'}_{i'}(q')}H_{\Bar{q}g}\ket{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(p)} & = -\,(2\pi)^{6}\, \delta^{(3)}(k-k')\, \delta^{(3)}(p-p'-q')\,\delta_{\alpha'\,\alpha}\,\delta_{\lambda_{1}\,\lambda'_{1}}\, \frac{g\,t^{a'}_{\beta\,\beta'}}{2\sqrt{2\,q'^{+}}} \notag \\ & \chi^{\dagger}_{\lambda_{2}}\,\left[\frac{2\,q'^{i'}}{q'^{+}}-\sigma^{i'}\,\frac{\sigma\cdot p'+m_{f}\,\sigma^{3}}{p'^{+}}-\frac{\sigma\cdot p+m_{f}\,\sigma^{3}}{p^{+}}\,\sigma^{i'}\right]\,\chi_{\lambda'_{2}}
\end{align}
With $p^{+}=(1-\zeta)q^{+}$, $p'^{+}=\eta q^{+}$, $q'^{+}=\xi q^{+}$ and $\Tilde{p}=q'-\frac{\xi}{1-\zeta}\,p$ the terms proportional to $p$ cancel and
$$\chi^{\dagger}_{\lambda}\left[\ldots\right]\chi_{\lambda_{2}} = \frac{\tau^{i'n}_{\lambda\lambda_{2}}(\xi,\eta)\,\Tilde{p}^{n}+\xi\,\om\,\mu^{i'}_{\lambda\lambda_{2}}}{\xi\,\eta\,q^{+}},\hspace{3em} \om = \frac{\xi\,m_{f}}{1-\zeta},$$
because the mass terms combine into $m_{f}\,\sigma^{3}\sigma^{i'}\left(\frac{1}{p'^{+}}-\frac{1}{p^{+}}\right) = \frac{\xi^{2}m_{f}}{(1-\zeta)}\,\frac{\sigma^{3}\sigma^{i'}}{\xi\,\eta\,q^{+}}$. Again the mass term flips the antiquark helicity.
\item  Instantaneous contribution (unchanged: the $\gamma^{+}$ in the instantaneous propagator removes the mass)
\begin{align}
    \bra{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(s)\,g^{b}_{j}(p)}\,H_{\gamma\rightarrow q\Bar{q}g}\ket{\gamma} & = -(2\pi)^{3}\, \delta^{(3)}(q-k-p-s)\, \frac{g\,e\,e_{f}\,\delta_{\lambda_{1}\lambda_{2}}\,t^{b}_{\alpha\beta}}{4\,\sqrt{q^{+}p^{+}}}\, \frac{\delta^{ij}+2i\,\epsilon^{ij}\lambda_{1}}{p^{+}+s^{+}}
\end{align}
\end{enumerate}
\subsection{Notations}
$$\Tilde{k} = k - \zeta\,q,\hspace{2em}\, \Tilde{p} = p - \frac{\xi}{1-\zeta}\, (q-k),\hspace{2em}\,\Tilde{Q}^{2} = \zeta(1-\zeta)\, Q^{2}$$
$$\eps^{2} = \Tilde{Q}^{2}+m^{2}_{f} = \zeta\,(1-\zeta)\,Q^{2}+m^{2}_{f},\hspace{2em} \om = \frac{\xi\,m_{f}}{1-\zeta},\hspace{2em} \Ef^{2} = Q^{2}+m^{2}_{f}\left(\frac{1}{\zeta}+\frac{1}{\eta}\right),\hspace{2em} a = \frac{\xi\,\eta}{\zeta\,(1-\zeta)^{2}}$$
$$\varphi^{ij}_{\lambda_{1}\lambda}(\zeta) = \delta_{\lambda_{1}\lambda}\left[(2\zeta -1)\delta^{ij} +2i\epsilon^{ij}\lambda_{1}\right],\hspace{3em} \tau^{mn}_{\lambda \lambda_{2}}(\xi,\eta) = \delta_{\lambda \lambda_{2}}\,\left[(2(1-\zeta)-\xi)\delta^{mn}+2i\,\xi\,\epsilon^{mn}\,\lambda\right] $$
$$\mu^{i}_{\lambda\lambda'} = \delta_{\lambda',-\lambda}\left[2\lambda\,\delta^{i1}-i\,\delta^{i2}\right] = \chi^{\dagger}_{\lambda}\,\sigma^{3}\sigma^{i}\,\chi_{\lambda'}\hspace{3em}(\text{helicity-flip vertex; } \textstyle\sum_{i}\mu^{i}_{\lambda\lambda'}\mu^{i*}_{\lambda\lambda'} = 2\,\delta_{\lambda',-\lambda})$$
Here $\eps$ is the usual mass-shifted photon scale (not the Levi-Civita symbol $\epsilon^{ij}$). The photon vertex and the antiquark-gluon vertex are
$$\text{photon: }\ \varphi^{ij}_{\lambda_{1}\lambda}(\zeta)\,\Tilde{k}^{j}+m_{f}\,\mu^{i}_{\lambda_{1}\lambda},\hspace{3em} \text{gluon off the antiquark: }\ \tau^{mn}_{\lambda\lambda_{2}}(\xi,\eta)\,\Tilde{p}^{n}+\xi\,\om\,\mu^{m}_{\lambda\lambda_{2}}.$$
The energy denominator is as follows:
$$E_{q}+E_{\Bar{q}}-E_{\gamma} = \frac{\Tilde{k}^{2}+\Tilde{Q}^{2}+m^{2}_{f}}{2\zeta\, (1-\zeta)\,q^{+}} = \frac{\Tilde{k}^{2}+\eps^{2}}{2\zeta\, (1-\zeta)\,q^{+}}$$
$$E_{q_{1}}+E_{\Bar{q}_{1}}+E_{g_{1}}-E_{\gamma} = \frac{(1-\zeta)\, (\Tilde{p}^{2}+\om^{2})}{2\,\xi\, \eta\,q^{+}} + \frac{(\Tilde{k}^{2}+\eps^{2})}{2\zeta\,(1-\zeta)\,q^{+}} = \frac{\zeta\,(1-\zeta)^{2}\,(\Tilde{p}^{2}+\om^{2})+\xi\,\eta\,(\Tilde{k}^{2}+\eps^{2})}{2\,\zeta\,(1-\zeta)\,\xi\,\eta\,q^{+}}$$
\begin{align*}E_{q_{1}}+E_{\Bar{q}_{1}}+E_{g_{1}}-E_{q}-E_{\Bar{q}} & = \frac{p^{2}}{2q^{+}\,\xi}+\frac{(q-k-p)^{2}+m^{2}_{f}}{2q^{+}\, \eta}-\frac{(q-k)^{2}+m^{2}_{f}}{2q^{+}\, (1-\zeta)} \\ & = \frac{\left[(1-\zeta)\,p - \xi\,(q-k)\right]^{2}}{2\xi\,(1-\zeta)\,\eta\,q^{+}}+\frac{m^{2}_{f}}{2q^{+}}\left(\frac{1}{\eta}-\frac{1}{1-\zeta}\right) = \frac{(1-\zeta)^{2}\,\Tilde{p}^{2}+\xi^{2}m^{2}_{f}}{2\xi\,(1-\zeta)\,\eta\,q^{+}}\end{align*}
Simplifying it we get
$$E_{q_{1}}+E_{\Bar{q}_{1}}+E_{g_{1}}-E_{q}-E_{\Bar{q}} = \frac{(1-\zeta)\, (\Tilde{p}^{2}+\om^{2})}{2\xi\,\eta\,q^{+}}$$
The gluon emission therefore sees the ``dead-cone'' scale $\om=\xi m_{f}/(1-\zeta)$, and when only the three-parton denominator is present (instantaneous term) the total scale is
$$\zeta(1-\zeta)^{2}\,\om^{2}+\xi\,\eta\,\eps^{2} = \zeta\,(1-\zeta)\,\xi\,\eta\,\Ef^{2},\hspace{3em} \Ef^{2} = Q^{2}+\frac{m^{2}_{f}}{\zeta}+\frac{m^{2}_{f}}{\eta}.$$

\section{Outgoing component of wave function through LCPT}
\begin{enumerate}
\item Anti quark emits a gluon before the shock wave
\begin{figure}[h]
    \centering
    \wavefig{Wave functions/1.jpeg}{Fig.~1: $\Bar{q}\rightarrow\Bar{q}g$ before the shock wave}
    \label{fig:1}
\end{figure}
\begin{align}
    \ket{\gamma}^{(1)}_{q\Bar{q}g} & = \ket{q_{2}\Bar{q}_{2}g_{2}}\,\bra{q_{2}\Bar{q}_{2}g_{2}}\,S\,\ket{q_{1}\Bar{q}_{1}g_{1}}\times \frac{\bra{q_{1}\Bar{q}_{1}g_{1}}H_{\Bar{q}g}\ket{q\Bar{q}}}{E_{\Bar{q}_{1}}+E_{q_{1}}+E_{g_{1}}-E_{\gamma}} \times \frac{\bra{q\Bar{q}}H_{\gamma\rightarrow q\Bar{q}}\ket{\gamma}}{E_{q}+E_{\Bar{q}}-E_{\gamma}}
\end{align}
\begin{align}
    \ket{\gamma}^{(1)}_{q\Bar{q}g} & = \int\, d^{3}k''\, d^{3}p''\, d^{3}r\,d^{3}k'\, d^{3}p'\, d^{3}q'\, d^{3}k\, d^{3}p\, \ket{q^{\alpha''}_{\lambda''_{1}}(k'')\,\Bar{q}^{\beta''}_{\lambda''_{2}}(p'')\,g^{b}_{j}(r)}\,\bra{q^{\alpha''}_{\lambda''_{1}}(k'')\,\Bar{q}^{\beta''}_{\lambda''_{2}}(p'')\,g^{b}_{j}(r)}\,S\,\ket{q^{\alpha'}_{\lambda'_{1}}(k')\,\Bar{q}^{\beta'}_{\lambda'_{2}}(p')\,g^{a'}_{i'}(q')}\times\notag \\ & \frac{\bra{q^{\alpha'}_{\lambda'_{1}}(k')\,\Bar{q}^{\beta'}_{\lambda'_{2}}(p')\,g^{a'}_{i'}(q')}H_{\Bar{q}g}\ket{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(p)}}{E_{\Bar{q}}(p')+E_{q}(k')+E_{g}(q')-E_{\gamma}} \times \frac{\bra{q^{\alpha}_{\lambda_{1}}(k)\,\Bar{q}^{\beta}_{\lambda_{2}}(p)}H_{\gamma\rightarrow q\Bar{q}}\ket{\gamma_{\lambda}(q)}}{E_{\Bar{q}}(p)+E_{q}(k)-E_{\gamma}(q,Q)}
\end{align}
We will first evaluate the matrix elements without scattering (i.e., $S = 1$) [Eq. (4.21) in 2211.04837]. With the massive vertices and energy denominators of Sec.~1,
\begin{align}
    \ket{\gamma_{i}\,(Q,q^{+},q)}^{(1\,a)}_{q\Bar{q}g} & = -\,\frac{1}{(2\pi)^{6}}\, \int\, d^{2}\Tilde{k}\, d^{2}\Tilde{p}\, \int_{0}^{1}\, d\zeta\, \int_{0}^{1-\zeta}\, d\xi\,\ket{q^{\alpha}_{\lambda_{1}}(\zeta q^{+},k)\Bar{q}^{\beta}_{\lambda_{2}}(\eta\,q^{+},q-k-p)g^{a}_{m}(\xi q^{+},p)} \frac{e\,e_{f}\,g\,q^{+}\,\zeta(1-\zeta)}{2\sqrt{\xi}\,(\Tilde{k}^{2}+\eps^{2})}\notag \\ & \frac{\left[\varphi^{ij}_{\lambda_{1}\lambda}(\zeta)\, \Tilde{k}^{j}+m_{f}\,\mu^{i}_{\lambda_{1}\lambda}\right]\,\left[\tau^{mn}_{\lambda \lambda_{2}}(\xi,\eta)\, \Tilde{p}^{n}+\xi\,\om\,\mu^{m}_{\lambda\lambda_{2}}\right]\, t^{a}_{\alpha\beta}}{\left[\zeta(1-\zeta)^{2}\, (\Tilde{p}^{2}+\om^{2}) +\xi\,\eta\,(\Tilde{k}^{2}+\eps^{2})\right]}
\end{align}
(a sum over the intermediate antiquark helicity $\lambda$ is understood). Compared to the massless case: $\Tilde{Q}^{2}\rightarrow\eps^{2}$, $\Tilde{p}^{2}\rightarrow\Tilde{p}^{2}+\om^{2}$, and each vertex acquires its helicity-flip mass term. The product of the two brackets therefore contains four helicity paths:
$$\underbrace{\varphi\,\tau\;\Tilde{k}^{j}\Tilde{p}^{n}}_{\text{no flip}},\hspace{2em} \underbrace{\xi\,\om\,\varphi\,\mu\;\Tilde{k}^{j}}_{\text{flip at the gluon vertex}},\hspace{2em} \underbrace{m_{f}\,\mu\,\tau\;\Tilde{p}^{n}}_{\text{flip at the photon vertex}},\hspace{2em} \underbrace{m_{f}\,\xi\,\om\,\mu\,\mu}_{\text{two flips}} .$$
The first and the last have $\lambda_{2}=\lambda_{1}$; the two middle ones have $\lambda_{2}=-\lambda_{1}$.

However, Wilson lines can be obtained only if the wave function is expressed in transverse coordinates. Therefore, we perform a Fourier transform to rewrite the wave function in transverse coordinate space. Performing Fourier transform for transverse co-ordinates,
\begin{align}
   \ket{q^{\alpha}_{\lambda_{1}}(k)\Bar{q}^{\beta}_{\lambda_{2}}(q-k-p)g^{a}_{m}(p)} & = \int_{x,y,z}\, e^{ik\cdot x}\, e^{ip\cdot z}\, e^{i(q-k-p)\cdot y}\,\ket{q^{\alpha}_{\lambda_{1}}(x)\Bar{q}^{\beta}_{\lambda_{2}}(y)g^{a}_{m}(z)}
\end{align}
Rewriting $k$, $p$ in terms of $\Tilde{k}$ and $\Tilde{p}$ respectively, we get
\begin{align}
    e^{i\left[\zeta\,x +\eta\, y + \xi\,z\right]\cdot q}\,e^{-i\,(y-z)\cdot \Tilde{p}}\, e^{i\left[(1-\zeta)x-\eta y-\xi z\right]\cdot \frac{\Tilde{k}}{(1-\zeta)}} & = e^{i\,\left(\zeta\,x+\eta\,y + \xi\,z\right)\cdot q}\,e^{-i\,(y-z)\cdot \Tilde{p}}\, e^{i\,\left(x-\frac{\eta\,y + \xi\,z}{(1-\zeta)}\right)\cdot \Tilde{k}} \notag \\ & = e^{i\,w\cdot q}\, e^{-i\,Y\cdot \Tilde{p}}\, e^{i\,R\cdot \Tilde{k}}.
\end{align}
The transverse coordinates of the photon $(w)$ and the intermediate antiquark ($y'$, before emitting the gluon) are fixed by centre of energy conservation at the two vertices (this is kinematics and does not depend on the mass):
\begin{align}
    w & = \zeta\,x +(1-\zeta)\,y' \notag \\ & = \zeta\,x + \eta\,y + \xi\,z,
\end{align}
and
\begin{align}
    y' & = \frac{\eta\,y + \xi\,z}{(1-\zeta)}.
\end{align}
The relative distances are defined as
$$R = x - y'; \hspace{2em} Y = y-z$$
where we used
$$(1-\zeta)\,x - \eta\,y -\xi\,z = (1-\zeta)\, (x-y') = (1-\zeta)\,R$$
In the massless case we used the following Fourier transform (note the sign of the exponent: with $e^{+ip\cdot Z}$ the result is as written; our phase above is $e^{-i\Tilde{p}\cdot Y}$, i.e.\ $Z=-Y$):
\begin{align}
    \int\, \frac{d^{2}k}{2\pi}\, \frac{d^{2}p}{2\pi}\, \frac{k^{i}p^{j}}{k^{2}\, \left[ak^{2}+bp^{2}\right]}\,e^{ik\cdot X + ip\cdot Z} & = -\frac{X^{i}Z^{j}}{Z^{2}\,\left[bX^{2}+aZ^{2}\right]}
\end{align}
With masses the two denominators contain two different scales, $\eps$ and $\om$, and the transform is no longer a single Bessel function. We need four transforms, one for each helicity path. Define
\begin{align}\label{eq:FTdef}
    (1-\zeta)\int\, \frac{d^{2}\Tilde{k}}{2\pi}\, \frac{d^{2}\Tilde{p}}{2\pi}\, \frac{e^{i\Tilde{k}\cdot R}\,e^{-i\Tilde{p}\cdot Y}\;\left\{\Tilde{k}^{j}\Tilde{p}^{n},\ \Tilde{k}^{j},\ \Tilde{p}^{n},\ 1\right\}}{(\Tilde{k}^{2}+\eps^{2})\,\left[\zeta(1-\zeta)^{2}(\Tilde{p}^{2}+\om^{2})+\xi\,\eta\,(\Tilde{k}^{2}+\eps^{2})\right]} & = \left\{R^{j}Y^{n}\,\G{11},\ \ i\,R^{j}\,\G{10},\ \ -i\,Y^{n}\,\G{01},\ \ \G{00}\right\}
\end{align}
To evaluate them we use Schwinger parameters,
$$\frac{1}{(\Tilde{k}^{2}+\eps^{2})\left[\zeta(1-\zeta)^{2}(\Tilde{p}^{2}+\om^{2})+\xi\eta(\Tilde{k}^{2}+\eps^{2})\right]} = \frac{1}{\xi\,\eta}\int_{0}^{\infty}du\int_{0}^{\infty}dv\; e^{-(u+v)(\Tilde{k}^{2}+\eps^{2})-\frac{v}{a}(\Tilde{p}^{2}+\om^{2})},$$
and the Gaussian integrals
$$\int\frac{d^{2}\Tilde{k}}{2\pi}\,e^{i\Tilde{k}\cdot R-U\Tilde{k}^{2}}\left\{1,\Tilde{k}^{j}\right\} = \frac{e^{-R^{2}/4U}}{2U}\left\{1,\frac{iR^{j}}{2U}\right\},\hspace{2em} \int\frac{d^{2}\Tilde{p}}{2\pi}\,e^{-i\Tilde{p}\cdot Y-S\Tilde{p}^{2}}\left\{1,\Tilde{p}^{n}\right\} = \frac{e^{-Y^{2}/4S}}{2S}\left\{1,-\frac{iY^{n}}{2S}\right\},$$
with $U=u+v$ and $S=v/a$. Writing $S = xU/a$ with $x\in(0,1)$ ($x$ is a Feynman parameter) and using $\int_{0}^{\infty}dU\,U^{\nu-1}e^{-AU-B/U} = 2\,(B/A)^{\nu/2}K_{\nu}(2\sqrt{AB})$, every transform becomes a one-dimensional integral:
\begin{align}\label{eq:Gdef}
    \G{\alpha\beta}(R,Y) & = \frac{a^{\beta}}{2\,\zeta\,(1-\zeta)}\int_{0}^{1}\frac{dx}{x^{1+\beta}}\left(\frac{A_{x}}{\Delta_{x}}\right)^{\frac{\alpha+\beta}{2}}K_{\alpha+\beta}\!\left(\sqrt{A_{x}\,\Delta_{x}}\right),\hspace{2em} \alpha,\beta\in\{0,1\}, \notag \\
    A_{x} & = \eps^{2}+\frac{x\,\om^{2}}{a} = \zeta(1-\zeta)\,Q^{2}+m^{2}_{f}\left(1+\frac{x\,\zeta\,\xi}{\eta}\right),\hspace{3em} \Delta_{x} = R^{2}+\frac{a\,Y^{2}}{x}.
\end{align}
Explicitly,
\begin{align}
    \G{11} & = \frac{a}{2\zeta(1-\zeta)}\int_{0}^{1}\frac{dx}{x^{2}}\,\frac{A_{x}}{\Delta_{x}}\,K_{2}\!\left(\sqrt{A_{x}\Delta_{x}}\right), & \G{10} & = \frac{1}{2\zeta(1-\zeta)}\int_{0}^{1}\frac{dx}{x}\,\sqrt{\frac{A_{x}}{\Delta_{x}}}\,K_{1}\!\left(\sqrt{A_{x}\Delta_{x}}\right), \notag \\
    \G{01} & = \frac{a}{2\zeta(1-\zeta)}\int_{0}^{1}\frac{dx}{x^{2}}\,\sqrt{\frac{A_{x}}{\Delta_{x}}}\,K_{1}\!\left(\sqrt{A_{x}\Delta_{x}}\right), & \G{00} & = \frac{1}{2\zeta(1-\zeta)}\int_{0}^{1}\frac{dx}{x}\,K_{0}\!\left(\sqrt{A_{x}\Delta_{x}}\right).
\end{align}
The end point $x=1$ is the three-parton configuration: $A_{1}\Delta_{1} = \Ef^{2}\,\mathcal{D}^{2}$, with $\mathcal{D}$ defined below. For $m_{f}=0$, $A_{x}=\Tilde{Q}^{2}$ does not depend on $x$; substituting $s^{2}=R^{2}+aY^{2}/x$ and using $\frac{d}{ds}\left[\frac{K_{1}(\Tilde{Q}s)}{s}\right] = -\frac{\Tilde{Q}\,K_{2}(\Tilde{Q}s)}{s}$ gives back the massless closed form,
$$\G{11}\Big|_{m_{f}=0} = \frac{\Tilde{Q}\,K_{1}\!\left(\Tilde{Q}\sqrt{R^{2}+aY^{2}}\right)}{\zeta(1-\zeta)\,Y^{2}\sqrt{R^{2}+aY^{2}}} = \frac{1}{Y^{2}}\,\frac{Q\,K_{1}(Q\,\mathcal{D})}{\mathcal{D}},\hspace{3em} \G{10},\,\G{01},\,\G{00}\ \text{ appear multiplied by } m_{f}.$$
where $a = \frac{\xi\, \eta}{\left[\zeta\,(1-\zeta)^{2}\right]}$ and $K_{\nu}$ is the modified Bessel function of the second kind. Now, implementing all these steps in the outgoing component of wave function, we get
\begin{align}
    \ket{\gamma_{i}\,(Q,q^{+},q)}^{(1\,a)}_{q\Bar{q}g} &  = -\,\frac{e\,e_{f}\,g\,q^{+}\,}{2\,(2\pi)^{4}}\, \int_{x,y,z} \int_{0}^{1}\, d\zeta\, \int_{0}^{1-\zeta}\, d\xi\,\ket{q^{\alpha}_{\lambda_{1}}(\zeta q^{+},x)\Bar{q}^{\beta}_{\lambda_{2}}(\eta\,q^{+},y)g^{a}_{m}(\xi q^{+},z)}\, \frac{\zeta}{\sqrt{\xi}}\, \delta^{(2)}(w-c)\,\Psi^{im}_{\lambda_{1}\lambda_{2}}\!\left[\mathcal{G}\right] t^{a}_{\alpha\beta}

\section{Cross section}
\subsection{Direct contributions - Anti quark contribution}
The cross section for trijet production is given by [(5.2)]
\begin{align}
    \frac{d^{3}N}{d^{3}k_{1}\,d^{3}k_{2}\, d^{3}k_{3}}\, (2\pi)\,\delta\left(q^{+}-k^{+}_{1}-k^{+}_{2}-k^{+}_{3}\right) & =  \frac{1}{2}\,\int_{w,w'}\, e^{-iq\cdot (w-w')}
\bra{\gamma^{i}_{T}(Q,q^{+},w')}\,\mathcal{N}_{q}(k_{1})\,\mathcal{N}_{\Bar{q}}(k_{2})\,\mathcal{N}_{g}(k_{3})\,\ket{\gamma^{i}_{T}(Q,q^{+},w)}_{q\Bar{q}g}
\end{align}
Inserting Eq.~\eqref{eq:qbarfull} in the amplitude and in the complex conjugate amplitude (barred coordinates) gives
\begin{align}\label{eq:dirqbar}
     \frac{d^{3}N^{\text{direct}}}{d^{3}k_{1}\,d^{3}k_{2}\, d^{3}k_{3}} & = \frac{e^{2}\,g^{2}\,\zeta^{2}\,C_{F}\,N_{c}}{8\, (2\pi)^{12}\,\xi\,q^{+\,2}}\, \delta\left(q^{+}-k^{+}_{1}-k^{+}_{2}-k^{+}_{3}\right)\, \int_{x,y,z,\Bar{x},\Bar{y},\Bar{z}}\, e^{-i\,k_{1}\cdot (x-\Bar{x})}\, e^{-i\,k_{2}\cdot (y-\Bar{y})}\, e^{-i\,k_{3}\cdot (z-\Bar{z})}\;\sum_{f}\,e^{2}_{f}\notag \\ & \bigg[ \Hs\!\left[\mathcal{G};\Bar{\mathcal{G}}\right]\,\mathcal{W}(x,y,z,\Bar{x},\Bar{y},\Bar{z})-\Hs\!\left[\mathcal{A};\Bar{\mathcal{G}}\right]\,\mathcal{W}(x,y',y',\Bar{x},\Bar{y},\Bar{z})\notag \\ & -\Hs\!\left[\mathcal{G};\Bar{\mathcal{A}}\right]\,\mathcal{W}(x,y,z,\Bar{x},\Bar{y}',\Bar{y}')+ \Hs\!\left[\mathcal{A};\Bar{\mathcal{A}}\right]\,\mathcal{W}(x,y',y',\Bar{x},\Bar{y}',\Bar{y}')\bigg]
\end{align}
where
$$\Hs\!\left[F;\Bar{F}\right] = \sum_{\lambda_{1},\lambda_{2}}\sum_{l,r}\M^{lr}_{\lambda_{1}\lambda_{2}}\!\left[F\right](x,y,z)\;\M^{lr\,*}_{\lambda_{1}\lambda_{2}}\!\left[\Bar{F}\right](\Bar{x},\Bar{y},\Bar{z}),$$
$\M[\mathcal{G}]$ is the full vertex function \eqref{eq:Mdef} (regular $+$ instantaneous) and $\M[\mathcal{A}] = \M|_{\text{after}}$ (regular only, i.e.\ $\GE\rightarrow 0$). Barred functions are evaluated at $(\Bar{R},\Bar{Y})$, $\Bar{F}^{(\alpha\beta)}=F^{(\alpha\beta)}(\Bar{R},\Bar{Y})$, $\Bar{\mathcal{G}}_{E}=\GE(\Bar{x},\Bar{y},\Bar{z})$. Compared to the massless formula the factor $Q^{2}$ in front is now inside the radial functions ($\frac{Q^{2}K_{1}K_{1}}{\mathcal{D}\Bar{\mathcal{D}}} = \frac{QK_{1}}{\mathcal{D}}\frac{QK_{1}}{\Bar{\mathcal{D}}}$) and the flavour sum is inside the integral. We write $\Hs$ in parts, exactly as in the massless case, plus the new helicity-flip pieces (index convention: $i\leftrightarrow R$, $m\leftrightarrow Y$, $j\leftrightarrow\Bar{R}$, $n\leftrightarrow\Bar{Y}$; $A\cdot B$ is the transverse scalar product).
\begin{itemize}
    \item Regular $\times$ Regular
    \begin{align}
        \Hs_{\text{reg}\times\text{reg}} & = R^{i}\,Y^{m}\,\Bar{R}^{j}\,\Bar{Y}^{n}\,\Big\{8\left[\eta^{2}+(1-\zeta)^{2}\right]\,\left[\zeta^{2}+(1-\zeta)^{2}\right]\,\delta^{ij}\,\delta^{mn} -8\,\xi\,(2\zeta-1)\,\left(2(1-\zeta)-\xi\right)\,\epsilon^{ij}\,\epsilon^{mn}\Big\}\,F^{(11)}\Bar{F}^{(11)} \notag \\
        & + 8\,m^{2}_{f}\left[\eta^{2}+(1-\zeta)^{2}\right]\,(Y\cdot\Bar{Y})\,F^{(01)}\Bar{F}^{(01)} + 8\,\xi^{2}\om^{2}\left[\zeta^{2}+(1-\zeta)^{2}\right]\,(R\cdot\Bar{R})\,F^{(10)}\Bar{F}^{(10)} + 8\,m^{2}_{f}\,\xi^{2}\om^{2}\,F^{(00)}\Bar{F}^{(00)} \notag \\
        & - 8\,\zeta\,\eta\,m_{f}\,\xi\,\om\,\Big[(R\cdot Y)\,F^{(11)}\Bar{F}^{(00)} + (\Bar{R}\cdot\Bar{Y})\,F^{(00)}\Bar{F}^{(11)} + (R\cdot\Bar{Y})\,F^{(10)}\Bar{F}^{(01)} + (Y\cdot\Bar{R})\,F^{(01)}\Bar{F}^{(10)}\Big]
    \end{align}
    The first line is the massless result (helicity conserving at both vertices); the second line is the square of the paths with one helicity flip ($m_{f}^{2}$ times the massless gluon-vertex factor $\eta^{2}+(1-\zeta)^{2}$, and $\xi^{2}\om^{2}$ times the massless photon-vertex factor $\zeta^{2}+(1-\zeta)^{2}$) and of the double flip; the third line is the interference between the paths with the same final helicities (no-flip with double-flip, and the two single-flip paths).
    \item Regular $\times$ Instantaneous $+$ h.c.
    \begin{align}\label{eq:RIqbar}
        \Hs_{\text{reg}\times\text{inst}} & = -\,\frac{8\,\zeta\,\xi\,\eta^{2}}{(1-\zeta)}\,\Big[(R\cdot Y)\,F^{(11)}\,\Bar{\mathcal{G}}_{E}+(\Bar{R}\cdot\Bar{Y})\,\GE\,\Bar{F}^{(11)}\Big] + \frac{8\,\xi^{2}\eta\,m_{f}\,\om}{(1-\zeta)}\,\Big[F^{(00)}\,\Bar{\mathcal{G}}_{E}+\GE\,\Bar{F}^{(00)}\Big]
    \end{align}
    (only the helicity-conserving paths interfere with the instantaneous term, which has $\lambda_{2}=\lambda_{1}$). The first bracket follows from
    $$\sum_{l,r}\varphi^{li}_{\lambda_{1}\lambda_{1}}(\zeta)\,\tau^{rm}_{\lambda_{1}\lambda_{1}}(\xi,\eta)\left(\delta^{lr}+2i\epsilon^{lr}\lambda_{1}\right)^{*} = 4\,\zeta\,\eta\left(\delta^{im}-2i\,\epsilon^{im}\lambda_{1}\right)\hspace{2em}\Rightarrow\hspace{2em}\sum_{\lambda_{1}}\ \rightarrow\ 8\,\zeta\,\eta\,\delta^{im}.$$
    \item Instantaneous $\times$ Instantaneous
    \begin{align}
        \Hs_{\text{inst}\times\text{inst}} & =  8\left[\frac{\xi\, \eta}{(1-\zeta)}\right]^{2}\,\GE\,\Bar{\mathcal{G}}_{E}
    \end{align}
    \item For the terms where one side is ``after the shock wave'' use $F=\mathcal{A}$ on that side and drop its $\GE$ (so $\Hs[\mathcal{A};\Bar{\mathcal{G}}]$ keeps only the $\Bar{\mathcal{G}}_{E}$ terms of the Regular $\times$ Instantaneous line, and $\Hs[\mathcal{A};\Bar{\mathcal{A}}]$ is Regular $\times$ Regular only).
\end{itemize}
\textbf{Massless limit.} For $m_{f}=0$: $F^{(11)}=\frac{1}{Y^{2}}\frac{QK_{1}(Q\mathcal{D})}{\mathcal{D}}$, $\GE=\frac{QK_{1}(Q\mathcal{D})}{\mathcal{D}}$ and the mass lines drop. With the common factor $\frac{R^{i}Y^{m}\Bar{R}^{j}\Bar{Y}^{n}}{Y^{2}\Bar{Y}^{2}}\frac{Q^{2}K_{1}K_{1}}{\mathcal{D}\Bar{\mathcal{D}}}$ taken out as in the massless note,
\begin{align}
    \Phi^{lirm}_{\lambda_{1}\lambda}\Phi^{ljrn\,*}_{\lambda_{1}\lambda}\Big|_{\text{reg}\times\text{inst}} & = -\,\frac{8\,\zeta\,\xi\,\eta^{2}}{(1-\zeta)}\,\delta^{im}\,\delta^{jn}\left[\frac{Y^{2}}{R\cdot Y}+\frac{\Bar{Y}^{2}}{\Bar{R}\cdot\Bar{Y}}\right], \hspace{3em}
    \Phi^{lirm}_{\lambda_{1}\lambda}\Phi^{ljrn\,*}_{\lambda_{1}\lambda}\Big|_{\text{inst}\times\text{inst}} = 8\,\delta^{im}\,\delta^{jn}\left[\frac{\xi\,\eta}{(1-\zeta)}\right]^{2}\frac{Y^{2}}{R\cdot Y}\,\frac{\Bar{Y}^{2}}{\Bar{R}\cdot\Bar{Y}},
\end{align}
and the Regular $\times$ Regular part is the massless one unchanged. (These two lines replace the corresponding massless expressions, see Appendix~\ref{app:changes}.)

$\mathcal{W}(x,y,z,\Bar{x},\Bar{y},\Bar{z})$ is the Wilson line structure (independent of the quark mass). This has 4 terms:

\begin{itemize}
    \item $\mathcal{W}(x,y,z,\Bar{x},\Bar{y},\Bar{z})$
    \begin{align}
        \mathcal{W}(x,y,z,\Bar{x},\Bar{y},\Bar{z}) & = S_{q\Bar{q}g\,q\Bar{q}g}(x,y,z,\Bar{x},\Bar{y},\Bar{z}) -S_{q\Bar{q}g}(x,y,z) - S^{\dagger}_{q\Bar{q}g}(\Bar{x},\Bar{y},\Bar{z}) + 1
    \end{align}
where the Wilson line averaging is given by 
$$S_{q\Bar{q}g\,q\Bar{q}g}(x,y,z,\Bar{x},\Bar{y},\Bar{z}) = \frac{1}{C_{F}\,N_{c}}\, \langle \left[U^{\dagger}(\Bar{z})U(z)\right]^{ab}\, \text{Tr}\left[V(\Bar{y})t^{a}V^{\dagger}(\Bar{x})V(x)t^{b}V^{\dagger}(y)\right] \rangle$$

$$S_{q\Bar{q}g\,q\Bar{q}g}(x,y,z,\Bar{x},\Bar{y},\Bar{z}) = Q(x,z,\Bar{z},\Bar{x})\, Q(z,y,\Bar{y},\Bar{z})$$
$$S_{q\Bar{q}g}(x,y,z) = S(x,z)\, S(z,y)$$
$$S^{\dagger}_{q\Bar{q}g}(\Bar{x},\Bar{y},\Bar{z}) = S(\Bar{z},\Bar{x})\, S(\Bar{y},\Bar{z})$$
Using Gaussian approximation to reduce quadrupole in to product of dipoles, we get
\begin{align}
    Q(x,z,\Bar{z},\Bar{x}) & = \exp{\left[-\frac{C_{F}}{2}\,\left(\Gamma(x-z)+\Gamma(\Bar{z}-\Bar{x})\right)\right]}-\frac{F(x,z;\Bar{z},\Bar{x})}{F(x,\Bar{z};z,\Bar{x})}\,\Bigg\{\exp{\left[-\frac{C_{F}}{2}\,\left(\Gamma(x-z)+\Gamma(\Bar{z}-\Bar{x})\right)\right]}\notag \\ & -\exp{\left[-\frac{C_{F}}{2}\,\left(\Gamma(x-\Bar{x})+\Gamma(\Bar{z}-z)\right)\right]}\Bigg\}
\end{align}
where $$\exp{\left[-\frac{C_{F}}{2}\,\left(\Gamma(x-z)+\Gamma(\Bar{z}-\Bar{x})\right)\right]} = S(x,z)\,S(\Bar{z},\Bar{x})$$
$$\exp{\left[-\frac{C_{F}}{2}\,\left(\Gamma(x-\Bar{x})+\Gamma(\Bar{z}-z)\right)\right]} = S(x,\Bar{x})\, S(\Bar{z},z)$$
So, 
\begin{align}
    Q(x,z,\Bar{z},\Bar{x}) & = S(x,z)\,S(\Bar{z},\Bar{x}) - \frac{F(x,z;\Bar{z},\Bar{x})}{F(x,\Bar{z};z,\Bar{x})}\, \left[S(x,z)\,S(\Bar{z},\Bar{x})-S(x,\Bar{x})\, S(\Bar{z},z)\right]
\end{align}
Similarly, 
\begin{align}
    Q(z,y,\Bar{y},\Bar{z}) & = S(z,y)\,S(\Bar{y},\Bar{z}) - \frac{F(z,y;\Bar{y},\Bar{z})}{F(z,\Bar{y};y,\Bar{z})}\,\left[S(z,y)\,S(\Bar{y},\Bar{z})-S(z,\Bar{z})\,S(\Bar{y},y)\right]
\end{align}
\item $\mathcal{W}(x,y',y',\Bar{x},\Bar{y},\Bar{z})$
\begin{align}
        \mathcal{W}(x,y',y',\Bar{x},\Bar{y},\Bar{z}) & = S_{q\Bar{q}g\,q\Bar{q}g}(x,y',y',\Bar{x},\Bar{y},\Bar{z}) -S_{q\Bar{q}g}(x,y',y') - S^{\dagger}_{q\Bar{q}g}(\Bar{x},\Bar{y},\Bar{z}) + 1
    \end{align}
where the Wilson line averaging is given by 
$$S_{q\Bar{q}g\,q\Bar{q}g}(x,y',y',\Bar{x},\Bar{y},\Bar{z}) =  \frac{1}{C_{F}\,N_{c}}\, \langle \left[U^{\dagger}(\Bar{z})U(y')\right]^{ab}\, \text{Tr}\left[V(\Bar{y})t^{a}V^{\dagger}(\Bar{x})V(x)t^{b}V^{\dagger}(y')\right] \rangle$$

At large $N_{c}$ limit, we get
$$S_{q\Bar{q}g\,q\Bar{q}g}(x,y',y',\Bar{x},\Bar{y},\Bar{z}) = Q(x,y',\Bar{z},\Bar{x})\, Q(y',y',\Bar{y},\Bar{z}) = Q(x,y',\Bar{z},\Bar{x})\, S(\Bar{y},\Bar{z})$$
$$S_{q\Bar{q}g}(x,y',y') = S(x,y')\, S(y',y') = S(x,y') $$
$$S^{\dagger}_{q\Bar{q}g}(\Bar{x},\Bar{y},\Bar{z}) = S(\Bar{z},\Bar{x})\, S(\Bar{y},\Bar{z})$$
Using Gaussian approximation,
\begin{align}
    Q(x,y',\Bar{z},\Bar{x}) & = S(x,y')\,S(\Bar{z},\Bar{x}) - \frac{F(x,y';\Bar{z},\Bar{x})}{F(x,\Bar{z};y',\Bar{x})}\, \left[S(x,y')\,S(\Bar{z},\Bar{x})-S(x,\Bar{x})\, S(\Bar{z},y')\right]
\end{align}
\item $\mathcal{W}(x,y,z,\Bar{x},\Bar{y}',\Bar{y}')$
\begin{align}
        \mathcal{W}(x,y,z,\Bar{x},\Bar{y}',\Bar{y}') & = S_{q\Bar{q}g\,q\Bar{q}g}(x,y,z,\Bar{x},\Bar{y}',\Bar{y}') -S_{q\Bar{q}g}(x,y,z) - S^{\dagger}_{q\Bar{q}g}(\Bar{x},\Bar{y}',\Bar{y}') + 1
    \end{align}
where the Wilson line averaging is given by 
$$S_{q\Bar{q}g\,q\Bar{q}g}(x,y,z,\Bar{x},\Bar{y}',\Bar{y}') = \frac{1}{C_{F}\,N_{c}}\, \langle \left[U^{\dagger}(\Bar{y}')U(z)\right]^{ab}\, \text{Tr}\left[V(\Bar{y})t^{a}V^{\dagger}(\Bar{x})V(x)t^{b}V^{\dagger}(y)\right] \rangle$$

\subsection{Direct contribution - Quark contributions}
We need not explicitly derive the direct contribution for quarks. Instead we do the following changes from the cross section of the anti quark sector
$$\zeta \rightarrow \eta, R\rightarrow \Tilde{R}, Y\rightarrow X, y'\rightarrow x', x\leftrightarrow y,\hspace{2em} \eps\rightarrow\epsq,\ \om\rightarrow\omq,\ \mathcal{G}\rightarrow\Tilde{\mathcal{G}},\ \mathcal{A}\rightarrow\Tilde{\mathcal{A}}$$
and use the quark-emission vertex function $\Mq$ of Eq.~\eqref{eq:Mqdef}:
\begin{align}
     \frac{d^{3}N^{\text{direct}}}{d^{3}k_{1}\,d^{3}k_{2}\, d^{3}k_{3}} & = \frac{e^{2}\,g^{2}\,\eta^{2}\,C_{F}\,N_{c}}{8\, (2\pi)^{12}\,\xi\,q^{+\,2}}\, \delta\left(q^{+}-k^{+}_{1}-k^{+}_{2}-k^{+}_{3}\right)\, \int_{x,y,z,\Bar{x},\Bar{y},\Bar{z}}\, e^{-i\,k_{1}\cdot (x-\Bar{x})}\, e^{-i\,k_{2}\cdot (y-\Bar{y})}\, e^{-i\,k_{3}\cdot (z-\Bar{z})}\;\sum_{f}\,e^{2}_{f}\notag \\ & \bigg[ \Tilde{\Hs}\!\left[\Tilde{\mathcal{G}};\Bar{\Tilde{\mathcal{G}}}\right]\mathcal{W}(y,x,z,\Bar{y},\Bar{x},\Bar{z})-\Tilde{\Hs}\!\left[\Tilde{\mathcal{A}};\Bar{\Tilde{\mathcal{G}}}\right]\mathcal{W}(y,x',x',\Bar{y},\Bar{x},\Bar{z})\notag \\ & -\Tilde{\Hs}\!\left[\Tilde{\mathcal{G}};\Bar{\Tilde{\mathcal{A}}}\right]\mathcal{W}(y,x,z,\Bar{y},\Bar{x}',\Bar{x}')+ \Tilde{\Hs}\!\left[\Tilde{\mathcal{A}};\Bar{\Tilde{\mathcal{A}}}\right]\mathcal{W}(y,x',x',\Bar{y},\Bar{x}',\Bar{x}')\bigg]
\end{align}
where $\Tilde{\Hs}[F;\Bar{F}] = \sum_{\lambda_{1}\lambda_{2}}\sum_{l,r}\Mq^{lr}_{\lambda_{1}\lambda_{2}}[F](x,y,z)\,\Mq^{lr\,*}_{\lambda_{1}\lambda_{2}}[\Bar{F}](\Bar{x},\Bar{y},\Bar{z})$ (after the shock wave: $\GE\rightarrow 0$). We write it in parts (index convention $i\leftrightarrow\Tilde{R}$, $m\leftrightarrow X$, $j\leftrightarrow\Bar{\Tilde{R}}$, $n\leftrightarrow\Bar{X}$):
\begin{itemize}
    \item Regular $\times$ Regular
    \begin{align}
        \Tilde{\Hs}_{\text{reg}\times\text{reg}} & = \Tilde{R}^{i}\,X^{m}\,\Bar{\Tilde{R}}^{j}\,\Bar{X}^{n}\,\Big\{8\left[\zeta^{2}+(1-\eta)^{2}\right]\,\left[\eta^{2}+(1-\eta)^{2}\right]\,\delta^{ij}\,\delta^{mn} -8\,\xi\,(2\eta-1)\,\left(2(1-\eta)-\xi\right)\,\epsilon^{ij}\,\epsilon^{mn}\Big\}\,F^{(11)}\Bar{F}^{(11)} \notag \\
        & + 8\,m^{2}_{f}\left[\zeta^{2}+(1-\eta)^{2}\right]\,(X\cdot\Bar{X})\,F^{(01)}\Bar{F}^{(01)} + 8\,\xi^{2}\omq^{2}\left[\eta^{2}+(1-\eta)^{2}\right]\,(\Tilde{R}\cdot\Bar{\Tilde{R}})\,F^{(10)}\Bar{F}^{(10)} + 8\,m^{2}_{f}\,\xi^{2}\omq^{2}\,F^{(00)}\Bar{F}^{(00)} \notag \\
        & + 8\,\zeta\,\eta\,m_{f}\,\xi\,\omq\,\Big[(\Tilde{R}\cdot X)\,F^{(11)}\Bar{F}^{(00)} + (\Bar{\Tilde{R}}\cdot\Bar{X})\,F^{(00)}\Bar{F}^{(11)} + (\Tilde{R}\cdot\Bar{X})\,F^{(10)}\Bar{F}^{(01)} + (X\cdot\Bar{\Tilde{R}})\,F^{(01)}\Bar{F}^{(10)}\Big]
    \end{align}
    (the relative sign of the last line with respect to the antiquark case comes from $R\rightarrow -\Tilde{R}$.)

\section{Summary of the massive functions}\label{app:functions}
All quantities for one flavour $f$; $\eta=1-\zeta-\xi$, $a=\frac{\xi\eta}{\zeta(1-\zeta)^{2}}$, $\Tilde{a}=\frac{\xi\zeta}{\eta(1-\eta)^{2}}$.
\begin{center}
\begin{tabular}{||l|l|l||}
\hline

\hline\hline
photon scale & $\eps^{2}=\zeta(1-\zeta)Q^{2}+m^{2}_{f}$ & $\epsq^{2}=\eta(1-\eta)Q^{2}+m^{2}_{f}$ \\
dead-cone scale & $\om=\xi m_{f}/(1-\zeta)$ & $\omq=\xi m_{f}/(1-\eta)$ \\
three-parton scale & \multicolumn{2}{c||}{$\Ef^{2}=Q^{2}+m^{2}_{f}(1/\zeta+1/\eta)$, \quad $\mathcal{D}^{2}=\zeta\eta\,r^{2}_{xy}+\zeta\xi\,r^{2}_{xz}+\eta\xi\,r^{2}_{yz}$} \\
before SW (regular) & $\G{\alpha\beta}(R,Y)$, Eq.~\eqref{eq:Gdef} & $\Gq{\alpha\beta}(\Tilde{R},X)$: Eq.~\eqref{eq:Gdef} with $\zeta\rightarrow\eta$, $a\rightarrow\Tilde{a}$ \\
after SW & $\Ga{\alpha\beta}(R,Y)$, Eq.~\eqref{eq:Adef} & $\Gaq{\alpha\beta}(\Tilde{R},X)$ \\
instantaneous & \multicolumn{2}{c||}{$\GE=\Ef K_{1}(\Ef\mathcal{D})/\mathcal{D}$ (the same function in both channels)} \\
\hline
\end{tabular}
\end{center}
The superscript $(\alpha\beta)$ records whether the photon vertex ($\alpha$) and the gluon vertex ($\beta$) are helicity conserving (1, momentum in the numerator) or helicity flipping (0, mass in the numerator). Properties:
\begin{enumerate}
    \item $m_{f}\rightarrow 0$: $\G{11}\rightarrow\frac{1}{Y^{2}}\frac{QK_{1}(Q\mathcal{D})}{\mathcal{D}}$, $\Ga{11}\rightarrow\frac{1}{Y^{2}}\frac{Q^{2}K_{1}(\Tilde{Q}R)}{\Tilde{Q}R}$, $\GE\rightarrow\frac{QK_{1}(Q\mathcal{D})}{\mathcal{D}}$; all other functions appear multiplied by $m_{f}$ or $\om\propto m_{f}$, so the massless note is recovered.
    \item The regular functions are \emph{not} a single Bessel function of a shifted scale: the two energy denominators have different mass shifts ($\eps$ and $\om$), and the Feynman parameter $x$ interpolates between the two-body scale ($x\rightarrow 0$) and the three-body scale $\Ef$ ($x=1$). Only the instantaneous term is exactly $Q\rightarrow\Ef$.
    \item $Y\rightarrow 0$ (gluon collinear to the antiquark): $Y^{2}\G{11}$ and $Y^{2}\Ga{11}$ tend to the same limit $\frac{\eps K_{1}(\eps R)}{\zeta(1-\zeta)R}$, $Y^{2}\G{01}$ and $Y^{2}\Ga{01}$ to $\frac{K_{0}(\eps R)}{\zeta(1-\zeta)}$, and $\G{10}-\Ga{10}$, $\G{00}-\Ga{00}$ stay finite. Together with $\mathcal{W}(x,y,z,\ldots)\rightarrow\mathcal{W}(x,y',y',\ldots)$ this is the same ultraviolet cancellation as in the massless case.
    \item Large distances are cut off by $\eps$ (and $\Ef$) even at $Q^{2}=0$; the gluon emission after the shock wave is cut off at $Y\gtrsim 1/\om$ (dead cone).
\end{enumerate}
